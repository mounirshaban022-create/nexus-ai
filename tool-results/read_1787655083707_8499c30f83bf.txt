   345→interface ParsedToolCall {
   346→  tool: string
   347→  args: Record<string, unknown>
   348→}
   349→
   350→/** Balanced-brace JSON extraction (string-aware). */
   351→function extractJson(text: string): string | null {
   352→  const start = text.indexOf('{')
   353→  if (start === -1) return null
   354→  let depth = 0
   355→  let inString = false
   356→  let escaped = false
   357→  for (let i = start; i < text.length; i++) {
   358→    const ch = text[i]
   359→    if (inString) {
   360→      if (escaped) escaped = false
   361→      else if (ch === '\\') escaped = true
   362→      else if (ch === '"') inString = false
   363→      continue
   364→    }
   365→    if (ch === '"') inString = true
   366→    else if (ch === '{') depth++
   367→    else if (ch === '}') {
   368→      depth--
   369→      if (depth === 0) return text.slice(start, i + 1)
   370→    }
   371→  }
   372→  return null
   373→}
   374→
   375→function parseToolCall(text: string): ParsedToolCall | null {
   376→  const idx = text.indexOf('TOOL_CALL')
   377→  if (idx === -1) return null
   378→  const raw = text.slice(idx).replace(/```(?:json)?/g, '')
   379→  const candidate = extractJson(raw)
   380→  if (!candidate) return null
   381→  try {
   382→    const parsed = JSON.parse(candidate) as { tool?: unknown; args?: unknown }
   383→    if (typeof parsed.tool === 'string') {
   384→      return {
   385→        tool: parsed.tool,
   386→        args:
   387→          parsed.args && typeof parsed.args === 'object' && !Array.isArray(parsed.args)
   388→            ? (parsed.args as Record<string, unknown>)
   389→            : {},
   390→      }
   391→    }
   392→  } catch {
   393→    /* not valid */
   394→  }
   395→  return null
   396→}

   398→/**
   399→ * Phase 1 P2: parse one or more ARTIFACT_PATCH directives from a model
   400→ * response. Returns an array of patches (find/replace/note) targeting the
   401→ * open artifact. Returns [] when no patches are present.
   402→ *
   403→ * The model emits lines like:
   404→ *   ARTIFACT_PATCH: {"artifactId": "abc", "find": "X", "replace": "Y", "note": "..."}
   405→ *
   406→ * We extract each directive's JSON via the same balanced-brace extractor
   407→ * as TOOL_CALL — supports multiline values (escaped newlines) and skips
   408→ * malformed lines (returns the well-formed ones only).
   409→ */
   410→function parseArtifactPatches(text: string): Array<{
   411→  artifactId: string
   412→  find: string
   413→  replace: string
   414→  note?: string
   415→}> {
   416→  const patches: Array<{ artifactId: string; find: string; replace: string; note?: string }> = []
   417→  let searchFrom = 0
   418→  while (true) {
   419→    const idx = text.indexOf('ARTIFACT_PATCH', searchFrom)
   420→    if (idx === -1) break
   421→    // Skip the "ARTIFACT_PATCH" label and any trailing whitespace/colon
   422→    let cursor = idx + 'ARTIFACT_PATCH'.length
   423→    while (cursor < text.length && /[\s:]/.test(text[cursor])) cursor++
   424→    const candidate = extractJson(text.slice(cursor))
   425→    if (!candidate) {
   426→      searchFrom = idx + 1
   427→      continue
   428→    }
   429→    try {
   430→      const parsed = JSON.parse(candidate) as {
   431→        artifactId?: unknown
   432→        find?: unknown
   433→        replace?: unknown
   434→        note?: unknown
   435→      }
   436→      if (
   437→        typeof parsed.artifactId === 'string' &&
   438→        typeof parsed.find === 'string' &&
   439→        typeof parsed.replace === 'string'
   440→      ) {
   441→        patches.push({
   442→          artifactId: parsed.artifactId,
   443→          find: parsed.find,
   444→          replace: parsed.replace,
   445→          note: typeof parsed.note === 'string' ? parsed.note : undefined,
   446→        })
   447→      }
   448→    } catch {
   449→      /* malformed — skip */
   450→    }
   451→    searchFrom = cursor + candidate.length
   452→  }
   453→  return patches
   454→}

   456→function stripToolCall(text: string): string {
   457→  return text
   458→    .replace(/```(?:json)?\s*TOOL_CALL[\s\S]*?```\s*/g, '')
   459→    .replace(/TOOL_CALL\s*[:=][\s\S]*$/g, '')
   460→    .replace(/```(?:json)?\s*ARTIFACT_PATCH[\s\S]*?```\s*/g, '')
   461→    .replace(/ARTIFACT_PATCH\s*[:=][\s\S]*$/g, '')
   462→    .trim()
   463→}

   465→/**
   466→ * Phase 1 P1: detect a "remember: ..." prefix and extract the durable fact.
   467→ * Returns null when the message is not a memory request. Supports English
   468→ * ("remember:", "remember that ...", "remember to ...") and Arabic
   469→ * ("تذكّر:" / "تذكر:" / "تذكّر أنني ...") prefixes. The match is
   470→ * case-insensitive and tolerant of leading whitespace. This is the OPT-IN
   471→ * extraction path — NEXUS never silently extracts memory from arbitrary chat.
   472→ *
   473→ * Disambiguation: for English "remember ..." WITHOUT a colon and WITHOUT
   474→ * "that"/"to", we do NOT match — too ambiguous (could be "remember when we
   475→ * went to the park?"). For Arabic, the colon or the "أن"-prefix serves the
   476→ * same role.
   477→ */
   478→function parseRememberDirective(message: string): string | null {
   479→  // English: explicit colon OR "that"/"to" disambiguator
   480→  const enColon = message.match(/^\s*remember\s*:\s*(.+)$/i)
   481→  if (enColon) return enColon[1].trim().slice(0, 600)
   482→  const enThat = message.match(/^\s*remember\s+(?:that|to)\s+(.+)$/i)
   483→  if (enThat) return enThat[1].trim().slice(0, 600)
   484→  // Arabic: explicit colon OR "أن"-prefix (أن / أنني / أنك / أننا / إلخ)
   485→  // تذكّر (with shadda) and تذكر (without) are both accepted via alternation
   486→  // — the shadda (U+0651) is a combining mark that doesn't behave well in
   487→  // a character class.
   488→  const arColon = message.match(/^\s*(?:تذكّر|تذكر)\s*:\s*(.+)$/u)
   489→  if (arColon) return arColon[1].trim().slice(0, 600)
   490→  const arAnn = message.match(/^\s*(?:تذكّر|تذكر)\s+أن\p{L}*\s+(.+)$/u)
   491→  if (arAnn) return arAnn[1].trim().slice(0, 600)
   492→  return null
   493→}

   514→  if (toolId === 'generate_image') {

   518→      method: 'POST',

   527→    }
   528→  }
   529→
   530→  if (toolId === 'create_document') {
   531→    const format = String(args.format ?? 'docx').toLowerCase()
   532→    const title = String(args.title ?? 'Document').slice(0, 200)
   533→    const content = String(args.content ?? '')
   534→    if (!content) throw new Error('content required')
   535→
   536→    // Build blocks from plain content (headings via lines starting with #)
   537→    const blocks: Array<Record<string, unknown>> = []
   538→    for (const line of content.split('\n')) {
   539→      const trimmed = line.trim()
   540→      if (!trimmed) continue
   541→      if (trimmed.startsWith('## ')) blocks.push({ type: 'heading', text: trimmed.slice(3), level: 2 })
   542→      else if (trimmed.startsWith('# ')) blocks.push({ type: 'heading', text: trimmed.slice(2), level: 1 })
   543→      else if (/^[-•*]\s/.test(trimmed)) {
   544→        const last = blocks[blocks.length - 1]
   545→        if (last?.type === 'bullets') (last.items as string[]).push(trimmed.replace(/^[-•*]\s/, ''))
   546→        else blocks.push({ type: 'bullets', items: [trimmed.replace(/^[-•*]\s/, '')] })
   547→      } else {
   548→        blocks.push({ type: 'paragraph', text: trimmed })
   549→      }
   550→    }
   551→    if (format === 'pptx') {
   552→      // Convert to slides: each heading starts a slide
   553→      const slides: Array<Record<string, unknown>> = []

   555→      for (const b of blocks) {
   556→        if (b.type === 'heading') {
   557→          if (current) slides.push({ type: 'slide', ...current })
   558→          current = { title: b.text as string, bullets: [] }
   559→        } else if (current) {
   560→          const items = b.type === 'bullets' ? (b.items as string[]) : [b.text as string]
   561→          current.bullets.push(...items.slice(0, 5))
   562→        }
   563→      }
   564→      if (current) slides.push({ type: 'slide', ...current })
   565→      blocks.length = 0
   566→      blocks.push(...(slides.length ? slides : [{ type: 'paragraph', text: content }]))
   567→    }
   568→
   569→    const res = await fetch(`${origin}/api/office/create`, {
   570→      method: 'POST',
   571→      headers: { 'Content-Type': 'application/json' },
   572→      body: JSON.stringify({
   573→        format: ['docx', 'xlsx', 'pptx'].includes(format) ? format : 'docx',
   574→        title,
   575→        blocks: blocks.slice(0, 40),
   576→      }),
   577→    })

   579→    if (!res.ok) throw new Error(data.error || 'Document creation failed')
   580→    // Phase 1 P2: include the source content + a stable artifactId so the
   581→    // ArtifactPanel can render the editable text and the AI can target
   582→    // ARTIFACT_PATCH directives at this artifact across follow-up messages.

   585→      result: {
   586→        downloadUrl: `${data.file.url}?download=1&title=${encodeURIComponent(title)}`,
   587→        format: data.file.format,
   588→        note: 'Document created — give the user the download link.',
   589→      },
   590→      attachment: {
   591→        type: 'document',
   592→        url: `${data.file.url}?download=1&title=${encodeURIComponent(title)}`,
   593→        title,
   594→        format: data.file.format,
   595→        size: data.file.size,
   596→        artifactId,
   597→        sourceContent: content,
   598→      },
   599→    }
   600→  }
   601→
   602→  if (toolId === 'edit_document') {
   603→    // AI-DRIVEN DOCUMENT EDITING: applies natural-language edit

   605→    if (!activeDoc) {
   606→      throw new Error('No document is attached in this conversation. Ask the user to attach one first (paperclip button).')
   607→    }

   609→    if (!instructions) throw new Error('instructions required')

   611→
   612→    // Step 1: AI applies the edits → new markdown
   613→    const editedMarkdown = await smartChat(
   614→      [
   615→        {
   616→          role: 'assistant',
   617→          content:
   618→            'You are NEXUS\'s document editor. Apply the requested edits to the document below. ' +
   619→            'Return the COMPLETE edited document in Markdown — every section, nothing omitted, with all your changes applied. ' +
   620→            'Preserve the original structure and formatting unless the instruction says otherwise. ' +
   621→            'Respond ONLY with the edited Markdown document.',
   622→        },
   623→        {
   624→          role: 'user',
   625→          content: `DOCUMENT "${activeDoc.title}":\n\n${activeDoc.text.slice(0, 40000)}\n\nEDIT INSTRUCTIONS: ${instructions}`,
   626→        },
   627→      ],
   628→      { maxTokens: 4000, task: 'documents' }
   629→    )
   630→    const cleaned = editedMarkdown
   631→      .replace(/^```(?:markdown|md)?\s*\n?/i, '')
   632→      .replace(/\n?```\s*$/i, '')

   639→      headers: { 'Content-Type': 'application/json' },
   640→      body: JSON.stringify({ format: 'docx', title: newTitle, markdown: cleaned }),
   641→    })

   644→    const artifactId = `doc-edit-${data.file.id ?? Date.now()}`

   646→      result: {
   647→        downloadUrl: `${data.file.url}?download=1&title=${encodeURIComponent(newTitle)}`,

   654→        title: newTitle,
   655→        format: 'docx',
   656→        size: data.file.size,
   657→        artifactId,
   658→        sourceContent: cleaned,
   659→      },
   660→    }
   661→  }
   662→
   663→  if (toolId === 'pdf_operation') {
   664→    // STIRLING-PDF OPERATIONS on the attached PDF — real binary edits.

   666→      throw new Error('No PDF is attached in this conversation. Ask the user to attach one first (paperclip button).')
   667→    }
   668→    if (activeDoc.format !== 'pdf') {
   669→      throw new Error(`The attached file is a ${activeDoc.format.toUpperCase()}, not a PDF. PDF operations need a PDF attachment.`)
   670→    }
   671→    const operation = String(args.operation ?? '').trim()
   672→    const allowed = ['rotate', 'removePages', 'rearrange', 'split', 'watermark', 'singlePage', 'toHtml', 'toImages', 'info']
   673→    if (!allowed.includes(operation)) {

   684→      method: 'POST',
   685→      headers: { 'Content-Type': 'application/json' },
   686→      body: JSON.stringify({ operation, file: activeDoc.dataUrl, params }),
   687→    })
   688→    const data = await res.json()
   689→    if (!res.ok) throw new Error(data.error || `PDF ${operation} failed`)
   690→    if (data.info) {
   691→      return { result: { info: data.info, note: 'PDF analysis result — report it to the user.' } }

   702→        type: 'document',
   703→        url: `${f.url}?download=1&title=${encodeURIComponent(label)}`,

   724→        name?: string

   726→        rows?: unknown[][]
   727→        formulas?: Array<{ row: number; col: number; formula: string }>

   731→        (Array.isArray(r) ? r : []).map((c) => {

   759→          40,
   760→          Math.max(
   761→            10,
   762→            ...data.slice(0, 40).map((row) => String((row as unknown[])[i] ?? '').length + 2)

   809→    const r = data.result

   811→    // ArtifactPanel can show the code editor and the AI can target
   812→    // ARTIFACT_PATCH directives at this code artifact.
   813→    const artifactId = `code-${Date.now()}`

   815→      result: {
   816→        stdout: r.stdout?.slice(0, 3000),

   913→      const pathMod = await import('path')

   915→      await mkdir(dir, { recursive: true })
   916→      const id = randomUUID()
   917→      const file = pathMod.join(dir, `${id}.png`)
   918→      await runCli(['screenshot', '--path', file])
   919→      const screenshotUrl = `/browser-shots/${id}.png`
   920→      return {
   921→        result: { action, screenshotUrl, note: 'Screenshot taken — show it inline.' },
   922→        attachment: { type: 'image', url: screenshotUrl, title: 'Browser screenshot' },
   923→      }
   924→    }
   925→    throw new Error(`Unknown browser action "${action}". Use open|read|click|fill|press|screenshot.`)

  1359→          // NEXUS One: announce the agent takeover (or hand-back to NEXUS)
  1360→          // so the UI can render the live routing chip. Only emitted for
  1361→          // AUTO-routed sessions — pinned sessions already show their agent.
  1362→          if (routingDecision) {
  1363→            const decisionMeta = routingDecision.agentSlug ? getAgentMeta(routingDecision.agentSlug) : null
  1364→            send({
  1365→              type: 'agent_assign',
  1366→              agentSlug: routingDecision.agentSlug,
  1367→              name: routingDecision.agentName,
  1368→              division: routingDecision.division,
  1369→              emoji: routingDecision.emoji,
  1370→              color: routingDecision.color,
  1371→              reason: routingDecision.reason,
  1372→              elapsedMs: routingDecision.elapsedMs,
  1373→              pinned: false,
  1374→              divisionLabel: decisionMeta ? getDivision(decisionMeta.division)?.label ?? null : null,
  1375→            })
  1376→          }
  1377→
  1378→          // LLM conversation: system + history + tool exchanges
  1379→          const llmMessages: Array<{ role: string; content: string }> = [
  1380→            { role: 'assistant', content: systemPrompt },
  1381→            ...history.slice(-MAX_HISTORY).map((m) => ({ role: m.role, content: m.content })),
  1382→          ]
  1383→          // Include the new user message (history was fetched before insert)
  1384→          llmMessages.push({ role: 'user', content: effectiveMessage })
  1385→
  1386→          // SESSION DOCUMENT MEMORY: persist the attachment context as a
  1387→          // DATABASE chat message when first attached, so it survives
  1388→          // server restarts AND appears in history for follow-ups. For
  1389→          // follow-ups with no new attachment, restore from the DB history
  1390→          // (search for the attachment marker) or the in-memory cache.
  1391→          if (!attachment && session?.id) {
  1392→            // Try in-memory first (fast path)
  1393→            let stored = sessionDocs.get(session.id) ?? null
  1394→            if (!stored) {
  1395→              // Fall back to the DB: look for the attachment marker message
  1396→              const marker = history.find(
  1397→                (m) => m.role === 'user' && m.content.startsWith('[The user attached the document "')
  1398→              )
  1399→              if (marker) {
  1400→                const filenameMatch = marker.content.match(/"([^"]+)"/)
  1401→                const textMatch = marker.content.match(/DOCUMENT "[^"]+":\n([\s\S]*?)\n(You can answer|$)/)
  1402→                stored = {
  1403→                  title: filenameMatch?.[1] ?? 'document',
  1404→                  text: (textMatch?.[1] ?? marker.content).slice(0, 40000),
  1405→                  format: marker.content.includes('pdf_operation') ? 'pdf' : 'doc',
  1406→                  filename: filenameMatch?.[1] ?? 'document',
  1407→                  dataUrl: '', // not available from history — tools will ask to re-attach for binary ops
  1408→                }
  1409→                // Cache for future requests
  1410→                sessionDocs.set(session.id, stored)
  1411→              }
  1412→            }
  1413→            if (stored) {
  1414→              activeDoc = stored
  1415→              attachmentContextMessage =
  1416→                `[CONTEXT — the user attached the document "${stored.filename}" earlier in this conversation. Its full content remains available below.\n\n` +
  1417→                `DOCUMENT "${stored.title}":\n${stored.text.slice(0, 24000)}\n` +
  1418→                `Continue answering follow-up questions about this document from its content. The edit_document and ${stored.format === 'pdf' ? 'pdf_operation' : 'file'} tools can still operate on it.]`
  1419→            }
  1420→          } else if (attachment && session?.id && activeDoc) {
  1421→            // Remember the newly-attached document for the rest of the session
  1422→            sessionDocs.set(session.id, activeDoc)
  1423→            // ALSO persist as a DB message so it survives restarts
  1424→            try {
  1425→              await db.chatMessage.create({
  1426→                data: {
  1427→                  sessionId: session.id,
  1428→                  role: 'user',
  1429→                  content: attachmentContextMessage.slice(0, 30000),
  1430→                },
  1431→              })
  1432→            } catch {
  1433→              /* best-effort persistence */
  1434→            }
  1435→          }
  1436→
  1437→          // Document attachment context: inject the parsed document content
  1438→          // BEFORE the user's message so the AI sees it as context.
  1439→          if (attachmentContextMessage) {
  1440→            llmMessages.splice(llmMessages.length - 1, 0, {
  1441→              role: 'user',
  1442→              content: attachmentContextMessage,
  1443→            })
  1444→          }
  1445→
  1446→          // Optional thinking phase
  1447→          if (thinkingEnabled) {
  1448→            try {
  1449→              const thinking = (await smartChat([
  1450→                {
  1451→                  role: 'assistant',
  1452→                  content:
  1453→                    'You are the reasoning engine of NEXUS AI. Given a question, produce a SHORT internal analysis (max 80 words, telegraphic) covering: what the user needs, which tools to use (if any), and the answer structure. Output ONLY the analysis.',
  1454→                },
  1455→                { role: 'user', content: trimmed },
  1456→              ], { maxTokens: 300, task: 'reasoning' })).trim().slice(0, 900)
  1457→              if (thinking) {
  1458→                llmMessages.push({
  1459→                  role: 'user',
  1460→                  content: `[internal analysis: ${thinking}]`,
  1461→                })
  1462→              }
  1463→            } catch {
  1464→              /* best-effort */
  1465→            }
  1466→          }
  1467→
  1468→          const attachments: Array<Record<string, unknown>> = []
  1469→          let toolCallsUsed = 0
  1470→
  1471→          // Determine if we can stream from a user-connected provider
  1472→          // (their provider answers non-streamed via smartChat; otherwise
  1473→          // the free multi-AI pool streams token-by-token directly).
  1474→          const hasUserProvider = !!(await getActiveAiProvider())
  1475→
  1476→          for (let step = 0; step <= MAX_TOOL_CALLS; step++) {
  1477→            const isLast = step === MAX_TOOL_CALLS
  1478→
  1479→            // STREAMING PATH — emit assistant_start + assistant_delta chunks
  1480→            // as tokens arrive. Primary engine: the FREE MULTI-AI POOL
  1481→            // (LLM7 → Kilo Code → OVHcloud, task-routed) — no single
  1482→            // dependency, every engine has its own rate-limit budget.
  1483→            let content: string
  1484→            let didStream = false
  1485→            if (!hasUserProvider) {
  1486→              // Open a new assistant message bubble on the client
  1487→              send({ type: 'assistant_start', id: `as-${Date.now()}-${step}` })
  1488→              try {
  1489→                const r = await streamAnonymousFallbackChat(
  1490→                  llmMessages,
  1491→                  (delta) => send({ type: 'assistant_delta', delta }),
  1492→                  { maxTokens: 4000, timeoutMs: 60_000, task: 'chat' }
  1493→                )
  1494→                content = r.content
  1495→                console.log(`[api/chat] served by free AI pool: ${r.providerId}/${r.model}`)
  1496→              } catch (poolErr) {
  1497→                console.error(
  1498→                  '[api/chat] free AI pool failed, trying non-streamed smartChat:',
  1499→                  poolErr instanceof Error ? poolErr.message : poolErr
  1500→                )
  1501→                // Last resort: non-streamed smartChat (user provider →
  1502→                // free pool chain) and emit the visible prelude as one chunk.
  1503→                content = await smartChat(llmMessages, { maxTokens: 4000, task: 'chat' })
  1504→                // Emit only the visible prelude (text before any
  1505→                // TOOL_CALL / ARTIFACT_PATCH directive) — matches the
  1506→                // peek-buffer behaviour of the normal streaming path.
  1507→                const prelude = stripToolCall(content)
  1508→                if (prelude) send({ type: 'assistant_delta', delta: prelude })
  1509→              }
  1510→              didStream = true
  1511→              if (!content.trim()) throw new Error('Empty model response.')
  1512→            } else {
  1513→              content = await smartChat(llmMessages, { maxTokens: 4000, task: 'chat' })
  1514→              if (!content.trim()) throw new Error('Empty model response.')
  1515→            }
  1516→
  1517→            const toolCall = isLast ? null : parseToolCall(content)
  1518→
  1519→            if (toolCall && toolCallsUsed < MAX_TOOL_CALLS) {
  1520→              toolCallsUsed += 1
  1521→
  1522→              // If we streamed, the prelude (text before TOOL_CALL) is
  1523→              // already visible to the user — that's the desired UX
  1524→              // ("Let me search for that..." → tool runs → final answer).
  1525→              // If we didn't stream (external provider), emit the prelude now.
  1526→              if (!didStream) {
  1527→                const prelude = stripToolCall(content)
  1528→                send({ type: 'assistant_start', id: `as-${Date.now()}-${step}` })
  1529→                if (prelude) send({ type: 'assistant_delta', delta: prelude })
  1530→                send({ type: 'assistant_end', attachments: [] })
  1531→              } else {
  1532→                // Close the streamed bubble (may contain a prelude, may be empty)
  1533→                send({ type: 'assistant_end', attachments: [] })
  1534→              }
  1535→
  1536→              send({ type: 'tool_start', tool: toolCall.tool, args: toolCall.args, index: toolCallsUsed })
  1537→
  1538→              // Heartbeat: image generation can take 60s+ on the upstream
  1539→              // provider. Without intermediate events the user perceives a
  1540→              // freeze ("the app freezes with every action"). Emit a
  1541→              // tool_progress ping every 5s while the tool is running so the
  1542→              // client knows the stream is alive and can show "still working".
  1543→              const toolStartedAt = Date.now()
  1544→              const heartbeat = setInterval(() => {
  1545→                const elapsedMs = Date.now() - toolStartedAt
  1546→                send({
  1547→                  type: 'tool_progress',
  1548→                  tool: toolCall.tool,
  1549→                  index: toolCallsUsed,
  1550→                  elapsedMs,
  1551→                  message: `Still working on ${toolCall.tool}… ${Math.round(elapsedMs / 1000)}s`,
  1552→                })
  1553→              }, 5000)
  1554→
  1555→              let ok = true
  1556→              let result: unknown
  1557→              let attachment: Record<string, unknown> | undefined
  1558→              try {
  1559→                const executed = await executeChatTool(req, toolCall.tool, toolCall.args, activeDoc)
  1560→                result = executed.result
  1561→                attachment = executed.attachment
  1562→              } catch (error) {
  1563→                ok = false
  1564→                result = { error: error instanceof Error ? error.message : 'Tool failed.' }
  1565→              } finally {
  1566→                clearInterval(heartbeat)
  1567→              }
  1568→
  1569→              if (attachment) attachments.push(attachment)
  1570→              send({ type: 'tool_result', tool: toolCall.tool, ok, result, index: toolCallsUsed })
  1571→
  1572→              const resultJson = JSON.stringify(result).slice(0, 4000)
  1573→              llmMessages.push({ role: 'assistant', content })
  1574→              llmMessages.push({
  1575→                role: 'user',
  1576→                content: `TOOL_RESULT (${toolCall.tool}):\n${resultJson}\n\nContinue: use another TOOL_CALL if needed, or give your final answer.`,
  1577→              })
  1578→            } else {
  1579→              // Phase 1 P2: parse any ARTIFACT_PATCH directives from the
  1580→              // response BEFORE computing the visible final answer. Each
  1581→              // patch is emitted as an `artifact_patch` event for the client
  1582→              // to apply to the open artifact (ChatGPT-Canvas style targeted
  1583→              // edit, instead of regenerating the whole artifact). Only
  1584→              // patches targeting the currently-open artifact id are emitted
  1585→              // — guards against the model addressing a stale or fabricated id.
  1586→              if (openArtifact) {
  1587→                const patches = parseArtifactPatches(content)
  1588→                for (const patch of patches) {
  1589→                  if (patch.artifactId === openArtifact.artifactId) {
  1590→                    send({
  1591→                      type: 'artifact_patch',
  1592→                      artifactId: patch.artifactId,
  1593→                      find: patch.find,
  1594→                      replace: patch.replace,
  1595→                      note: patch.note,
  1596→                    })
  1597→                  }
  1598→                }
  1599→              }
  1600→
  1601→              // Final answer: strip both TOOL_CALL and ARTIFACT_PATCH
  1602→              // directives from the visible prose (leaving the model's
  1603→              // one-line acknowledgement like "Done — shortened the intro.").
  1604→              const finalAnswer = stripToolCall(content) || 'I could not complete that. Try rephrasing.'
  1605→              const assistantMessage = await db.chatMessage.create({
  1606→                data: {
  1607→                  sessionId: session!.id,
  1608→                  role: 'assistant',
  1609→                  content: finalAnswer,
  1610→                  // NEXUS One: persist attachments so image/video/document
  1611→                  // cards survive session resume.
  1612→                  attachments: attachments.length ? JSON.stringify(attachments) : null,
  1613→                },
  1614→              })
  1615→              await db.chatSession.update({
  1616→                where: { id: session!.id },
  1617→                data: { updatedAt: new Date() },
  1618→              })
  1619→              // Mirror the assistant message + session timestamp to Supabase
  1620→              if (verifiedUserId) {
  1621→                void supabaseUpsert('chat_messages', {
  1622→                  id: assistantMessage.id,
  1623→                  session_id: session!.id,
  1624→                  role: 'assistant',
  1625→                  content: finalAnswer,
  1626→                }, { onConflict: 'id' })
  1627→                void supabaseUpsert('chat_sessions', {
  1628→                  id: session!.id,
  1629→                  user_id: verifiedUserId,
  1630→                  title: session!.title,
  1631→                  kind: 'chat',
  1632→                  updated_at: new Date().toISOString(),
  1633→                }, { onConflict: 'id' })
  1634→              }
  1635→
  1636→              if (didStream) {
  1637→                // The streamed content IS the final answer. Just close the
  1638→                // bubble with attachments + emit done.
  1639→                send({ type: 'assistant_end', attachments })
  1640→              } else {
  1641→                // External provider path — emit the full content as one delta.
  1642→                send({ type: 'assistant_start', id: `as-${Date.now()}-${step}` })
  1643→                send({ type: 'assistant_delta', delta: finalAnswer })
  1644→                send({ type: 'assistant_end', attachments })
  1645→              }
  1646→              send({ type: 'done', sessionId: session!.id })
  1647→              break
  1648→            }
  1649→          }
  1650→        } catch (error) {
  1651→          console.error('[api/chat] stream error:', error)
  1652→          send({
  1653→            type: 'error',
  1654→            message: error instanceof Error ? error.message : 'Chat failed.',
  1655→          })
  1656→        } finally {
  1657→          controller.close()
  1658→        }
  1659→      },
  1660→    })
  1661→
  1662→    return new NextResponse(stream, {
  1663→      status: 200,
  1664→      headers: {
  1665→        'Content-Type': 'application/x-ndjson; charset=utf-8',
  1666→        'Cache-Control': 'no-cache, no-transform',
  1667→        'X-Accel-Buffering': 'no',
  1668→      },
  1669→    })
  1670→  } catch (error) {
  1671→    console.error('[api/chat] POST error:', error)
  1672→    const message = error instanceof Error ? error.message : 'Chat failed.'
  1673→    return NextResponse.json({ error: message }, { status: 500 })
  1674→  }

   256→/** Streams a chat completion from the built-in ZAI engine, calling `onDelta`
   257→ *  with each token-chunk as it arrives. Returns the full accumulated content.
   258→ *
   259→ *  PEEK LOGIC — avoids showing a half-formed TOOL_CALL to the user:
   260→ *    1. We buffer the first ~24 chars locally before emitting any delta.
   261→ *    2. If the buffered text starts to look like a TOOL_CALL, we keep
   262→ *       buffering (no deltas) until the stream ends, then return the full
   263→ *       content for tool-call parsing.
   264→ *    3. Otherwise, we flush the peeked prefix as a single delta, then emit
   265→ *       subsequent chunks as they arrive.
   266→ *
   267→ *  This is what makes the AI "feel fast" — the user sees the first token
   268→ *  within ~200ms instead of waiting 4-20s for the full response. */
   269→async function streamZaiChat(
   270→  messages: Array<{ role: string; content: string }>,
   271→  onDelta: (delta: string) => void,
   272→  signal?: { aborted: boolean }
   273→): Promise<string> {
   274→  const zai = await getZAI()
   275→  const streamBody = await zai.chat.completions.create({
   276→    messages: messages.map((m) => ({ role: m.role as any, content: m.content })),
   277→    stream: true,
   278→    max_tokens: 4096,
   279→    temperature: 0.7,
   280→    thinking: { type: 'disabled' },
   281→  })
   282→
   283→  // streamBody is a ReadableStream (SSE format) when stream:true is honored
   284→  if (!(streamBody instanceof ReadableStream)) {
   285→    // Some providers return a JSON object instead of a stream. Fall back.
   286→    const fallback = streamBody as { choices?: Array<{ message?: { content?: string } }> }
   287→    const content = fallback.choices?.[0]?.message?.content ?? ''
   288→    if (content) onDelta(content)
   289→    return content
   290→  }
   291→
   292→  const reader = (streamBody as ReadableStream<Uint8Array>).getReader()
   293→  const decoder = new TextDecoder()
   294→  let buf = ''
   295→  let fullContent = ''
   296→  // Phase 1 P2: 'tool' mode also covers ARTIFACT_PATCH — we buffer the
   297→  // directive silently so the user doesn't see raw JSON streaming in, then
   298→  // parse it after the stream completes (same shape as TOOL_CALL).
   299→  let mode: 'peeking' | 'streaming' | 'tool' = 'peeking'
   300→  const PEEK_LEN = 24
   301→
   302→  while (true) {
   303→    if (signal?.aborted) break
   304→    const { done, value } = await reader.read()
   305→    if (done) break
   306→    buf += decoder.decode(value, { stream: true })
   307→
   308→    // SSE: lines starting with "data:"
   309→    const lines = buf.split('\n')
   310→    buf = lines.pop() ?? ''
   311→
   312→    for (const line of lines) {
   313→      const trimmed = line.trim()
   314→      if (!trimmed.startsWith('data:')) continue
   315→      const data = trimmed.slice(5).trim()
   316→      if (!data || data === '[DONE]') continue
   317→      let deltaText = ''
   318→      try {
   319→        const json = JSON.parse(data)
   320→        deltaText = json?.choices?.[0]?.delta?.content ?? ''
   321→      } catch {
   322→        // Some providers send plain-text chunks; treat the line itself as delta
   323→        if (data && !data.startsWith('{')) deltaText = data
   324→      }
   325→      if (!deltaText) continue
   326→      fullContent += deltaText
   327→
   328→      if (mode === 'peeking') {
   329→        // Check if this looks like a tool-call OR artifact-patch forming
   330→        if (
   331→          fullContent.startsWith('TOOL_CALL') ||
   332→          fullContent.includes('TOOL_CALL') ||
   333→          fullContent.startsWith('ARTIFACT_PATCH') ||
   334→          fullContent.includes('ARTIFACT_PATCH')
   335→        ) {
   336→          mode = 'tool'
   337→          // Don't emit deltas — wait for the full content to parse as a directive
   338→        } else if (fullContent.length >= PEEK_LEN) {
   339→          // Safe to assume this is a final answer — flush the peeked prefix
   340→          mode = 'streaming'
   341→          onDelta(fullContent)
   342→        }
   343→        // else: keep peeking
   344→      } else if (mode === 'streaming') {
   345→        // Emit just the new chunk (prefix already flushed)
   346→        onDelta(deltaText)
   347→      }
   348→      // mode === 'tool': silently accumulate
   349→    }
   350→  }
   351→
   352→  return fullContent
   353→}

   355→interface ParsedToolCall {
   356→  tool: string
   357→  args: Record<string, unknown>
   358→}
   359→
   360→/** Balanced-brace JSON extraction (string-aware). */
   361→function extractJson(text: string): string | null {
   362→  const start = text.indexOf('{')
   363→  if (start === -1) return null
   364→  let depth = 0
   365→  let inString = false
   366→  let escaped = false
   367→  for (let i = start; i < text.length; i++) {
   368→    const ch = text[i]
   369→    if (inString) {
   370→      if (escaped) escaped = false
   371→      else if (ch === '\\') escaped = true
   372→      else if (ch === '"') inString = false
   373→      continue
   374→    }
   375→    if (ch === '"') inString = true
   376→    else if (ch === '{') depth++
   377→    else if (ch === '}') {
   378→      depth--
   379→      if (depth === 0) return text.slice(start, i + 1)
   380→    }
   381→  }
   382→  return null
   383→}
   384→
   385→function parseToolCall(text: string): ParsedToolCall | null {
   386→  const idx = text.indexOf('TOOL_CALL')
   387→  if (idx === -1) return null
   388→  const raw = text.slice(idx).replace(/```(?:json)?/g, '')
   389→  const candidate = extractJson(raw)
   390→  if (!candidate) return null
   391→  try {
   392→    const parsed = JSON.parse(candidate) as { tool?: unknown; args?: unknown }
   393→    if (typeof parsed.tool === 'string') {
   394→      return {
   395→        tool: parsed.tool,
   396→        args:
   397→          parsed.args && typeof parsed.args === 'object' && !Array.isArray(parsed.args)
   398→            ? (parsed.args as Record<string, unknown>)
   399→            : {},
   400→      }
   401→    }
   402→  } catch {
   403→    /* not valid */
   404→  }
   405→  return null
   406→}
   407→
   408→/**
   409→ * Phase 1 P2: parse one or more ARTIFACT_PATCH directives from a model
   410→ * response. Returns an array of patches (find/replace/note) targeting the
   411→ * open artifact. Returns [] when no patches are present.
   412→ *
   413→ * The model emits lines like:
   414→ *   ARTIFACT_PATCH: {"artifactId": "abc", "find": "X", "replace": "Y", "note": "..."}
   415→ *
   416→ * We extract each directive's JSON via the same balanced-brace extractor
   417→ * as TOOL_CALL — supports multiline values (escaped newlines) and skips
   418→ * malformed lines (returns the well-formed ones only).
   419→ */
   420→function parseArtifactPatches(text: string): Array<{
   421→  artifactId: string
   422→  find: string
   423→  replace: string
   424→  note?: string
   425→}> {
   426→  const patches: Array<{ artifactId: string; find: string; replace: string; note?: string }> = []
   427→  let searchFrom = 0
   428→  while (true) {
   429→    const idx = text.indexOf('ARTIFACT_PATCH', searchFrom)
   430→    if (idx === -1) break
   431→    // Skip the "ARTIFACT_PATCH" label and any trailing whitespace/colon
   432→    let cursor = idx + 'ARTIFACT_PATCH'.length
   433→    while (cursor < text.length && /[\s:]/.test(text[cursor])) cursor++
   434→    const candidate = extractJson(text.slice(cursor))
   435→    if (!candidate) {
   436→      searchFrom = idx + 1
   437→      continue
   438→    }
   439→    try {
   440→      const parsed = JSON.parse(candidate) as {
   441→        artifactId?: unknown
   442→        find?: unknown
   443→        replace?: unknown
   444→        note?: unknown
   445→      }
   446→      if (
   447→        typeof parsed.artifactId === 'string' &&
   448→        typeof parsed.find === 'string' &&
   449→        typeof parsed.replace === 'string'
   450→      ) {
   451→        patches.push({
   452→          artifactId: parsed.artifactId,
   453→          find: parsed.find,
   454→          replace: parsed.replace,
   455→          note: typeof parsed.note === 'string' ? parsed.note : undefined,
   456→        })
   457→      }
   458→    } catch {
   459→      /* malformed — skip */
   460→    }
   461→    searchFrom = cursor + candidate.length
   462→  }
   463→  return patches
   464→}

   466→function stripToolCall(text: string): string {
   467→  return text
   468→    .replace(/```(?:json)?\s*TOOL_CALL[\s\S]*?```\s*/g, '')
   469→    .replace(/TOOL_CALL\s*[:=][\s\S]*$/g, '')
   470→    .replace(/```(?:json)?\s*ARTIFACT_PATCH[\s\S]*?```\s*/g, '')
   471→    .replace(/ARTIFACT_PATCH\s*[:=][\s\S]*$/g, '')
   472→    .trim()
   473→}

   475→/**
   476→ * Phase 1 P1: detect a "remember: ..." prefix and extract the durable fact.
   477→ * Returns null when the message is not a memory request. Supports English
   478→ * ("remember:", "remember that ...", "remember to ...") and Arabic
   479→ * ("تذكّر:" / "تذكر:" / "تذكّر أنني ...") prefixes. The match is
   480→ * case-insensitive and tolerant of leading whitespace. This is the OPT-IN
   481→ * extraction path — NEXUS never silently extracts memory from arbitrary chat.
   482→ *
   483→ * Disambiguation: for English "remember ..." WITHOUT a colon and WITHOUT
   484→ * "that"/"to", we do NOT match — too ambiguous (could be "remember when we
   485→ * went to the park?"). For Arabic, the colon or the "أن"-prefix serves the
   486→ * same role.
   487→ */
   488→function parseRememberDirective(message: string): string | null {
   489→  // English: explicit colon OR "that"/"to" disambiguator
   490→  const enColon = message.match(/^\s*remember\s*:\s*(.+)$/i)
   491→  if (enColon) return enColon[1].trim().slice(0, 600)
   492→  const enThat = message.match(/^\s*remember\s+(?:that|to)\s+(.+)$/i)
   493→  if (enThat) return enThat[1].trim().slice(0, 600)
   494→  // Arabic: explicit colon OR "أن"-prefix (أن / أنني / أنك / أننا / إلخ)
   495→  // تذكّر (with shadda) and تذكر (without) are both accepted via alternation
   496→  // — the shadda (U+0651) is a combining mark that doesn't behave well in
   497→  // a character class.
   498→  const arColon = message.match(/^\s*(?:تذكّر|تذكر)\s*:\s*(.+)$/u)
   499→  if (arColon) return arColon[1].trim().slice(0, 600)
   500→  const arAnn = message.match(/^\s*(?:تذكّر|تذكر)\s+أن\p{L}*\s+(.+)$/u)
   501→  if (arAnn) return arAnn[1].trim().slice(0, 600)
   502→  return null
   503→}

   505→/** Executes a chat-ability tool (image/doc/code), returning an attachment. */
   506→async function executeChatTool(
   507→  toolId: string,
   508→  args: Record<string, unknown>
   509→): Promise<{ result: unknown; attachment?: Record<string, unknown> }> {
   510→  const origin = 'http://localhost:3000'
   511→
   512→  if (toolId === 'generate_image') {
   513→    const prompt = String(args.prompt ?? '').slice(0, 2000)
   514→    if (!prompt) throw new Error('prompt required')
   515→    const res = await fetch(`${origin}/api/image`, {
   516→      method: 'POST',
   517→      headers: { 'Content-Type': 'application/json' },
   518→      body: JSON.stringify({ prompt, size: '1024x1024' }),
   519→    })
   520→    const data = await res.json()
   521→    if (!res.ok) throw new Error(data.error || 'Image generation failed')
   522→    return {
   523→      result: { imageUrl: data.image.url, note: 'Image generated — show it inline.' },
   524→      attachment: { type: 'image', url: data.image.url, title: prompt.slice(0, 80) },
   525→    }
   526→  }
   527→
   528→  if (toolId === 'create_document') {
   529→    const format = String(args.format ?? 'docx').toLowerCase()
   530→    const title = String(args.title ?? 'Document').slice(0, 200)
   531→    const content = String(args.content ?? '')
   532→    if (!content) throw new Error('content required')
   533→
   534→    // Build blocks from plain content (headings via lines starting with #)
   535→    const blocks: Array<Record<string, unknown>> = []
   536→    for (const line of content.split('\n')) {
   537→      const trimmed = line.trim()
   538→      if (!trimmed) continue
   539→      if (trimmed.startsWith('## ')) blocks.push({ type: 'heading', text: trimmed.slice(3), level: 2 })
   540→      else if (trimmed.startsWith('# ')) blocks.push({ type: 'heading', text: trimmed.slice(2), level: 1 })
   541→      else if (/^[-•*]\s/.test(trimmed)) {
   542→        const last = blocks[blocks.length - 1]
   543→        if (last?.type === 'bullets') (last.items as string[]).push(trimmed.replace(/^[-•*]\s/, ''))
   544→        else blocks.push({ type: 'bullets', items: [trimmed.replace(/^[-•*]\s/, '')] })
   545→      } else {
   546→        blocks.push({ type: 'paragraph', text: trimmed })
   547→      }
   548→    }
   549→    if (format === 'pptx') {
   550→      // Convert to slides: each heading starts a slide
   551→      const slides: Array<Record<string, unknown>> = []
   552→      let current: { title: string; bullets: string[] } | null = null
   553→      for (const b of blocks) {
   554→        if (b.type === 'heading') {
   555→          if (current) slides.push({ type: 'slide', ...current })
   556→          current = { title: b.text as string, bullets: [] }
   557→        } else if (current) {
   558→          const items = b.type === 'bullets' ? (b.items as string[]) : [b.text as string]
   559→          current.bullets.push(...items.slice(0, 5))
   560→        }
   561→      }
   562→      if (current) slides.push({ type: 'slide', ...current })
   563→      blocks.length = 0
   564→      blocks.push(...(slides.length ? slides : [{ type: 'paragraph', text: content }]))
   565→    }
   566→
   567→    const res = await fetch(`${origin}/api/office/create`, {
   568→      method: 'POST',
   569→      headers: { 'Content-Type': 'application/json' },
   570→      body: JSON.stringify({
   571→        format: ['docx', 'xlsx', 'pptx'].includes(format) ? format : 'docx',
   572→        title,
   573→        blocks: blocks.slice(0, 40),
   574→      }),
   575→    })
   576→    const data = await res.json()
   577→    if (!res.ok) throw new Error(data.error || 'Document creation failed')
   578→    // Phase 1 P2: include the source content + a stable artifactId so the
   579→    // ArtifactPanel can render the editable text and the AI can target
   580→    // ARTIFACT_PATCH directives at this artifact across follow-up messages.
   581→    const artifactId = `doc-${data.file.id ?? Date.now()}`
   582→    return {
   583→      result: {
   584→        downloadUrl: `${data.file.url}?download=1&title=${encodeURIComponent(title)}`,
   585→        format: data.file.format,
   586→        note: 'Document created — give the user the download link.',
   587→      },
   588→      attachment: {
   589→        type: 'document',
   590→        url: `${data.file.url}?download=1&title=${encodeURIComponent(title)}`,
   591→        title,
   592→        format: data.file.format,
   593→        size: data.file.size,
   594→        artifactId,
   595→        sourceContent: content,
   596→      },
   597→    }
   598→  }
   599→
   600→  if (toolId === 'run_code') {
   601→    const language = String(args.language ?? 'javascript').toLowerCase()
   602→    const code = String(args.code ?? '')
   603→    if (!code) throw new Error('code required')
   604→    const res = await fetch(`${origin}/api/code/run`, {
   605→      method: 'POST',
   606→      headers: { 'Content-Type': 'application/json' },
   607→      body: JSON.stringify({
   608→        language: ['javascript', 'typescript', 'python'].includes(language) ? language : 'javascript',
   609→        code,
   610→      }),
   611→    })
   612→    const data = await res.json()
   613→    if (!res.ok) throw new Error(data.error || 'Code execution failed')
   614→    const r = data.result
   615→    // Phase 1 P2: include the source code + a stable artifactId so the
   616→    // ArtifactPanel can show the code editor and the AI can target
   617→    // ARTIFACT_PATCH directives at this code artifact.
   618→    const artifactId = `code-${Date.now()}`
   619→    return {
   620→      result: {
   621→        stdout: r.stdout?.slice(0, 3000),
   622→        stderr: r.stderr?.slice(0, 1500),
   623→        exitCode: r.exitCode,
   624→        note: 'Code executed — include the output in your answer.',
   625→      },
   626→      attachment: {
   627→        type: 'code',
   628→        language,
   629→        stdout: (r.stdout ?? '').slice(0, 2000),
   630→        stderr: (r.stderr ?? '').slice(0, 500),
   631→        exitCode: r.exitCode,
   632→        artifactId,
   633→        sourceContent: code,
   634→      },
   635→    }
   636→  }
   637→
   638→  // Data connectors pass through
   639→  const connector = CONNECTOR_MAP.get(toolId)
   640→  if (!connector) return { result: { error: `Unknown tool "${toolId}"` } }
   641→  const validated = validateConnectorArgs(connector, args)
   642→  if (!validated.ok) return { result: { error: validated.error } }
   643→  const result = await connector.execute(validated.args)
   644→  // Attach search results richly
   645→  if (toolId === 'web_search' && Array.isArray((result as { results?: unknown[] }).results)) {
   646→    const results = (result as { results: Array<{ title: string; url: string; snippet: string }> }).results
   647→    return {
   648→      result,
   649→      attachment: {
   650→        type: 'search',
   651→        results: results.slice(0, 5).map((r) => ({ title: r.title, url: r.url, snippet: r.snippet?.slice(0, 140) })),
   652→      },
   653→    }
   654→  }
   655→  return { result }
   656→}

   960→                attachment = executed.attachment
   961→              } catch (error) {

   963→                result = { error: error instanceof Error ? error.message : 'Tool failed.' }
   964→              } finally {
   965→                clearInterval(heartbeat)
   966→              }
   967→

   969→              send({ type: 'tool_result', tool: toolCall.tool, ok, result, index: toolCallsUsed })
   970→

   972→              llmMessages.push({ role: 'assistant', content })
   973→              llmMessages.push({
   974→                role: 'user',
   975→                content: `TOOL_RESULT (${toolCall.tool}):\n${resultJson}\n\nContinue: use another TOOL_CALL if needed, or give your final answer.`,
   976→              })
   977→            } else {
   978→              // Phase 1 P2: parse any ARTIFACT_PATCH directives from the
   979→              // response BEFORE computing the visible final answer. Each
   980→              // patch is emitted as an `artifact_patch` event for the client
   981→              // to apply to the open artifact (ChatGPT-Canvas style targeted
   982→              // edit, instead of regenerating the whole artifact). Only
   983→              // patches targeting the currently-open artifact id are emitted
   984→              // — guards against the model addressing a stale or fabricated id.
   985→              if (openArtifact) {
   986→                const patches = parseArtifactPatches(content)
   987→                for (const patch of patches) {
   988→                  if (patch.artifactId === openArtifact.artifactId) {
   989→                    send({
   990→                      type: 'artifact_patch',
   991→                      artifactId: patch.artifactId,
   992→                      find: patch.find,
   993→                      replace: patch.replace,
   994→                      note: patch.note,
   995→                    })
   996→                  }
   997→                }
   998→              }
   999→
  1000→              // Final answer: strip both TOOL_CALL and ARTIFACT_PATCH
  1001→              // directives from the visible prose (leaving the model's
  1002→              // one-line acknowledgement like "Done — shortened the intro.").
  1003→              const finalAnswer = stripToolCall(content) || 'I could not complete that. Try rephrasing.'
  1004→              await db.chatMessage.create({
  1005→                data: { sessionId: session!.id, role: 'assistant', content: finalAnswer },
  1006→              })
  1007→              await db.chatSession.update({
  1008→                where: { id: session!.id },
  1009→                data: { updatedAt: new Date() },
  1010→              })
  1011→
  1012→              if (didStream) {
  1013→                // The streamed content IS the final answer. Just close the
  1014→                // bubble with attachments + emit done.
  1015→                send({ type: 'assistant_end', attachments })
  1016→              } else {
  1017→                // External provider path — emit the full content as one delta.
  1018→                send({ type: 'assistant_start', id: `as-${Date.now()}-${step}` })
  1019→                send({ type: 'assistant_delta', delta: finalAnswer })
  1020→                send({ type: 'assistant_end', attachments })
  1021→              }
  1022→              send({ type: 'done', sessionId: session!.id })
  1023→              break
  1024→            }
  1025→          }
  1026→        } catch (error) {
  1027→          console.error('[api/chat] stream error:', error)
  1028→          send({
  1029→            type: 'error',
  1030→            message: error instanceof Error ? error.message : 'Chat failed.',
  1031→          })
  1032→        } finally {
  1033→          controller.close()
  1034→        }
  1035→      },
  1036→    })

  1039→      status: 200,
  1040→      headers: {
  1041→        'Content-Type': 'application/x-ndjson; charset=utf-8',
  1042→        'Cache-Control': 'no-cache, no-transform',
  1043→        'X-Accel-Buffering': 'no',
  1044→      },
  1045→    })
  1046→  } catch (error) {
  1047→    console.error('[api/chat] POST error:', error)
  1048→    const message = error instanceof Error ? error.message : 'Chat failed.'

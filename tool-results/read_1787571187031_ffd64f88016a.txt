   126→   126→/** Streams a chat completion from the built-in ZAI engine, calling `onDelta`
   127→   127→ *  with each token-chunk as it arrives. Returns the full accumulated content.
   128→   128→ *
   129→   129→ *  PEEK LOGIC — avoids showing a half-formed TOOL_CALL to the user:
   130→   130→ *    1. We buffer the first ~24 chars locally before emitting any delta.
   131→   131→ *    2. If the buffered text starts to look like a TOOL_CALL, we keep
   132→   132→ *       buffering (no deltas) until the stream ends, then return the full
   133→   133→ *       content for tool-call parsing.
   134→   134→ *    3. Otherwise, we flush the peeked prefix as a single delta, then emit
   135→   135→ *       subsequent chunks as they arrive.
   136→   136→ *
   137→   137→ *  This is what makes the AI "feel fast" — the user sees the first token
   138→   138→ *  within ~200ms instead of waiting 4-20s for the full response. */
   139→   139→async function streamZaiChat(
   140→   140→  messages: Array<{ role: string; content: string }>,
   141→   141→  onDelta: (delta: string) => void,
   142→   142→  signal?: { aborted: boolean }
   143→   143→): Promise<string> {
   144→   144→  const zai = await getZAI()
   145→   145→  const streamBody = await zai.chat.completions.create({
   146→   146→    messages: messages.map((m) => ({ role: m.role as any, content: m.content })),
   147→   147→    stream: true,
   148→   148→    max_tokens: 4096,
   149→   149→    temperature: 0.7,
   150→   150→    thinking: { type: 'disabled' },
   151→   151→  })
   152→   152→
   153→   153→  // streamBody is a ReadableStream (SSE format) when stream:true is honored
   154→   154→  if (!(streamBody instanceof ReadableStream)) {
   155→   155→    // Some providers return a JSON object instead of a stream. Fall back.
   156→   156→    const fallback = streamBody as { choices?: Array<{ message?: { content?: string } }> }
   157→   157→    const content = fallback.choices?.[0]?.message?.content ?? ''
   158→   158→    if (content) onDelta(content)
   159→   159→    return content
   160→   160→  }
   161→   161→
   162→   162→  const reader = (streamBody as ReadableStream<Uint8Array>).getReader()
   163→   163→  const decoder = new TextDecoder()
   164→   164→  let buf = ''
   165→   165→  let fullContent = ''
   166→   166→  let mode: 'peeking' | 'streaming' | 'tool' = 'peeking'
   167→   167→  const PEEK_LEN = 24
   168→   168→
   169→   169→  while (true) {
   170→   170→    if (signal?.aborted) break
   171→   171→    const { done, value } = await reader.read()
   172→   172→    if (done) break
   173→   173→    buf += decoder.decode(value, { stream: true })
   174→   174→
   175→   175→    // SSE: lines starting with "data:"
   176→   176→    const lines = buf.split('\n')
   177→   177→    buf = lines.pop() ?? ''
   178→   178→
   179→   179→    for (const line of lines) {
   180→   180→      const trimmed = line.trim()
   181→   181→      if (!trimmed.startsWith('data:')) continue
   182→   182→      const data = trimmed.slice(5).trim()
   183→   183→      if (!data || data === '[DONE]') continue
   184→   184→      let deltaText = ''
   185→   185→      try {
   186→   186→        const json = JSON.parse(data)
   187→   187→        deltaText = json?.choices?.[0]?.delta?.content ?? ''
   188→   188→      } catch {
   189→   189→        // Some providers send plain-text chunks; treat the line itself as delta
   190→   190→        if (data && !data.startsWith('{')) deltaText = data
   191→   191→      }
   192→   192→      if (!deltaText) continue
   193→   193→      fullContent += deltaText
   194→   194→
   195→   195→      if (mode === 'peeking') {
   196→   196→        // Check if this looks like a tool-call forming
   197→   197→        if (fullContent.startsWith('TOOL_CALL') || fullContent.includes('TOOL_CALL')) {
   198→   198→          mode = 'tool'
   199→   199→          // Don't emit deltas — wait for the full content to parse as a tool call
   200→   200→        } else if (fullContent.length >= PEEK_LEN) {
   201→   201→          // Safe to assume this is a final answer — flush the peeked prefix
   202→   202→          mode = 'streaming'
   203→   203→          onDelta(fullContent)
   204→   204→        }
   205→   205→        // else: keep peeking
   206→   206→      } else if (mode === 'streaming') {
   207→   207→        // Emit just the new chunk (prefix already flushed)
   208→   208→        onDelta(deltaText)
   209→   209→      }
   210→   210→      // mode === 'tool': silently accumulate
   211→   211→    }
   212→   212→  }
   213→   213→
   214→   214→  return fullContent
   215→   215→}

   217→   217→interface ParsedToolCall {
   218→   218→  tool: string
   219→   219→  args: Record<string, unknown>
   220→   220→}
   221→   221→
   222→   222→/** Balanced-brace JSON extraction (string-aware). */
   223→   223→function extractJson(text: string): string | null {
   224→   224→  const start = text.indexOf('{')
   225→   225→  if (start === -1) return null
   226→   226→  let depth = 0
   227→   227→  let inString = false
   228→   228→  let escaped = false
   229→   229→  for (let i = start; i < text.length; i++) {
   230→   230→    const ch = text[i]
   231→   231→    if (inString) {
   232→   232→      if (escaped) escaped = false
   233→   233→      else if (ch === '\\') escaped = true
   234→   234→      else if (ch === '"') inString = false
   235→   235→      continue
   236→   236→    }
   237→   237→    if (ch === '"') inString = true
   238→   238→    else if (ch === '{') depth++
   239→   239→    else if (ch === '}') {
   240→   240→      depth--
   241→   241→      if (depth === 0) return text.slice(start, i + 1)
   242→   242→    }
   243→   243→  }
   244→   244→  return null
   245→   245→}
   246→   246→
   247→   247→function parseToolCall(text: string): ParsedToolCall | null {
   248→   248→  const idx = text.indexOf('TOOL_CALL')
   249→   249→  if (idx === -1) return null
   250→   250→  const raw = text.slice(idx).replace(/```(?:json)?/g, '')
   251→   251→  const candidate = extractJson(raw)
   252→   252→  if (!candidate) return null
   253→   253→  try {
   254→   254→    const parsed = JSON.parse(candidate) as { tool?: unknown; args?: unknown }
   255→   255→    if (typeof parsed.tool === 'string') {
   256→   256→      return {
   257→   257→        tool: parsed.tool,
   258→   258→        args:
   259→   259→          parsed.args && typeof parsed.args === 'object' && !Array.isArray(parsed.args)
   260→   260→            ? (parsed.args as Record<string, unknown>)
   261→   261→            : {},
   262→   262→      }
   263→   263→    }
   264→   264→  } catch {
   265→   265→    /* not valid */
   266→   266→  }
   267→   267→  return null
   268→   268→}

   277→   277→/** Executes a chat-ability tool (image/doc/code), returning an attachment. */
   278→   278→async function executeChatTool(
   279→   279→  toolId: string,
   280→   280→  args: Record<string, unknown>
   281→   281→): Promise<{ result: unknown; attachment?: Record<string, unknown> }> {
   282→   282→  const origin = 'http://localhost:3000'
   283→   283→
   284→   284→  if (toolId === 'generate_image') {
   285→   285→    const prompt = String(args.prompt ?? '').slice(0, 2000)
   286→   286→    if (!prompt) throw new Error('prompt required')
   287→   287→    const res = await fetch(`${origin}/api/image`, {
   288→   288→      method: 'POST',
   289→   289→      headers: { 'Content-Type': 'application/json' },
   290→   290→      body: JSON.stringify({ prompt, size: '1024x1024' }),
   291→   291→    })
   292→   292→    const data = await res.json()
   293→   293→    if (!res.ok) throw new Error(data.error || 'Image generation failed')
   294→   294→    return {
   295→   295→      result: { imageUrl: data.image.url, note: 'Image generated — show it inline.' },
   296→   296→      attachment: { type: 'image', url: data.image.url, title: prompt.slice(0, 80) },
   297→   297→    }
   298→   298→  }
   299→   299→
   300→   300→  if (toolId === 'create_document') {
   301→   301→    const format = String(args.format ?? 'docx').toLowerCase()
   302→   302→    const title = String(args.title ?? 'Document').slice(0, 200)
   303→   303→    const content = String(args.content ?? '')
   304→   304→    if (!content) throw new Error('content required')
   305→   305→
   306→   306→    // Build blocks from plain content (headings via lines starting with #)
   307→   307→    const blocks: Array<Record<string, unknown>> = []
   308→   308→    for (const line of content.split('\n')) {
   309→   309→      const trimmed = line.trim()
   310→   310→      if (!trimmed) continue
   311→   311→      if (trimmed.startsWith('## ')) blocks.push({ type: 'heading', text: trimmed.slice(3), level: 2 })
   312→   312→      else if (trimmed.startsWith('# ')) blocks.push({ type: 'heading', text: trimmed.slice(2), level: 1 })
   313→   313→      else if (/^[-•*]\s/.test(trimmed)) {
   314→   314→        const last = blocks[blocks.length - 1]
   315→   315→        if (last?.type === 'bullets') (last.items as string[]).push(trimmed.replace(/^[-•*]\s/, ''))
   316→   316→        else blocks.push({ type: 'bullets', items: [trimmed.replace(/^[-•*]\s/, '')] })
   317→   317→      } else {
   318→   318→        blocks.push({ type: 'paragraph', text: trimmed })
   319→   319→      }
   320→   320→    }
   321→   321→    if (format === 'pptx') {
   322→   322→      // Convert to slides: each heading starts a slide
   323→   323→      const slides: Array<Record<string, unknown>> = []
   324→   324→      let current: { title: string; bullets: string[] } | null = null
   325→   325→      for (const b of blocks) {
   326→   326→        if (b.type === 'heading') {
   327→   327→          if (current) slides.push({ type: 'slide', ...current })
   328→   328→          current = { title: b.text as string, bullets: [] }
   329→   329→        } else if (current) {
   330→   330→          const items = b.type === 'bullets' ? (b.items as string[]) : [b.text as string]
   331→   331→          current.bullets.push(...items.slice(0, 5))
   332→   332→        }
   333→   333→      }
   334→   334→      if (current) slides.push({ type: 'slide', ...current })
   335→   335→      blocks.length = 0
   336→   336→      blocks.push(...(slides.length ? slides : [{ type: 'paragraph', text: content }]))
   337→   337→    }
   338→   338→
   339→   339→    const res = await fetch(`${origin}/api/office/create`, {
   340→   340→      method: 'POST',
   341→   341→      headers: { 'Content-Type': 'application/json' },
   342→   342→      body: JSON.stringify({
   343→   343→        format: ['docx', 'xlsx', 'pptx'].includes(format) ? format : 'docx',
   344→   344→        title,
   345→   345→        blocks: blocks.slice(0, 40),
   346→   346→      }),
   347→   347→    })
   348→   348→    const data = await res.json()
   349→   349→    if (!res.ok) throw new Error(data.error || 'Document creation failed')
   350→   350→    return {
   351→   351→      result: {
   352→   352→        downloadUrl: `${data.file.url}?download=1&title=${encodeURIComponent(title)}`,
   353→   353→        format: data.file.format,
   354→   354→        note: 'Document created — give the user the download link.',
   355→   355→      },
   356→   356→      attachment: {
   357→   357→        type: 'document',
   358→   358→        url: `${data.file.url}?download=1&title=${encodeURIComponent(title)}`,
   359→   359→        title,
   360→   360→        format: data.file.format,
   361→   361→        size: data.file.size,
   362→   362→      },
   363→   363→    }
   364→   364→  }
   365→   365→
   366→   366→  if (toolId === 'run_code') {
   367→   367→    const language = String(args.language ?? 'javascript').toLowerCase()
   368→   368→    const code = String(args.code ?? '')
   369→   369→    if (!code) throw new Error('code required')
   370→   370→    const res = await fetch(`${origin}/api/code/run`, {
   371→   371→      method: 'POST',
   372→   372→      headers: { 'Content-Type': 'application/json' },
   373→   373→      body: JSON.stringify({
   374→   374→        language: ['javascript', 'typescript', 'python'].includes(language) ? language : 'javascript',
   375→   375→        code,
   376→   376→      }),
   377→   377→    })
   378→   378→    const data = await res.json()
   379→   379→    if (!res.ok) throw new Error(data.error || 'Code execution failed')
   380→   380→    const r = data.result
   381→   381→    return {
   382→   382→      result: {
   383→   383→        stdout: r.stdout?.slice(0, 3000),
   384→   384→        stderr: r.stderr?.slice(0, 1500),
   385→   385→        exitCode: r.exitCode,
   386→   386→        note: 'Code executed — include the output in your answer.',
   387→   387→      },
   388→   388→      attachment: {
   389→   389→        type: 'code',
   390→   390→        language,
   391→   391→        stdout: (r.stdout ?? '').slice(0, 2000),
   392→   392→        stderr: (r.stderr ?? '').slice(0, 500),
   393→   393→        exitCode: r.exitCode,
   394→   394→      },
   395→   395→    }
   396→   396→  }
   397→   397→
   398→   398→  // Data connectors pass through
   399→   399→  const connector = CONNECTOR_MAP.get(toolId)
   400→   400→  if (!connector) return { result: { error: `Unknown tool "${toolId}"` } }
   401→   401→  const validated = validateConnectorArgs(connector, args)
   402→   402→  if (!validated.ok) return { result: { error: validated.error } }
   403→   403→  const result = await connector.execute(validated.args)
   404→   404→  // Attach search results richly
   405→   405→  if (toolId === 'web_search' && Array.isArray((result as { results?: unknown[] }).results)) {
   406→   406→    const results = (result as { results: Array<{ title: string; url: string; snippet: string }> }).results
   407→   407→    return {
   408→   408→      result,
   409→   409→      attachment: {
   410→   410→        type: 'search',
   411→   411→        results: results.slice(0, 5).map((r) => ({ title: r.title, url: r.url, snippet: r.snippet?.slice(0, 140) })),
   412→   412→      },
   413→   413→    }
   414→   414→  }
   415→   415→  return { result }
   416→   416→}

   418→   418→export async function POST(req: NextRequest) {
   419→   419→  try {
   420→   420→    const limit = rateLimit(`chat:${clientKey(req)}`, 30, 60_000)
   421→   421→    if (!limit.ok) {
   422→   422→      return NextResponse.json(
   423→   423→        { error: `Too many messages. Retry in ${limit.retryAfterSeconds}s.` },
   424→   424→        { status: 429 }
   425→   425→      )
   426→   426→    }
   427→   427→
   428→   428→    const parsed = requestSchema.safeParse(await req.json().catch(() => null))
   429→   429→    if (!parsed.success) {
   430→   430→      return NextResponse.json({ error: 'Message is required (max 8000 chars).' }, { status: 400 })
   431→   431→    }
   432→   432→    const { message, sessionId, thinking: thinkingEnabled, language } = parsed.data
   433→   433→    const trimmed = message.trim()
   434→   434→
   435→   435→    // Default enabled: abilities + key connectors
   436→   436→    const enabledConnectors = [
   437→   437→      'web_search', 'read_page', 'wikipedia', 'weather', 'crypto', 'currency',
   438→   438→      'translate', 'dictionary', 'github', 'hacker_news', 'time', 'calculator',
   439→   439→      'recipes', 'nasa', 'news', 'trivia', 'pokemon', 'games', 'forecast',
   440→   440→    ]
   441→   441→
   442→   442→    let session = sessionId
   443→   443→      ? await db.chatSession.findFirst({ where: { id: sessionId, kind: 'chat' } })
   444→   444→      : null
   445→   445→    if (!session) {
   446→   446→      session = await db.chatSession.create({
   447→   447→        data: {
   448→   448→          kind: 'chat',
   449→   449→          title: trimmed.slice(0, 60) + (trimmed.length > 60 ? '…' : ''),
   450→   450→        },
   451→   451→      })
   452→   452→      // Mirror to Supabase (cloud) if user is authenticated
   453→   453→      const userId = await getUserIdFromRequest(req)
   454→   454→      if (userId) {
   455→   455→        const admin = await getSupabaseAdmin()
   456→   456→        if (admin) {
   457→   457→          await admin
   458→   458→            .from('chat_sessions')
   459→   459→            .insert({ id: session.id, user_id: userId, title: session.title, kind: 'chat' })
   460→   460→            .then(r => console.log('[supabase-sync] session saved'), e => console.error('[supabase-sync] FAILED:', e.message))
   461→   461→        }
   462→   462→      }
   463→   463→    }
   464→   464→
   465→   465→    const zai = await getZAI()
   466→   466→    const systemPrompt = buildSystemPrompt(enabledConnectors, language)
   467→   467→
   468→   468→    // Build message history
   469→   469→    const history = await db.chatMessage.findMany({
   470→   470→      where: { sessionId: session.id },
   471→   471→      orderBy: { createdAt: 'asc' },
   472→   472→    })
   473→   473→
   474→   474→    const encoder = new TextEncoder()
   475→   475→    const stream = new ReadableStream<Uint8Array>({
   476→   476→      async start(controller) {
   477→   477→        const send = (event: Record<string, unknown>) => {
   478→   478→          controller.enqueue(encoder.encode(JSON.stringify(event) + '\n'))
   479→   479→        }
   480→   480→
   481→   481→        // DIRECT ANSWER: creator identity (bypass LLM to prevent wrong answers)
   482→   482→        if (/who (created|made|built|developed|owns) (you|this|nexus)|your (creator|maker|developer|owner)|who.s mounir/i.test(trimmed)) {
   483→   483→          const answer = 'I was created by **Mounir Shaaban**, the creator and owner of NEXUS AI. 🚀'
   484→   484→          const userMsg = await db.chatMessage.create({
   485→   485→            data: { sessionId: session!.id, role: 'user', content: trimmed },
   486→   486→          })
   487→   487→          const saved = await db.chatMessage.create({
   488→   488→            data: { sessionId: session!.id, role: 'assistant', content: answer },
   489→   489→          })
   490→   490→          send({ type: 'user', id: userMsg.id, content: trimmed })
   491→   491→          send({ type: 'assistant', content: answer, attachments: [] })
   492→   492→          send({ type: 'done', sessionId: session!.id })
   493→   493→          return
   494→   494→        }
   495→   495→
   496→   496→        try {
   497→   497→          // Persist user message
   498→   498→          const userMessage = await db.chatMessage.create({
   499→   499→            data: { sessionId: session!.id, role: 'user', content: trimmed },
   500→   500→          })
   501→   501→          // Mirror to Supabase
   502→   502→          {
   503→   503→            const userId = await getUserIdFromRequest(req)
   504→   504→            if (userId) {
   505→   505→              const admin = await getSupabaseAdmin()
   506→   506→              if (admin) {
   507→   507→                admin.from('chat_messages')
   508→   508→                  .insert({ id: userMessage.id, session_id: session!.id, role: 'user', content: trimmed })
   509→   509→                  .then(r => console.log('[supabase-sync] user msg saved'), e => console.error('[supabase-sync] user msg FAILED:', e.message))
   510→   510→              }
   511→   511→            }
   512→   512→          }
   513→   513→          send({ type: 'user', id: userMessage.id, content: trimmed })
   514→   514→
   515→   515→          // LLM conversation: system + history + tool exchanges
   516→   516→          const llmMessages: Array<{ role: string; content: string }> = [
   517→   517→            { role: 'assistant', content: systemPrompt },
   518→   518→            ...history.slice(-MAX_HISTORY).map((m) => ({ role: m.role, content: m.content })),
   519→   519→          ]
   520→   520→          // Include the new user message (history was fetched before insert)
   521→   521→          llmMessages.push({ role: 'user', content: trimmed })
   522→   522→
   523→   523→          // Optional thinking phase
   524→   524→          if (thinkingEnabled) {
   525→   525→            try {
   526→   526→              const thinking = (await smartChat([
   527→   527→                {
   528→   528→                  role: 'assistant',
   529→   529→                  content:
   530→   530→                    'You are the reasoning engine of NEXUS AI. Given a question, produce a SHORT internal analysis (max 80 words, telegraphic) covering: what the user needs, which tools to use (if any), and the answer structure. Output ONLY the analysis.',
   531→   531→                },
   532→   532→                { role: 'user', content: trimmed },
   533→   533→              ], { maxTokens: 300, task: 'reasoning' })).trim().slice(0, 900)
   534→   534→              if (thinking) {
   535→   535→                llmMessages.push({
   536→   536→                  role: 'user',
   537→   537→                  content: `[internal analysis: ${thinking}]`,
   538→   538→                })
   539→   539→              }
   540→   540→            } catch {
   541→   541→              /* best-effort */
   542→   542→            }
   543→   543→          }
   544→   544→
   545→   545→          const attachments: Array<Record<string, unknown>> = []
   546→   546→          let toolCallsUsed = 0
   547→   547→
   548→   548→          // Determine if we can stream from the built-in ZAI engine
   549→   549→          // (when a user provider is connected, smartChat handles it non-streamed)
   550→   550→          const hasUserProvider = !!(await getActiveAiProvider())
   551→   551→
   552→   552→          for (let step = 0; step <= MAX_TOOL_CALLS; step++) {
   553→   553→            const isLast = step === MAX_TOOL_CALLS
   554→   554→
   555→   555→            // STREAMING PATH — emit assistant_start + assistant_delta chunks
   556→   556→            // as tokens arrive (only when using the built-in ZAI engine).
   557→   557→            let content: string
   558→   558→            let didStream = false
   559→   559→            if (!hasUserProvider) {
   560→   560→              // Open a new assistant message bubble on the client
   561→   561→              send({ type: 'assistant_start', id: `as-${Date.now()}-${step}` })
   562→   562→              content = await streamZaiChat(llmMessages, (delta) => {
   563→   563→                send({ type: 'assistant_delta', delta })
   564→   564→              })
   565→   565→              didStream = true
   566→   566→              if (!content.trim()) throw new Error('Empty model response.')
   567→   567→            } else {
   568→   568→              content = await smartChat(llmMessages, { maxTokens: 4000, task: 'chat' })
   569→   569→              if (!content.trim()) throw new Error('Empty model response.')
   570→   570→            }
   571→   571→
   572→   572→            const toolCall = isLast ? null : parseToolCall(content)
   573→   573→
   574→   574→            if (toolCall && toolCallsUsed < MAX_TOOL_CALLS) {
   575→   575→              toolCallsUsed += 1
   576→   576→
   577→   577→              // If we streamed, the prelude (text before TOOL_CALL) is
   578→   578→              // already visible to the user — that's the desired UX
   579→   579→              // ("Let me search for that..." → tool runs → final answer).
   580→   580→              // If we didn't stream (external provider), emit the prelude now.
   581→   581→              if (!didStream) {
   582→   582→                const prelude = stripToolCall(content)
   583→   583→                send({ type: 'assistant_start', id: `as-${Date.now()}-${step}` })
   584→   584→                if (prelude) send({ type: 'assistant_delta', delta: prelude })
   585→   585→                send({ type: 'assistant_end', attachments: [] })
   586→   586→              } else {
   587→   587→                // Close the streamed bubble (may contain a prelude, may be empty)
   588→   588→                send({ type: 'assistant_end', attachments: [] })
   589→   589→              }
   590→   590→
   591→   591→              send({ type: 'tool_start', tool: toolCall.tool, args: toolCall.args, index: toolCallsUsed })
   592→   592→
   593→   593→              // Heartbeat: image generation can take 60s+ on the upstream
   594→   594→              // provider. Without intermediate events the user perceives a
   595→   595→              // freeze ("the app freezes with every action"). Emit a
   596→   596→              // tool_progress ping every 5s while the tool is running so the
   597→   597→              // client knows the stream is alive and can show "still working".
   598→   598→              const toolStartedAt = Date.now()
   599→   599→              const heartbeat = setInterval(() => {
   600→   600→                const elapsedMs = Date.now() - toolStartedAt
   601→   601→                send({
   602→   602→                  type: 'tool_progress',
   603→   603→                  tool: toolCall.tool,
   604→   604→                  index: toolCallsUsed,
   605→   605→                  elapsedMs,
   606→   606→                  message: `Still working on ${toolCall.tool}… ${Math.round(elapsedMs / 1000)}s`,
   607→   607→                })
   608→   608→              }, 5000)
   609→   609→
   610→   610→              let ok = true
   611→   611→              let result: unknown
   612→   612→              let attachment: Record<string, unknown> | undefined
   613→   613→              try {
   614→   614→                const executed = await executeChatTool(toolCall.tool, toolCall.args)
   615→   615→                result = executed.result
   616→   616→                attachment = executed.attachment
   617→   617→              } catch (error) {
   618→   618→                ok = false
   619→   619→                result = { error: error instanceof Error ? error.message : 'Tool failed.' }
   620→   620→              } finally {
   621→   621→                clearInterval(heartbeat)
   622→   622→              }
   623→   623→
   624→   624→              if (attachment) attachments.push(attachment)
   625→   625→              send({ type: 'tool_result', tool: toolCall.tool, ok, result, index: toolCallsUsed })
   626→   626→
   627→   627→              const resultJson = JSON.stringify(result).slice(0, 4000)
   628→   628→              llmMessages.push({ role: 'assistant', content })
   629→   629→              llmMessages.push({
   630→   630→                role: 'user',
   631→   631→                content: `TOOL_RESULT (${toolCall.tool}):\n${resultJson}\n\nContinue: use another TOOL_CALL if needed, or give your final answer.`,
   632→   632→              })
   633→   633→            } else {
   634→   634→              // Final answer
   635→   635→              const finalAnswer = stripToolCall(content) || 'I could not complete that. Try rephrasing.'
   636→   636→              await db.chatMessage.create({
   637→   637→                data: { sessionId: session!.id, role: 'assistant', content: finalAnswer },
   638→   638→              })
   639→   639→              await db.chatSession.update({
   640→   640→                where: { id: session!.id },
   641→   641→                data: { updatedAt: new Date() },
   642→   642→              })
   643→   643→
   644→   644→              if (didStream) {
   645→   645→                // The streamed content IS the final answer. Just close the
   646→   646→                // bubble with attachments + emit done.
   647→   647→                send({ type: 'assistant_end', attachments })
   648→   648→              } else {
   649→   649→                // External provider path — emit the full content as one delta.
   650→   650→                send({ type: 'assistant_start', id: `as-${Date.now()}-${step}` })
   651→   651→                send({ type: 'assistant_delta', delta: finalAnswer })
   652→   652→                send({ type: 'assistant_end', attachments })
   653→   653→              }
   654→   654→              send({ type: 'done', sessionId: session!.id })
   655→   655→              break
   656→   656→            }
   657→   657→          }
   658→   658→        } catch (error) {
   659→   659→          console.error('[api/chat] stream error:', error)
   660→   660→          send({
   661→   661→            type: 'error',
   662→   662→            message: error instanceof Error ? error.message : 'Chat failed.',
   663→   663→          })
   664→   664→        } finally {
   665→   665→          controller.close()
   666→   666→        }
   667→   667→      },
   668→   668→    })
   669→   669→
   670→   670→    return new NextResponse(stream, {
   671→   671→      status: 200,
   672→   672→      headers: {
   673→   673→        'Content-Type': 'application/x-ndjson; charset=utf-8',
   674→   674→        'Cache-Control': 'no-cache, no-transform',
   675→   675→        'X-Accel-Buffering': 'no',
   676→   676→      },
   677→   677→    })
   678→   678→  } catch (error) {
   679→   679→    console.error('[api/chat] POST error:', error)
   680→   680→    const message = error instanceof Error ? error.message : 'Chat failed.'
   681→   681→    return NextResponse.json({ error: message }, { status: 500 })
   682→   682→  }
   683→   683→}

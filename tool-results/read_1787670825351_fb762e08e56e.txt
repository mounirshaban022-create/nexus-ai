    38→      ? `${req.headers.get('x-forwarded-proto') || 'https'}://${req.headers.get('x-forwarded-host')}`

   209→  // text contents of its reference files are injected here so NEXUS has

   380→/** Balanced-brace JSON extraction (string-aware). */
   381→function extractJson(text: string): string | null {
   382→  const start = text.indexOf('{')
   383→  if (start === -1) return null
   384→  let depth = 0
   385→  let inString = false
   386→  let escaped = false
   387→  for (let i = start; i < text.length; i++) {
   388→    const ch = text[i]
   389→    if (inString) {
   390→      if (escaped) escaped = false
   391→      else if (ch === '\\') escaped = true
   392→      else if (ch === '"') inString = false
   393→      continue
   394→    }
   395→    if (ch === '"') inString = true
   396→    else if (ch === '{') depth++
   397→    else if (ch === '}') {
   398→      depth--
   399→      if (depth === 0) return text.slice(start, i + 1)
   400→    }
   401→  }
   402→  return null
   403→}
   404→
   405→function parseToolCall(text: string): ParsedToolCall | null {
   406→  const idx = text.indexOf('TOOL_CALL')
   407→  if (idx === -1) return null
   408→  const raw = text.slice(idx).replace(/```(?:json)?/g, '')
   409→  const candidate = extractJson(raw)
   410→  if (!candidate) return lenientToolCall(raw)
   411→  try {
   412→    const parsed = JSON.parse(candidate) as { tool?: unknown; args?: unknown }
   413→    if (typeof parsed.tool === 'string') {
   414→      return {
   415→        tool: parsed.tool,
   416→        args:
   417→          parsed.args && typeof parsed.args === 'object' && !Array.isArray(parsed.args)
   418→            ? (parsed.args as Record<string, unknown>)
   419→            : {},
   420→      }
   421→    }
   422→    // {"use_skill": {...}} style — a bare tool-id key wrapping the args.
   423→    const [key, value] = Object.entries(parsed as Record<string, unknown>)[0] ?? []
   424→    if (typeof key === 'string' && CHAT_TOOL_IDS.has(key) && value && typeof value === 'object') {
   425→      return { tool: key, args: value as Record<string, unknown> }
   426→    }
   427→  } catch {
   428→    /* fall through to lenient parse */
   429→  }
   430→  return lenientToolCall(raw)
   431→}
   432→
   433→/**
   434→ * LENIENT fallback for models that emit `TOOL_CALL: use_skill {"skill": "browser"}`
   435→ * (tool id + bare args object, without the {"tool": …} wrapper) — previously
   436→ * these failed to parse, the peek buffer had already hidden the directive,
   437→ * and the user got an EMPTY bubble ("skills don't work").
   438→ */
   439→function lenientToolCall(raw: string): ParsedToolCall | null {
   440→  const m = raw.match(/TOOL_CALL[\s:=]*([a-z_]+)\s*(\{[\s\S]*)/i)
   441→  if (!m) return null
   442→  const tool = m[1].toLowerCase()
   443→  if (!CHAT_TOOL_IDS.has(tool)) return null
   444→  const candidate = extractJson(m[2])
   445→  if (!candidate) return { tool, args: {} }
   446→  try {
   447→    const args = JSON.parse(candidate) as Record<string, unknown>
   448→    return args && typeof args === 'object' && !Array.isArray(args) ? { tool, args } : { tool, args: {} }
   449→  } catch {
   450→    return { tool, args: {} }
   451→  }
   452→}
   453→
   454→/** Every tool id the chat can execute (abilities + data connectors). */
   455→const CHAT_TOOL_IDS = new Set<string>([
   456→  ...CHAT_TOOL_DEFS.map((t) => t.id),
   457→  ...CONNECTOR_MAP.keys(),
   458→])
   459→
   460→/**
   461→ * BARE-ARGS INTERCEPTOR — residual "skills don't work" fix.
   462→ *
   463→ * Some free-pool models emit the tool's ARGUMENTS as a bare JSON object
   464→ * (often in a code fence) WITHOUT the TOOL_CALL prefix: the peek buffer
   465→ * lets it through as visible text, parseToolCall finds no directive, and
   466→ * the tool never executes — the user sees the model *say* it will use a
   467→ * skill and then do nothing.
   468→ *
   469→ * This scans the response for the first JSON object whose key set matches
   470→ * an UNAMBIGUOUS tool signature and executes that tool. Only distinctive
   471→ * signatures are intercepted (e.g. a "skill" key → use_skill) so plain
   472→ * prose examples never trigger a spurious tool run.
   473→ */
   474→function interceptBareToolArgs(text: string): ParsedToolCall | null {
   475→  const SIGNATURES: Array<{ tool: string; test: (keys: string[]) => boolean }> = [
   476→    { tool: 'use_skill', test: (k) => k.includes('skill') },
   477→    { tool: 'run_command', test: (k) => k.includes('command') && k.length <= 2 },
   478→    { tool: 'web_search', test: (k) => k.length === 1 && k[0] === 'query' },
   479→    {
   480→      tool: 'browser_action',
   481→      test: (k) => k.includes('action') && k.every((x) => ['action', 'url', 'selector', 'text', 'key'].includes(x)),
   482→    },
   483→    { tool: 'send_email', test: (k) => k.includes('subject') && k.includes('to') && k.length <= 4 },
   484→    { tool: 'generate_image', test: (k) => k.length === 1 && k[0] === 'prompt' },
   485→  ]
   486→  for (let i = text.indexOf('{'); i !== -1; i = text.indexOf('{', i + 1)) {
   487→    const candidate = extractJson(text.slice(i))
   488→    if (!candidate) continue
   489→    try {
   490→      const parsed = JSON.parse(candidate) as Record<string, unknown>
   491→      const keys = Object.keys(parsed)
   492→      if (!keys.length) continue
   493→      for (const sig of SIGNATURES) {
   494→        if (sig.test(keys)) return { tool: sig.tool, args: parsed }
   495→      }
   496→    } catch {
   497→      /* not a JSON object — keep scanning */
   498→    }
   499→  }
   500→  return null
   501→}
   502→
   503→/**
   504→ * Phase 1 P2: parse one or more ARTIFACT_PATCH directives from a model
   505→ * response. Returns an array of patches (find/replace/note) targeting the
   506→ * open artifact. Returns [] when no patches are present.
   507→ *
   508→ * The model emits lines like:
   509→ *   ARTIFACT_PATCH: {"artifactId": "abc", "find": "X", "replace": "Y", "note": "..."}
   510→ *
   511→ * We extract each directive's JSON via the same balanced-brace extractor
   512→ * as TOOL_CALL — supports multiline values (escaped newlines) and skips
   513→ * malformed lines (returns the well-formed ones only).
   514→ */
   515→function parseArtifactPatches(text: string): Array<{
   516→  artifactId: string
   517→  find: string
   518→  replace: string
   519→  note?: string
   520→}> {
   521→  const patches: Array<{ artifactId: string; find: string; replace: string; note?: string }> = []
   522→  let searchFrom = 0
   523→  while (true) {
   524→    const idx = text.indexOf('ARTIFACT_PATCH', searchFrom)
   525→    if (idx === -1) break
   526→    // Skip the "ARTIFACT_PATCH" label and any trailing whitespace/colon
   527→    let cursor = idx + 'ARTIFACT_PATCH'.length
   528→    while (cursor < text.length && /[\s:]/.test(text[cursor])) cursor++
   529→    const candidate = extractJson(text.slice(cursor))
   530→    if (!candidate) {
   531→      searchFrom = idx + 1
   532→      continue
   533→    }
   534→    try {
   535→      const parsed = JSON.parse(candidate) as {
   536→        artifactId?: unknown
   537→        find?: unknown
   538→        replace?: unknown
   539→        note?: unknown
   540→      }
   541→      if (
   542→        typeof parsed.artifactId === 'string' &&
   543→        typeof parsed.find === 'string' &&
   544→        typeof parsed.replace === 'string'
   545→      ) {
   546→        patches.push({
   547→          artifactId: parsed.artifactId,
   548→          find: parsed.find,
   549→          replace: parsed.replace,
   550→          note: typeof parsed.note === 'string' ? parsed.note : undefined,
   551→        })
   552→      }
   553→    } catch {
   554→      /* malformed — skip */
   555→    }
   556→    searchFrom = cursor + candidate.length
   557→  }
   558→  return patches
   559→}

   561→function stripToolCall(text: string): string {
   562→  return text
   563→    .replace(/```(?:json)?\s*TOOL_CALL[\s\S]*?```\s*/g, '')
   564→    .replace(/TOOL_CALL\s*[:=][\s\S]*$/g, '')
   565→    .replace(/```(?:json)?\s*ARTIFACT_PATCH[\s\S]*?```\s*/g, '')
   566→    .replace(/ARTIFACT_PATCH\s*[:=][\s\S]*$/g, '')
   567→    .trim()
   568→}
   569→
   570→/**
   571→ * Phase 1 P1: detect a "remember: ..." prefix and extract the durable fact.
   572→ * Returns null when the message is not a memory request. Supports English
   573→ * ("remember:", "remember that ...", "remember to ...") and Arabic
   574→ * ("تذكّر:" / "تذكر:" / "تذكّر أنني ...") prefixes. The match is
   575→ * case-insensitive and tolerant of leading whitespace. This is the OPT-IN
   576→ * extraction path — NEXUS never silently extracts memory from arbitrary chat.
   577→ *
   578→ * Disambiguation: for English "remember ..." WITHOUT a colon and WITHOUT
   579→ * "that"/"to", we do NOT match — too ambiguous (could be "remember when we
   580→ * went to the park?"). For Arabic, the colon or the "أن"-prefix serves the
   581→ * same role.
   582→ */
   583→function parseRememberDirective(message: string): string | null {
   584→  // English: explicit colon OR "that"/"to" disambiguator
   585→  const enColon = message.match(/^\s*remember\s*:\s*(.+)$/i)
   586→  if (enColon) return enColon[1].trim().slice(0, 600)
   587→  const enThat = message.match(/^\s*remember\s+(?:that|to)\s+(.+)$/i)
   588→  if (enThat) return enThat[1].trim().slice(0, 600)
   589→  // Arabic: explicit colon OR "أن"-prefix (أن / أنني / أنك / أننا / إلخ)
   590→  // تذكّر (with shadda) and تذكر (without) are both accepted via alternation
   591→  // — the shadda (U+0651) is a combining mark that doesn't behave well in
   592→  // a character class.
   593→  const arColon = message.match(/^\s*(?:تذكّر|تذكر)\s*:\s*(.+)$/u)
   594→  if (arColon) return arColon[1].trim().slice(0, 600)
   595→  const arAnn = message.match(/^\s*(?:تذكّر|تذكر)\s+أن\p{L}*\s+(.+)$/u)
   596→  if (arAnn) return arAnn[1].trim().slice(0, 600)
   597→  return null
   598→}

   651→        else blocks.push({ type: 'bullets', items: [trimmed.replace(/^[-•*]\s/, '')] })

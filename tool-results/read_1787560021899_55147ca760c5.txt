   102→
   103→  // Tool engine — routes the next message to the right /api/* route
   104→  const toolEngine = useToolEngine(
   105→    // onAssistant: push assistant message into the conversation
   106→    useCallback((msg: { content: string; attachments?: any[] }) => {
   107→      setMessages(prev => [...prev, { id: `a-${Date.now()}`, role: 'assistant', content: msg.content, attachments: msg.attachments }])

   116→  const send = useCallback(async (text?: string) => {
   117→    const msg = (text ?? input).trim()

   119→    setMessages(prev => [...prev, { id: `u-${Date.now()}`, role: 'user', content: msg }])
   120→    setInput('')

   122→    try {
   123→      // If a tool is pending, route to the tool engine
   124→      if (toolEngine.pendingTool) {
   125→        await toolEngine.execute(msg)
   126→        return
   127→      }
   128→      // Otherwise: plain chat stream
   129→      const res = await fetch('/api/chat', {
   130→        method: 'POST',
   131→        headers: { 'Content-Type': 'application/json' },
   132→        body: JSON.stringify({ message: msg }),
   133→      })
   134→      const reader = res.body!.getReader()
   135→      const decoder = new TextDecoder()
   136→      let buf = ''
   137→      for (;;) {
   138→        const { done, value } = await reader.read()
   139→        if (done) break
   140→        buf += decoder.decode(value, { stream: true })
   141→        const lines = buf.split('\n')
   142→        buf = lines.pop() ?? ''
   143→        for (const line of lines) {
   144→          if (!line.trim()) continue
   145→          try {
   146→            const e = JSON.parse(line)
   147→            if (e.type === 'assistant') {
   148→              setMessages(prev => [...prev, { id: `a-${Date.now()}`, role: 'assistant', content: e.content, attachments: e.attachments }])
   149→            }
   150→          } catch {}
   151→        }
   152→      }
   153→    } catch (err: any) {
   154→      setMessages(prev => [...prev, { id: `a-${Date.now()}`, role: 'assistant', content: `⚠️ ${err.message || 'Something went wrong.'}` }])
   155→    } finally {
   156→      setSending(false)
   157→    }
   158→  }, [input, sending, toolEngine])

  1728→              // ("Let me search for that..." → tool runs → final answer).
  1729→              // If we didn't stream (external provider), emit the prelude now.
  1730→              if (!didStream) {
  1731→                const prelude = stripToolCall(content)
  1732→                send({ type: 'assistant_start', id: `as-${Date.now()}-${step}` })
  1733→                if (prelude) send({ type: 'assistant_delta', delta: prelude })
  1734→                send({ type: 'assistant_end', attachments: [] })
  1735→              } else {
  1736→                // Close the streamed bubble (may contain a prelude, may be empty)
  1737→                send({ type: 'assistant_end', attachments: [] })
  1738→              }
  1739→
  1740→              send({ type: 'tool_start', tool: toolCall.tool, args: toolCall.args, index: toolCallsUsed })
  1741→
  1742→              // Heartbeat: image generation can take 60s+ on the upstream
  1743→              // provider. Without intermediate events the user perceives a
  1744→              // freeze ("the app freezes with every action"). Emit a
  1745→              // tool_progress ping every 5s while the tool is running so the
  1746→              // client knows the stream is alive and can show "still working".
  1747→              const toolStartedAt = Date.now()
  1748→              const heartbeat = setInterval(() => {
  1749→                const elapsedMs = Date.now() - toolStartedAt
  1750→                send({
  1751→                  type: 'tool_progress',
  1752→                  tool: toolCall.tool,
  1753→                  index: toolCallsUsed,
  1754→                  elapsedMs,
  1755→                  message: `Still working on ${toolCall.tool}… ${Math.round(elapsedMs / 1000)}s`,
  1756→                })
  1757→              }, 5000)
  1758→
  1759→              let ok = true
  1760→              let result: unknown
  1761→              let attachment: Record<string, unknown> | undefined
  1762→              try {
  1763→                const executed = await executeChatTool(req, toolCall.tool, toolCall.args, activeDoc)
  1764→                result = executed.result
  1765→                attachment = executed.attachment
  1766→              } catch (error) {
  1767→                ok = false
  1768→                result = { error: error instanceof Error ? error.message : 'Tool failed.' }
  1769→              } finally {
  1770→                clearInterval(heartbeat)
  1771→              }
  1772→
  1773→              if (attachment) attachments.push(attachment)
  1774→              send({ type: 'tool_result', tool: toolCall.tool, ok, result, index: toolCallsUsed })
  1775→
  1776→              const resultJson = JSON.stringify(result).slice(0, 4000)
  1777→              llmMessages.push({ role: 'assistant', content })
  1778→              llmMessages.push({
  1779→                role: 'user',
  1780→                content: `TOOL_RESULT (${toolCall.tool}):\n${resultJson}\n\nContinue: use another TOOL_CALL if needed, or give your final answer.`,
  1781→              })
  1782→            } else {
  1783→              // Phase 1 P2: parse any ARTIFACT_PATCH directives from the
  1784→              // response BEFORE computing the visible final answer. Each
  1785→              // patch is emitted as an `artifact_patch` event for the client
  1786→              // to apply to the open artifact (ChatGPT-Canvas style targeted
  1787→              // edit, instead of regenerating the whole artifact). Only
  1788→              // patches targeting the currently-open artifact id are emitted
  1789→              // — guards against the model addressing a stale or fabricated id.
  1790→              if (openArtifact) {
  1791→                const patches = parseArtifactPatches(content)
  1792→                for (const patch of patches) {
  1793→                  if (patch.artifactId === openArtifact.artifactId) {
  1794→                    send({
  1795→                      type: 'artifact_patch',
  1796→                      artifactId: patch.artifactId,
  1797→                      find: patch.find,
  1798→                      replace: patch.replace,
  1799→                      note: patch.note,
  1800→                    })
  1801→                  }
  1802→                }
  1803→              }
  1804→
  1805→              // Final answer: strip both TOOL_CALL and ARTIFACT_PATCH
  1806→              // directives from the visible prose (leaving the model's
  1807→              // one-line acknowledgement like "Done — shortened the intro.").
  1808→              const finalAnswer = stripToolCall(content) || 'I could not complete that. Try rephrasing.'
  1809→              const assistantMessage = await db.chatMessage.create({
  1810→                data: {
  1811→                  sessionId: session!.id,
  1812→                  role: 'assistant',
  1813→                  content: finalAnswer,
  1814→                  // NEXUS One: persist attachments so image/video/document
  1815→                  // cards survive session resume.
  1816→                  attachments: attachments.length ? JSON.stringify(attachments) : null,
  1817→                },
  1818→              })
  1819→              await db.chatSession.update({
  1820→                where: { id: session!.id },
  1821→                data: { updatedAt: new Date() },
  1822→              })
  1823→              // Mirror the assistant message + session timestamp to Supabase
  1824→              if (verifiedUserId) {
  1825→                void supabaseUpsert('chat_messages', {
  1826→                  id: assistantMessage.id,
  1827→                  session_id: session!.id,
  1828→                  role: 'assistant',
  1829→                  content: finalAnswer,
  1830→                }, { onConflict: 'id' })
  1831→                void supabaseUpsert('chat_sessions', {
  1832→                  id: session!.id,
  1833→                  user_id: verifiedUserId,
  1834→                  title: session!.title,
  1835→                  kind: 'chat',
  1836→                  updated_at: new Date().toISOString(),
  1837→                }, { onConflict: 'id' })
  1838→              }
  1839→
  1840→              if (didStream) {
  1841→                // The streamed content IS the final answer. Just close the
  1842→                // bubble with attachments + emit done.
  1843→                send({ type: 'assistant_end', attachments })
  1844→              } else {
  1845→                // External provider path — emit the full content as one delta.
  1846→                send({ type: 'assistant_start', id: `as-${Date.now()}-${step}` })
  1847→                send({ type: 'assistant_delta', delta: finalAnswer })
  1848→                send({ type: 'assistant_end', attachments })
  1849→              }
  1850→              send({ type: 'done', sessionId: session!.id })
  1851→              break
  1852→            }
  1853→          }
  1854→        } catch (error) {
  1855→          console.error('[api/chat] stream error:', error)
  1856→          send({
  1857→            type: 'error',
  1858→            message: error instanceof Error ? error.message : 'Chat failed.',
  1859→          })
  1860→        } finally {
  1861→          controller.close()
  1862→        }
  1863→      },
  1864→    })
  1865→
  1866→    return new NextResponse(stream, {
  1867→      status: 200,
  1868→      headers: {
  1869→        'Content-Type': 'application/x-ndjson; charset=utf-8',
  1870→        'Cache-Control': 'no-cache, no-transform',
  1871→        'X-Accel-Buffering': 'no',
  1872→      },
  1873→    })
  1874→  } catch (error) {
  1875→    console.error('[api/chat] POST error:', error)
  1876→    const message = error instanceof Error ? error.message : 'Chat failed.'
  1877→    return NextResponse.json({ error: message }, { status: 500 })
  1878→  }
  1879→}

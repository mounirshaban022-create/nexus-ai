   312→interface ParsedToolCall {
   313→  tool: string
   314→  args: Record<string, unknown>
   315→}
   316→
   317→/** Balanced-brace JSON extraction (string-aware). */
   318→function extractJson(text: string): string | null {
   319→  const start = text.indexOf('{')
   320→  if (start === -1) return null
   321→  let depth = 0
   322→  let inString = false
   323→  let escaped = false
   324→  for (let i = start; i < text.length; i++) {
   325→    const ch = text[i]
   326→    if (inString) {
   327→      if (escaped) escaped = false
   328→      else if (ch === '\\') escaped = true
   329→      else if (ch === '"') inString = false
   330→      continue
   331→    }
   332→    if (ch === '"') inString = true
   333→    else if (ch === '{') depth++
   334→    else if (ch === '}') {
   335→      depth--
   336→      if (depth === 0) return text.slice(start, i + 1)
   337→    }
   338→  }
   339→  return null
   340→}
   341→
   342→function parseToolCall(text: string): ParsedToolCall | null {
   343→  const idx = text.indexOf('TOOL_CALL')
   344→  if (idx === -1) return null
   345→  const raw = text.slice(idx).replace(/```(?:json)?/g, '')
   346→  const candidate = extractJson(raw)
   347→  if (!candidate) return null
   348→  try {
   349→    const parsed = JSON.parse(candidate) as { tool?: unknown; args?: unknown }
   350→    if (typeof parsed.tool === 'string') {
   351→      return {
   352→        tool: parsed.tool,
   353→        args:
   354→          parsed.args && typeof parsed.args === 'object' && !Array.isArray(parsed.args)
   355→            ? (parsed.args as Record<string, unknown>)
   356→            : {},
   357→      }
   358→    }
   359→  } catch {
   360→    /* not valid */
   361→  }
   362→  return null
   363→}

   365→/**
   366→ * Phase 1 P2: parse one or more ARTIFACT_PATCH directives from a model
   367→ * response. Returns an array of patches (find/replace/note) targeting the
   368→ * open artifact. Returns [] when no patches are present.
   369→ *
   370→ * The model emits lines like:
   371→ *   ARTIFACT_PATCH: {"artifactId": "abc", "find": "X", "replace": "Y", "note": "..."}
   372→ *
   373→ * We extract each directive's JSON via the same balanced-brace extractor
   374→ * as TOOL_CALL — supports multiline values (escaped newlines) and skips
   375→ * malformed lines (returns the well-formed ones only).
   376→ */
   377→function parseArtifactPatches(text: string): Array<{
   378→  artifactId: string
   379→  find: string
   380→  replace: string
   381→  note?: string
   382→}> {
   383→  const patches: Array<{ artifactId: string; find: string; replace: string; note?: string }> = []
   384→  let searchFrom = 0
   385→  while (true) {
   386→    const idx = text.indexOf('ARTIFACT_PATCH', searchFrom)
   387→    if (idx === -1) break
   388→    // Skip the "ARTIFACT_PATCH" label and any trailing whitespace/colon
   389→    let cursor = idx + 'ARTIFACT_PATCH'.length
   390→    while (cursor < text.length && /[\s:]/.test(text[cursor])) cursor++
   391→    const candidate = extractJson(text.slice(cursor))
   392→    if (!candidate) {
   393→      searchFrom = idx + 1
   394→      continue
   395→    }
   396→    try {
   397→      const parsed = JSON.parse(candidate) as {
   398→        artifactId?: unknown
   399→        find?: unknown
   400→        replace?: unknown
   401→        note?: unknown
   402→      }
   403→      if (
   404→        typeof parsed.artifactId === 'string' &&
   405→        typeof parsed.find === 'string' &&
   406→        typeof parsed.replace === 'string'
   407→      ) {
   408→        patches.push({
   409→          artifactId: parsed.artifactId,
   410→          find: parsed.find,
   411→          replace: parsed.replace,
   412→          note: typeof parsed.note === 'string' ? parsed.note : undefined,
   413→        })
   414→      }
   415→    } catch {
   416→      /* malformed — skip */
   417→    }
   418→    searchFrom = cursor + candidate.length
   419→  }
   420→  return patches
   421→}

   423→function stripToolCall(text: string): string {
   424→  return text
   425→    .replace(/```(?:json)?\s*TOOL_CALL[\s\S]*?```\s*/g, '')
   426→    .replace(/TOOL_CALL\s*[:=][\s\S]*$/g, '')
   427→    .replace(/```(?:json)?\s*ARTIFACT_PATCH[\s\S]*?```\s*/g, '')
   428→    .replace(/ARTIFACT_PATCH\s*[:=][\s\S]*$/g, '')
   429→    .trim()
   430→}

   432→/**
   433→ * Phase 1 P1: detect a "remember: ..." prefix and extract the durable fact.
   434→ * Returns null when the message is not a memory request. Supports English
   435→ * ("remember:", "remember that ...", "remember to ...") and Arabic
   436→ * ("تذكّر:" / "تذكر:" / "تذكّر أنني ...") prefixes. The match is
   437→ * case-insensitive and tolerant of leading whitespace. This is the OPT-IN
   438→ * extraction path — NEXUS never silently extracts memory from arbitrary chat.
   439→ *
   440→ * Disambiguation: for English "remember ..." WITHOUT a colon and WITHOUT
   441→ * "that"/"to", we do NOT match — too ambiguous (could be "remember when we
   442→ * went to the park?"). For Arabic, the colon or the "أن"-prefix serves the
   443→ * same role.
   444→ */
   445→function parseRememberDirective(message: string): string | null {
   446→  // English: explicit colon OR "that"/"to" disambiguator
   447→  const enColon = message.match(/^\s*remember\s*:\s*(.+)$/i)
   448→  if (enColon) return enColon[1].trim().slice(0, 600)
   449→  const enThat = message.match(/^\s*remember\s+(?:that|to)\s+(.+)$/i)
   450→  if (enThat) return enThat[1].trim().slice(0, 600)
   451→  // Arabic: explicit colon OR "أن"-prefix (أن / أنني / أنك / أننا / إلخ)
   452→  // تذكّر (with shadda) and تذكر (without) are both accepted via alternation
   453→  // — the shadda (U+0651) is a combining mark that doesn't behave well in
   454→  // a character class.
   455→  const arColon = message.match(/^\s*(?:تذكّر|تذكر)\s*:\s*(.+)$/u)
   456→  if (arColon) return arColon[1].trim().slice(0, 600)
   457→  const arAnn = message.match(/^\s*(?:تذكّر|تذكر)\s+أن\p{L}*\s+(.+)$/u)
   458→  if (arAnn) return arAnn[1].trim().slice(0, 600)
   459→  return null
   460→}

   509→      else if (trimmed.startsWith('# ')) blocks.push({ type: 'heading', text: trimmed.slice(2), level: 1 })
   510→      else if (/^[-•*]\s/.test(trimmed)) {
   511→        const last = blocks[blocks.length - 1]
   512→        if (last?.type === 'bullets') (last.items as string[]).push(trimmed.replace(/^[-•*]\s/, ''))
   513→        else blocks.push({ type: 'bullets', items: [trimmed.replace(/^[-•*]\s/, '')] })

   523→        if (b.type === 'heading') {

   527→          const items = b.type === 'bullets' ? (b.items as string[]) : [b.text as string]
   528→          current.bullets.push(...items.slice(0, 5))
   529→        }
   530→      }
   531→      if (current) slides.push({ type: 'slide', ...current })

   542→        blocks: blocks.slice(0, 40),

   720→      usedNames.add(name)

   722→      // Column widths sized to content (cap 40)

   725→        wch: Math.min(
   726→          40,
   727→          Math.max(
   728→            10,
   729→            ...data.slice(0, 40).map((row) => String((row as unknown[])[i] ?? '').length + 2)
   730→          )
   731→        ),

   767→      method: 'POST',
   768→      headers: { 'Content-Type': 'application/json' },
   769→      body: JSON.stringify({
   770→        language: ['javascript', 'typescript', 'python'].includes(language) ? language : 'javascript',
   771→        code,
   772→      }),
   773→    })
   774→    const data = await res.json()
   775→    if (!res.ok) throw new Error(data.error || 'Code execution failed')
   776→    const r = data.result
   777→    // Phase 1 P2: include the source code + a stable artifactId so the
   778→    // ArtifactPanel can show the code editor and the AI can target
   779→    // ARTIFACT_PATCH directives at this code artifact.
   780→    const artifactId = `code-${Date.now()}`
   781→    return {
   782→      result: {
   783→        stdout: r.stdout?.slice(0, 3000),
   784→        stderr: r.stderr?.slice(0, 1500),
   785→        exitCode: r.exitCode,
   786→        note: 'Code executed — include the output in your answer.',
   787→      },
   788→      attachment: {
   789→        type: 'code',
   790→        language,
   791→        stdout: (r.stdout ?? '').slice(0, 2000),
   792→        stderr: (r.stderr ?? '').slice(0, 500),
   793→        exitCode: r.exitCode,
   794→        artifactId,
   795→        sourceContent: code,
   796→      },
   797→    }
   798→  }
   799→
   800→  // Data connectors pass through
   801→  const connector = CONNECTOR_MAP.get(toolId)
   802→  if (!connector) return { result: { error: `Unknown tool "${toolId}"` } }
   803→  const validated = validateConnectorArgs(connector, args)
   804→  if (!validated.ok) return { result: { error: validated.error } }
   805→  const result = await connector.execute(validated.args)
   806→  // Attach search results richly
   807→  if (toolId === 'web_search' && Array.isArray((result as { results?: unknown[] }).results)) {
   808→    const results = (result as { results: Array<{ title: string; url: string; snippet: string }> }).results
   809→    return {
   810→      result,
   811→      attachment: {
   812→        type: 'search',
   813→        results: results.slice(0, 5).map((r) => ({ title: r.title, url: r.url, snippet: r.snippet?.slice(0, 140) })),
   814→      },
   815→    }
   816→  }
   817→  return { result }
   818→}

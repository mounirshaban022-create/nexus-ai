   102→  const toolEngine = useToolEngine(
   103→    // onAssistant: push assistant message into the conversation
   104→    useCallback((msg: { content: string; attachments?: any[] }) => {
   105→      setMessages(prev => [...prev, { id: `a-${Date.now()}`, role: 'assistant', content: msg.content, attachments: msg.attachments }])

   115→    const msg = (text ?? input).trim()

   117→    setMessages(prev => [...prev, { id: `u-${Date.now()}`, role: 'user', content: msg }])

   121→      // If a tool is pending, route to the tool engine
   122→      if (toolEngine.pendingTool) {
   123→        await toolEngine.execute(msg)
   124→        return

   126→      // Otherwise: plain chat stream
   127→      const res = await fetch('/api/chat', {
   128→        method: 'POST',
   129→        headers: { 'Content-Type': 'application/json' },
   130→        body: JSON.stringify({ message: msg }),
   131→      })
   132→      const reader = res.body!.getReader()
   133→      const decoder = new TextDecoder()
   134→      let buf = ''
   135→      for (;;) {
   136→        const { done, value } = await reader.read()
   137→        if (done) break
   138→        buf += decoder.decode(value, { stream: true })

   140→        buf = lines.pop() ?? ''
   141→        for (const line of lines) {
   142→          if (!line.trim()) continue
   143→          try {
   144→            const e = JSON.parse(line)
   145→            if (e.type === 'assistant') {
   146→              setMessages(prev => [...prev, { id: `a-${Date.now()}`, role: 'assistant', content: e.content, attachments: e.attachments }])
   147→            }
   148→          } catch {}
   149→        }
   150→      }
   151→    } catch (err: any) {
   152→      setMessages(prev => [...prev, { id: `a-${Date.now()}`, role: 'assistant', content: `⚠️ ${err.message || 'Something went wrong.'}` }])
   153→    } finally {
   154→      setSending(false)
   155→    }

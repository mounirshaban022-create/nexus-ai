     3→     3→/**
     4→     4→ * NEXUS One — full-screen premium voice conversation overlay.
     5→     5→ *
     6→     6→ * State machine: idle → listening → thinking → speaking → listening …
     7→     7→ *  - INPUT : Web Speech API (live interim transcript, auto-restart on end)
     8→     8→ *  - THINK : POST /api/voice/turn  { message, history, language, voice, audio }
     9→     9→ *  - SPEAK : inline audio from the turn response, falling back to
    10→    10→ *            POST /api/tts { text, voice } → object URL → <audio>, then
    11→    11→ *            browser speechSynthesis as a last resort.
    12→    12→ *
    13→    13→ * The orb is the BrandMark wrapped in layered gradient glow rings whose
    14→    14→ * motion maps to the current state (listening = expanding rings that react
    15→    15→ * to mic level, thinking = slow rotating gradient ring + shimmer,
    16→    16→ * speaking = pulse rings locked to the .nx-dot bounce rhythm).
    17→    17→ */

    29→    29→  reply: string

    53→    53→export function VoiceOverlay({ open, onOpenChange }: { open: boolean; onOpenChange: (o: boolean) => void }) {
    54→    54→  const [state, setState] = useState<VoiceState>('idle')
    55→    55→  const [supported, setSupported] = useState(true)
    56→    56→  const [muted, setMuted] = useState(false)
    57→    57→  const [lang, setLang] = useState('en-US')
    58→    58→  const [ttsVoice, setTtsVoice] = useState('en-US-AriaNeural')
    59→    59→  const [interim, setInterim] = useState('')
    60→    60→  const [micLevel, setMicLevel] = useState(0)
    61→    61→  const [turns, setTurns] = useState<Turn[]>([])
    62→    62→  const [error, setError] = useState('')
    63→    63→  const [showLangs, setShowLangs] = useState(false)
    64→    64→  const [textInput, setTextInput] = useState('')
    65→    65→
    66→    66→  const recognitionRef = useRef<any>(null)
    67→    67→  const historyRef = useRef<Array<{ role: 'user' | 'assistant'; content: string }>>([])
    68→    68→  const audioRef = useRef<HTMLAudioElement | null>(null)
    69→    69→  const objectUrlRef = useRef<string | null>(null)
    70→    70→  const stateRef = useRef<VoiceState>('idle')
    71→    71→  const mutedRef = useRef(false)
    72→    72→  const startListeningRef = useRef<((langOverride?: string) => void) | null>(null)
    73→    73→
    74→    74→  const setStateSafe = useCallback((s: VoiceState) => {
    75→    75→    stateRef.current = s
    76→    76→    setState(s)
    77→    77→  }, [])
    78→    78→
    79→    79→  /* --------------------------------------------------------------- */
    80→    80→  /* Audio plumbing                                                   */
    81→    81→  /* --------------------------------------------------------------- */
    82→    82→
    83→    83→  const stopSpeaking = useCallback(() => {
    84→    84→    const el = audioRef.current
    85→    85→    if (el) {
    86→    86→      try {
    87→    87→        el.pause()
    88→    88→        el.removeAttribute('src')
    89→    89→        el.load()
    90→    90→      } catch {
    91→    91→        /* already stopped */
    92→    92→      }
    93→    93→    }
    94→    94→    if (objectUrlRef.current) {
    95→    95→      URL.revokeObjectURL(objectUrlRef.current)
    96→    96→      objectUrlRef.current = null
    97→    97→    }
    98→    98→  }, [])
    99→    99→
   100→   100→  /** Plays an object URL through the shared <audio> element. Resolves on end. */
   101→   101→  const playUrl = useCallback((url: string): Promise<void> => {
   102→   102→    return new Promise<void>((resolve) => {
   103→   103→      const el = audioRef.current
   104→   104→      if (!el) return resolve()
   105→   105→      el.src = url
   106→   106→      el.onended = () => resolve()
   107→   107→      el.onerror = () => resolve()
   108→   108→      el.play().catch(() => resolve())
   109→   109→    })
   110→   110→  }, [])
   111→   111→
   112→   112→  /**
   113→   113→   * Speaks a reply. Prefers the inline audio returned by /api/voice/turn
   114→   114→   * (saves a round-trip), then POST /api/tts, then speechSynthesis.
   115→   115→   */
   116→   116→  const speak = useCallback(
   117→   117→    async (text: string, inlineAudio?: string | null, audioFormat?: string): Promise<void> => {
   118→   118→      setStateSafe('speaking')
   119→   119→      let url: string | null = null
   120→   120→      try {
   121→   121→        if (inlineAudio) {
   122→   122→          const binary = atob(inlineAudio)
   123→   123→          const bytes = new Uint8Array(binary.length)
   124→   124→          for (let i = 0; i < binary.length; i++) bytes[i] = binary.charCodeAt(i)
   125→   125→          const blob = new Blob([bytes], { type: audioFormat === 'mp3' ? 'audio/mpeg' : 'audio/wav' })
   126→   126→          url = URL.createObjectURL(blob)
   127→   127→        } else {
   128→   128→          const res = await fetch('/api/tts', {
   129→   129→            method: 'POST',
   130→   130→            headers: { 'Content-Type': 'application/json' },
   131→   131→            body: JSON.stringify({ text, voice: ttsVoice, speed: 1.0 }),
   132→   132→          })
   133→   133→          if (!res.ok) throw new Error('TTS failed')
   134→   134→          const blob = await res.blob()
   135→   135→          url = URL.createObjectURL(blob)
   136→   136→        }
   137→   137→        if (objectUrlRef.current) URL.revokeObjectURL(objectUrlRef.current)
   138→   138→        objectUrlRef.current = url
   139→   139→        await playUrl(url)
   140→   140→      } catch {
   141→   141→        // Last resort: the browser's built-in synthesizer.
   142→   142→        try {
   143→   143→          await new Promise<void>((resolve) => {
   144→   144→            const utter = new SpeechSynthesisUtterance(text)
   145→   145→            utter.lang = lang
   146→   146→            utter.onend = () => resolve()
   147→   147→            utter.onerror = () => resolve()
   148→   148→            window.speechSynthesis.speak(utter)
   149→   149→            setTimeout(resolve, Math.min(30000, text.length * 100))
   150→   150→          })
   151→   151→        } catch {
   152→   152→          /* continue silently — the text is still in the log */
   153→   153→        }
   154→   154→      }
   155→   155→    },
   156→   156→    [ttsVoice, lang, playUrl, setStateSafe]
   157→   157→  )
   158→   158→
   159→   159→  /* --------------------------------------------------------------- */
   160→   160→  /* One conversation turn: think (LLM) → speak (TTS)                  */
   161→   161→  /* --------------------------------------------------------------- */
   162→   162→
   163→   163→  const think = useCallback(
   164→   164→    async (userText: string): Promise<{ reply?: string; audio?: string | null; audioFormat?: string }> => {
   165→   165→      const res = await fetch('/api/voice/turn', {
   166→   166→        method: 'POST',
   167→   167→        headers: { 'Content-Type': 'application/json' },
   168→   168→        body: JSON.stringify({
   169→   169→          message: userText,
   170→   170→          history: historyRef.current.slice(-6),
   171→   171→          language: lang.split('-')[0],
   172→   172→          voice: ttsVoice,
   173→   173→          audio: SILENT_WAV,
   174→   174→        }),
   175→   175→      })
   176→   176→      if (!res.ok) throw new Error('AI failed to respond — try again.')
   177→   177→      return await res.json()
   178→   178→    },
   179→   179→    [lang, ttsVoice]
   180→   180→  )
   181→   181→
   182→   182→  const handleUtterance = useCallback(
   183→   183→    async (text: string) => {
   184→   184→      const clean = text.trim()
   185→   185→      if (!clean) return
   186→   186→      stopRecognitionRef.current?.()
   187→   187→      setStateSafe('thinking')
   188→   188→      setInterim('')
   189→   189→      setError('')
   190→   190→      try {
   191→   191→        historyRef.current.push({ role: 'user', content: clean })
   192→   192→        const data = await think(clean)
   193→   193→        const reply = data.reply?.trim() || 'Sorry, I did not get that.'
   194→   194→        historyRef.current.push({ role: 'assistant', content: reply })
   195→   195→        setTurns((prev) => [...prev.slice(-5), { id: crypto.randomUUID(), user: clean, reply }])
   196→   196→        await speak(reply, data.audio, data.audioFormat)
   197→   197→      } catch (e) {
   198→   198→        setError(e instanceof Error ? e.message : 'Something went wrong — try again.')
   199→   199→      } finally {
   200→   200→        // Auto-reset to listening for the next turn.
   201→   201→        startListeningRef.current?.()
   202→   202→      }
   203→   203→    },
   204→   204→    [think, speak, setStateSafe]
   205→   205→  )
   206→   206→
   207→   207→  /* --------------------------------------------------------------- */
   208→   208→  /* Web Speech API input (ported from the proven omni voice mode)    */
   209→   209→  /* --------------------------------------------------------------- */
   210→   210→
   211→   211→  const stopRecognition = useCallback(() => {
   212→   212→    if (recognitionRef.current) {
   213→   213→      try {
   214→   214→        const pulse = recognitionRef.current.__pulse as ReturnType<typeof setInterval> | undefined
   215→   215→        if (pulse) clearInterval(pulse)
   216→   216→        recognitionRef.current.onend = null
   217→   217→        recognitionRef.current.onresult = null
   218→   218→        recognitionRef.current.onerror = null
   219→   219→        recognitionRef.current.stop()
   220→   220→      } catch {
   221→   221→        /* already stopped */
   222→   222→      }
   223→   223→      recognitionRef.current = null
   224→   224→      setMicLevel(0)
   225→   225→    }
   226→   226→  }, [])
   227→   227→
   228→   228→  const stopRecognitionRef = useRef(stopRecognition)
   229→   229→  useEffect(() => {
   230→   230→    stopRecognitionRef.current = stopRecognition
   231→   231→  }, [stopRecognition])
   232→   232→
   233→   233→  const startListening = useCallback(
   234→   234→    (langOverride?: string) => {
   235→   235→      const SR = (window as any).SpeechRecognition || (window as any).webkitSpeechRecognition
   236→   236→      if (!SR) {
   237→   237→        setSupported(false)
   238→   238→        setStateSafe('idle')
   239→   239→        return
   240→   240→      }
   241→   241→      if (mutedRef.current) return
   242→   242→      stopSpeaking()
   243→   243→
   244→   244→      const rec = new SR()
   245→   245→      rec.lang = langOverride ?? lang
   246→   246→      rec.continuous = true
   247→   247→      rec.interimResults = true
   248→   248→
   249→   249→      rec.onresult = (event: any) => {
   250→   250→        let finalTranscript = ''
   251→   251→        let interimText = ''
   252→   252→        for (let i = event.resultIndex; i < event.results.length; i++) {
   253→   253→          const r = event.results[i]
   254→   254→          if (r.isFinal) finalTranscript += r[0].transcript
   255→   255→          else interimText += r[0].transcript
   256→   256→        }
   257→   257→        if (interimText) {
   258→   258→          setInterim(interimText)
   259→   259→          setMicLevel(0.3 + Math.random() * 0.4) // active-speech visual
   260→   260→        }
   261→   261→        if (finalTranscript.trim()) {
   262→   262→          handleUtterance(finalTranscript)
   263→   263→        }
   264→   264→      }
   265→   265→
   266→   266→      rec.onerror = (event: any) => {
   267→   267→        if (event.error === 'not-allowed' || event.error === 'permission-denied') {
   268→   268→          setError('Microphone blocked — allow mic access for this site, then reopen voice mode.')
   269→   269→          setStateSafe('idle')
   270→   270→        } else if (event.error === 'audio-capture' || event.error === 'service-not-allowed' || event.error === 'network') {
   271→   271→          setError('Could not reach the speech service — check your connection.')
   272→   272→          setStateSafe('idle')
   273→   273→        }
   274→   274→        // 'no-speech' and 'aborted' are normal — keep listening.
   275→   275→      }
   276→   276→
   277→   277→      rec.onend = () => {
   278→   278→        // Chrome stops after silence — restart while still in listening state.
   279→   279→        if (stateRef.current === 'listening' && recognitionRef.current && !mutedRef.current) {
   280→   280→          try {
   281→   281→            recognitionRef.current.start()
   282→   282→          } catch {
   283→   283→            /* restart race — ignore */
   284→   284→          }
   285→   285→        }
   286→   286→      }
   287→   287→
   288→   288→      recognitionRef.current = rec
   289→   289→      try {
   290→   290→        rec.start()
   291→   291→        setStateSafe('listening')
   292→   292→        setError('')
   293→   293→        const pulse = setInterval(() => {
   294→   294→          setMicLevel((v) => Math.max(0.05, v * 0.85))
   295→   295→        }, 150)
   296→   296→        rec.__pulse = pulse
   297→   297→      } catch {
   298→   298→        /* already started */
   299→   299→      }
   300→   300→    },
   301→   301→    [lang, handleUtterance, stopSpeaking, setStateSafe]
   302→   302→  )
   303→   303→
   304→   304→  useEffect(() => {
   305→   305→    startListeningRef.current = startListening
   306→   306→  }, [startListening])
   307→   307→
   308→   308→  /* --------------------------------------------------------------- */
   309→   309→  /* Overlay lifecycle                                                */
   310→   310→  /* --------------------------------------------------------------- */
   311→   311→
   312→   312→  // Fresh conversation every time the overlay opens; teardown on close.
   313→   313→  useEffect(() => {
   314→   314→    if (!open) return
   315→   315→    const t = setTimeout(() => {
   316→   316→      setSupported(true)
   317→   317→      setMuted(false)
   318→   318→      mutedRef.current = false
   319→   319→      setTurns([])
   320→   320→      setInterim('')
   321→   321→      setError('')
   322→   322→      setTextInput('')
   323→   323→      setShowLangs(false)
   324→   324→      historyRef.current = []
   325→   325→      setStateSafe('idle')
   326→   326→      startListeningRef.current?.()
   327→   327→    }, 60)
   328→   328→    return () => {
   329→   329→      clearTimeout(t)
   330→   330→      stopRecognition()
   331→   331→      stopSpeaking()
   332→   332→    }
   333→   333→  }, [open, setStateSafe, stopRecognition, stopSpeaking])
   334→   334→
   335→   335→  // Escape closes the overlay.
   336→   336→  useEffect(() => {
   337→   337→    if (!open) return
   338→   338→    const onKey = (e: KeyboardEvent) => {
   339→   339→      if (e.key === 'Escape') onOpenChange(false)
   340→   340→    }
   341→   341→    window.addEventListener('keydown', onKey)
   342→   342→    return () => window.removeEventListener('keydown', onKey)
   343→   343→  }, [open, onOpenChange])
   344→   344→
   345→   345→  /* --------------------------------------------------------------- */
   346→   346→  /* Controls                                                         */
   347→   347→  /* --------------------------------------------------------------- */
   348→   348→
   349→   349→  const toggleMute = () => {
   350→   350→    if (muted) {
   351→   351→      setMuted(false)
   352→   352→      mutedRef.current = false
   353→   353→      setError('')
   354→   354→      startListening()
   355→   355→    } else {
   356→   356→      setMuted(true)
   357→   357→      mutedRef.current = true
   358→   358→      stopRecognition()
   359→   359→      setInterim('')
   360→   360→      setStateSafe('idle')
   361→   361→    }
   362→   362→  }
   363→   363→
   364→   364→  const pickLanguage = (l: (typeof LANGUAGES)[number]) => {
   365→   365→    setLang(l.code)
   366→   366→    setTtsVoice(l.voice)
   367→   367→    setShowLangs(false)
   368→   368→    // Restart recognition in the new language when mid-conversation.
   369→   369→    if (stateRef.current === 'listening') {
   370→   370→      stopRecognition()
   371→   371→      setTimeout(() => startListeningRef.current?.(l.code), 150)
   372→   372→    }
   373→   373→  }
   374→   374→
   375→   375→  const sendText = () => {
   376→   376→    const text = textInput.trim()
   377→   377→    if (!text || stateRef.current === 'thinking' || stateRef.current === 'speaking') return
   378→   378→    setTextInput('')
   379→   379→    void handleUtterance(text)
   380→   380→  }
   381→   381→
   382→   382→  /* --------------------------------------------------------------- */
   383→   383→  /* Render                                                           */
   384→   384→  /* --------------------------------------------------------------- */
   385→   385→
   386→   386→  if (!open) return null
   387→   387→
   388→   388→  const statusLabel =
   389→   389→    state === 'listening'
   390→   390→      ? 'Listening…'
   391→   391→      : state === 'thinking'
   392→   392→        ? 'Thinking…'
   393→   393→        : state === 'speaking'
   394→   394→          ? 'Speaking…'
   395→   395→          : muted
   396→   396→            ? 'Mic muted'
   397→   397→            : supported
   398→   398→              ? 'Starting…'
   399→   399→              : 'Text mode'
   400→   400→
   401→   401→  const recentTurns = turns.slice(-VISIBLE_TURNS)
   402→   402→
   403→   403→  return (
   404→   404→    <div
   405→   405→      role="dialog"
   406→   406→      aria-modal="true"
   407→   407→      aria-label="NEXUS voice mode"
   408→   408→      className="fixed inset-0 z-50 flex flex-col text-zinc-100"
   409→   409→      style={{
   410→   410→        background:
   411→   411→          'linear-gradient(168deg, rgba(9,9,11,0.97) 0%, rgba(20,8,12,0.96) 55%, rgba(9,9,11,0.98) 100%)',
   412→   412→        backdropFilter: 'blur(24px)',
   413→   413→        WebkitBackdropFilter: 'blur(24px)',
   414→   414→      }}
   415→   415→    >
   416→   416→      {/* Top bar */}
   417→   417→      <div className="flex items-center justify-between px-5 pt-5 sm:px-8">
   418→   418→        <div className="flex items-center gap-2.5">
   419→   419→          <BrandMark size={22} />
   420→   420→          <span className="text-xs font-semibold uppercase tracking-[0.2em] text-zinc-500">Voice</span>
   421→   421→        </div>
   422→   422→        <button
   423→   423→          onClick={() => onOpenChange(false)}
   424→   424→          aria-label="Close voice mode"
   425→   425→          className="rounded-full p-2.5 text-zinc-400 transition hover:bg-white/10 hover:text-zinc-100"
   426→   426→        >
   427→   427→          <X className="h-5 w-5" />
   428→   428→        </button>
   429→   429→      </div>
   430→   430→
   431→   431→      {/* Center stage */}
   432→   432→      <div className="flex min-h-0 flex-1 flex-col items-center justify-center px-6 py-4">
   433→   433→        {/* ---- The orb ---- */}
   434→   434→        <div className="relative flex items-center justify-center">
   435→   435→          {/* ambient brand glow */}
   436→   436→          <div
   437→   437→            aria-hidden
   438→   438→            className="absolute h-[300px] w-[300px] rounded-full"
   439→   439→            style={{
   440→   440→              background:
   441→   441→                'radial-gradient(closest-side, rgba(255,90,95,0.20), rgba(245,166,35,0.08) 55%, transparent 75%)',
   442→   442→              filter: 'blur(30px)',
   443→   443→            }}
   444→   444→          />
   445→   445→
   446→   446→          {/* listening — expanding rings that breathe with the mic */}
   447→   447→          {state === 'listening' &&
   448→   448→            [0, 1, 2].map((i) => (
   449→   449→              <motion.span
   450→   450→                key={`ring-l-${i}`}
   451→   451→                aria-hidden
   452→   452→                className="absolute rounded-full border"
   453→   453→                style={{ borderColor: 'rgba(255,90,95,0.45)', width: 192, height: 192 }}
   454→   454→                initial={{ scale: 1, opacity: 0.5 }}
   455→   455→                animate={{ scale: 1.55, opacity: 0 }}
   456→   456→                transition={{ duration: 1.9, repeat: Infinity, ease: 'easeOut', delay: i * 0.63 }}
   457→   457→              />
   458→   458→            ))}
   459→   459→
   460→   460→          {/* speaking — faster pulse locked to the nx-dot bounce rhythm */}
   461→   461→          {state === 'speaking' &&
   462→   462→            [0, 1].map((i) => (
   463→   463→              <motion.span
   464→   464→                key={`ring-s-${i}`}
   465→   465→                aria-hidden

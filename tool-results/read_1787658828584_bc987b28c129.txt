   401→  1650→              streamedSoFar += delta
   402→  1651→              send({ type: 'assistant_delta', delta })
   403→  1652→            }
   404→  1653→
   405→  1654→            // 1) User's connected provider — streamed, with model rotation
   406→  1655→            //    left to streamExternalChatCompletion's fallback chain.
   407→  1656→            if (userProvider) {
   408→  1657→              try {
   409→  1658→                content = await streamExternalChatCompletion(
   410→  1659→                  userProvider,
   411→  1660→                  llmMessages as Parameters<typeof streamExternalChatCompletion>[1],
   412→  1661→                  emitDelta,
   413→  1662→                  { maxTokens: 4000, timeoutMs: 60_000 }
   414→  1663→                )
   415→  1664→                didStream = true
   416→  1665→                console.log(`[api/chat] served by user provider: ${userProvider.providerId}/${userProvider.defaultModel}`)
   417→  1666→              } catch (provErr) {
   418→  1667→                // Partial text already visible → keep it, no retry (would
   419→  1668→                // duplicate on screen).
   420→  1669→                if (streamedSoFar.trim()) {
   421→  1670→                  content = streamedSoFar
   422→  1671→                  didStream = true
   423→  1672→                } else {
   424→  1673→                  console.error(
   425→  1674→                    '[api/chat] user provider failed, racing the free AI pool:',
   426→  1675→                    provErr instanceof Error ? provErr.message : provErr
   427→  1676→                  )
   428→  1677→                }
   429→  1678→              }
   430→  1679→            }
   431→  1680→
   432→  1681→            // 2) FREE MULTI-AI POOL — raced in parallel (kilo ∥ llm7 ∥ ovh),
   433→  1682→            //    first token wins. Also the fallback when no user provider
   434→  1683→            //    or the user provider failed before emitting anything.
   435→  1684→            if (!didStream) {
   436→  1685→              try {
   437→  1686→                const r = await streamAnonymousFallbackChat(
   438→  1687→                  llmMessages as Parameters<typeof streamAnonymousFallbackChat>[0],
   439→  1688→                  emitDelta,
   440→  1689→                  {
   441→  1690→                    maxTokens: 4000,
   442→  1691→                    timeoutMs: 60_000,
   443→  1692→                    task: 'chat',
   444→  1693→                  }
   445→  1694→                )
   446→  1695→                content = r.content
   447→  1696→                didStream = true
   448→  1697→                console.log(`[api/chat] served by free AI pool: ${r.providerId}/${r.model}`)
   449→  1698→              } catch (poolErr) {
   450→  1699→                // Mid-race partial output is already visible → keep it.
   451→  1700→                if (streamedSoFar.trim()) {
   452→  1701→                  content = streamedSoFar
   453→  1702→                  didStream = true
   454→  1703→                } else {
   455→  1704→                  console.error(
   456→  1705→                    '[api/chat] free AI pool failed, trying non-streamed smartChat:',
   457→  1706→                    poolErr instanceof Error ? poolErr.message : poolErr
   458→  1707→                  )
   459→  1708→                  // Last resort: non-streamed smartChat (user provider →
   460→  1709→                  // free pool chain) and emit the visible prelude as one chunk.
   461→  1710→                  content = await smartChat(llmMessages, { maxTokens: 4000, task: 'chat' })
   462→  1711→                  // Emit only the visible prelude (text before any
   463→  1712→                  // TOOL_CALL / ARTIFACT_PATCH directive) — matches the
   464→  1713→                  // peek-buffer behaviour of the normal streaming path.
   465→  1714→                  const prelude = stripToolCall(content)
   466→  1715→                  if (prelude) send({ type: 'assistant_delta', delta: prelude })
   467→  1716→                }
   468→  1717→              }
   469→  1718→            }
   470→  1719→            if (!content.trim()) throw new Error('Empty model response.')
   471→  1720→
   472→  1721→            const toolCall = isLast ? null : parseToolCall(content)
   473→  1722→
   474→  1723→            if (toolCall && toolCallsUsed < MAX_TOOL_CALLS) {
   475→  1724→              toolCallsUsed += 1
   476→  1725→
   477→  1726→              // If we streamed, the prelude (text before TOOL_CALL) is
   478→  1727→              // already visible to the user — that's the desired UX
   479→  1728→              // ("Let me search for that..." → tool runs → final answer).
   480→  1729→              // If we didn't stream (external provider), emit the prelude now.
   481→  1730→              if (!didStream) {
   482→  1731→                const prelude = stripToolCall(content)
   483→  1732→                send({ type: 'assistant_start', id: `as-${Date.now()}-${step}` })
   484→  1733→                if (prelude) send({ type: 'assistant_delta', delta: prelude })
   485→  1734→                send({ type: 'assistant_end', attachments: [] })
   486→  1735→              } else {
   487→  1736→                // Close the streamed bubble (may contain a prelude, may be empty)
   488→  1737→                send({ type: 'assistant_end', attachments: [] })
   489→  1738→              }
   490→  1739→
   491→  1740→              send({ type: 'tool_start', tool: toolCall.tool, args: toolCall.args, index: toolCallsUsed })
   492→  1741→
   493→  1742→              // Heartbeat: image generation can take 60s+ on the upstream
   494→  1743→              // provider. Without intermediate events the user perceives a
   495→  1744→              // freeze ("the app freezes with every action"). Emit a
   496→  1745→              // tool_progress ping every 5s while the tool is running so the
   497→  1746→              // client knows the stream is alive and can show "still working".
   498→  1747→              const toolStartedAt = Date.now()
   499→  1748→              const heartbeat = setInterval(() => {
   500→  1749→                const elapsedMs = Date.now() - toolStartedAt
   501→  1750→                send({
   502→  1751→                  type: 'tool_progress',
   503→  1752→                  tool: toolCall.tool,
   504→  1753→                  index: toolCallsUsed,
   505→  1754→                  elapsedMs,
   506→  1755→                  message: `Still working on ${toolCall.tool}… ${Math.round(elapsedMs / 1000)}s`,
   507→  1756→                })
   508→  1757→              }, 5000)
   509→  1758→
   510→  1759→              let ok = true
   511→  1760→              let result: unknown
   512→  1761→              let attachment: Record<string, unknown> | undefined
   513→  1762→              try {
   514→  1763→                const executed = await executeChatTool(req, toolCall.tool, toolCall.args, activeDoc)
   515→  1764→                result = executed.result
   516→  1765→                attachment = executed.attachment
   517→  1766→              } catch (error) {
   518→  1767→                ok = false
   519→  1768→                result = { error: error instanceof Error ? error.message : 'Tool failed.' }
   520→  1769→              } finally {
   521→  1770→                clearInterval(heartbeat)
   522→  1771→              }
   523→  1772→
   524→  1773→              if (attachment) attachments.push(attachment)
   525→  1774→              send({ type: 'tool_result', tool: toolCall.tool, ok, result, index: toolCallsUsed })
   526→  1775→
   527→  1776→              const resultJson = JSON.stringify(result).slice(0, 4000)
   528→  1777→              llmMessages.push({ role: 'assistant', content })
   529→  1778→              llmMessages.push({
   530→  1779→                role: 'user',
   531→  1780→                content: `TOOL_RESULT (${toolCall.tool}):\n${resultJson}\n\nContinue: use another TOOL_CALL if needed, or give your final answer.`,
   532→  1781→              })
   533→  1782→            } else {
   534→  1783→              // Phase 1 P2: parse any ARTIFACT_PATCH directives from the
   535→  1784→              // response BEFORE computing the visible final answer. Each
   536→  1785→              // patch is emitted as an `artifact_patch` event for the client
   537→  1786→              // to apply to the open artifact (ChatGPT-Canvas style targeted
   538→  1787→              // edit, instead of regenerating the whole artifact). Only
   539→  1788→              // patches targeting the currently-open artifact id are emitted
   540→  1789→              // — guards against the model addressing a stale or fabricated id.
   541→  1790→              if (openArtifact) {
   542→  1791→                const patches = parseArtifactPatches(content)
   543→  1792→                for (const patch of patches) {
   544→  1793→                  if (patch.artifactId === openArtifact.artifactId) {
   545→  1794→                    send({
   546→  1795→                      type: 'artifact_patch',
   547→  1796→                      artifactId: patch.artifactId,
   548→  1797→                      find: patch.find,
   549→  1798→                      replace: patch.replace,
   550→  1799→                      note: patch.note,
   551→  1800→                    })
   552→  1801→                  }
   553→  1802→                }
   554→  1803→              }
   555→  1804→
   556→  1805→              // Final answer: strip both TOOL_CALL and ARTIFACT_PATCH
   557→  1806→              // directives from the visible prose (leaving the model's
   558→  1807→              // one-line acknowledgement like "Done — shortened the intro.").
   559→  1808→              const finalAnswer = stripToolCall(content) || 'I could not complete that. Try rephrasing.'
   560→  1809→              const assistantMessage = await db.chatMessage.create({
   561→  1810→                data: {
   562→  1811→                  sessionId: session!.id,
   563→  1812→                  role: 'assistant',
   564→  1813→                  content: finalAnswer,
   565→  1814→                  // NEXUS One: persist attachments so image/video/document
   566→  1815→                  // cards survive session resume.
   567→  1816→                  attachments: attachments.length ? JSON.stringify(attachments) : null,
   568→  1817→                },
   569→  1818→              })
   570→  1819→              await db.chatSession.update({
   571→  1820→                where: { id: session!.id },
   572→  1821→                data: { updatedAt: new Date() },
   573→  1822→              })
   574→  1823→              // Mirror the assistant message + session timestamp to Supabase
   575→  1824→              if (verifiedUserId) {
   576→  1825→                void supabaseUpsert('chat_messages', {
   577→  1826→                  id: assistantMessage.id,
   578→  1827→                  session_id: session!.id,
   579→  1828→                  role: 'assistant',
   580→  1829→                  content: finalAnswer,
   581→  1830→                }, { onConflict: 'id' })

   583→  1832→                  id: session!.id,
   584→  1833→                  user_id: verifiedUserId,
   585→  1834→                  title: session!.title,
   586→  1835→                  kind: 'chat',
   587→  1836→                  updated_at: new Date().toISOString(),
   588→  1837→                }, { onConflict: 'id' })
   589→  1838→              }
   590→  1839→
   591→  1840→              if (didStream) {
   592→  1841→                // The streamed content IS the final answer. Just close the
   593→  1842→                // bubble with attachments + emit done.
   594→  1843→                send({ type: 'assistant_end', attachments })
   595→  1844→              } else {
   596→  1845→                // External provider path — emit the full content as one delta.
   597→  1846→                send({ type: 'assistant_start', id: `as-${Date.now()}-${step}` })
   598→  1847→                send({ type: 'assistant_delta', delta: finalAnswer })
   599→  1848→                send({ type: 'assistant_end', attachments })
   600→  1849→              }
   601→  1850→              send({ type: 'done', sessionId: session!.id })
   602→  1851→              break
   603→  1852→            }
   604→  1853→          }
   605→  1854→        } catch (error) {
   606→  1855→          console.error('[api/chat] stream error:', error)
   607→  1856→          send({
   608→  1857→            type: 'error',
   609→  1858→            message: error instanceof Error ? error.message : 'Chat failed.',
   610→  1859→          })
   611→  1860→        } finally {
   612→  1861→          controller.close()
   613→  1862→        }
   614→  1863→      },
   615→  1864→    })
   616→  1865→
   617→  1866→    return new NextResponse(stream, {
   618→  1867→      status: 200,
   619→  1868→      headers: {
   620→  1869→        'Content-Type': 'application/x-ndjson; charset=utf-8',

   623→  1872→      },

   625→  1874→  } catch (error) {
   626→  1875→    console.error('[api/chat] POST error:', error)
   627→  1876→    const message = error instanceof Error ? error.message : 'Chat failed.'
   628→  1877→    return NextResponse.json({ error: message }, { status: 500 })
   629→  1878→  }

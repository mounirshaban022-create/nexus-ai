   194→/** Streams a chat completion from the built-in ZAI engine, calling `onDelta`
   195→ *  with each token-chunk as it arrives. Returns the full accumulated content.
   196→ *
   197→ *  PEEK LOGIC — avoids showing a half-formed TOOL_CALL to the user:
   198→ *    1. We buffer the first ~24 chars locally before emitting any delta.
   199→ *    2. If the buffered text starts to look like a TOOL_CALL, we keep
   200→ *       buffering (no deltas) until the stream ends, then return the full
   201→ *       content for tool-call parsing.
   202→ *    3. Otherwise, we flush the peeked prefix as a single delta, then emit
   203→ *       subsequent chunks as they arrive.
   204→ *
   205→ *  This is what makes the AI "feel fast" — the user sees the first token
   206→ *  within ~200ms instead of waiting 4-20s for the full response. */
   207→async function streamZaiChat(
   208→  messages: Array<{ role: string; content: string }>,
   209→  onDelta: (delta: string) => void,
   210→  signal?: { aborted: boolean }
   211→): Promise<string> {
   212→  const zai = await getZAI()
   213→  const streamBody = await zai.chat.completions.create({
   214→    messages: messages.map((m) => ({ role: m.role as any, content: m.content })),
   215→    stream: true,
   216→    max_tokens: 4096,
   217→    temperature: 0.7,
   218→    thinking: { type: 'disabled' },
   219→  })
   220→
   221→  // streamBody is a ReadableStream (SSE format) when stream:true is honored
   222→  if (!(streamBody instanceof ReadableStream)) {
   223→    // Some providers return a JSON object instead of a stream. Fall back.
   224→    const fallback = streamBody as { choices?: Array<{ message?: { content?: string } }> }
   225→    const content = fallback.choices?.[0]?.message?.content ?? ''
   226→    if (content) onDelta(content)
   227→    return content
   228→  }
   229→
   230→  const reader = (streamBody as ReadableStream<Uint8Array>).getReader()
   231→  const decoder = new TextDecoder()
   232→  let buf = ''
   233→  let fullContent = ''
   234→  // Phase 1 P2: 'tool' mode also covers ARTIFACT_PATCH — we buffer the
   235→  // directive silently so the user doesn't see raw JSON streaming in, then
   236→  // parse it after the stream completes (same shape as TOOL_CALL).
   237→  let mode: 'peeking' | 'streaming' | 'tool' = 'peeking'
   238→  const PEEK_LEN = 24
   239→
   240→  while (true) {
   241→    if (signal?.aborted) break
   242→    const { done, value } = await reader.read()
   243→    if (done) break
   244→    buf += decoder.decode(value, { stream: true })
   245→
   246→    // SSE: lines starting with "data:"
   247→    const lines = buf.split('\n')
   248→    buf = lines.pop() ?? ''
   249→
   250→    for (const line of lines) {
   251→      const trimmed = line.trim()
   252→      if (!trimmed.startsWith('data:')) continue
   253→      const data = trimmed.slice(5).trim()
   254→      if (!data || data === '[DONE]') continue
   255→      let deltaText = ''
   256→      try {
   257→        const json = JSON.parse(data)
   258→        deltaText = json?.choices?.[0]?.delta?.content ?? ''
   259→      } catch {
   260→        // Some providers send plain-text chunks; treat the line itself as delta
   261→        if (data && !data.startsWith('{')) deltaText = data
   262→      }
   263→      if (!deltaText) continue
   264→      fullContent += deltaText
   265→
   266→      if (mode === 'peeking') {
   267→        // Check if this looks like a tool-call OR artifact-patch forming
   268→        if (
   269→          fullContent.startsWith('TOOL_CALL') ||
   270→          fullContent.includes('TOOL_CALL') ||
   271→          fullContent.startsWith('ARTIFACT_PATCH') ||
   272→          fullContent.includes('ARTIFACT_PATCH')
   273→        ) {
   274→          mode = 'tool'
   275→          // Don't emit deltas — wait for the full content to parse as a directive
   276→        } else if (fullContent.length >= PEEK_LEN) {
   277→          // Safe to assume this is a final answer — flush the peeked prefix
   278→          mode = 'streaming'
   279→          onDelta(fullContent)
   280→        }
   281→        // else: keep peeking
   282→      } else if (mode === 'streaming') {
   283→        // Emit just the new chunk (prefix already flushed)
   284→        onDelta(deltaText)
   285→      }
   286→      // mode === 'tool': silently accumulate
   287→    }
   288→  }
   289→
   290→  return fullContent
   291→}

   298→/** Balanced-brace JSON extraction (string-aware). */
   299→function extractJson(text: string): string | null {
   300→  const start = text.indexOf('{')
   301→  if (start === -1) return null
   302→  let depth = 0
   303→  let inString = false
   304→  let escaped = false
   305→  for (let i = start; i < text.length; i++) {
   306→    const ch = text[i]
   307→    if (inString) {
   308→      if (escaped) escaped = false
   309→      else if (ch === '\\') escaped = true
   310→      else if (ch === '"') inString = false
   311→      continue
   312→    }
   313→    if (ch === '"') inString = true
   314→    else if (ch === '{') depth++
   315→    else if (ch === '}') {
   316→      depth--
   317→      if (depth === 0) return text.slice(start, i + 1)
   318→    }
   319→  }
   320→  return null
   321→}
   322→
   323→function parseToolCall(text: string): ParsedToolCall | null {
   324→  const idx = text.indexOf('TOOL_CALL')
   325→  if (idx === -1) return null
   326→  const raw = text.slice(idx).replace(/```(?:json)?/g, '')
   327→  const candidate = extractJson(raw)
   328→  if (!candidate) return null
   329→  try {
   330→    const parsed = JSON.parse(candidate) as { tool?: unknown; args?: unknown }
   331→    if (typeof parsed.tool === 'string') {
   332→      return {
   333→        tool: parsed.tool,
   334→        args:
   335→          parsed.args && typeof parsed.args === 'object' && !Array.isArray(parsed.args)
   336→            ? (parsed.args as Record<string, unknown>)
   337→            : {},
   338→      }
   339→    }
   340→  } catch {
   341→    /* not valid */
   342→  }
   343→  return null
   344→}

   346→/**
   347→ * Phase 1 P2: parse one or more ARTIFACT_PATCH directives from a model
   348→ * response. Returns an array of patches (find/replace/note) targeting the
   349→ * open artifact. Returns [] when no patches are present.
   350→ *
   351→ * The model emits lines like:
   352→ *   ARTIFACT_PATCH: {"artifactId": "abc", "find": "X", "replace": "Y", "note": "..."}
   353→ *
   354→ * We extract each directive's JSON via the same balanced-brace extractor
   355→ * as TOOL_CALL — supports multiline values (escaped newlines) and skips
   356→ * malformed lines (returns the well-formed ones only).
   357→ */
   358→function parseArtifactPatches(text: string): Array<{
   359→  artifactId: string
   360→  find: string
   361→  replace: string
   362→  note?: string
   363→}> {
   364→  const patches: Array<{ artifactId: string; find: string; replace: string; note?: string }> = []
   365→  let searchFrom = 0
   366→  while (true) {
   367→    const idx = text.indexOf('ARTIFACT_PATCH', searchFrom)
   368→    if (idx === -1) break
   369→    // Skip the "ARTIFACT_PATCH" label and any trailing whitespace/colon
   370→    let cursor = idx + 'ARTIFACT_PATCH'.length
   371→    while (cursor < text.length && /[\s:]/.test(text[cursor])) cursor++
   372→    const candidate = extractJson(text.slice(cursor))
   373→    if (!candidate) {
   374→      searchFrom = idx + 1
   375→      continue
   376→    }
   377→    try {
   378→      const parsed = JSON.parse(candidate) as {
   379→        artifactId?: unknown
   380→        find?: unknown
   381→        replace?: unknown
   382→        note?: unknown
   383→      }
   384→      if (
   385→        typeof parsed.artifactId === 'string' &&
   386→        typeof parsed.find === 'string' &&
   387→        typeof parsed.replace === 'string'
   388→      ) {
   389→        patches.push({
   390→          artifactId: parsed.artifactId,
   391→          find: parsed.find,
   392→          replace: parsed.replace,
   393→          note: typeof parsed.note === 'string' ? parsed.note : undefined,
   394→        })
   395→      }
   396→    } catch {
   397→      /* malformed — skip */
   398→    }
   399→    searchFrom = cursor + candidate.length
   400→  }
   401→  return patches
   402→}

   404→function stripToolCall(text: string): string {
   405→  return text
   406→    .replace(/```(?:json)?\s*TOOL_CALL[\s\S]*?```\s*/g, '')
   407→    .replace(/TOOL_CALL\s*[:=][\s\S]*$/g, '')
   408→    .replace(/```(?:json)?\s*ARTIFACT_PATCH[\s\S]*?```\s*/g, '')
   409→    .replace(/ARTIFACT_PATCH\s*[:=][\s\S]*$/g, '')
   410→    .trim()
   411→}

   413→/**
   414→ * Phase 1 P1: detect a "remember: ..." prefix and extract the durable fact.
   415→ * Returns null when the message is not a memory request. Supports English
   416→ * ("remember:", "remember that ...", "remember to ...") and Arabic
   417→ * ("تذكّر:" / "تذكر:" / "تذكّر أنني ...") prefixes. The match is
   418→ * case-insensitive and tolerant of leading whitespace. This is the OPT-IN
   419→ * extraction path — NEXUS never silently extracts memory from arbitrary chat.
   420→ *
   421→ * Disambiguation: for English "remember ..." WITHOUT a colon and WITHOUT
   422→ * "that"/"to", we do NOT match — too ambiguous (could be "remember when we
   423→ * went to the park?"). For Arabic, the colon or the "أن"-prefix serves the
   424→ * same role.
   425→ */
   426→function parseRememberDirective(message: string): string | null {
   427→  // English: explicit colon OR "that"/"to" disambiguator
   428→  const enColon = message.match(/^\s*remember\s*:\s*(.+)$/i)
   429→  if (enColon) return enColon[1].trim().slice(0, 600)
   430→  const enThat = message.match(/^\s*remember\s+(?:that|to)\s+(.+)$/i)
   431→  if (enThat) return enThat[1].trim().slice(0, 600)
   432→  // Arabic: explicit colon OR "أن"-prefix (أن / أنني / أنك / أننا / إلخ)
   433→  // تذكّر (with shadda) and تذكر (without) are both accepted via alternation
   434→  // — the shadda (U+0651) is a combining mark that doesn't behave well in
   435→  // a character class.
   436→  const arColon = message.match(/^\s*(?:تذكّر|تذكر)\s*:\s*(.+)$/u)
   437→  if (arColon) return arColon[1].trim().slice(0, 600)
   438→  const arAnn = message.match(/^\s*(?:تذكّر|تذكر)\s+أن\p{L}*\s+(.+)$/u)
   439→  if (arAnn) return arAnn[1].trim().slice(0, 600)
   440→  return null
   441→}

   443→/** Executes a chat-ability tool (image/doc/code), returning an attachment. */
   444→async function executeChatTool(
   445→  toolId: string,
   446→  args: Record<string, unknown>
   447→): Promise<{ result: unknown; attachment?: Record<string, unknown> }> {
   448→  const origin = 'http://localhost:3000'
   449→
   450→  if (toolId === 'generate_image') {
   451→    const prompt = String(args.prompt ?? '').slice(0, 2000)
   452→    if (!prompt) throw new Error('prompt required')
   453→    const res = await fetch(`${origin}/api/image`, {
   454→      method: 'POST',
   455→      headers: { 'Content-Type': 'application/json' },
   456→      body: JSON.stringify({ prompt, size: '1024x1024' }),
   457→    })
   458→    const data = await res.json()
   459→    if (!res.ok) throw new Error(data.error || 'Image generation failed')
   460→    return {
   461→      result: { imageUrl: data.image.url, note: 'Image generated — show it inline.' },
   462→      attachment: { type: 'image', url: data.image.url, title: prompt.slice(0, 80) },
   463→    }
   464→  }
   465→
   466→  if (toolId === 'create_document') {
   467→    const format = String(args.format ?? 'docx').toLowerCase()
   468→    const title = String(args.title ?? 'Document').slice(0, 200)
   469→    const content = String(args.content ?? '')
   470→    if (!content) throw new Error('content required')
   471→
   472→    // Build blocks from plain content (headings via lines starting with #)
   473→    const blocks: Array<Record<string, unknown>> = []
   474→    for (const line of content.split('\n')) {
   475→      const trimmed = line.trim()
   476→      if (!trimmed) continue
   477→      if (trimmed.startsWith('## ')) blocks.push({ type: 'heading', text: trimmed.slice(3), level: 2 })
   478→      else if (trimmed.startsWith('# ')) blocks.push({ type: 'heading', text: trimmed.slice(2), level: 1 })
   479→      else if (/^[-•*]\s/.test(trimmed)) {
   480→        const last = blocks[blocks.length - 1]
   481→        if (last?.type === 'bullets') (last.items as string[]).push(trimmed.replace(/^[-•*]\s/, ''))
   482→        else blocks.push({ type: 'bullets', items: [trimmed.replace(/^[-•*]\s/, '')] })
   483→      } else {
   484→        blocks.push({ type: 'paragraph', text: trimmed })
   485→      }
   486→    }
   487→    if (format === 'pptx') {
   488→      // Convert to slides: each heading starts a slide
   489→      const slides: Array<Record<string, unknown>> = []
   490→      let current: { title: string; bullets: string[] } | null = null
   491→      for (const b of blocks) {
   492→        if (b.type === 'heading') {
   493→          if (current) slides.push({ type: 'slide', ...current })
   494→          current = { title: b.text as string, bullets: [] }
   495→        } else if (current) {
   496→          const items = b.type === 'bullets' ? (b.items as string[]) : [b.text as string]
   497→          current.bullets.push(...items.slice(0, 5))
   498→        }
   499→      }
   500→      if (current) slides.push({ type: 'slide', ...current })
   501→      blocks.length = 0
   502→      blocks.push(...(slides.length ? slides : [{ type: 'paragraph', text: content }]))
   503→    }
   504→
   505→    const res = await fetch(`${origin}/api/office/create`, {
   506→      method: 'POST',
   507→      headers: { 'Content-Type': 'application/json' },
   508→      body: JSON.stringify({
   509→        format: ['docx', 'xlsx', 'pptx'].includes(format) ? format : 'docx',
   510→        title,
   511→        blocks: blocks.slice(0, 40),
   512→      }),
   513→    })
   514→    const data = await res.json()
   515→    if (!res.ok) throw new Error(data.error || 'Document creation failed')
   516→    // Phase 1 P2: include the source content + a stable artifactId so the
   517→    // ArtifactPanel can render the editable text and the AI can target
   518→    // ARTIFACT_PATCH directives at this artifact across follow-up messages.
   519→    const artifactId = `doc-${data.file.id ?? Date.now()}`
   520→    return {
   521→      result: {
   522→        downloadUrl: `${data.file.url}?download=1&title=${encodeURIComponent(title)}`,
   523→        format: data.file.format,
   524→        note: 'Document created — give the user the download link.',
   525→      },
   526→      attachment: {
   527→        type: 'document',
   528→        url: `${data.file.url}?download=1&title=${encodeURIComponent(title)}`,
   529→        title,
   530→        format: data.file.format,
   531→        size: data.file.size,
   532→        artifactId,
   533→        sourceContent: content,
   534→      },
   535→    }
   536→  }
   537→
   538→  if (toolId === 'run_code') {
   539→    const language = String(args.language ?? 'javascript').toLowerCase()
   540→    const code = String(args.code ?? '')
   541→    if (!code) throw new Error('code required')
   542→    const res = await fetch(`${origin}/api/code/run`, {
   543→      method: 'POST',
   544→      headers: { 'Content-Type': 'application/json' },
   545→      body: JSON.stringify({
   546→        language: ['javascript', 'typescript', 'python'].includes(language) ? language : 'javascript',
   547→        code,
   548→      }),
   549→    })
   550→    const data = await res.json()
   551→    if (!res.ok) throw new Error(data.error || 'Code execution failed')
   552→    const r = data.result
   553→    // Phase 1 P2: include the source code + a stable artifactId so the
   554→    // ArtifactPanel can show the code editor and the AI can target
   555→    // ARTIFACT_PATCH directives at this code artifact.
   556→    const artifactId = `code-${Date.now()}`
   557→    return {
   558→      result: {
   559→        stdout: r.stdout?.slice(0, 3000),
   560→        stderr: r.stderr?.slice(0, 1500),
   561→        exitCode: r.exitCode,
   562→        note: 'Code executed — include the output in your answer.',
   563→      },
   564→      attachment: {
   565→        type: 'code',
   566→        language,
   567→        stdout: (r.stdout ?? '').slice(0, 2000),
   568→        stderr: (r.stderr ?? '').slice(0, 500),
   569→        exitCode: r.exitCode,
   570→        artifactId,
   571→        sourceContent: code,
   572→      },
   573→    }
   574→  }
   575→
   576→  // Data connectors pass through
   577→  const connector = CONNECTOR_MAP.get(toolId)
   578→  if (!connector) return { result: { error: `Unknown tool "${toolId}"` } }
   579→  const validated = validateConnectorArgs(connector, args)
   580→  if (!validated.ok) return { result: { error: validated.error } }
   581→  const result = await connector.execute(validated.args)
   582→  // Attach search results richly
   583→  if (toolId === 'web_search' && Array.isArray((result as { results?: unknown[] }).results)) {
   584→    const results = (result as { results: Array<{ title: string; url: string; snippet: string }> }).results
   585→    return {
   586→      result,
   587→      attachment: {
   588→        type: 'search',
   589→        results: results.slice(0, 5).map((r) => ({ title: r.title, url: r.url, snippet: r.snippet?.slice(0, 140) })),
   590→      },
   591→    }
   592→  }
   593→  return { result }
   594→}

   596→export async function POST(req: NextRequest) {
   597→  try {
   598→    const limit = rateLimit(`chat:${clientKey(req)}`, 30, 60_000)
   599→    if (!limit.ok) {
   600→      return NextResponse.json(
   601→        { error: `Too many messages. Retry in ${limit.retryAfterSeconds}s.` },
   602→        { status: 429 }
   603→      )
   604→    }
   605→
   606→    const parsed = requestSchema.safeParse(await req.json().catch(() => null))
   607→    if (!parsed.success) {
   608→      return NextResponse.json({ error: 'Message is required (max 8000 chars).' }, { status: 400 })
   609→    }
   610→    const { message, sessionId, thinking: thinkingEnabled, language, openArtifact } = parsed.data
   611→    const trimmed = message.trim()
   612→
   613→    // Phase 0 Bug 2: derive the verified user id ONCE (JWT verification is
   614→    // not free) and reuse for both session ownership (Bug 3) and Supabase
   615→    // cloud sync. null = guest.
   616→    const verifiedUserId = await getUserIdFromRequest(req)
   617→
   618→    // Default enabled: abilities + key connectors
   619→    const enabledConnectors = [
   620→      'web_search', 'read_page', 'wikipedia', 'weather', 'crypto', 'currency',
   621→      'translate', 'dictionary', 'github', 'hacker_news', 'time', 'calculator',
   622→      'recipes', 'nasa', 'news', 'trivia', 'pokemon', 'games', 'forecast',
   623→    ]
   624→
   625→    // Phase 0 Bug 3: ownership check.
   626→    //   - Authenticated user (verifiedUserId = "abc"): may only resume
   627→    //     sessions where userId = "abc".
   628→    //   - Guest (verifiedUserId = null): may only resume sessions where
   629→    //     userId is null (guest sessions). Any guest can resume any guest
   630→    //     session by id — that is the documented guest behaviour.
   631→    //   - If a client supplies an authenticated user's sessionId without a
   632→    //     valid cookie for that user, the lookup returns null and a fresh
   633→    //     session is created — preventing cross-user takeover.
   634→    let session = sessionId
   635→      ? await db.chatSession.findFirst({
   636→          where: { id: sessionId, kind: 'chat', userId: verifiedUserId },
   637→        })
   638→      : null
   639→    if (!session) {
   640→      session = await db.chatSession.create({
   641→        data: {
   642→          kind: 'chat',
   643→          title: trimmed.slice(0, 60) + (trimmed.length > 60 ? '…' : ''),
   644→          // Phase 0 Bug 3: stamp ownership on creation.
   645→          userId: verifiedUserId,
   646→        },
   647→      })
   648→      // Mirror to Supabase (cloud) if user is authenticated
   649→      if (verifiedUserId) {
   650→        const admin = await getSupabaseAdmin()
   651→        if (admin) {
   652→          await admin
   653→            .from('chat_sessions')
   654→            .insert({ id: session.id, user_id: verifiedUserId, title: session.title, kind: 'chat' })
   655→            .then(r => console.log('[supabase-sync] session saved'), e => console.error('[supabase-sync] FAILED:', e.message))
   656→        }
   657→      }
   658→    }
   659→
   660→    const zai = await getZAI()
   661→    // Phase 1 P1: fetch the verified user's durable memories (top 25 by
   662→    // recency) and inject them into the system prompt. Guests and users
   663→    // with no memories get the unmodified prompt.
   664→    const userMemories = verifiedUserId
   665→      ? await db.userMemory.findMany({
   666→          where: { userId: verifiedUserId },
   667→          orderBy: { createdAt: 'desc' },
   668→          take: 25,
   669→          select: { content: true },
   670→        })
   671→      : []
   672→    const systemPrompt = buildSystemPrompt(enabledConnectors, language, userMemories, openArtifact)
   673→
   674→    // Phase 1 P1: detect a "remember: ..." directive from the user. This is
   675→    // the opt-in extraction path — we never silently extract. When the user
   676→    // types "remember: I'm a morning person" the durable fact is saved to
   677→    // the UserMemory table for future sessions, scoped to this user only.
   678→    // The assistant still answers the message normally afterward.
   679→    const rememberFact = verifiedUserId ? parseRememberDirective(trimmed) : null
   680→    if (rememberFact && verifiedUserId) {
   681→      // Cap total memories per user (matches the /api/memory POST guard).
   682→      const memCount = await db.userMemory.count({ where: { userId: verifiedUserId } })
   683→      if (memCount < 500) {
   684→        await db.userMemory.create({
   685→          data: {
   686→            userId: verifiedUserId,
   687→            content: rememberFact,
   688→            sourceSessionId: session.id,
   689→          },
   690→        }).catch((e) => console.error('[memory] save failed:', e))
   691→      }
   692→    }
   693→
   694→    // Build message history
   695→    const history = await db.chatMessage.findMany({
   696→      where: { sessionId: session.id },
   697→      orderBy: { createdAt: 'asc' },
   698→    })
   699→
   700→    const encoder = new TextEncoder()
   701→    const stream = new ReadableStream<Uint8Array>({
   702→      async start(controller) {
   703→        const send = (event: Record<string, unknown>) => {
   704→          controller.enqueue(encoder.encode(JSON.stringify(event) + '\n'))
   705→        }
   706→
   707→        // DIRECT ANSWER: creator identity (bypass LLM to prevent wrong answers)
   708→        if (/who (created|made|built|developed|owns) (you|this|nexus)|your (creator|maker|developer|owner)|who.s mounir/i.test(trimmed)) {
   709→          const answer = 'I was created by **Mounir Shaaban**, the creator and owner of NEXUS AI. 🚀'
   710→          const userMsg = await db.chatMessage.create({
   711→            data: { sessionId: session!.id, role: 'user', content: trimmed },
   712→          })
   713→          const saved = await db.chatMessage.create({
   714→            data: { sessionId: session!.id, role: 'assistant', content: answer },
   715→          })
   716→          send({ type: 'user', id: userMsg.id, content: trimmed })
   717→          send({ type: 'assistant', content: answer, attachments: [] })
   718→          send({ type: 'done', sessionId: session!.id })
   719→          return
   720→        }
   721→
   722→        try {
   723→          // Persist user message
   724→          const userMessage = await db.chatMessage.create({
   725→            data: { sessionId: session!.id, role: 'user', content: trimmed },
   726→          })
   727→          // Mirror to Supabase (Phase 0 Bug 2: verifiedUserId replaces the
   728→          // forged-header path; the cloud row is only written for an
   729→          // authenticated user, with their verified id)
   730→          {
   731→            if (verifiedUserId) {
   732→              const admin = await getSupabaseAdmin()
   733→              if (admin) {
   734→                admin.from('chat_messages')
   735→                  .insert({ id: userMessage.id, session_id: session!.id, role: 'user', content: trimmed })
   736→                  .then(r => console.log('[supabase-sync] user msg saved'), e => console.error('[supabase-sync] user msg FAILED:', e.message))
   737→              }
   738→            }
   739→          }
   740→          send({ type: 'user', id: userMessage.id, content: trimmed })
   741→
   742→          // LLM conversation: system + history + tool exchanges
   743→          const llmMessages: Array<{ role: string; content: string }> = [
   744→            { role: 'assistant', content: systemPrompt },
   745→            ...history.slice(-MAX_HISTORY).map((m) => ({ role: m.role, content: m.content })),
   746→          ]
   747→          // Include the new user message (history was fetched before insert)
   748→          llmMessages.push({ role: 'user', content: trimmed })
   749→
   750→          // Optional thinking phase
   751→          if (thinkingEnabled) {
   752→            try {
   753→              const thinking = (await smartChat([
   754→                {
   755→                  role: 'assistant',
   756→                  content:
   757→                    'You are the reasoning engine of NEXUS AI. Given a question, produce a SHORT internal analysis (max 80 words, telegraphic) covering: what the user needs, which tools to use (if any), and the answer structure. Output ONLY the analysis.',
   758→                },
   759→                { role: 'user', content: trimmed },
   760→              ], { maxTokens: 300, task: 'reasoning' })).trim().slice(0, 900)
   761→              if (thinking) {
   762→                llmMessages.push({
   763→                  role: 'user',
   764→                  content: `[internal analysis: ${thinking}]`,
   765→                })
   766→              }
   767→            } catch {
   768→              /* best-effort */
   769→            }
   770→          }
   771→
   772→          const attachments: Array<Record<string, unknown>> = []
   773→          let toolCallsUsed = 0
   774→
   775→          // Determine if we can stream from the built-in ZAI engine
   776→          // (when a user provider is connected, smartChat handles it non-streamed)
   777→          const hasUserProvider = !!(await getActiveAiProvider())
   778→
   779→          for (let step = 0; step <= MAX_TOOL_CALLS; step++) {
   780→            const isLast = step === MAX_TOOL_CALLS
   781→
   782→            // STREAMING PATH — emit assistant_start + assistant_delta chunks
   783→            // as tokens arrive (only when using the built-in ZAI engine).
   784→            let content: string
   785→            let didStream = false
   786→            if (!hasUserProvider) {
   787→              // Open a new assistant message bubble on the client
   788→              send({ type: 'assistant_start', id: `as-${Date.now()}-${step}` })
   789→              content = await streamZaiChat(llmMessages, (delta) => {
   790→                send({ type: 'assistant_delta', delta })
   791→              })
   792→              didStream = true
   793→              if (!content.trim()) throw new Error('Empty model response.')
   794→            } else {
   795→              content = await smartChat(llmMessages, { maxTokens: 4000, task: 'chat' })
   796→              if (!content.trim()) throw new Error('Empty model response.')
   797→            }
   798→
   799→            const toolCall = isLast ? null : parseToolCall(content)
   800→
   801→            if (toolCall && toolCallsUsed < MAX_TOOL_CALLS) {
   802→              toolCallsUsed += 1
   803→
   804→              // If we streamed, the prelude (text before TOOL_CALL) is
   805→              // already visible to the user — that's the desired UX
   806→              // ("Let me search for that..." → tool runs → final answer).
   807→              // If we didn't stream (external provider), emit the prelude now.
   808→              if (!didStream) {
   809→                const prelude = stripToolCall(content)
   810→                send({ type: 'assistant_start', id: `as-${Date.now()}-${step}` })
   811→                if (prelude) send({ type: 'assistant_delta', delta: prelude })
   812→                send({ type: 'assistant_end', attachments: [] })
   813→              } else {
   814→                // Close the streamed bubble (may contain a prelude, may be empty)
   815→                send({ type: 'assistant_end', attachments: [] })
   816→              }
   817→
   818→              send({ type: 'tool_start', tool: toolCall.tool, args: toolCall.args, index: toolCallsUsed })
   819→
   820→              // Heartbeat: image generation can take 60s+ on the upstream
   821→              // provider. Without intermediate events the user perceives a
   822→              // freeze ("the app freezes with every action"). Emit a
   823→              // tool_progress ping every 5s while the tool is running so the
   824→              // client knows the stream is alive and can show "still working".
   825→              const toolStartedAt = Date.now()
   826→              const heartbeat = setInterval(() => {
   827→                const elapsedMs = Date.now() - toolStartedAt
   828→                send({
   829→                  type: 'tool_progress',
   830→                  tool: toolCall.tool,
   831→                  index: toolCallsUsed,
   832→                  elapsedMs,
   833→                  message: `Still working on ${toolCall.tool}… ${Math.round(elapsedMs / 1000)}s`,
   834→                })
   835→              }, 5000)
   836→
   837→              let ok = true
   838→              let result: unknown
   839→              let attachment: Record<string, unknown> | undefined
   840→              try {
   841→                const executed = await executeChatTool(toolCall.tool, toolCall.args)
   842→                result = executed.result
   843→                attachment = executed.attachment
   844→              } catch (error) {
   845→                ok = false
   846→                result = { error: error instanceof Error ? error.message : 'Tool failed.' }
   847→              } finally {
   848→                clearInterval(heartbeat)
   849→              }
   850→
   851→              if (attachment) attachments.push(attachment)
   852→              send({ type: 'tool_result', tool: toolCall.tool, ok, result, index: toolCallsUsed })
   853→
   854→              const resultJson = JSON.stringify(result).slice(0, 4000)
   855→              llmMessages.push({ role: 'assistant', content })
   856→              llmMessages.push({
   857→                role: 'user',
   858→                content: `TOOL_RESULT (${toolCall.tool}):\n${resultJson}\n\nContinue: use another TOOL_CALL if needed, or give your final answer.`,
   859→              })
   860→            } else {
   861→              // Phase 1 P2: parse any ARTIFACT_PATCH directives from the
   862→              // response BEFORE computing the visible final answer. Each
   863→              // patch is emitted as an `artifact_patch` event for the client
   864→              // to apply to the open artifact (ChatGPT-Canvas style targeted
   865→              // edit, instead of regenerating the whole artifact). Only
   866→              // patches targeting the currently-open artifact id are emitted
   867→              // — guards against the model addressing a stale or fabricated id.
   868→              if (openArtifact) {
   869→                const patches = parseArtifactPatches(content)
   870→                for (const patch of patches) {
   871→                  if (patch.artifactId === openArtifact.artifactId) {
   872→                    send({
   873→                      type: 'artifact_patch',
   874→                      artifactId: patch.artifactId,
   875→                      find: patch.find,
   876→                      replace: patch.replace,
   877→                      note: patch.note,
   878→                    })
   879→                  }
   880→                }
   881→              }
   882→
   883→              // Final answer: strip both TOOL_CALL and ARTIFACT_PATCH
   884→              // directives from the visible prose (leaving the model's
   885→              // one-line acknowledgement like "Done — shortened the intro.").
   886→              const finalAnswer = stripToolCall(content) || 'I could not complete that. Try rephrasing.'
   887→              await db.chatMessage.create({
   888→                data: { sessionId: session!.id, role: 'assistant', content: finalAnswer },
   889→              })
   890→              await db.chatSession.update({
   891→                where: { id: session!.id },
   892→                data: { updatedAt: new Date() },
   893→              })
   894→
   895→              if (didStream) {
   896→                // The streamed content IS the final answer. Just close the
   897→                // bubble with attachments + emit done.
   898→                send({ type: 'assistant_end', attachments })
   899→              } else {
   900→                // External provider path — emit the full content as one delta.
   901→                send({ type: 'assistant_start', id: `as-${Date.now()}-${step}` })
   902→                send({ type: 'assistant_delta', delta: finalAnswer })
   903→                send({ type: 'assistant_end', attachments })
   904→              }
   905→              send({ type: 'done', sessionId: session!.id })
   906→              break
   907→            }
   908→          }
   909→        } catch (error) {
   910→          console.error('[api/chat] stream error:', error)
   911→          send({
   912→            type: 'error',
   913→            message: error instanceof Error ? error.message : 'Chat failed.',
   914→          })
   915→        } finally {
   916→          controller.close()
   917→        }
   918→      },
   919→    })
   920→
   921→    return new NextResponse(stream, {
   922→      status: 200,
   923→      headers: {
   924→        'Content-Type': 'application/x-ndjson; charset=utf-8',
   925→        'Cache-Control': 'no-cache, no-transform',
   926→        'X-Accel-Buffering': 'no',
   927→      },
   928→    })
   929→  } catch (error) {
   930→    console.error('[api/chat] POST error:', error)
   931→    const message = error instanceof Error ? error.message : 'Chat failed.'
   932→    return NextResponse.json({ error: message }, { status: 500 })
   933→  }
   934→}

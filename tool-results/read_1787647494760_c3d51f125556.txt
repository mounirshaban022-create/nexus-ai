   293→interface ParsedToolCall {
   294→  tool: string
   295→  args: Record<string, unknown>
   296→}

   298→/** Balanced-brace JSON extraction (string-aware). */
   299→function extractJson(text: string): string | null {
   300→  const start = text.indexOf('{')
   301→  if (start === -1) return null
   302→  let depth = 0
   303→  let inString = false
   304→  let escaped = false
   305→  for (let i = start; i < text.length; i++) {
   306→    const ch = text[i]
   307→    if (inString) {
   308→      if (escaped) escaped = false
   309→      else if (ch === '\\') escaped = true
   310→      else if (ch === '"') inString = false
   311→      continue
   312→    }
   313→    if (ch === '"') inString = true
   314→    else if (ch === '{') depth++
   315→    else if (ch === '}') {
   316→      depth--
   317→      if (depth === 0) return text.slice(start, i + 1)
   318→    }
   319→  }
   320→  return null
   321→}
   322→
   323→function parseToolCall(text: string): ParsedToolCall | null {
   324→  const idx = text.indexOf('TOOL_CALL')
   325→  if (idx === -1) return null
   326→  const raw = text.slice(idx).replace(/```(?:json)?/g, '')
   327→  const candidate = extractJson(raw)
   328→  if (!candidate) return null
   329→  try {
   330→    const parsed = JSON.parse(candidate) as { tool?: unknown; args?: unknown }
   331→    if (typeof parsed.tool === 'string') {
   332→      return {
   333→        tool: parsed.tool,
   334→        args:
   335→          parsed.args && typeof parsed.args === 'object' && !Array.isArray(parsed.args)
   336→            ? (parsed.args as Record<string, unknown>)
   337→            : {},
   338→      }
   339→    }
   340→  } catch {
   341→    /* not valid */
   342→  }
   343→  return null
   344→}

   346→/**
   347→ * Phase 1 P2: parse one or more ARTIFACT_PATCH directives from a model
   348→ * response. Returns an array of patches (find/replace/note) targeting the
   349→ * open artifact. Returns [] when no patches are present.
   350→ *
   351→ * The model emits lines like:
   352→ *   ARTIFACT_PATCH: {"artifactId": "abc", "find": "X", "replace": "Y", "note": "..."}
   353→ *
   354→ * We extract each directive's JSON via the same balanced-brace extractor
   355→ * as TOOL_CALL — supports multiline values (escaped newlines) and skips
   356→ * malformed lines (returns the well-formed ones only).
   357→ */
   358→function parseArtifactPatches(text: string): Array<{
   359→  artifactId: string
   360→  find: string
   361→  replace: string
   362→  note?: string
   363→}> {
   364→  const patches: Array<{ artifactId: string; find: string; replace: string; note?: string }> = []
   365→  let searchFrom = 0
   366→  while (true) {
   367→    const idx = text.indexOf('ARTIFACT_PATCH', searchFrom)
   368→    if (idx === -1) break
   369→    // Skip the "ARTIFACT_PATCH" label and any trailing whitespace/colon
   370→    let cursor = idx + 'ARTIFACT_PATCH'.length
   371→    while (cursor < text.length && /[\s:]/.test(text[cursor])) cursor++
   372→    const candidate = extractJson(text.slice(cursor))
   373→    if (!candidate) {
   374→      searchFrom = idx + 1
   375→      continue
   376→    }
   377→    try {
   378→      const parsed = JSON.parse(candidate) as {
   379→        artifactId?: unknown
   380→        find?: unknown
   381→        replace?: unknown
   382→        note?: unknown
   383→      }
   384→      if (
   385→        typeof parsed.artifactId === 'string' &&
   386→        typeof parsed.find === 'string' &&
   387→        typeof parsed.replace === 'string'
   388→      ) {
   389→        patches.push({
   390→          artifactId: parsed.artifactId,
   391→          find: parsed.find,
   392→          replace: parsed.replace,
   393→          note: typeof parsed.note === 'string' ? parsed.note : undefined,
   394→        })
   395→      }
   396→    } catch {
   397→      /* malformed — skip */
   398→    }
   399→    searchFrom = cursor + candidate.length
   400→  }
   401→  return patches
   402→}

   404→function stripToolCall(text: string): string {
   405→  return text
   406→    .replace(/```(?:json)?\s*TOOL_CALL[\s\S]*?```\s*/g, '')
   407→    .replace(/TOOL_CALL\s*[:=][\s\S]*$/g, '')
   408→    .replace(/```(?:json)?\s*ARTIFACT_PATCH[\s\S]*?```\s*/g, '')
   409→    .replace(/ARTIFACT_PATCH\s*[:=][\s\S]*$/g, '')
   410→    .trim()
   411→}

   413→/**
   414→ * Phase 1 P1: detect a "remember: ..." prefix and extract the durable fact.
   415→ * Returns null when the message is not a memory request. Supports English
   416→ * ("remember:", "remember that ...", "remember to ...") and Arabic
   417→ * ("تذكّر:" / "تذكر:" / "تذكّر أنني ...") prefixes. The match is
   418→ * case-insensitive and tolerant of leading whitespace. This is the OPT-IN
   419→ * extraction path — NEXUS never silently extracts memory from arbitrary chat.
   420→ *
   421→ * Disambiguation: for English "remember ..." WITHOUT a colon and WITHOUT
   422→ * "that"/"to", we do NOT match — too ambiguous (could be "remember when we
   423→ * went to the park?"). For Arabic, the colon or the "أن"-prefix serves the
   424→ * same role.
   425→ */
   426→function parseRememberDirective(message: string): string | null {
   427→  // English: explicit colon OR "that"/"to" disambiguator
   428→  const enColon = message.match(/^\s*remember\s*:\s*(.+)$/i)
   429→  if (enColon) return enColon[1].trim().slice(0, 600)
   430→  const enThat = message.match(/^\s*remember\s+(?:that|to)\s+(.+)$/i)
   431→  if (enThat) return enThat[1].trim().slice(0, 600)
   432→  // Arabic: explicit colon OR "أن"-prefix (أن / أنني / أنك / أننا / إلخ)
   433→  // تذكّر (with shadda) and تذكر (without) are both accepted via alternation
   434→  // — the shadda (U+0651) is a combining mark that doesn't behave well in
   435→  // a character class.
   436→  const arColon = message.match(/^\s*(?:تذكّر|تذكر)\s*:\s*(.+)$/u)
   437→  if (arColon) return arColon[1].trim().slice(0, 600)
   438→  const arAnn = message.match(/^\s*(?:تذكّر|تذكر)\s+أن\p{L}*\s+(.+)$/u)
   439→  if (arAnn) return arAnn[1].trim().slice(0, 600)
   440→  return null
   441→}

  1011→    // types "remember: I'm a morning person" the durable fact is saved to
  1012→    // the UserMemory table for future sessions, scoped to this user only.
  1013→    // The assistant still answers the message normally afterward.
  1014→    const rememberFact = verifiedUserId ? parseRememberDirective(trimmed) : null
  1015→    if (rememberFact && verifiedUserId) {
  1016→      // Cap total memories per user (matches the /api/memory POST guard).
  1017→      const memCount = await db.userMemory.count({ where: { userId: verifiedUserId } })

  1020→          data: {
  1021→            userId: verifiedUserId,
  1022→            content: rememberFact,
  1023→            sourceSessionId: session.id,
  1024→          },
  1025→        }).catch((e) => console.error('[memory] save failed:', e))

  1031→      where: { sessionId: session.id },
  1032→      orderBy: { createdAt: 'asc' },
  1033→    })
  1034→
  1035→    const encoder = new TextEncoder()
  1036→    const stream = new ReadableStream<Uint8Array>({
  1037→      async start(controller) {
  1038→        const send = (event: Record<string, unknown>) => {
  1039→          controller.enqueue(encoder.encode(JSON.stringify(event) + '\n'))
  1040→        }
  1041→
  1042→        // DIRECT ANSWER: creator identity (bypass LLM to prevent wrong answers)
  1043→        if (/who (created|made|built|developed|owns) (you|this|nexus)|your (creator|maker|developer|owner)|who.s mounir|where.*(creator|maker|from)|mounir|mansoura|who made this app|who built this/i.test(trimmed)) {
  1044→          const answer =
  1045→            'I was created by **Mounir Shaaban** — the creator and owner of NEXUS AI. ' +
  1046→            "He's from **Mansoura, Egypt**. 🇪🇬🚀 If you'd like, tell me what you want to build today!" 
  1047→          const userMsg = await db.chatMessage.create({
  1048→            data: { sessionId: session!.id, role: 'user', content: trimmed || '[attached file]' },
  1049→          })
  1050→          const saved = await db.chatMessage.create({
  1051→            data: { sessionId: session!.id, role: 'assistant', content: answer },
  1052→          })
  1053→          send({ type: 'user', id: userMsg.id, content: trimmed })
  1054→          send({ type: 'assistant', content: answer, attachments: [] })

  1057→        }
  1058→
  1059→        try {
  1060→          // Persist user message
  1061→          const userMessage = await db.chatMessage.create({
  1062→            data: { sessionId: session!.id, role: 'user', content: trimmed || '[attached file]' },
  1063→          })
  1064→          // Mirror to Supabase (Phase 0 Bug 2: verifiedUserId replaces the
  1065→          // forged-header path; the cloud row is only written for an
  1066→          // authenticated user, with their verified id)
  1067→          if (verifiedUserId) {
  1068→            void supabaseUpsert('chat_messages', {
  1069→              id: userMessage.id,
  1070→              session_id: session!.id,
  1071→              role: 'user',
  1072→              content: trimmed,
  1073→            }, { onConflict: 'id' })
  1074→          }
  1075→          send({ type: 'user', id: userMessage.id, content: trimmed })
  1076→
  1077→          // LLM conversation: system + history + tool exchanges
  1078→          const llmMessages: Array<{ role: string; content: string }> = [
  1079→            { role: 'assistant', content: systemPrompt },
  1080→            ...history.slice(-MAX_HISTORY).map((m) => ({ role: m.role, content: m.content })),
  1081→          ]
  1082→          // Include the new user message (history was fetched before insert)
  1083→          llmMessages.push({ role: 'user', content: effectiveMessage })
  1084→
  1085→          // SESSION DOCUMENT MEMORY: persist the attachment context as a
  1086→          // DATABASE chat message when first attached, so it survives
  1087→          // server restarts AND appears in history for follow-ups. For
  1088→          // follow-ups with no new attachment, restore from the DB history
  1089→          // (search for the attachment marker) or the in-memory cache.
  1090→          if (!attachment && session?.id) {
  1091→            // Try in-memory first (fast path)
  1092→            let stored = sessionDocs.get(session.id) ?? null
  1093→            if (!stored) {
  1094→              // Fall back to the DB: look for the attachment marker message
  1095→              const marker = history.find(
  1096→                (m) => m.role === 'user' && m.content.startsWith('[The user attached the document "')
  1097→              )
  1098→              if (marker) {
  1099→                const filenameMatch = marker.content.match(/"([^"]+)"/)
  1100→                const textMatch = marker.content.match(/DOCUMENT "[^"]+":\n([\s\S]*?)\n(You can answer|$)/)
  1101→                stored = {
  1102→                  title: filenameMatch?.[1] ?? 'document',
  1103→                  text: (textMatch?.[1] ?? marker.content).slice(0, 40000),
  1104→                  format: marker.content.includes('pdf_operation') ? 'pdf' : 'doc',
  1105→                  filename: filenameMatch?.[1] ?? 'document',
  1106→                  dataUrl: '', // not available from history — tools will ask to re-attach for binary ops
  1107→                }
  1108→                // Cache for future requests
  1109→                sessionDocs.set(session.id, stored)
  1110→              }
  1111→            }
  1112→            if (stored) {
  1113→              activeDoc = stored
  1114→              attachmentContextMessage =
  1115→                `[CONTEXT — the user attached the document "${stored.filename}" earlier in this conversation. Its full content remains available below.\n\n` +
  1116→                `DOCUMENT "${stored.title}":\n${stored.text.slice(0, 24000)}\n` +
  1117→                `Continue answering follow-up questions about this document from its content. The edit_document and ${stored.format === 'pdf' ? 'pdf_operation' : 'file'} tools can still operate on it.]`
  1118→            }

  1120→            // Remember the newly-attached document for the rest of the session
  1121→            sessionDocs.set(session.id, activeDoc)
  1122→            // ALSO persist as a DB message so it survives restarts
  1123→            try {
  1124→              await db.chatMessage.create({
  1125→                data: {
  1126→                  sessionId: session.id,
  1127→                  role: 'user',
  1128→                  content: attachmentContextMessage.slice(0, 30000),
  1129→                },
  1130→              })
  1131→            } catch {
  1132→              /* best-effort persistence */
  1133→            }
  1134→          }
  1135→

  1137→          // BEFORE the user's message so the AI sees it as context.
  1138→          if (attachmentContextMessage) {
  1139→            llmMessages.splice(llmMessages.length - 1, 0, {
  1140→              role: 'user',
  1141→              content: attachmentContextMessage,

  1149→                {

  1151→                  content:
  1152→                    'You are the reasoning engine of NEXUS AI. Given a question, produce a SHORT internal analysis (max 80 words, telegraphic) covering: what the user needs, which tools to use (if any), and the answer structure. Output ONLY the analysis.',
  1153→                },

  1169→

  1179→            // as tokens arrive. Primary engine: the FREE MULTI-AI POOL

  1315→                  session_id: session!.id,

  1317→                  content: finalAnswer,

  1321→                  user_id: verifiedUserId,

  1337→              }
  1338→              send({ type: 'done', sessionId: session!.id })

  1340→            }
  1341→          }
  1342→        } catch (error) {
  1343→          console.error('[api/chat] stream error:', error)
  1344→          send({
  1345→            type: 'error',
  1346→            message: error instanceof Error ? error.message : 'Chat failed.',
  1347→          })
  1348→        } finally {
  1349→          controller.close()
  1350→        }
  1351→      },
  1352→    })
  1353→
  1354→    return new NextResponse(stream, {
  1355→      status: 200,
  1356→      headers: {
  1357→        'Content-Type': 'application/x-ndjson; charset=utf-8',
  1358→        'Cache-Control': 'no-cache, no-transform',
  1359→        'X-Accel-Buffering': 'no',
  1360→      },
  1361→    })
  1362→  } catch (error) {
  1363→    console.error('[api/chat] POST error:', error)
  1364→    const message = error instanceof Error ? error.message : 'Chat failed.'
  1365→    return NextResponse.json({ error: message }, { status: 500 })
  1366→  }

   123→export function NexusChat(props: NexusChatProps) {
   124→  const { pinnedAgent, onPinnedAgentChange, onSessionCreated } = props
   125→
   126→  const [messages, setMessages] = useState<NxMsg[]>([])
   127→  const [streaming, setStreaming] = useState(false)
   128→  const [sessionId, setSessionId] = useState<string | undefined>(props.sessionId)
   129→  const [input, setInput] = useState(props.prefill ?? '')
   130→  const [attach, setAttach] = useState<{ filename: string; dataUrl: string } | null>(null)
   131→  const [attachError, setAttachError] = useState('')
   132→  const [assign, setAssign] = useState<AgentAssignEvent | null>(null)
   133→
   134→  const scrollRef = useRef<HTMLDivElement>(null)
   135→  const textareaRef = useRef<HTMLTextAreaElement>(null)
   136→  const fileRef = useRef<HTMLInputElement>(null)
   137→  const stickToBottomRef = useRef(true)
   138→  const streamIdRef = useRef<string | null>(null)
   139→  const pendingDeltaRef = useRef('')
   140→  const flushTimerRef = useRef<ReturnType<typeof setInterval> | null>(null)
   141→  const abortRef = useRef<AbortController | null>(null)
   142→  const assignRef = useRef<AgentAssignEvent | null>(null)
   143→  const seqRef = useRef(0)
   144→
   145→  const nextId = (prefix: string) => `${prefix}-${Date.now()}-${seqRef.current++}`
   146→
   147→  /* ---------- session resume + one-time pin sync ---------- */
   148→  // The session to resume is captured at MOUNT: page.tsx remounts the chat
   149→  // (chatEpoch key) whenever the user switches sessions, and later
   150→  // props.sessionId changes (e.g. when THIS chat creates its own session)
   151→  // must NOT re-trigger a refetch that would clobber the live transcript
   152→  // (handoff pills, tool cards and attachments are stream-only state).
   153→  const [resumeId] = useState(props.sessionId)
   154→  useEffect(() => {
   155→    if (!resumeId) return
   156→    let cancelled = false
   157→    ;(async () => {
   158→      try {
   159→        const res = await fetch(`/api/chat/sessions/${encodeURIComponent(resumeId)}`)
   160→        if (!res.ok) return
   161→        const data = (await res.json().catch(() => null)) as { session?: unknown } | null
   162→        const s =
   163→          data && typeof data.session === 'object' ? (data.session as Record<string, unknown>) : null
   164→        if (cancelled || !s) return
   165→        const rowsRaw = Array.isArray(s.messages) ? s.messages : []
   166→        const sessionAgentSlug = typeof s.agentSlug === 'string' ? s.agentSlug : null
   167→        const rows: NxMsg[] = rowsRaw
   168→          .filter((m): m is Record<string, unknown> => !!m && typeof m === 'object')
   169→          .map((m, i): NxMsg => {
   170→            const role = m.role === 'user' ? 'user' : m.role === 'tool' ? 'tool' : 'assistant'
   171→            return {
   172→              id: typeof m.id === 'string' ? m.id : `m-${i}`,
   173→              role,
   174→              content: typeof m.content === 'string' ? m.content : '',
   175→              toolName: typeof m.toolName === 'string' ? m.toolName : null,
   176→              toolData: typeof m.toolData === 'string' ? m.toolData : null,
   177→              toolStatus: role === 'tool' ? 'done' : undefined,
   178→              agentSlug: role === 'assistant' ? sessionAgentSlug : undefined,
   179→              // NEXUS One: attachments persisted server-side (image/video/
   180→              // document/code cards) survive resume.
   181→              attachments: Array.isArray(m.attachments) ? m.attachments : undefined,
   182→            }
   183→          })
   184→        setMessages(rows)
   185→        setSessionId(typeof s.id === 'string' ? s.id : resumeId)
   186→        // Sync page-level pin state ONCE after the session loads.
   187→        onPinnedAgentChange(s.agentPinned === true && sessionAgentSlug ? sessionAgentSlug : null)
   188→      } catch {
   189→        /* failed to load — fall back to an empty transcript */
   190→      }
   191→    })()
   192→    return () => {
   193→      cancelled = true
   194→    }
   195→    // Mount-per-session resume (page.tsx remounts via chatEpoch); the pin
   196→    // sync must fire exactly once per loaded session.
   197→  }, [resumeId])
   198→
   199→  /* ---------- abort any live stream on unmount ---------- */
   200→  useEffect(() => {
   201→    return () => {
   202→      abortRef.current?.abort()
   203→      if (flushTimerRef.current) clearInterval(flushTimerRef.current)
   204→    }
   205→  }, [])
   206→
   207→  /* ---------- smart auto-scroll (sticks while near the bottom) ---------- */
   208→  useEffect(() => {
   209→    const el = scrollRef.current
   210→    if (!el || !stickToBottomRef.current) return
   211→    el.scrollTop = el.scrollHeight
   212→  }, [messages, streaming])
   213→
   214→  const onScroll = () => {
   215→    const el = scrollRef.current
   216→    if (!el) return
   217→    const distFromBottom = el.scrollHeight - el.scrollTop - el.clientHeight
   218→    stickToBottomRef.current = distFromBottom <= 150
   219→  }
   220→
   221→  /* ---------- textarea auto-resize (rows 1 → 6) ---------- */
   222→  useEffect(() => {
   223→    const el = textareaRef.current
   224→    if (!el) return
   225→    el.style.height = 'auto'
   226→    el.style.height = `${Math.min(el.scrollHeight, 144)}px`
   227→  }, [input])
   228→
   229→  /* ---------- streaming helpers ---------- */
   230→
   231→  const appendDelta = useCallback((delta: string) => {
   232→    const id = streamIdRef.current
   233→    if (!id || !delta) return
   234→    setMessages((prev) => prev.map((m) => (m.id === id ? { ...m, content: m.content + delta } : m)))
   235→  }, [])
   236→
   237→  /** Deltas are buffered and flushed every 80ms so react-markdown is not
   238→   * re-parsed on every single token — keeps long streams silky. */
   239→  const startDeltaFlusher = useCallback(() => {
   240→    if (flushTimerRef.current) return
   241→    flushTimerRef.current = setInterval(() => {
   242→      if (!pendingDeltaRef.current) return
   243→      const chunk = pendingDeltaRef.current
   244→      pendingDeltaRef.current = ''
   245→      appendDelta(chunk)
   246→    }, 80)
   247→  }, [appendDelta])
   248→
   249→  const stopDeltaFlusher = useCallback(() => {
   250→    if (flushTimerRef.current) {
   251→      clearInterval(flushTimerRef.current)
   252→      flushTimerRef.current = null
   253→    }
   254→    if (pendingDeltaRef.current) {
   255→      const chunk = pendingDeltaRef.current
   256→      pendingDeltaRef.current = ''
   257→      appendDelta(chunk)
   258→    }
   259→  }, [appendDelta])
   260→
   261→  /** Patch the live tool chip that matches (toolName, index). */
   262→  const updateTool = useCallback(
   263→    (tool: string, index: number, patch: Partial<NxMsg>) => {
   264→      setMessages((prev) => {
   265→        let target = -1
   266→        for (let i = prev.length - 1; i >= 0; i--) {
   267→          const m = prev[i]
   268→          if (m.role === 'tool' && m.toolName === tool && m.toolIndex === index) {
   269→            target = i
   270→            break
   271→          }
   272→        }
   273→        if (target === -1) {
   274→          for (let i = prev.length - 1; i >= 0; i--) {
   275→            const m = prev[i]
   276→            if (m.role === 'tool' && m.toolName === tool && m.toolStatus === 'running') {
   277→              target = i
   278→              break
   279→            }
   280→          }
   281→        }
   282→        if (target === -1) return prev
   283→        const copy = [...prev]
   284→        copy[target] = { ...copy[target], ...patch }
   285→        return copy
   286→      })
   287→    },
   288→    []
   289→  )
   290→
   291→  /* ---------- send ---------- */
   292→
   293→  const send = useCallback(
   294→    async (text: string) => {
   295→      const trimmed = text.trim()
   296→      const hasAttach = attach != null
   297→      if (streaming) return
   298→      if (!trimmed && !hasAttach) return
   299→
   300→      const userMsg: NxMsg = {
   301→        id: nextId('local'),
   302→        role: 'user',
   303→        content: trimmed,
   304→        ...(hasAttach ? { attachName: attach!.filename } : {}),
   305→      }
   306→      setMessages((prev) => [...prev, userMsg])
   307→      setInput('')
   308→      setAttach(null)
   309→      setAttachError('')
   310→      setStreaming(true)
   311→      stickToBottomRef.current = true
   312→
   313→      abortRef.current = new AbortController()
   314→      try {
   315→        const res = await fetch('/api/chat', {
   316→          method: 'POST',
   317→          headers: { 'Content-Type': 'application/json' },
   318→          body: JSON.stringify({
   319→            message: trimmed,
   320→            ...(sessionId ? { sessionId } : {}),
   321→            // '' = auto-routing; a slug pins the session persona (idempotent).
   322→            agentSlug: pinnedAgent ?? '',
   323→            ...(hasAttach
   324→              ? { attachment: { filename: attach!.filename, dataUrl: attach!.dataUrl } }
   325→              : {}),
   326→          }),
   327→          signal: abortRef.current.signal,
   328→        })
   329→
   330→        if (!res.ok || !res.body) {
   331→          const data = (await res.json().catch(() => null)) as { error?: string } | null
   332→          throw new Error(data?.error || 'Chat request failed.')
   333→        }
   334→
   335→        // Consume the NDJSON stream.
   336→        const reader = res.body.getReader()
   337→        const decoder = new TextDecoder()
   338→        let buffer = ''
   339→
   340→        for (;;) {
   341→          const { done, value } = await reader.read()
   342→          if (done) break
   343→          buffer += decoder.decode(value, { stream: true })
   344→          const lines = buffer.split('\n')
   345→          buffer = lines.pop() ?? ''
   346→          for (const line of lines) {
   347→            const text = line.trim()
   348→            if (!text) continue
   349→            let event: ChatStreamEvent
   350→            try {
   351→              event = JSON.parse(text) as ChatStreamEvent
   352→            } catch {
   353→              continue // partial / malformed line — skip
   354→            }
   355→
   356→            switch (event.type) {
   357→              case 'user': {
   358→                setMessages((prev) =>
   359→                  prev.map((m) =>
   360→                    m.id === userMsg.id ? { ...m, id: event.id, content: event.content } : m
   361→                  )
   362→                )
   363→                break
   364→              }
   365→              case 'agent_assign': {
   366→                assignRef.current = event
   367→                setAssign(event)
   368→                setMessages((prev) => [
   369→                  ...prev,
   370→                  { id: nextId('assign'), role: 'agent', content: '', assign: event },
   371→                ])
   372→                break
   373→              }
   374→              case 'assistant_start': {
   375→                streamIdRef.current = event.id
   376→                const slug = pinnedAgent ?? assignRef.current?.agentSlug ?? null
   377→                setMessages((prev) => [
   378→                  ...prev,
   379→                  { id: event.id, role: 'assistant', content: '', streaming: true, agentSlug: slug },
   380→                ])
   381→                break
   382→              }
   383→              case 'assistant_delta': {
   384→                pendingDeltaRef.current += event.delta
   385→                startDeltaFlusher()
   386→                break
   387→              }
   388→              case 'assistant_end': {
   389→                stopDeltaFlusher()
   390→                const id = streamIdRef.current
   391→                streamIdRef.current = null
   392→                if (id) {
   393→                  const attachments = Array.isArray(event.attachments) ? event.attachments : []
   394→                  setMessages((prev) =>
   395→                    prev.map((m) =>
   396→                      m.id === id
   397→                        ? {
   398→                            ...m,
   399→                            streaming: false,
   400→                            ...(attachments.length > 0 ? { attachments } : {}),
   401→                          }
   402→                        : m
   403→                    )
   404→                  )
   405→                }
   406→                break
   407→              }
   408→              case 'assistant': {
   409→                // Full-message fallback (non-streaming providers).
   410→                stopDeltaFlusher()
   411→                const id = streamIdRef.current
   412→                streamIdRef.current = null
   413→                const attachments = Array.isArray(event.attachments) ? event.attachments : []
   414→                if (id) {
   415→                  setMessages((prev) =>
   416→                    prev.map((m) =>
   417→                      m.id === id
   418→                        ? {
   419→                            ...m,
   420→                            content: event.content,
   421→                            streaming: false,
   422→                            ...(attachments.length > 0 ? { attachments } : {}),
   423→                          }
   424→                        : m
   425→                    )
   426→                  )
   427→                } else {
   428→                  setMessages((prev) => [
   429→                    ...prev,
   430→                    {
   431→                      id: nextId('a'),
   432→                      role: 'assistant',
   433→                      content: event.content,
   434→                      agentSlug: pinnedAgent ?? assignRef.current?.agentSlug ?? null,
   435→                      ...(attachments.length > 0 ? { attachments } : {}),
   436→                    },
   437→                  ])
   438→                }
   439→                break
   440→              }
   441→              case 'tool_start': {
   442→                setMessages((prev) => [
   443→                  ...prev,
   444→                  {
   445→                    id: nextId(`tool-${event.index}`),
   446→                    role: 'tool',
   447→                    content: '',
   448→                    toolName: event.tool,
   449→                    toolIndex: event.index,
   450→                    toolStatus: 'running',
   451→                    toolArgs: event.args,
   452→                  },
   453→                ])
   454→                break
   455→              }
   456→              case 'tool_progress': {
   457→                updateTool(event.tool, event.index, { toolMessage: event.message })
   458→                break
   459→              }
   460→              case 'tool_result': {
   461→                let data: string | null = null
   462→                try {
   463→                  data = JSON.stringify(event.result)
   464→                } catch {
   465→                  data = null
   466→                }
   467→                updateTool(event.tool, event.index, {
   468→                  toolStatus: event.ok ? 'done' : 'error',
   469→                  toolData: data,
   470→                })
   471→                break
   472→              }
   473→              case 'done': {
   474→                setSessionId(event.sessionId)
   475→                onSessionCreated(event.sessionId)
   476→                break
   477→              }
   478→              case 'error': {
   479→                throw new Error(event.message)
   480→              }
   481→            }
   482→          }
   483→        }
   484→      } catch (error) {
   485→        if (error instanceof DOMException && error.name === 'AbortError') return
   486→        setMessages((prev) => [
   487→          ...prev,
   488→          {
   489→            id: nextId('err'),
   490→            role: 'assistant',
   491→            content:
   492→              error instanceof Error
   493→                ? `Something went wrong: ${error.message}`
   494→                : 'Something went wrong. Please try again.',
   495→            isError: true,
   496→            agentSlug: pinnedAgent ?? assignRef.current?.agentSlug ?? null,
   497→          },
   498→        ])
   499→      } finally {
   500→        stopDeltaFlusher()
   501→        streamIdRef.current = null
   502→        setStreaming(false)
   503→        setMessages((prev) =>
   504→          prev.map((m) =>
   505→            m.streaming ? { ...m, streaming: false } : m.toolStatus === 'running' ? { ...m, toolStatus: 'done' } : m
   506→          )
   507→        )
   508→      }
   509→    },
   510→    [attach, sessionId, streaming, pinnedAgent, startDeltaFlusher, stopDeltaFlusher, updateTool, onSessionCreated]
   511→  )
   512→
   513→  /* ---------- attachments (composer) ---------- */
   514→
   515→  const onPickFile = (e: React.ChangeEvent<HTMLInputElement>) => {
   516→    const file = e.target.files?.[0]
   517→    e.target.value = '' // allow re-picking the same file
   518→    if (!file) return
   519→    if (file.size > 14 * 1024 * 1024) {
   520→      setAttachError('That file is too large (max 14 MB).')
   521→      return
   522→    }
   523→    const reader = new FileReader()
   524→    reader.onload = () => {
   525→      const dataUrl = typeof reader.result === 'string' ? reader.result : ''
   526→      if (!dataUrl) {
   527→        setAttachError('Could not read that file.')
   528→        return
   529→      }
   530→      setAttach({ filename: file.name, dataUrl })
   531→      setAttachError('')
   532→    }
   533→    reader.onerror = () => setAttachError('Could not read that file.')
   534→    reader.readAsDataURL(file)
   535→  }
   536→
   537→  /* ---------- keyboard ---------- */
   538→
   539→  const onKeyDown = (e: React.KeyboardEvent<HTMLTextAreaElement>) => {
   540→    if (e.key === 'Enter' && !e.shiftKey && !e.nativeEvent.isComposing) {
   541→      e.preventDefault()
   542→      void send(input)
   543→    }
   544→  }
   545→
   546→  const stop = () => abortRef.current?.abort()
   547→
   548→  /* ---------- derived ---------- */
   549→
   550→  const headerAgent = agentOrNexus(pinnedAgent ?? assign?.agentSlug)
   551→  const headerDivision = DIVISION_MAP[headerAgent.division]
   552→  const isPinned = pinnedAgent != null
   553→  const showWelcome = messages.length === 0 && !streaming
   554→  const canSend = (input.trim().length > 0 || attach != null) && !streaming
   555→  const placeholder = `Message ${headerAgent.name}…`
   556→
   557→  const applyCapability = (c: Capability) => {
   558→    if (c.voice) {
   559→      props.onOpenVoice()
   560→      return
   561→    }
   562→    if (c.whatsapp) {
   563→      props.onOpenWhatsApp()
   564→      return
   565→    }
   566→    if (c.prompt) {
   567→      setInput(c.prompt)
   568→      textareaRef.current?.focus()
   569→    }
   570→  }
   571→
   572→  /* Index of the newest handoff pill — it shimmers while its reply streams. */
   573→  let lastAssignIdx = -1
   574→  for (let i = messages.length - 1; i >= 0; i--) {
   575→    if (messages[i].role === 'agent') {
   576→      lastAssignIdx = i
   577→      break
   578→    }
   579→  }
   580→
   581→  return (
   582→    <div className="flex h-[calc(100dvh-56px)] flex-col md:h-screen">
   583→      {/* ---------- Header: current agent identity ---------- */}
   584→      <header className="flex h-14 shrink-0 items-center gap-3 border-b border-white/8 px-3 sm:px-4">
   585→        <div className="flex min-w-0 flex-1 items-center gap-2.5">
   586→          {isPinned ? (
   587→            <>
   588→              <AgentAvatar agent={headerAgent} size={34} />
   589→              <div className="min-w-0">
   590→                <p className="flex items-center gap-1.5 text-sm font-semibold text-zinc-100">
   591→                  <span className="truncate">{headerAgent.name}</span>
   592→                  <span className="flex shrink-0 items-center gap-1 rounded-full bg-white/8 px-1.5 py-0.5 text-[10px] font-medium text-zinc-400">
   593→                    <Pin className="h-2.5 w-2.5" aria-hidden />
   594→                    pinned
   595→                  </span>

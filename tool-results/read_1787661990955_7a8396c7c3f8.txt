     1→'use client'
     2→
     3→/**
     4→ * NEXUS One — full-screen premium voice conversation overlay.
     5→ *
     6→ * State machine: idle → listening → thinking → speaking → listening …
     7→ *  - INPUT : Web Speech API (live interim transcript, auto-restart on end)
     8→ *  - THINK : POST /api/voice/turn  { message, history, language, voice, audio }
     9→ *  - SPEAK : inline audio from the turn response, falling back to
    10→ *            POST /api/tts { text, voice } → object URL → <audio>, then
    11→ *            browser speechSynthesis as a last resort.
    12→ *
    13→ * The orb is the BrandMark wrapped in layered gradient glow rings whose
    14→ * motion maps to the current state (listening = expanding rings that react
    15→ * to mic level, thinking = slow rotating gradient ring + shimmer,
    16→ * speaking = pulse rings locked to the .nx-dot bounce rhythm).
    17→ */
    18→
    19→import { useCallback, useEffect, useRef, useState } from 'react'
    20→import { AnimatePresence, motion } from 'framer-motion'
    21→import { Globe, Mic, MicOff, PhoneOff, Send, X } from 'lucide-react'
    22→import { BrandMark } from './shared'
    23→
    24→type VoiceState = 'idle' | 'listening' | 'thinking' | 'speaking'
    25→
    26→interface Turn {
    27→  id: string
    28→  user: string
    29→  reply: string
    30→}
    31→
    32→const LANGUAGES = [
    33→  { code: 'en-US', label: 'English', voice: 'en-US-AriaNeural' },
    34→  { code: 'ar-AE', label: 'العربية', voice: 'ar-AE-FatimaNeural' },
    35→  { code: 'ar-SA', label: 'العربية (SA)', voice: 'ar-SA-ZariyahNeural' },
    36→  { code: 'fr-FR', label: 'Français', voice: 'fr-FR-DeniseNeural' },
    37→  { code: 'es-ES', label: 'Español', voice: 'es-ES-ElviraNeural' },
    38→  { code: 'de-DE', label: 'Deutsch', voice: 'de-DE-KatjaNeural' },
    39→  { code: 'hi-IN', label: 'हिन्दी', voice: 'hi-IN-SwaraNeural' },
    40→  { code: 'ja-JP', label: '日本語', voice: 'ja-JP-NanamiNeural' },
    41→  { code: 'zh-CN', label: '中文', voice: 'zh-CN-XiaoxiaoNeural' },
    42→  { code: 'pt-BR', label: 'Português', voice: 'pt-BR-FranciscaNeural' },
    43→  { code: 'ru-RU', label: 'Русский', voice: 'ru-RU-SvetlanaNeural' },
    44→  { code: 'tr-TR', label: 'Türkçe', voice: 'tr-TR-EmelNeural' },
    45→]
    46→
    47→/** Tiny silent WAV so the /api/voice/turn zod schema passes on the text path. */
    48→const SILENT_WAV = 'UklGRiQAAABXQVZFZm10IBAAAAABAAEAQB8AAEAfAAABAAgAZGF0YQAAAAA='
    49→
    50→/** The last N turns rendered in the compact conversation log. */
    51→const VISIBLE_TURNS = 4
    52→
    53→export function VoiceOverlay({ open, onOpenChange }: { open: boolean; onOpenChange: (o: boolean) => void }) {
    54→  const [state, setState] = useState<VoiceState>('idle')
    55→  const [supported, setSupported] = useState(true)
    56→  const [muted, setMuted] = useState(false)
    57→  const [lang, setLang] = useState('en-US')
    58→  const [ttsVoice, setTtsVoice] = useState('en-US-AriaNeural')
    59→  const [interim, setInterim] = useState('')
    60→  const [micLevel, setMicLevel] = useState(0)
    61→  const [turns, setTurns] = useState<Turn[]>([])
    62→  const [error, setError] = useState('')
    63→  const [showLangs, setShowLangs] = useState(false)
    64→  const [textInput, setTextInput] = useState('')
    65→
    66→  const recognitionRef = useRef<any>(null)
    67→  const historyRef = useRef<Array<{ role: 'user' | 'assistant'; content: string }>>([])
    68→  const audioRef = useRef<HTMLAudioElement | null>(null)
    69→  const objectUrlRef = useRef<string | null>(null)
    70→  const stateRef = useRef<VoiceState>('idle')
    71→  const mutedRef = useRef(false)
    72→  const startListeningRef = useRef<((langOverride?: string) => void) | null>(null)
    73→
    74→  const setStateSafe = useCallback((s: VoiceState) => {
    75→    stateRef.current = s
    76→    setState(s)
    77→  }, [])
    78→
    79→  /* --------------------------------------------------------------- */
    80→  /* Audio plumbing                                                   */
    81→  /* --------------------------------------------------------------- */
    82→
    83→  const stopSpeaking = useCallback(() => {
    84→    const el = audioRef.current
    85→    if (el) {
    86→      try {
    87→        el.pause()
    88→        el.removeAttribute('src')
    89→        el.load()
    90→      } catch {
    91→        /* already stopped */
    92→      }
    93→    }
    94→    if (objectUrlRef.current) {
    95→      URL.revokeObjectURL(objectUrlRef.current)
    96→      objectUrlRef.current = null
    97→    }
    98→  }, [])
    99→
   100→  /** Plays an object URL through the shared <audio> element. Resolves on end. */
   101→  const playUrl = useCallback((url: string): Promise<void> => {
   102→    return new Promise<void>((resolve) => {
   103→      const el = audioRef.current
   104→      if (!el) return resolve()
   105→      el.src = url
   106→      el.onended = () => resolve()
   107→      el.onerror = () => resolve()
   108→      el.play().catch(() => resolve())
   109→    })
   110→  }, [])
   111→
   112→  /**
   113→   * Speaks a reply. Prefers the inline audio returned by /api/voice/turn
   114→   * (saves a round-trip), then POST /api/tts, then speechSynthesis.
   115→   */
   116→  const speak = useCallback(
   117→    async (text: string, inlineAudio?: string | null, audioFormat?: string): Promise<void> => {
   118→      setStateSafe('speaking')
   119→      let url: string | null = null
   120→      try {
   121→        if (inlineAudio) {
   122→          const binary = atob(inlineAudio)
   123→          const bytes = new Uint8Array(binary.length)
   124→          for (let i = 0; i < binary.length; i++) bytes[i] = binary.charCodeAt(i)
   125→          const blob = new Blob([bytes], { type: audioFormat === 'mp3' ? 'audio/mpeg' : 'audio/wav' })
   126→          url = URL.createObjectURL(blob)
   127→        } else {
   128→          const res = await fetch('/api/tts', {
   129→            method: 'POST',
   130→            headers: { 'Content-Type': 'application/json' },
   131→            body: JSON.stringify({ text, voice: ttsVoice, speed: 1.0 }),
   132→          })
   133→          if (!res.ok) throw new Error('TTS failed')
   134→          const blob = await res.blob()
   135→          url = URL.createObjectURL(blob)
   136→        }
   137→        if (objectUrlRef.current) URL.revokeObjectURL(objectUrlRef.current)
   138→        objectUrlRef.current = url
   139→        await playUrl(url)
   140→      } catch {
   141→        // Last resort: the browser's built-in synthesizer.
   142→        try {
   143→          await new Promise<void>((resolve) => {
   144→            const utter = new SpeechSynthesisUtterance(text)
   145→            utter.lang = lang
   146→            utter.onend = () => resolve()
   147→            utter.onerror = () => resolve()
   148→            window.speechSynthesis.speak(utter)
   149→            setTimeout(resolve, Math.min(30000, text.length * 100))
   150→          })
   151→        } catch {
   152→          /* continue silently — the text is still in the log */
   153→        }
   154→      }
   155→    },
   156→    [ttsVoice, lang, playUrl, setStateSafe]
   157→  )
   158→
   159→  /* --------------------------------------------------------------- */
   160→  /* One conversation turn: think (LLM) → speak (TTS)                  */
   161→  /* --------------------------------------------------------------- */
   162→
   163→  const think = useCallback(
   164→    async (userText: string): Promise<{ reply?: string; audio?: string | null; audioFormat?: string }> => {
   165→      const res = await fetch('/api/voice/turn', {
   166→        method: 'POST',
   167→        headers: { 'Content-Type': 'application/json' },
   168→        body: JSON.stringify({
   169→          message: userText,
   170→          history: historyRef.current.slice(-6),
   171→          language: lang.split('-')[0],
   172→          voice: ttsVoice,
   173→          audio: SILENT_WAV,
   174→        }),
   175→      })
   176→      if (!res.ok) throw new Error('AI failed to respond — try again.')
   177→      return await res.json()
   178→    },
   179→    [lang, ttsVoice]
   180→  )
   181→
   182→  const handleUtterance = useCallback(
   183→    async (text: string) => {
   184→      const clean = text.trim()
   185→      if (!clean) return
   186→      stopRecognitionRef.current?.()
   187→      setStateSafe('thinking')
   188→      setInterim('')
   189→      setError('')
   190→      try {
   191→        historyRef.current.push({ role: 'user', content: clean })
   192→        const data = await think(clean)
   193→        const reply = data.reply?.trim() || 'Sorry, I did not get that.'
   194→        historyRef.current.push({ role: 'assistant', content: reply })
   195→        setTurns((prev) => [...prev.slice(-5), { id: crypto.randomUUID(), user: clean, reply }])
   196→        await speak(reply, data.audio, data.audioFormat)
   197→      } catch (e) {
   198→        setError(e instanceof Error ? e.message : 'Something went wrong — try again.')
   199→      } finally {
   200→        // Auto-reset to listening for the next turn.
   201→        startListeningRef.current?.()
   202→      }
   203→    },
   204→    [think, speak, setStateSafe]
   205→  )
   206→
   207→  /* --------------------------------------------------------------- */
   208→  /* Web Speech API input (ported from the proven omni voice mode)    */
   209→  /* --------------------------------------------------------------- */
   210→
   211→  const stopRecognition = useCallback(() => {
   212→    if (recognitionRef.current) {
   213→      try {
   214→        const pulse = recognitionRef.current.__pulse as ReturnType<typeof setInterval> | undefined
   215→        if (pulse) clearInterval(pulse)
   216→        recognitionRef.current.onend = null
   217→        recognitionRef.current.onresult = null
   218→        recognitionRef.current.onerror = null
   219→        recognitionRef.current.stop()
   220→      } catch {
   221→        /* already stopped */
   222→      }
   223→      recognitionRef.current = null
   224→      setMicLevel(0)
   225→    }
   226→  }, [])
   227→
   228→  const stopRecognitionRef = useRef(stopRecognition)
   229→  useEffect(() => {
   230→    stopRecognitionRef.current = stopRecognition
   231→  }, [stopRecognition])
   232→
   233→  const startListening = useCallback(
   234→    (langOverride?: string) => {
   235→      const SR = (window as any).SpeechRecognition || (window as any).webkitSpeechRecognition
   236→      if (!SR) {
   237→        setSupported(false)
   238→        setStateSafe('idle')
   239→        return
   240→      }
   241→      if (mutedRef.current) return
   242→      stopSpeaking()
   243→
   244→      const rec = new SR()
   245→      rec.lang = langOverride ?? lang
   246→      rec.continuous = true
   247→      rec.interimResults = true
   248→
   249→      rec.onresult = (event: any) => {
   250→        let finalTranscript = ''
   251→        let interimText = ''
   252→        for (let i = event.resultIndex; i < event.results.length; i++) {
   253→          const r = event.results[i]
   254→          if (r.isFinal) finalTranscript += r[0].transcript
   255→          else interimText += r[0].transcript
   256→        }
   257→        if (interimText) {
   258→          setInterim(interimText)
   259→          setMicLevel(0.3 + Math.random() * 0.4) // active-speech visual
   260→        }
   261→        if (finalTranscript.trim()) {
   262→          handleUtterance(finalTranscript)
   263→        }
   264→      }
   265→
   266→      rec.onerror = (event: any) => {
   267→        if (event.error === 'not-allowed' || event.error === 'permission-denied') {
   268→          setError('Microphone blocked — allow mic access for this site, then reopen voice mode.')
   269→          setStateSafe('idle')
   270→        } else if (event.error === 'audio-capture' || event.error === 'service-not-allowed' || event.error === 'network') {
   271→          setError('Could not reach the speech service — check your connection.')
   272→          setStateSafe('idle')
   273→        }
   274→        // 'no-speech' and 'aborted' are normal — keep listening.
   275→      }
   276→
   277→      rec.onend = () => {
   278→        // Chrome stops after silence — restart while still in listening state.
   279→        if (stateRef.current === 'listening' && recognitionRef.current && !mutedRef.current) {
   280→          try {
   281→            recognitionRef.current.start()
   282→          } catch {
   283→            /* restart race — ignore */
   284→          }
   285→        }
   286→      }
   287→
   288→      recognitionRef.current = rec
   289→      try {
   290→        rec.start()
   291→        setStateSafe('listening')
   292→        setError('')
   293→        const pulse = setInterval(() => {
   294→          setMicLevel((v) => Math.max(0.05, v * 0.85))
   295→        }, 150)
   296→        rec.__pulse = pulse
   297→      } catch {
   298→        /* already started */
   299→      }
   300→    },
   301→    [lang, handleUtterance, stopSpeaking, setStateSafe]
   302→  )
   303→
   304→  useEffect(() => {
   305→    startListeningRef.current = startListening
   306→  }, [startListening])
   307→
   308→  /* --------------------------------------------------------------- */
   309→  /* Overlay lifecycle                                                */
   310→  /* --------------------------------------------------------------- */
   311→
   312→  // Fresh conversation every time the overlay opens; teardown on close.
   313→  useEffect(() => {
   314→    if (!open) return
   315→    const t = setTimeout(() => {
   316→      setSupported(true)
   317→      setMuted(false)
   318→      mutedRef.current = false
   319→      setTurns([])
   320→      setInterim('')
   321→      setError('')
   322→      setTextInput('')
   323→      setShowLangs(false)
   324→      historyRef.current = []
   325→      setStateSafe('idle')
   326→      startListeningRef.current?.()
   327→    }, 60)
   328→    return () => {
   329→      clearTimeout(t)
   330→      stopRecognition()
   331→      stopSpeaking()
   332→    }
   333→  }, [open, setStateSafe, stopRecognition, stopSpeaking])
   334→
   335→  // Escape closes the overlay.
   336→  useEffect(() => {
   337→    if (!open) return
   338→    const onKey = (e: KeyboardEvent) => {
   339→      if (e.key === 'Escape') onOpenChange(false)
   340→    }
   341→    window.addEventListener('keydown', onKey)
   342→    return () => window.removeEventListener('keydown', onKey)
   343→  }, [open, onOpenChange])
   344→
   345→  /* --------------------------------------------------------------- */
   346→  /* Controls                                                         */
   347→  /* --------------------------------------------------------------- */
   348→
   349→  const toggleMute = () => {
   350→    if (muted) {
   351→      setMuted(false)
   352→      mutedRef.current = false
   353→      setError('')
   354→      startListening()
   355→    } else {
   356→      setMuted(true)
   357→      mutedRef.current = true
   358→      stopRecognition()
   359→      setInterim('')
   360→      setStateSafe('idle')
   361→    }
   362→  }
   363→
   364→  const pickLanguage = (l: (typeof LANGUAGES)[number]) => {
   365→    setLang(l.code)
   366→    setTtsVoice(l.voice)
   367→    setShowLangs(false)
   368→    // Restart recognition in the new language when mid-conversation.
   369→    if (stateRef.current === 'listening') {
   370→      stopRecognition()
   371→      setTimeout(() => startListeningRef.current?.(l.code), 150)
   372→    }
   373→  }
   374→
   375→  const sendText = () => {
   376→    const text = textInput.trim()
   377→    if (!text || stateRef.current === 'thinking' || stateRef.current === 'speaking') return
   378→    setTextInput('')
   379→    void handleUtterance(text)
   380→  }
   381→
   382→  /* --------------------------------------------------------------- */
   383→  /* Render                                                           */
   384→  /* --------------------------------------------------------------- */
   385→
   386→  if (!open) return null
   387→
   388→  const statusLabel =
   389→    state === 'listening'
   390→      ? 'Listening…'
   391→      : state === 'thinking'
   392→        ? 'Thinking…'
   393→        : state === 'speaking'
   394→          ? 'Speaking…'
   395→          : muted
   396→            ? 'Mic muted'
   397→            : supported
   398→              ? 'Starting…'
   399→              : 'Text mode'
   400→
   401→  const recentTurns = turns.slice(-VISIBLE_TURNS)
   402→
   403→  return (
   404→    <div
   405→      role="dialog"
   406→      aria-modal="true"
   407→      aria-label="NEXUS voice mode"
   408→      className="fixed inset-0 z-50 flex flex-col text-zinc-100"
   409→      style={{
   410→        background:
   411→          'linear-gradient(168deg, rgba(9,9,11,0.97) 0%, rgba(20,8,12,0.96) 55%, rgba(9,9,11,0.98) 100%)',
   412→        backdropFilter: 'blur(24px)',
   413→        WebkitBackdropFilter: 'blur(24px)',
   414→      }}
   415→    >
   416→      {/* Top bar */}
   417→      <div className="flex items-center justify-between px-5 pt-5 sm:px-8">
   418→        <div className="flex items-center gap-2.5">
   419→          <BrandMark size={22} />
   420→          <span className="text-xs font-semibold uppercase tracking-[0.2em] text-zinc-500">Voice</span>
   421→        </div>
   422→        <button
   423→          onClick={() => onOpenChange(false)}
   424→          aria-label="Close voice mode"
   425→          className="rounded-full p-2.5 text-zinc-400 transition hover:bg-white/10 hover:text-zinc-100"
   426→        >
   427→          <X className="h-5 w-5" />
   428→        </button>
   429→      </div>
   430→
   431→      {/* Center stage */}
   432→      <div className="flex min-h-0 flex-1 flex-col items-center justify-center px-6 py-4">
   433→        {/* ---- The orb ---- */}
   434→        <div className="relative flex items-center justify-center">
   435→          {/* ambient brand glow */}
   436→          <div
   437→            aria-hidden
   438→            className="absolute h-[300px] w-[300px] rounded-full"
   439→            style={{
   440→              background:
   441→                'radial-gradient(closest-side, rgba(255,90,95,0.20), rgba(245,166,35,0.08) 55%, transparent 75%)',
   442→              filter: 'blur(30px)',
   443→            }}
   444→          />
   445→
   446→          {/* listening — expanding rings that breathe with the mic */}
   447→          {state === 'listening' &&
   448→            [0, 1, 2].map((i) => (
   449→              <motion.span
   450→                key={`ring-l-${i}`}
   451→                aria-hidden
   452→                className="absolute rounded-full border"
   453→                style={{ borderColor: 'rgba(255,90,95,0.45)', width: 192, height: 192 }}
   454→                initial={{ scale: 1, opacity: 0.5 }}
   455→                animate={{ scale: 1.55, opacity: 0 }}
   456→                transition={{ duration: 1.9, repeat: Infinity, ease: 'easeOut', delay: i * 0.63 }}
   457→              />
   458→            ))}
   459→
   460→          {/* speaking — faster pulse locked to the nx-dot bounce rhythm */}
   461→          {state === 'speaking' &&
   462→            [0, 1].map((i) => (
   463→              <motion.span
   464→                key={`ring-s-${i}`}
   465→                aria-hidden
   466→                className="absolute rounded-full border"
   467→                style={{ borderColor: 'rgba(245,166,35,0.5)', width: 192, height: 192 }}
   468→                initial={{ scale: 1, opacity: 0.55 }}
   469→                animate={{ scale: 1.38, opacity: 0 }}
   470→                transition={{ duration: 1.2, repeat: Infinity, ease: 'easeOut', delay: i * 0.6 }}
   471→              />
   472→            ))}
   473→
   474→          {/* thinking — slow rotating gradient ring */}
   475→          {state === 'thinking' && (
   476→            <div
   477→              aria-hidden
   478→              className="absolute animate-spin rounded-full"
   479→              style={{
   480→                width: 208,
   481→                height: 208,
   482→                animationDuration: '5.5s',
   483→                background:
   484→                  'conic-gradient(from 0deg, rgba(245,166,35,0) 0deg, rgba(245,166,35,0.65) 80deg, rgba(255,90,95,0.8) 180deg, rgba(255,42,104,0.65) 280deg, rgba(255,42,104,0) 360deg)',
   485→                WebkitMask: 'radial-gradient(farthest-side, transparent calc(100% - 4px), #000 calc(100% - 3px))',
   486→                mask: 'radial-gradient(farthest-side, transparent calc(100% - 4px), #000 calc(100% - 3px))',
   487→              }}
   488→            />
   489→          )}
   490→
   491→          {/* the disc — speaking pulse + mic-level scale on separate layers */}
   492→          <motion.div
   493→            className="relative z-10"
   494→            animate={state === 'speaking' ? { scale: [1, 1.045, 1] } : { scale: 1 }}
   495→            transition={
   496→              state === 'speaking'
   497→                ? { duration: 1.2, repeat: Infinity, ease: 'easeInOut' }
   498→                : { duration: 0.3 }
   499→            }
   500→          >
   501→            <div
   502→              style={
   503→                state === 'listening'
   504→                  ? {
   505→                      transform: `scale(${1 + Math.min(0.16, micLevel * 0.3)})`,
   506→                      transition: 'transform 0.18s ease-out',
   507→                    }
   508→                  : undefined
   509→              }
   510→            >
   511→              <div
   512→                className="relative flex items-center justify-center overflow-hidden rounded-full border border-white/10"
   513→                style={{
   514→                  width: 184,
   515→                  height: 184,
   516→                  background:
   517→                    'radial-gradient(circle at 50% 32%, rgba(255,90,95,0.18), rgba(24,10,14,0.92) 72%)',
   518→                  boxShadow:
   519→                    '0 0 90px -18px rgba(255,90,95,0.45), inset 0 1px 0 rgba(255,255,255,0.09)',
   520→                }}
   521→              >
   522→                {state === 'thinking' && <div aria-hidden className="nx-shimmer absolute inset-0 rounded-full" />}
   523→                <BrandMark size={116} className="relative z-10" />
   524→              </div>
   525→            </div>
   526→          </motion.div>
   527→        </div>
   528→
   529→        {/* status */}
   530→        <div className="mt-8 flex items-center gap-2" aria-live="polite">
   531→          <p className="text-sm font-medium text-zinc-300">{statusLabel}</p>
   532→          {state === 'speaking' && (
   533→            <span className="flex items-end gap-1" aria-hidden>
   534→              <span className="nx-dot h-1.5 w-1.5 rounded-full bg-[#ff5a5f]" />
   535→              <span className="nx-dot h-1.5 w-1.5 rounded-full bg-[#ff5a5f]" />
   536→              <span className="nx-dot h-1.5 w-1.5 rounded-full bg-[#ff5a5f]" />
   537→            </span>
   538→          )}
   539→        </div>
   540→        {error && <p className="mt-1.5 max-w-sm text-center text-xs text-red-400/90">{error}</p>}
   541→
   542→        {/* live transcript / text fallback */}
   543→        <div className="mt-3 flex min-h-[30px] w-full max-w-md flex-col items-center">
   544→          {state === 'listening' && interim ? (
   545→            <motion.p
   546→              key={interim.slice(0, 24)}
   547→              initial={{ opacity: 0.4 }}
   548→              animate={{ opacity: 1 }}
   549→              className="text-center text-lg italic leading-snug text-zinc-500"
   550→            >
   551→              {interim}
   552→            </motion.p>
   553→          ) : null}
   554→
   555→          {!supported && (
   556→            <div className="nx-rise w-full">
   557→              <p className="text-center text-sm leading-relaxed text-zinc-400">
   558→                Your browser doesn&apos;t support live voice — try Chrome. Type below and NEXUS will
   559→                still think and speak.
   560→              </p>
   561→              <form
   562→                onSubmit={(e) => {
   563→                  e.preventDefault()
   564→                  sendText()
   565→                }}
   566→                className="nx-composer mt-3 flex items-center gap-2 rounded-2xl p-1.5 pl-4"
   567→              >
   568→                <input
   569→                  value={textInput}
   570→                  onChange={(e) => setTextInput(e.target.value)}
   571→                  placeholder="Type a message…"
   572→                  aria-label="Type a message for NEXUS"
   573→                  className="flex-1 bg-transparent py-2 text-sm text-zinc-100 outline-none placeholder:text-zinc-600"
   574→                />
   575→                <button
   576→                  type="submit"
   577→                  disabled={!textInput.trim() || state === 'thinking' || state === 'speaking'}
   578→                  aria-label="Send message"
   579→                  className="nx-gradient-surface flex h-9 w-9 shrink-0 items-center justify-center rounded-xl disabled:opacity-40"
   580→                >
   581→                  <Send className="h-4 w-4" />
   582→                </button>
   583→              </form>
   584→            </div>
   585→          )}
   586→        </div>
   587→
   588→        {/* compact conversation log — last turns, older fading */}
   589→        {recentTurns.length > 0 && (
   590→          <div className="nx-scroll mt-4 max-h-[20vh] w-full max-w-lg overflow-y-auto pr-1">
   591→            {recentTurns.map((t, i) => (
   592→              <div
   593→                key={t.id}
   594→                className="nx-rise mb-3 text-center last:mb-0"
   595→                style={{ opacity: recentTurns.length === 1 ? 1 : 0.3 + (0.7 * (i + 1)) / recentTurns.length }}
   596→              >
   597→                <p className="text-sm font-medium text-zinc-200">{t.user}</p>
   598→                <p className="mt-0.5 text-sm leading-relaxed text-zinc-400">{t.reply}</p>
   599→              </div>
   600→            ))}
   601→          </div>
   602→        )}
   603→      </div>
   604→
   605→      {/* Bottom controls */}
   606→      <div
   607→        className="relative z-10 flex items-center justify-between gap-4 px-6 sm:px-10"
   608→        style={{ paddingBottom: 'max(2rem, env(safe-area-inset-bottom))' }}
   609→      >
   610→        {/* language picker */}
   611→        <div className="relative">
   612→          <AnimatePresence>
   613→            {showLangs && (
   614→              <motion.div
   615→                initial={{ opacity: 0, y: 8, scale: 0.97 }}
   616→                animate={{ opacity: 1, y: 0, scale: 1 }}
   617→                exit={{ opacity: 0, y: 8, scale: 0.97 }}
   618→                transition={{ duration: 0.16 }}
   619→                className="nx-glow-card absolute bottom-14 left-0 z-20 w-72 p-2.5"
   620→                style={{ background: 'rgba(16,10,13,0.97)' }}
   621→                role="menu"
   622→                aria-label="Choose language"
   623→              >
   624→                <p className="px-1.5 pb-1.5 text-[10px] font-semibold uppercase tracking-widest text-zinc-500">
   625→                  Language
   626→                </p>
   627→                <div className="nx-scroll grid max-h-64 grid-cols-2 gap-1 overflow-y-auto">
   628→                  {LANGUAGES.map((l) => (
   629→                    <button
   630→                      key={l.code}
   631→                      onClick={() => pickLanguage(l)}
   632→                      role="menuitemradio"
   633→                      aria-checked={lang === l.code}
   634→                      className={`rounded-lg px-2.5 py-1.5 text-left text-xs transition ${
   635→                        lang === l.code
   636→                          ? 'font-semibold text-white'
   637→                          : 'text-zinc-400 hover:bg-white/5 hover:text-zinc-200'
   638→                      }`}
   639→                      style={lang === l.code ? { background: 'rgba(255,90,95,0.22)' } : undefined}
   640→                    >
   641→                      {l.label}
   642→                    </button>
   643→                  ))}
   644→                </div>
   645→              </motion.div>
   646→            )}
   647→          </AnimatePresence>
   648→          <button
   649→            onClick={() => setShowLangs((v) => !v)}
   650→            aria-label="Change language"
   651→            aria-expanded={showLangs}
   652→            className="flex h-11 w-11 items-center justify-center rounded-full border border-white/10 bg-white/5 text-zinc-300 transition hover:border-white/25 hover:bg-white/10"
   653→          >
   654→            <Globe className="h-5 w-5" />
   655→          </button>
   656→        </div>
   657→
   658→        <div className="flex items-center gap-3">
   659→          <button
   660→            onClick={toggleMute}
   661→            disabled={!supported}
   662→            aria-label={muted ? 'Unmute microphone' : 'Mute microphone'}
   663→            className={`flex h-12 w-12 items-center justify-center rounded-full border transition disabled:opacity-40 ${
   664→              muted
   665→                ? 'border-[#ff5a5f]/50 bg-[#ff5a5f]/15 text-[#ff8a8d]'
   666→                : 'border-white/15 bg-white/5 text-zinc-200 hover:bg-white/10'
   667→            }`}
   668→          >
   669→            {muted ? <MicOff className="h-5 w-5" /> : <Mic className="h-5 w-5" />}
   670→          </button>
   671→          <button
   672→            onClick={() => onOpenChange(false)}
   673→            aria-label="End voice session"
   674→            className="flex h-14 w-14 items-center justify-center rounded-full bg-[#e5484d] text-white shadow-[0_8px_30px_-6px_rgba(229,72,77,0.6)] transition hover:bg-[#f0555a]"
   675→          >
   676→            <PhoneOff className="h-6 w-6" />
   677→          </button>
   678→        </div>
   679→      </div>
   680→
   681→      <audio ref={audioRef} className="hidden" />
   682→    </div>
   683→  )

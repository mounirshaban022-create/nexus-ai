   123→export function NexusChat(props: NexusChatProps) {
   124→  const { pinnedAgent, onPinnedAgentChange, onSessionCreated } = props
   125→
   126→  const [messages, setMessages] = useState<NxMsg[]>([])
   127→  const [streaming, setStreaming] = useState(false)
   128→  const [sessionId, setSessionId] = useState<string | undefined>(props.sessionId)
   129→  const [input, setInput] = useState(props.prefill ?? '')
   130→  const [attach, setAttach] = useState<{ filename: string; dataUrl: string } | null>(null)
   131→  const [attachError, setAttachError] = useState('')
   132→  const [assign, setAssign] = useState<AgentAssignEvent | null>(null)
   133→
   134→  const scrollRef = useRef<HTMLDivElement>(null)
   135→  const textareaRef = useRef<HTMLTextAreaElement>(null)
   136→  const fileRef = useRef<HTMLInputElement>(null)
   137→  const stickToBottomRef = useRef(true)
   138→  const streamIdRef = useRef<string | null>(null)
   139→  const pendingDeltaRef = useRef('')
   140→  const flushTimerRef = useRef<ReturnType<typeof setInterval> | null>(null)
   141→  const abortRef = useRef<AbortController | null>(null)
   142→  const assignRef = useRef<AgentAssignEvent | null>(null)
   143→  const seqRef = useRef(0)
   144→  /** Live status line from the server ("Picking the right specialist…") —
   145→   *  shown next to the typing dots so waiting never feels dead. */
   146→  const [statusText, setStatusText] = useState('')
   147→  /** Typewriter reveal state — animates one-shot bursts word-by-word. */
   148→  const typewriterRef = useRef<{ timer: ReturnType<typeof setInterval> | null }>({ timer: null })
   149→  const scrollFrameRef = useRef<number | null>(null)
   150→
   151→  const nextId = (prefix: string) => `${prefix}-${Date.now()}-${seqRef.current++}`
   152→
   153→  /* ---------- session resume + one-time pin sync ---------- */
   154→  // The session to resume is captured at MOUNT: page.tsx remounts the chat
   155→  // (chatEpoch key) whenever the user switches sessions, and later
   156→  // props.sessionId changes (e.g. when THIS chat creates its own session)
   157→  // must NOT re-trigger a refetch that would clobber the live transcript
   158→  // (handoff pills, tool cards and attachments are stream-only state).
   159→  const [resumeId] = useState(props.sessionId)
   160→  useEffect(() => {
   161→    if (!resumeId) return
   162→    let cancelled = false
   163→    ;(async () => {
   164→      try {
   165→        const res = await fetch(`/api/chat/sessions/${encodeURIComponent(resumeId)}`)
   166→        if (!res.ok) return
   167→        const data = (await res.json().catch(() => null)) as { session?: unknown } | null
   168→        const s =
   169→          data && typeof data.session === 'object' ? (data.session as Record<string, unknown>) : null
   170→        if (cancelled || !s) return
   171→        const rowsRaw = Array.isArray(s.messages) ? s.messages : []
   172→        const sessionAgentSlug = typeof s.agentSlug === 'string' ? s.agentSlug : null
   173→        const rows: NxMsg[] = rowsRaw
   174→          .filter((m): m is Record<string, unknown> => !!m && typeof m === 'object')
   175→          .map((m, i): NxMsg => {
   176→            const role = m.role === 'user' ? 'user' : m.role === 'tool' ? 'tool' : 'assistant'
   177→            return {
   178→              id: typeof m.id === 'string' ? m.id : `m-${i}`,
   179→              role,
   180→              content: typeof m.content === 'string' ? m.content : '',
   181→              toolName: typeof m.toolName === 'string' ? m.toolName : null,
   182→              toolData: typeof m.toolData === 'string' ? m.toolData : null,
   183→              toolStatus: role === 'tool' ? 'done' : undefined,
   184→              agentSlug: role === 'assistant' ? sessionAgentSlug : undefined,
   185→              // NEXUS One: attachments persisted server-side (image/video/
   186→              // document/code cards) survive resume.
   187→              attachments: Array.isArray(m.attachments) ? m.attachments : undefined,
   188→            }
   189→          })
   190→        setMessages(rows)
   191→        setSessionId(typeof s.id === 'string' ? s.id : resumeId)
   192→        // Sync page-level pin state ONCE after the session loads.
   193→        onPinnedAgentChange(s.agentPinned === true && sessionAgentSlug ? sessionAgentSlug : null)
   194→      } catch {
   195→        /* failed to load — fall back to an empty transcript */
   196→      }
   197→    })()
   198→    return () => {
   199→      cancelled = true
   200→    }
   201→    // Mount-per-session resume (page.tsx remounts via chatEpoch); the pin
   202→    // sync must fire exactly once per loaded session.
   203→  }, [resumeId])
   204→
   205→  /* ---------- abort any live stream on unmount ---------- */
   206→  useEffect(() => {
   207→    return () => {
   208→      abortRef.current?.abort()
   209→      if (flushTimerRef.current) clearInterval(flushTimerRef.current)
   210→      if (typewriterRef.current.timer) clearInterval(typewriterRef.current.timer)
   211→      if (scrollFrameRef.current) cancelAnimationFrame(scrollFrameRef.current)
   212→    }
   213→  }, [])
   214→
   215→  /* ---------- smart auto-scroll (sticks while near the bottom) ----------
   216→   * SCROLL-GLITCH FIX — three hardening measures:
   217→   *   1. rAF throttling: the DOM height settles within one animation frame
   218→   *      before we pin to the bottom (no fighting mid-layout writes).
   219→   *   2. Hysteresis: stick again at ≤140px from the bottom, but only
   220→   *      un-stick beyond 260px — mobile rubber-banding at the bottom edge
   221→   *      no longer randomly detaches the auto-scroll.
   222→   *   3. `overflow-anchor:none` on .nx-scroll (globals.css) stops the
   223→   *      browser's own scroll-anchoring from fighting our scrollTop writes.
   224→   */
   225→  const scrollToEnd = useCallback(() => {
   226→    if (scrollFrameRef.current) cancelAnimationFrame(scrollFrameRef.current)
   227→    scrollFrameRef.current = requestAnimationFrame(() => {
   228→      scrollFrameRef.current = null
   229→      const el = scrollRef.current
   230→      if (!el || !stickToBottomRef.current) return
   231→      el.scrollTop = el.scrollHeight
   232→    })
   233→  }, [])
   234→
   235→  useEffect(() => {
   236→    if (!stickToBottomRef.current) return
   237→    scrollToEnd()
   238→  }, [messages, streaming, scrollToEnd])
   239→
   240→  /* Mobile keyboards + viewport resizes change the container height without
   241→   * a messages update — re-stick to the bottom so the newest content stays
   242→   * visible instead of being cut off under the composer. */
   243→  useEffect(() => {
   244→    const onResize = () => {
   245→      if (stickToBottomRef.current) scrollToEnd()
   246→    }
   247→    window.addEventListener('resize', onResize)
   248→    return () => window.removeEventListener('resize', onResize)
   249→  }, [scrollToEnd])
   250→
   251→  const onScroll = () => {
   252→    const el = scrollRef.current
   253→    if (!el) return
   254→    const distFromBottom = el.scrollHeight - el.scrollTop - el.clientHeight
   255→    // Hysteresis band: easily re-stick near the bottom, hard to un-stick.
   256→    if (distFromBottom <= 140) stickToBottomRef.current = true
   257→    else if (distFromBottom > 260) stickToBottomRef.current = false
   258→  }
   259→
   260→  /* ---------- textarea auto-resize (rows 1 → 6) ---------- */
   261→  useEffect(() => {
   262→    const el = textareaRef.current
   263→    if (!el) return
   264→    el.style.height = 'auto'
   265→    el.style.height = `${Math.min(el.scrollHeight, 144)}px`
   266→  }, [input])
   267→
   268→  /* ---------- streaming helpers ---------- */
   269→
   270→  const appendDelta = useCallback((delta: string) => {
   271→    const id = streamIdRef.current
   272→    if (!id || !delta) return
   273→    setMessages((prev) => prev.map((m) => (m.id === id ? { ...m, content: m.content + delta } : m)))
   274→  }, [])
   275→
   276→  /** Deltas are buffered and flushed every 80ms so react-markdown is not
   277→   * re-parsed on every single token — keeps long streams silky. */
   278→  const startDeltaFlusher = useCallback(() => {
   279→    if (flushTimerRef.current) return
   280→    flushTimerRef.current = setInterval(() => {
   281→      if (!pendingDeltaRef.current) return
   282→      const chunk = pendingDeltaRef.current
   283→      pendingDeltaRef.current = ''
   284→      appendDelta(chunk)
   285→    }, 80)
   286→  }, [appendDelta])
   287→
   288→  const stopDeltaFlusher = useCallback(() => {
   289→    if (flushTimerRef.current) {
   290→      clearInterval(flushTimerRef.current)
   291→      flushTimerRef.current = null
   292→    }
   293→    if (pendingDeltaRef.current) {
   294→      const chunk = pendingDeltaRef.current
   295→      pendingDeltaRef.current = ''
   296→      appendDelta(chunk)
   297→    }
   298→  }, [appendDelta])
   299→
   300→  /* ---------- typewriter reveal (the "alive" effect) ----------
   301→   * When the backend hands us a finished text in ONE burst (non-streaming
   302→   * fallback path, or a coalesced mega-delta), animating it word-by-word
   303→   * keeps the ChatGPT feel instead of a jarring wall of text. Reveal is
   304→   * chunked so even long answers finish in ~1.2s. */
   305→  const revealTypewriter = useCallback((id: string, fullText: string) => {
   306→    if (typewriterRef.current.timer) {
   307→      clearInterval(typewriterRef.current.timer)
   308→      typewriterRef.current.timer = null
   309→    }
   310→    if (!fullText) return
   311→    const total = fullText.length
   312→    // ~70 ticks × 16ms ≈ 1.1s regardless of length; min 2 chars/tick.
   313→    const chunk = Math.max(2, Math.ceil(total / 70))
   314→    let pos = 0
   315→    setMessages((prev) => prev.map((m) => (m.id === id ? { ...m, content: '' } : m)))
   316→    typewriterRef.current.timer = setInterval(() => {
   317→      pos = Math.min(total, pos + chunk)
   318→      const slice = fullText.slice(0, pos)
   319→      setMessages((prev) => prev.map((m) => (m.id === id ? { ...m, content: slice } : m)))
   320→      if (pos >= total) {
   321→        if (typewriterRef.current.timer) clearInterval(typewriterRef.current.timer)
   322→        typewriterRef.current.timer = null
   323→      }
   324→    }, 16)
   325→  }, [])
   326→
   327→  /** Patch the live tool chip that matches (toolName, index). */
   328→  const updateTool = useCallback(
   329→    (tool: string, index: number, patch: Partial<NxMsg>) => {
   330→      setMessages((prev) => {
   331→        let target = -1
   332→        for (let i = prev.length - 1; i >= 0; i--) {
   333→          const m = prev[i]
   334→          if (m.role === 'tool' && m.toolName === tool && m.toolIndex === index) {
   335→            target = i
   336→            break
   337→          }
   338→        }
   339→        if (target === -1) {
   340→          for (let i = prev.length - 1; i >= 0; i--) {
   341→            const m = prev[i]
   342→            if (m.role === 'tool' && m.toolName === tool && m.toolStatus === 'running') {
   343→              target = i
   344→              break
   345→            }
   346→          }
   347→        }
   348→        if (target === -1) return prev
   349→        const copy = [...prev]
   350→        copy[target] = { ...copy[target], ...patch }
   351→        return copy
   352→      })
   353→    },
   354→    []
   355→  )
   356→
   357→  /* ---------- send ---------- */
   358→
   359→  const send = useCallback(
   360→    async (text: string) => {
   361→      const trimmed = text.trim()
   362→      const hasAttach = attach != null
   363→      if (streaming) return
   364→      if (!trimmed && !hasAttach) return
   365→
   366→      const userMsg: NxMsg = {
   367→        id: nextId('local'),
   368→        role: 'user',
   369→        content: trimmed,
   370→        ...(hasAttach ? { attachName: attach!.filename } : {}),
   371→      }
   372→      setMessages((prev) => [...prev, userMsg])
   373→      setInput('')
   374→      setAttach(null)
   375→      setAttachError('')
   376→      setStreaming(true)
   377→      stickToBottomRef.current = true
   378→      setStatusText('')
   379→
   380→      abortRef.current = new AbortController()
   381→
   382→      // STREAM WATCHDOG (defense-in-depth): if the server ever fails to
   383→      // close the NDJSON stream, reader.read() hangs forever and the
   384→      // composer stays locked in "generating". This timer aborts the fetch
   385→      // after 90s of total silence and frees the UI. Declared OUTSIDE the
   386→      // try so the finally block can always disarm it.
   387→      let watchdog: ReturnType<typeof setTimeout> | null = null
   388→      const armWatchdog = () => {
   389→        if (watchdog) clearTimeout(watchdog)
   390→        watchdog = setTimeout(() => {
   391→          abortRef.current?.abort()
   392→        }, 90_000)
   393→      }
   394→      const disarmWatchdog = () => {
   395→        if (watchdog) clearTimeout(watchdog)
   396→        watchdog = null
   397→      }
   398→
   399→      try {
   400→        const res = await fetch('/api/chat', {
   401→          method: 'POST',
   402→          headers: { 'Content-Type': 'application/json' },
   403→          body: JSON.stringify({
   404→            message: trimmed,
   405→            ...(sessionId ? { sessionId } : {}),
   406→            // '' = auto-routing; a slug pins the session persona (idempotent).
   407→            agentSlug: pinnedAgent ?? '',
   408→            ...(hasAttach
   409→              ? { attachment: { filename: attach!.filename, dataUrl: attach!.dataUrl } }
   410→              : {}),
   411→          }),
   412→          signal: abortRef.current.signal,
   413→        })
   414→
   415→        if (!res.ok || !res.body) {
   416→          const data = (await res.json().catch(() => null)) as { error?: string } | null
   417→          throw new Error(data?.error || 'Chat request failed.')
   418→        }
   419→
   420→        // Consume the NDJSON stream.
   421→        const reader = res.body.getReader()
   422→        const decoder = new TextDecoder()
   423→        let buffer = ''
   424→        armWatchdog()
   425→
   426→        for (;;) {
   427→          const { done, value } = await reader.read()
   428→          if (done) break
   429→          armWatchdog()
   430→          buffer += decoder.decode(value, { stream: true })
   431→          const lines = buffer.split('\n')
   432→          buffer = lines.pop() ?? ''
   433→          for (const line of lines) {
   434→            const text = line.trim()
   435→            if (!text) continue
   436→            let event: ChatStreamEvent
   437→            try {
   438→              event = JSON.parse(text) as ChatStreamEvent
   439→            } catch {
   440→              continue // partial / malformed line — skip
   441→            }
   442→
   443→            switch (event.type) {
   444→              case 'user': {
   445→                setMessages((prev) =>
   446→                  prev.map((m) =>
   447→                    m.id === userMsg.id ? { ...m, id: event.id, content: event.content } : m
   448→                  )
   449→                )
   450→                break
   451→              }
   452→              case 'agent_assign': {
   453→                assignRef.current = event
   454→                setAssign(event)
   455→                setMessages((prev) => [
   456→                  ...prev,
   457→                  { id: nextId('assign'), role: 'agent', content: '', assign: event },
   458→                ])
   459→                break
   460→              }
   461→              case 'status': {
   462→                // Live server status line — displayed with the typing dots
   463→                // until the first delta lands.
   464→                setStatusText(event.message)
   465→                break
   466→              }
   467→              case 'assistant_start': {
   468→                streamIdRef.current = event.id
   469→                const slug = pinnedAgent ?? assignRef.current?.agentSlug ?? null
   470→                setMessages((prev) => [
   471→                  ...prev,
   472→                  { id: event.id, role: 'assistant', content: '', streaming: true, agentSlug: slug },
   473→                ])
   474→                break
   475→              }
   476→              case 'assistant_delta': {
   477→                setStatusText('')
   478→                const id = streamIdRef.current
   479→                // BURST DETECTION: a huge first delta means the backend
   480→                // handed us a finished text in one shot (non-streaming
   481→                // fallback) — reveal it with the typewriter instead of
   482→                // slamming a wall of text on screen.
   483→                if (id && event.delta.length > 160) {
   484→                  const bubbleEmpty = !messagesRef.current.find((m) => m.id === id)?.content
   485→                  if (bubbleEmpty && !typewriterRef.current.timer) {
   486→                    revealTypewriter(id, event.delta)
   487→                    break
   488→                  }
   489→                }
   490→                pendingDeltaRef.current += event.delta
   491→                startDeltaFlusher()
   492→                break
   493→              }
   494→              case 'assistant_end': {
   495→                stopDeltaFlusher()
   496→                const id = streamIdRef.current
   497→                streamIdRef.current = null
   498→                if (id) {
   499→                  const attachments = Array.isArray(event.attachments) ? event.attachments : []
   500→                  setMessages((prev) =>
   501→                    prev.map((m) =>
   502→                      m.id === id
   503→                        ? {
   504→                            ...m,
   505→                            streaming: false,
   506→                            ...(attachments.length > 0 ? { attachments } : {}),
   507→                          }
   508→                        : m
   509→                    )
   510→                  )
   511→                }
   512→                break
   513→              }
   514→              case 'assistant': {
   515→                // Full-message fallback (non-streaming providers).
   516→                stopDeltaFlusher()
   517→                const id = streamIdRef.current
   518→                streamIdRef.current = null
   519→                const attachments = Array.isArray(event.attachments) ? event.attachments : []
   520→                if (id) {
   521→                  // Typewriter reveal — the full text arrived in one shot;
   522→                  // animate it word-by-word so it feels alive.
   523→                  setMessages((prev) =>
   524→                    prev.map((m) =>
   525→                      m.id === id
   526→                        ? {
   527→                            ...m,
   528→                            streaming: false,
   529→                            ...(attachments.length > 0 ? { attachments } : {}),
   530→                          }
   531→                        : m
   532→                    )
   533→                  )
   534→                  revealTypewriter(id, event.content)
   535→                } else {
   536→                  const freshId = nextId('a')
   537→                  setMessages((prev) => [
   538→                    ...prev,
   539→                    {
   540→                      id: freshId,
   541→                      role: 'assistant',
   542→                      content: '',
   543→                      agentSlug: pinnedAgent ?? assignRef.current?.agentSlug ?? null,
   544→                      ...(attachments.length > 0 ? { attachments } : {}),
   545→                    },
   546→                  ])
   547→                  revealTypewriter(freshId, event.content)
   548→                }
   549→                break
   550→              }
   551→              case 'tool_start': {
   552→                setMessages((prev) => [
   553→                  ...prev,
   554→                  {
   555→                    id: nextId(`tool-${event.index}`),
   556→                    role: 'tool',
   557→                    content: '',
   558→                    toolName: event.tool,
   559→                    toolIndex: event.index,
   560→                    toolStatus: 'running',
   561→                    toolArgs: event.args,
   562→                  },
   563→                ])
   564→                break
   565→              }
   566→              case 'tool_progress': {
   567→                updateTool(event.tool, event.index, { toolMessage: event.message })
   568→                break
   569→              }
   570→              case 'tool_result': {
   571→                let data: string | null = null
   572→                try {
   573→                  data = JSON.stringify(event.result)
   574→                } catch {
   575→                  data = null
   576→                }
   577→                updateTool(event.tool, event.index, {
   578→                  toolStatus: event.ok ? 'done' : 'error',
   579→                  toolData: data,
   580→                })
   581→                break
   582→              }
   583→              case 'done': {
   584→                setSessionId(event.sessionId)
   585→                onSessionCreated(event.sessionId)
   586→                break
   587→              }
   588→              case 'error': {
   589→                throw new Error(event.message)
   590→              }
   591→            }
   592→          }
   593→        }
   594→      } catch (error) {
   595→        if (error instanceof DOMException && error.name === 'AbortError') return
   596→        setMessages((prev) => [
   597→          ...prev,
   598→          {
   599→            id: nextId('err'),
   600→            role: 'assistant',
   601→            content:
   602→              error instanceof Error
   603→                ? `Something went wrong: ${error.message}`
   604→                : 'Something went wrong. Please try again.',
   605→            isError: true,
   606→            agentSlug: pinnedAgent ?? assignRef.current?.agentSlug ?? null,
   607→          },
   608→        ])
   609→      } finally {

   611→        stopDeltaFlusher()
   612→        streamIdRef.current = null
   613→        setStreaming(false)
   614→        setStatusText('')
   615→        setMessages((prev) =>

   617→            m.streaming ? { ...m, streaming: false } : m.toolStatus === 'running' ? { ...m, toolStatus: 'done' } : m

   620→      }
   621→    },

     1→     1→'use client'
     2→     2→
     3→     3→import { useCallback, useEffect, useRef, useState } from 'react'
     4→     4→import { motion, AnimatePresence } from 'framer-motion'
     5→     5→import {
     6→     6→  Calendar,
     7→     7→  CheckCircle2,
     8→     8→  Cloud,
     9→     9→  Globe,
    10→    10→  Loader2,
    11→    11→  Mail,
    12→    12→  MessageSquare,
    13→    13→  Plus,
    14→    14→  ShieldCheck,
    15→    15→  Trash2,
    16→    16→  X,
    17→    17→  XCircle,
    18→    18→} from 'lucide-react'
    19→    19→import { Button } from '@/components/ui/button'
    20→    20→import { Input } from '@/components/ui/input'
    21→    21→import { Label } from '@/components/ui/label'
    22→    22→
    23→    23→/* ------------------------------------------------------------------ */
    24→    24→/* Types                                                               */
    25→    25→/* ------------------------------------------------------------------ */
    26→    26→
    27→    27→interface EmailPreset {
    28→    28→  id: string
    29→    29→  label: string
    30→    30→  imapHost: string
    31→    31→  imapPort: number
    32→    32→  smtpHost: string
    33→    33→  smtpPort: number
    34→    34→  smtpSecure: boolean
    35→    35→  hint: string
    36→    36→}
    37→    37→
    38→    38→interface EmailAccount {
    39→    39→  id: string
    40→    40→  label: string
    41→    41→  email: string
    42→    42→  imapHost: string
    43→    43→  smtpHost: string
    44→    44→  status: string // unverified | connected | error
    45→    45→  statusMessage: string
    46→    46→  createdAt: string
    47→    47→}
    48→    48→
    49→    49→interface ConnectPanelProps {
    50→    50→  open: boolean
    51→    51→  onClose: () => void
    52→    52→  /** Called when an account is added or removed — parent can refresh state */
    53→    53→  onAccountsChange?: () => void
    54→    54→}
    55→    55→
    56→    56→const EMAIL_REGEX = /^[^@\s]+@[^@\s]+\.[^@\s]+$/
    57→    57→
    58→    58→/* The "coming soon" apps shown alongside Email in the Connect grid */
    59→    59→const COMING_SOON_APPS = [
    60→    60→  { id: 'calendar', label: 'Calendar', icon: Calendar, desc: 'Google Calendar · iCloud' },
    61→    61→  { id: 'drive', label: 'Cloud Drive', icon: Cloud, desc: 'Drive · Dropbox · OneDrive' },
    62→    62→  { id: 'slack', label: 'Slack', icon: MessageSquare, desc: 'Team messaging' },
    63→    63→  { id: 'web', label: 'Web', icon: Globe, desc: 'Live web search & reader' },
    64→    64→]
    65→    65→
    66→    66→export function ConnectPanel({ open, onClose, onAccountsChange }: ConnectPanelProps) {
    67→    67→  const [accounts, setAccounts] = useState<EmailAccount[]>([])
    68→    68→  const [presets, setPresets] = useState<EmailPreset[]>([])
    69→    69→  const [loading, setLoading] = useState(false)
    70→    70→  const [showAddForm, setShowAddForm] = useState(false)
    71→    71→
    72→    72→  // Form state
    73→    73→  const [presetId, setPresetId] = useState<string>('gmail')
    74→    74→  const [label, setLabel] = useState('')
    75→    75→  const [email, setEmail] = useState('')
    76→    76→  const [fromName, setFromName] = useState('')
    77→    77→  const [username, setUsername] = useState('')
    78→    78→  const [password, setPassword] = useState('')
    79→    79→  const [imapHost, setImapHost] = useState('imap.gmail.com')
    80→    80→  const [imapPort, setImapPort] = useState(993)
    81→    81→  const [smtpHost, setSmtpHost] = useState('smtp.gmail.com')
    82→    82→  const [smtpPort, setSmtpPort] = useState(465)
    83→    83→  const [smtpSecure, setSmtpSecure] = useState(true)
    84→    84→  const [submitting, setSubmitting] = useState(false)
    85→    85→  const [formError, setFormError] = useState<string | null>(null)
    86→    86→  const [formSuccess, setFormSuccess] = useState<string | null>(null)
    87→    87→  const [deletingId, setDeletingId] = useState<string | null>(null)
    88→    88→
    89→    89→  const firstFieldRef = useRef<HTMLInputElement | null>(null)
    90→    90→
    91→    91→  /* ---------------- load accounts + presets ---------------- */
    92→    92→  const loadAccounts = useCallback(async () => {
    93→    93→    setLoading(true)
    94→    94→    try {
    95→    95→      const res = await fetch('/api/email/accounts')
    96→    96→      const data = await res.json()
    97→    97→      if (res.ok) {
    98→    98→        setAccounts(data.accounts ?? [])
    99→    99→        setPresets(data.presets ?? [])
   100→   100→      }
   101→   101→    } catch {
   102→   102→      /* best-effort */
   103→   103→    } finally {
   104→   104→      setLoading(false)
   105→   105→    }
   106→   106→  }, [])
   107→   107→
   108→   108→  useEffect(() => {
   109→   109→    if (open) {
   110→   110→      loadAccounts()
   111→   111→      setFormError(null)
   112→   112→      setFormSuccess(null)
   113→   113→      setShowAddForm(false)
   114→   114→    }
   115→   115→  }, [open, loadAccounts])
   116→   116→
   117→   117→  /* ---------------- escape + focus ---------------- */
   118→   118→  useEffect(() => {
   119→   119→    if (!open) return
   120→   120→    const onKey = (e: KeyboardEvent) => {
   121→   121→      if (e.key === 'Escape') onClose()
   122→   122→    }
   123→   123→    window.addEventListener('keydown', onKey)
   124→   124→    return () => window.removeEventListener('keydown', onKey)
   125→   125→  }, [open, onClose])
   126→   126→
   127→   127→  useEffect(() => {
   128→   128→    if (open && showAddForm && firstFieldRef.current) {
   129→   129→      const id = window.setTimeout(() => firstFieldRef.current?.focus(), 30)
   130→   130→      return () => window.clearTimeout(id)
   131→   131→    }
   132→   132→  }, [open, showAddForm])
   133→   133→
   134→   134→  /* ---------------- preset selection ---------------- */
   135→   135→  const applyPreset = useCallback((id: string) => {
   136→   136→    setPresetId(id)
   137→   137→    setFormError(null)
   138→   138→    setFormSuccess(null)
   139→   139→    const p = presets.find((x) => x.id === id)
   140→   140→    if (!p || id === 'custom') return
   141→   141→    setImapHost(p.imapHost)
   142→   142→    setImapPort(p.imapPort)
   143→   143→    setSmtpHost(p.smtpHost)
   144→   144→    setSmtpPort(p.smtpPort)
   145→   145→    setSmtpSecure(p.smtpSecure)
   146→   146→    // Auto-derive label + username from email if empty
   147→   147→    setEmail((prev) => prev)
   148→   148→    setUsername((prev) => prev || email)
   149→   149→  }, [presets, email])
   150→   150→
   151→   151→  /* ---------------- submit (connect account) ---------------- */
   152→   152→  const submit = useCallback(async () => {
   153→   153→    setFormError(null)
   154→   154→    setFormSuccess(null)
   155→   155→    if (!label.trim()) return setFormError('Give this account a label (e.g. "Work Gmail").')
   156→   156→    if (!EMAIL_REGEX.test(email)) return setFormError('Enter a valid email address.')
   157→   157→    if (!username.trim()) return setFormError('Username is required (usually your full email).')
   158→   158→    if (!password) return setFormError('Password / app password is required.')
   159→   159→    if (presetId === 'custom') {
   160→   160→      if (!imapHost || !imapPort || !smtpHost || !smtpPort) {
   161→   161→        return setFormError('Fill in the IMAP and SMTP host/port for your custom provider.')
   162→   162→      }
   163→   163→    }
   164→   164→    setSubmitting(true)
   165→   165→    try {
   166→   166→      const res = await fetch('/api/email/accounts', {
   167→   167→        method: 'POST',
   168→   168→        headers: { 'Content-Type': 'application/json' },
   169→   169→        body: JSON.stringify({
   170→   170→          label: label.trim(),
   171→   171→          email: email.trim(),
   172→   172→          fromName: fromName.trim(),
   173→   173→          imapHost,
   174→   174→          imapPort,
   175→   175→          smtpHost,
   176→   176→          smtpPort,
   177→   177→          smtpSecure,
   178→   178→          username: username.trim(),
   179→   179→          password,
   180→   180→        }),
   181→   181→      })
   182→   182→      const data = await res.json()
   183→   183→      if (!res.ok) {
   184→   184→        return setFormError(data.error || 'Could not connect that account.')
   185→   185→      }
   186→   186→      const v = data.verification || {}
   187→   187→      if (v.ok) {
   188→   188→        setFormSuccess(v.message || 'Connected successfully.')
   189→   189→        // Reset form + refresh list
   190→   190→        setLabel('')
   191→   191→        setFromName('')
   192→   192→        setPassword('')
   193→   193→        setShowAddForm(false)
   194→   194→        await loadAccounts()
   195→   195→        onAccountsChange?.()
   196→   196→      } else {
   197→   197→        setFormError(v.message || 'Verification failed — check your credentials.')
   198→   198→      }
   199→   199→    } catch (err: any) {
   200→   200→      setFormError(err?.message || 'Network error — please try again.')
   201→   201→    } finally {
   202→   202→      setSubmitting(false)
   203→   203→    }
   204→   204→  }, [label, email, fromName, username, password, presetId, imapHost, imapPort, smtpHost, smtpPort, smtpSecure, loadAccounts, onAccountsChange])
   205→   205→
   206→   206→  /* ---------------- delete account ---------------- */
   207→   207→  const removeAccount = useCallback(async (id: string) => {
   208→   208→    setDeletingId(id)
   209→   209→    try {
   210→   210→      const res = await fetch(`/api/email/accounts/${id}`, { method: 'DELETE' })
   211→   211→      if (res.ok) {
   212→   212→        await loadAccounts()
   213→   213→        onAccountsChange?.()
   214→   214→      }
   215→   215→    } catch {
   216→   216→      /* best-effort */
   217→   217→    } finally {
   218→   218→      setDeletingId(null)
   219→   219→    }
   220→   220→  }, [loadAccounts, onAccountsChange])
   221→   221→
   222→   222→  const selectedPreset = presets.find((p) => p.id === presetId)
   223→   223→  const connectedCount = accounts.filter((a) => a.status === 'connected').length
   224→   224→
   225→   225→  return (
   226→   226→    <AnimatePresence>
   227→   227→      {open && (
   228→   228→        <motion.div
   229→   229→          className="fixed inset-0 z-50 flex items-end justify-center bg-black/50 p-0 sm:items-center sm:p-4"
   230→   230→          initial={{ opacity: 0 }}
   231→   231→          animate={{ opacity: 1 }}
   232→   232→          exit={{ opacity: 0 }}
   233→   233→          onClick={onClose}
   234→   234→        >
   235→   235→          <motion.div
   236→   236→            initial={{ y: '100%', opacity: 0 }}
   237→   237→            animate={{ y: 0, opacity: 1 }}
   238→   238→            exit={{ y: '100%', opacity: 0 }}
   239→   239→            transition={{ type: 'spring', damping: 30, stiffness: 300 }}
   240→   240→            onClick={(e) => e.stopPropagation()}
   241→   241→            className="omni-scroll flex max-h-[92vh] w-full flex-col overflow-y-auto rounded-t-3xl border border-border bg-background sm:max-w-lg sm:rounded-3xl"
   242→   242→            role="dialog"
   243→   243→            aria-modal="true"
   244→   244→            aria-label="Connect your apps and email"
   245→   245→          >
   246→   246→            {/* Header */}
   247→   247→            <div className="sticky top-0 z-10 flex items-center justify-between border-b border-border bg-background/95 px-5 py-4 backdrop-blur">
   248→   248→              <div className="flex items-center gap-2.5">
   249→   249→                <span className="flex h-8 w-8 items-center justify-center rounded-full bg-gradient-to-br from-orange-500 to-rose-500 text-white">
   250→   250→                  <Plus className="h-4 w-4" aria-hidden />
   251→   251→                </span>
   252→   252→                <div>
   253→   253→                  <h2 className="text-base font-semibold">Connect</h2>
   254→   254→                  <p className="text-[11px] text-muted-foreground">
   255→   255→                    Link your email and apps to Nexus
   256→   256→                  </p>
   257→   257→                </div>
   258→   258→              </div>
   259→   259→              <button
   260→   260→                onClick={onClose}
   261→   261→                aria-label="Close"
   262→   262→                className="flex h-8 w-8 items-center justify-center rounded-full text-muted-foreground transition hover:bg-secondary"
   263→   263→              >
   264→   264→                <X className="h-4 w-4" />
   265→   265→              </button>
   266→   266→            </div>
   267→   267→
   268→   268→            <div className="px-5 py-5">
   269→   269→              {/* Connected apps grid */}
   270→   270→              <section>
   271→   271→                <h3 className="mb-2.5 text-[10px] font-semibold uppercase tracking-widest text-muted-foreground/60">
   272→   272→                  Apps
   273→   273→                </h3>
   274→   274→                <div className="grid grid-cols-2 gap-2.5 sm:grid-cols-3">
   275→   275→                  {/* Email — the active connector */}
   276→   276→                  <div
   277→   277→                    className={`relative flex flex-col items-start gap-2 overflow-hidden rounded-2xl border p-3 ${
   278→   278→                      connectedCount > 0
   279→   279→                        ? 'border-emerald-500/40 bg-emerald-500/5'
   280→   280→                        : 'border-border bg-card'
   281→   281→                    }`}
   282→   282→                  >
   283→   283→                    <span className="flex h-9 w-9 items-center justify-center rounded-xl bg-gradient-to-br from-orange-500 to-amber-500 text-white shadow-sm">
   284→   284→                      <Mail className="h-4 w-4" aria-hidden />
   285→   285→                    </span>
   286→   286→                    <div>
   287→   287→                      <p className="text-xs font-semibold">Email</p>
   288→   288→                      <p className="text-[10px] text-muted-foreground">
   289→   289→                        {connectedCount > 0 ? `${connectedCount} connected` : 'Not connected'}
   290→   290→                      </p>
   291→   291→                    </div>
   292→   292→                    {connectedCount > 0 && (
   293→   293→                      <span className="absolute right-2 top-2 flex h-4 w-4 items-center justify-center rounded-full bg-emerald-500 text-white">
   294→   294→                        <CheckCircle2 className="h-2.5 w-2.5" aria-hidden />
   295→   295→                      </span>
   296→   296→                    )}
   297→   297→                  </div>
   298→   298→                  {/* Coming-soon apps */}
   299→   299→                  {COMING_SOON_APPS.map((app) => {
   300→   300→                    const Icon = app.icon
   301→   301→                    return (
   302→   302→                      <div
   303→   303→                        key={app.id}
   304→   304→                        className="relative flex flex-col items-start gap-2 overflow-hidden rounded-2xl border border-dashed border-border/60 bg-secondary/20 p-3 opacity-60"
   305→   305→                      >
   306→   306→                        <span className="flex h-9 w-9 items-center justify-center rounded-xl bg-secondary text-muted-foreground">
   307→   307→                          <Icon className="h-4 w-4" aria-hidden />
   308→   308→                        </span>
   309→   309→                        <div>
   310→   310→                          <p className="text-xs font-semibold">{app.label}</p>
   311→   311→                          <p className="text-[10px] text-muted-foreground">Soon</p>
   312→   312→                        </div>
   313→   313→                      </div>
   314→   314→                    )
   315→   315→                  })}
   316→   316→                </div>
   317→   317→              </section>
   318→   318→
   319→   319→              {/* Email accounts list */}
   320→   320→              <section className="mt-6">
   321→   321→                <div className="mb-2.5 flex items-center justify-between">
   322→   322→                  <h3 className="text-[10px] font-semibold uppercase tracking-widest text-muted-foreground/60">
   323→   323→                    Email accounts
   324→   324→                  </h3>
   325→   325→                  <button
   326→   326→                    onClick={() => {
   327→   327→                      setShowAddForm((o) => !o)
   328→   328→                      setFormError(null)
   329→   329→                      setFormSuccess(null)
   330→   330→                    }}
   331→   331→                    className="flex items-center gap-1 rounded-full bg-primary px-2.5 py-1 text-[11px] font-medium text-primary-foreground transition hover:brightness-110"
   332→   332→                  >
   333→   333→                    <Plus className="h-3 w-3" aria-hidden />
   334→   334→                    {showAddForm ? 'Cancel' : 'Add account'}
   335→   335→                  </button>
   336→   336→                </div>
   337→   337→
   338→   338→                {loading ? (
   339→   339→                  <div className="flex items-center gap-2 rounded-xl border border-border bg-card px-4 py-4 text-xs text-muted-foreground">
   340→   340→                    <Loader2 className="h-3.5 w-3.5 animate-spin" /> Loading…
   341→   341→                  </div>
   342→   342→                ) : accounts.length === 0 ? (
   343→   343→                  <div className="rounded-xl border border-dashed border-border/70 bg-card/40 px-4 py-6 text-center">
   344→   344→                    <Mail className="mx-auto mb-2 h-6 w-6 text-muted-foreground/40" aria-hidden />
   345→   345→                    <p className="text-xs text-muted-foreground">
   346→   346→                      No email accounts yet. Add one to send and read email from Nexus.
   347→   347→                    </p>
   348→   348→                  </div>
   349→   349→                ) : (
   350→   350→                  <ul className="space-y-2">
   351→   351→                    {accounts.map((a) => {
   352→   352→                      const ok = a.status === 'connected'
   353→   353→                      return (
   354→   354→                        <li
   355→   355→                          key={a.id}
   356→   356→                          className="flex items-start gap-3 rounded-xl border border-border bg-card px-3.5 py-3"
   357→   357→                        >
   358→   358→                          <span
   359→   359→                            className={`mt-0.5 flex h-7 w-7 shrink-0 items-center justify-center rounded-full ${
   360→   360→                              ok ? 'bg-emerald-500/15 text-emerald-600' : 'bg-destructive/15 text-destructive'
   361→   361→                            }`}
   362→   362→                          >
   363→   363→                            {ok ? <CheckCircle2 className="h-3.5 w-3.5" /> : <XCircle className="h-3.5 w-3.5" />}
   364→   364→                          </span>
   365→   365→                          <div className="min-w-0 flex-1">
   366→   366→                            <p className="truncate text-sm font-medium">{a.label}</p>
   367→   367→                            <p className="truncate text-xs text-muted-foreground">{a.email}</p>
   368→   368→                            <p className="mt-0.5 truncate text-[10px] text-muted-foreground/70">
   369→   369→                              IMAP {a.imapHost} · SMTP {a.smtpHost}
   370→   370→                            </p>
   371→   371→                            {!ok && a.statusMessage && (
   372→   372→                              <p className="mt-1 text-[10px] text-destructive/90">{a.statusMessage}</p>
   373→   373→                            )}
   374→   374→                          </div>
   375→   375→                          <button
   376→   376→                            onClick={() => removeAccount(a.id)}
   377→   377→                            disabled={deletingId === a.id}
   378→   378→                            aria-label="Remove account"
   379→   379→                            className="flex h-7 w-7 shrink-0 items-center justify-center rounded-full text-muted-foreground transition hover:bg-destructive/10 hover:text-destructive disabled:opacity-50"
   380→   380→                          >
   381→   381→                            {deletingId === a.id ? (
   382→   382→                              <Loader2 className="h-3.5 w-3.5 animate-spin" />
   383→   383→                            ) : (
   384→   384→                              <Trash2 className="h-3.5 w-3.5" />
   385→   385→                            )}
   386→   386→                          </button>
   387→   387→                        </li>
   388→   388→                      )
   389→   389→                    })}
   390→   390→                  </ul>
   391→   391→                )}
   392→   392→              </section>
   393→   393→
   394→   394→              {/* Add-account form (collapsible) */}
   395→   395→              <AnimatePresence initial={false}>
   396→   396→                {showAddForm && (
   397→   397→                  <motion.section
   398→   398→                    initial={{ height: 0, opacity: 0 }}
   399→   399→                    animate={{ height: 'auto', opacity: 1 }}
   400→   400→                    exit={{ height: 0, opacity: 0 }}
   401→   401→                    className="overflow-hidden"
   402→   402→                  >
   403→   403→                    <div className="mt-4 rounded-2xl border border-border bg-card p-4">
   404→   404→                      <h4 className="mb-3 flex items-center gap-2 text-sm font-semibold">
   405→   405→                        <Mail className="h-4 w-4 text-primary" aria-hidden /> Connect an email account
   406→   406→                      </h4>
   407→   407→
   408→   408→                      {/* Provider preset picker */}
   409→   409→                      <Label className="mb-1.5 block text-[11px] text-muted-foreground">Provider</Label>
   410→   410→                      <div className="mb-3 flex flex-wrap gap-1.5">
   411→   411→                        {presets.map((p) => (
   412→   412→                          <button
   413→   413→                            key={p.id}
   414→   414→                            type="button"
   415→   415→                            onClick={() => applyPreset(p.id)}
   416→   416→                            className={`rounded-full border px-3 py-1.5 text-xs font-medium transition ${
   417→   417→                              presetId === p.id
   418→   418→                                ? 'border-primary bg-primary/10 text-primary'
   419→   419→                                : 'border-border bg-card text-muted-foreground hover:border-primary/40'
   420→   420→                            }`}
   421→   421→                          >
   422→   422→                            {p.label}
   423→   423→                          </button>
   424→   424→                        ))}
   425→   425→                      </div>
   426→   426→
   427→   427→                      {selectedPreset?.hint && (
   428→   428→                        <p className="mb-3 flex items-start gap-1.5 rounded-lg bg-amber-500/10 px-3 py-2 text-[11px] text-amber-700">
   429→   429→                          <ShieldCheck className="mt-0.5 h-3 w-3 shrink-0" aria-hidden />
   430→   430→                          <span>{selectedPreset.hint}</span>
   431→   431→                        </p>
   432→   432→                      )}
   433→   433→
   434→   434→                      <div className="space-y-3">
   435→   435→                        <Field>
   436→   436→                          <Label htmlFor="acct-label" className="mb-1.5 block text-[11px] text-muted-foreground">
   437→   437→                            Label
   438→   438→                          </Label>
   439→   439→                          <Input
   440→   440→                            id="acct-label"
   441→   441→                            ref={firstFieldRef}
   442→   442→                            value={label}
   443→   443→                            onChange={(e) => setLabel(e.target.value)}

    12→/**
    13→ * NEXUS Premium Voice Mode — ChatGPT-style full-screen voice conversation.
    14→ *
    15→ * Pipeline (proven, multi-layer, resilient):
    16→ *  - INPUT:  Browser-native Web Speech API (live transcription as you speak).
    17→ *            Falls back to MediaRecorder → /api/asr (server-side ASR) if
    18→ *            the browser can't do on-device recognition.
    19→ *  - THINK:  /api/voice/turn (single fast LLM call, voice persona prompt).
    20→ *  - OUTPUT: /api/tts (Microsoft neural voices, 300+ premium voices) OR
    21→ *            Kokoro in-browser neural voice OR browser speechSynthesis.
    22→ *
    23→ * Premium UI:
    24→ *  - Full-screen gradient backdrop with ambient grain.
    25→ *  - Large animated orb (gradient core + pulsing rings + waveform).
    26→ *  - Live interim transcript as you speak.
    27→ *  - Conversation transcript ribbon at the bottom.
    28→ *  - Voice picker (premium neural voices grouped by language).
    29→ *  - Language picker (12 languages).
    30→ */

    58→export function VoiceModeOverlay({ open, onClose }: { open: boolean; onClose: () => void }) {
    59→  const { toast } = useToast()
    60→  const [state, setState] = useState<VoiceState>('idle')
    61→  const [iframeBlocked, setIframeBlocked] = useState(false)
    62→  const [audioUnlocked, setAudioUnlocked] = useState(false)
    63→  const [lang, setLang] = useState('en-US')
    64→  const [ttsVoice, setTtsVoice] = useState('en-US-AriaNeural')
    65→  const [interim, setInterim] = useState('')
    66→  const [micLevel, setMicLevel] = useState(0)
    67→  const [turns, setTurns] = useState<Turn[]>([])
    68→  const [showVoicePicker, setShowVoicePicker] = useState(false)
    69→  const [showLangPicker, setShowLangPicker] = useState(false)
    70→  const [error, setError] = useState('')
    71→
    72→  const recognitionRef = useRef<any>(null)
    73→  const historyRef = useRef<Array<{ role: string; content: string }>>([])
    74→  const audioRef = useRef<HTMLAudioElement | null>(null)
    75→  const stateRef = useRef<VoiceState>('idle')
    76→  const audioCtxRef = useRef<AudioContext | null>(null)
    77→  const sourceRef = useRef<AudioBufferSourceNode | null>(null)
    78→  const scrollRef = useRef<HTMLDivElement | null>(null)
    79→
    80→  // Bug C: keep the overlay cheap when closed — callbacks no-op via this ref.
    81→  const openRef = useRef(open)
    82→  // Bug K: per-turn AbortController so in-flight think()/speak() can be cancelled on close.
    83→  const controllerRef = useRef<AbortController | null>(null)
    84→  // Bug E: SpeechRecognition auto-restart loop protection.
    85→  const restartCountRef = useRef(0)
    86→  const lastStartRef = useRef(0)
    87→  const manualStopRef = useRef(false)
    88→
    89→  const setStateSafe = useCallback((s: VoiceState) => {
    90→    stateRef.current = s
    91→    setState(s)
    92→  }, [])
    93→
    94→  // Bug C: keep openRef in sync so guards inside callbacks read the latest value.
    95→  useEffect(() => { openRef.current = open }, [open])
    96→
    97→  const stopSpeaking = useCallback(() => {
    98→    // Bug K: cancel any in-flight think()/speak() fetch.
    99→    controllerRef.current?.abort()
   100→    if (audioRef.current) { audioRef.current.pause(); audioRef.current = null }
   101→    if (sourceRef.current) { try { sourceRef.current.stop() } catch {} sourceRef.current = null }
   102→  }, [])
   103→
   104→  const stopRecognition = useCallback(() => {
   105→    // Bug E: mark as manually stopped so onend won't auto-restart.
   106→    manualStopRef.current = true
   107→    if (recognitionRef.current) {
   108→      try {
   109→        const pulse = (recognitionRef.current as any).__pulse
   110→        if (pulse) clearInterval(pulse)
   111→        recognitionRef.current.onend = null
   112→        recognitionRef.current.onresult = null
   113→        recognitionRef.current.onerror = null
   114→        recognitionRef.current.stop()
   115→      } catch {}
   116→      recognitionRef.current = null
   117→      setMicLevel(0)
   118→    }
   119→  }, [])
   120→
   121→  /* ---------- WebAudio unlock (autoplay policy) ---------- */
   122→  const unlockAudio = useCallback(() => {
   123→    try {
   124→      if (!audioCtxRef.current || audioCtxRef.current.state === 'closed') {
   125→        const AudioCtx = window.AudioContext || (window as any).webkitAudioContext
   126→        audioCtxRef.current = new AudioCtx()
   127→      }
   128→      if (audioCtxRef.current.state === 'suspended') {
   129→        audioCtxRef.current.resume().catch(() => {})
   130→      }
   131→      setAudioUnlocked(true)
   132→    } catch {}
   133→  }, [])
   134→
   135→  const playWebAudio = useCallback(async (arrayBuffer: ArrayBuffer): Promise<void> => {
   136→    unlockAudio()
   137→    const ctx = audioCtxRef.current
   138→    if (!ctx || ctx.state === 'closed') throw new Error('Audio context unavailable')
   139→    if (ctx.state === 'suspended') await ctx.resume().catch(() => {})
   140→    const audioBuffer = await ctx.decodeAudioData(arrayBuffer)
   141→    if (sourceRef.current) { try { sourceRef.current.stop() } catch {} }
   142→    await new Promise<void>((resolve) => {
   143→      const source = ctx.createBufferSource()
   144→      source.buffer = audioBuffer
   145→      source.connect(ctx.destination)
   146→      sourceRef.current = source
   147→      source.onended = () => resolve()
   148→      try { source.start() } catch { resolve() }
   149→    })
   150→  }, [unlockAudio])
   151→
   152→  /* ---------- SPEAK (multi-layer TTS — Kokoro removed for latency) ---------- */
   153→  const speak = useCallback(async (text: string): Promise<void> => {
   154→    // Bug C: no-op when overlay closed.
   155→    if (!openRef.current) return
   156→    // Layer 1: Edge neural TTS via our API (server-side Microsoft neural voices)
   157→    try {
   158→      setStateSafe('speaking')
   159→      const r = await safeJsonFetch('/api/tts', {
   160→        method: 'POST',
   161→        headers: { 'Content-Type': 'application/json' },
   162→        body: JSON.stringify({ text, voice: ttsVoice, speed: 1.0 }),
   163→        signal: controllerRef.current?.signal, // Bug K: cancel on close
   164→      }, { timeoutMs: 30_000, label: 'Voice synthesis' })
   165→      if (!r.ok) throw new Error(r.error || 'TTS failed')
   166→      // r.data is { audio: base64, format } OR raw audio blob?
   167→      // /api/tts returns audio bytes directly — re-fetch as blob if JSON.
   168→      // Handle both: if JSON with audio field, decode; else fetch as blob.
   169→      let arrayBuffer: ArrayBuffer
   170→      if (r.data && typeof r.data === 'object' && r.data.audio) {
   171→        const b64 = r.data.audio
   172→        const byteStr = atob(b64)
   173→        const bytes = new Uint8Array(byteStr.length)
   174→        for (let i = 0; i < byteStr.length; i++) bytes[i] = byteStr.charCodeAt(i)
   175→        arrayBuffer = bytes.buffer
   176→      } else {
   177→        // Fallback: fetch as blob directly
   178→        const blobRes = await fetch('/api/tts', {
   179→          method: 'POST',
   180→          headers: { 'Content-Type': 'application/json' },
   181→          body: JSON.stringify({ text, voice: ttsVoice, speed: 1.0 }),
   182→        })
   183→        const blob = await blobRes.blob()
   184→        arrayBuffer = await blob.arrayBuffer()
   185→      }
   186→      await playWebAudio(arrayBuffer)
   187→    } catch {
   188→      // Layer 2: browser speechSynthesis (works in restricted contexts)
   189→      try {
   190→        await new Promise<void>((resolve) => {
   191→          const utter = new SpeechSynthesisUtterance(text)
   192→          utter.lang = lang
   193→          utter.onend = () => resolve()
   194→          utter.onerror = () => resolve()
   195→          speechSynthesis.speak(utter)
   196→          setTimeout(resolve, Math.min(30000, text.length * 100))
   197→        })
   198→      } catch {}
   199→    }
   200→  }, [ttsVoice, lang, playWebAudio, setStateSafe])
   201→
   202→  /* ---------- THINK (one LLM turn via /api/voice/turn) ---------- */
   203→  // Bug A: returns the full server payload so the client can play server-generated
   204→  // TTS audio directly and skip the redundant client-side speak() pass.
   205→  const think = useCallback(async (userText: string): Promise<{
   206→    reply: string
   207→    audio: string | null
   208→    audioFormat: string
   209→  }> => {
   210→    // Bug C: no-op when overlay closed.
   211→    if (!openRef.current) return { reply: '', audio: null, audioFormat: 'wav' }
   212→    const r = await safeJsonFetch<any>('/api/voice/turn', {
   213→      method: 'POST',
   214→      headers: { 'Content-Type': 'application/json' },
   215→      body: JSON.stringify({
   216→        message: userText,
   217→        history: historyRef.current.slice(-6),
   218→        language: lang.split('-')[0],
   219→        voice: ttsVoice,
   220→        audio: 'UklGRiQAAABXQVZFZm10IBAAAAABAAEAQB8AAEAfAAABAAgAZGF0YQAAAAA=',
   221→      }),
   222→      signal: controllerRef.current?.signal, // Bug K: cancel on close
   223→    }, { timeoutMs: 60_000, label: 'AI reply' })
   224→    if (!r.ok || !r.data?.reply) throw new Error(r.error || 'AI had trouble responding. Try again.')
   225→    return {
   226→      reply: r.data.reply,
   227→      audio: r.data.audio ?? null,
   228→      audioFormat: r.data.audioFormat ?? 'wav',
   229→    }
   230→  }, [lang, ttsVoice])
   231→
   232→  /* ---------- HANDLE FINAL TRANSCRIPT ---------- */
   233→  const handleUtterance = useCallback(async (text: string) => {
   234→    // Bug C: no-op when overlay closed (also stops the think/speak chain).
   235→    if (!openRef.current) return
   236→    const clean = text.trim()
   237→    if (!clean) return
   238→    stopRecognition()
   239→    setStateSafe('thinking')
   240→    setInterim('')
   241→    setError('')
   242→    // Bug K: fresh AbortController for this turn so close/stop can cancel in-flight work.
   243→    controllerRef.current = new AbortController()
   244→    try {
   245→      historyRef.current.push({ role: 'user', content: clean })
   246→      // Bug A: think() returns the full { reply, audio, audioFormat } payload.
   247→      const { reply, audio } = await think(clean)
   248→      if (!reply) return // closed mid-flight
   249→      historyRef.current.push({ role: 'assistant', content: reply })
   250→      setTurns(prev => [...prev, { id: crypto.randomUUID(), user: clean, reply }])
   251→      if (audio) {
   252→        // Bug A: server already generated TTS audio — play it directly and SKIP speak().
   253→        setStateSafe('speaking')
   254→        const byteStr = atob(audio)
   255→        const bytes = new Uint8Array(byteStr.length)
   256→        for (let i = 0; i < byteStr.length; i++) bytes[i] = byteStr.charCodeAt(i)
   257→        await playWebAudio(bytes.buffer)
   258→      } else {
   259→        // No server audio — fall back to client-side TTS pipeline.
   260→        await speak(reply)
   261→      }
   262→    } catch (e) {
   263→      if ((e as Error)?.name === 'AbortError') return // Bug K: silent cancel
   264→      setError(e instanceof Error ? e.message : 'Something went wrong')
   265→      toast({ title: 'Voice error', description: 'Could not get a response.', variant: 'destructive' })
   266→    } finally {
   267→      startListening()
   268→    }
   269→    // NOTE: startListening is intentionally omitted from deps — it is declared
   270→    // below this callback (cycle: startListening → handleUtterance → startListening).
   271→    // Matching the original pattern keeps the closure stable without reordering.
   272→  }, [think, speak, playWebAudio, toast, setStateSafe, stopRecognition])
   273→
   274→  /* ---------- ASR FALLBACK (record + upload) ---------- */
   275→  const startAsrFallback = useCallback(() => {
   276→    try {
   277→      startAsrMode({ lang, ttsVoice, historyRef, setStateSafe, setTurns, setInterim, setError })
   278→    } catch {
   279→      setError('Voice is not available in this browser.')
   280→      setStateSafe('idle')
   281→    }
   282→  }, [lang, ttsVoice, setStateSafe])
   283→
   284→  /* ---------- LISTEN (Web Speech API) ---------- */
   285→  const startListening = useCallback(() => {
   286→    // Bug C: no-op when overlay closed.
   287→    if (!openRef.current) return
   288→    const SR = (window as any).SpeechRecognition || (window as any).webkitSpeechRecognition
   289→    if (!SR) {
   290→      // No Web Speech — go straight to ASR fallback
   291→      startAsrFallback()
   292→      return
   293→    }
   294→    stopSpeaking()
   295→    const rec = new SR()
   296→    rec.lang = lang
   297→    rec.continuous = true
   298→    rec.interimResults = true
   299→    rec.onresult = (event: any) => {
   300→      let finalTranscript = ''
   301→      let interimText = ''
   302→      for (let i = event.resultIndex; i < event.results.length; i++) {
   303→        const r = event.results[i]
   304→        if (r.isFinal) finalTranscript += r[0].transcript
   305→        else interimText += r[0].transcript
   306→      }
   307→      if (interimText) {
   308→        setInterim(interimText)
   309→        setMicLevel(0.3) // Bug D: fixed pulse level, no Math.random() jitter
   310→      }
   311→      if (finalTranscript.trim()) handleUtterance(finalTranscript)
   312→    }
   313→    rec.onerror = (event: any) => {
   314→      if (event.error === 'not-allowed') {
   315→        const inIframe = window.self !== window.top
   316→        setError(inIframe
   317→          ? 'This preview frame blocks the microphone. Open the app in a new tab for voice.'
   318→          : 'Microphone blocked — allow mic access in your browser settings.')
   319→        if (inIframe) setIframeBlocked(true)
   320→        setStateSafe('idle')
   321→      } else if (['audio-capture', 'service-not-allowed', 'network'].includes(event.error)) {
   322→        stopRecognition()
   323→        setStateSafe('idle')
   324→        setTimeout(() => startAsrFallback(), 50)
   325→      }
   326→    }
   327→    rec.onend = () => {
   328→      // Bug E: never auto-restart after a manual stop.
   329→      if (manualStopRef.current) return
   330→      if (stateRef.current !== 'listening' || !recognitionRef.current) return
   331→      // Count only consecutive IMMEDIATE restarts (gap < 1s between start and onend).
   332→      const elapsed = performance.now() - lastStartRef.current
   333→      if (elapsed < 1000) {
   334→        restartCountRef.current += 1
   335→        // After 10 immediate restarts, give up entirely.
   336→        if (restartCountRef.current >= 10) {
   337→          stopRecognition()
   338→          setStateSafe('idle')
   339→          toast({
   340→            title: 'Voice recognition unavailable',
   341→            description: 'Try the ASR fallback',
   342→            variant: 'destructive',
   343→          })
   344→          return
   345→        }
   346→        // After 5 immediate restarts, switch to the proven record/upload fallback.
   347→        if (restartCountRef.current >= 5) {
   348→          stopRecognition()
   349→          setStateSafe('idle')
   350→          setTimeout(() => startAsrFallback(), 50)
   351→          return
   352→        }
   353→      } else {
   354→        restartCountRef.current = 0
   355→      }
   356→      try {
   357→        recognitionRef.current.start()
   358→        lastStartRef.current = performance.now()
   359→      } catch {}
   360→    }
   361→    recognitionRef.current = rec
   362→    restartCountRef.current = 0
   363→    manualStopRef.current = false
   364→    try {
   365→      rec.start()
   366→      lastStartRef.current = performance.now()
   367→      setStateSafe('listening')
   368→      setError('')
   369→      // Bug D: throttle pulse to 500ms (was 150ms) to cut re-renders ~3.3×.
   370→      const pulse = setInterval(() => setMicLevel(v => Math.max(0.05, v * 0.85)), 500)
   371→      ;(rec as any).__pulse = pulse
   372→    } catch {}
   373→  }, [lang, handleUtterance, stopSpeaking, setStateSafe, startAsrFallback, stopRecognition, toast])
   374→
   375→  /* ---------- CONTROLS ---------- */
   376→  const toggle = useCallback(() => {
   377→    // Bug C: no-op when overlay closed.
   378→    if (!openRef.current) return
   379→    const s = stateRef.current
   380→    if (s === 'idle') { unlockAudio(); startListening() }
   381→    else if (s === 'listening') { stopRecognition(); stopAsrMode(); setStateSafe('idle'); setInterim('') }
   382→    else if (s === 'speaking') { stopSpeaking(); stopAsrMode(); startListening() }
   383→    else { stopSpeaking(); stopRecognition(); stopAsrMode(); setStateSafe('idle') }
   384→  }, [unlockAudio, startListening, stopRecognition, stopSpeaking, setStateSafe])
   385→
   386→  const endConversation = useCallback(() => {
   387→    stopRecognition(); stopAsrMode(); stopSpeaking()
   388→    setStateSafe('idle'); setTurns([]); setInterim('')
   389→    historyRef.current = []
   390→  }, [stopRecognition, stopSpeaking, setStateSafe])
   391→
   392→  /* ---------- AUTO-SCROLL TRANSCRIPT ---------- */
   393→  useEffect(() => {
   394→    if (scrollRef.current) scrollRef.current.scrollTop = scrollRef.current.scrollHeight
   395→  }, [turns, interim])
   396→
   397→  /* ---------- CLEANUP ON CLOSE ---------- */
   398→  useEffect(() => {
   399→    if (!open) {
   400→      stopRecognition(); stopAsrMode(); stopSpeaking()
   401→      // Bug K: cancel any in-flight think()/speak() fetch.
   402→      controllerRef.current?.abort()
   403→      setStateSafe('idle')
   404→    }
   405→  }, [open])
   406→
   407→  // Bug I + Bug K: on unmount, close the AudioContext and abort any in-flight turn.
   408→  useEffect(() => () => {
   409→    stopRecognition(); stopSpeaking()
   410→    audioCtxRef.current?.close?.().catch(() => {})
   411→    audioCtxRef.current = null
   412→    controllerRef.current?.abort()
   413→    controllerRef.current = null
   414→  }, [stopRecognition, stopSpeaking])
   415→
   416→  const stateMeta: Record<VoiceState, { label: string; sub: string }> = {
   417→    idle: { label: 'Tap to talk', sub: 'Natural conversation — I hear you as you speak' },
   418→    listening: { label: 'Listening…', sub: 'Speak naturally — I will respond when you pause' },
   419→    thinking: { label: 'Thinking…', sub: 'Crafting your reply' },
   420→    speaking: { label: 'Speaking', sub: 'Tap to interrupt' },
   421→  }
   422→  const meta = stateMeta[state]
   423→
   424→  const voicesForLang = (code: string) => {
   425→    const langPrefix = code.split('-')[0]
   426→    const edge = EDGE_VOICES.filter(v => v.id.startsWith(langPrefix))
   427→    return [...NEXUS_VOICES, ...edge]
   428→  }
   429→
   430→  return (
   431→    <AnimatePresence>
   432→      {open && (
   433→        <motion.div
   434→          initial={{ opacity: 0 }}
   435→          animate={{ opacity: 1 }}
   436→          exit={{ opacity: 0 }}
   437→          transition={{ duration: 0.25 }}
   438→          className="fixed inset-0 z-50 flex flex-col bg-background"
   439→          role="dialog"
   440→          aria-modal="true"
   441→          aria-label="Nexus Voice Mode"
   442→        >
   443→          {/* Ambient gradient backdrop — Bug D: single static gradient (no blur, much cheaper) */}
   444→          <div className="pointer-events-none absolute inset-0 overflow-hidden" aria-hidden>
   445→            <div className="absolute inset-0 bg-gradient-to-br from-primary/15 to-rose-500/10" />
   446→          </div>
   447→
   448→          {/* Top bar */}
   449→          <div className="relative z-10 flex items-center justify-between px-5 pt-[max(1rem,env(safe-area-inset-top))] pb-3">
   450→            <div className="flex items-center gap-2">
   451→              <Image src="/nexus-header-logo.png" alt="Nexus" width={96} height={32} className="h-7 w-auto" />
   452→              <span className="rounded-full bg-primary/10 px-2 py-0.5 text-[10px] font-semibold uppercase tracking-widest text-primary">Voice</span>
   453→            </div>
   454→            <button onClick={onClose} aria-label="Close voice mode"
   455→              className="flex h-10 w-10 items-center justify-center rounded-full bg-secondary/60 transition hover:bg-secondary">
   456→              <X className="h-5 w-5" />
   457→            </button>
   458→          </div>
   459→
   460→          {/* Voice picker sheet */}
   461→          <AnimatePresence>
   462→            {showVoicePicker && (
   463→              <motion.div
   464→                initial={{ opacity: 0, y: 20 }}
   465→                animate={{ opacity: 1, y: 0 }}
   466→                exit={{ opacity: 0, y: 20 }}
   467→                className="absolute inset-x-3 top-16 z-20 mx-auto max-w-md"
   468→              >
   469→                <div className="omni-scroll max-h-[60vh] overflow-y-auto rounded-3xl border border-border bg-card/95 p-4 shadow-2xl backdrop-blur-xl">
   470→                  <div className="mb-3 flex items-center justify-between">
   471→                    <p className="text-xs font-semibold uppercase tracking-widest text-muted-foreground">Premium voices</p>
   472→                    <button onClick={() => setShowVoicePicker(false)} className="text-muted-foreground hover:text-foreground"><X className="h-4 w-4" /></button>
   473→                  </div>
   474→                  <div className="space-y-4">
   475→                    <div>
   476→                      <p className="mb-2 text-[10px] font-medium uppercase tracking-widest text-muted-foreground/70">Nexus voices</p>

     1→'use client'
     2→
     3→import { useCallback, useEffect, useRef, useState } from 'react'
     4→import { motion, AnimatePresence } from 'framer-motion'
     5→import {
     6→  Calendar,
     7→  CheckCircle2,
     8→  Cloud,
     9→  Globe,
    10→  Loader2,
    11→  Mail,
    12→  MessageSquare,
    13→  Plus,
    14→  ShieldCheck,
    15→  Trash2,
    16→  X,
    17→  XCircle,
    18→} from 'lucide-react'
    19→import { Button } from '@/components/ui/button'
    20→import { Input } from '@/components/ui/input'
    21→import { Label } from '@/components/ui/label'
    22→
    23→/* ------------------------------------------------------------------ */
    24→/* Types                                                               */
    25→/* ------------------------------------------------------------------ */
    26→
    27→interface EmailPreset {
    28→  id: string
    29→  label: string
    30→  imapHost: string
    31→  imapPort: number
    32→  smtpHost: string
    33→  smtpPort: number
    34→  smtpSecure: boolean
    35→  hint: string
    36→}
    37→
    38→interface EmailAccount {
    39→  id: string
    40→  label: string
    41→  email: string
    42→  imapHost: string
    43→  smtpHost: string
    44→  status: string // unverified | connected | error
    45→  statusMessage: string
    46→  createdAt: string
    47→}
    48→
    49→interface ConnectPanelProps {
    50→  open: boolean
    51→  onClose: () => void
    52→  /** Called when an account is added or removed — parent can refresh state */
    53→  onAccountsChange?: () => void
    54→}
    55→
    56→const EMAIL_REGEX = /^[^@\s]+@[^@\s]+\.[^@\s]+$/
    57→
    58→/* The "coming soon" apps shown alongside Email in the Connect grid */
    59→const COMING_SOON_APPS = [
    60→  { id: 'calendar', label: 'Calendar', icon: Calendar, desc: 'Google Calendar · iCloud' },
    61→  { id: 'drive', label: 'Cloud Drive', icon: Cloud, desc: 'Drive · Dropbox · OneDrive' },
    62→  { id: 'slack', label: 'Slack', icon: MessageSquare, desc: 'Team messaging' },
    63→  { id: 'web', label: 'Web', icon: Globe, desc: 'Live web search & reader' },
    64→]
    65→
    66→export function ConnectPanel({ open, onClose, onAccountsChange }: ConnectPanelProps) {
    67→  const [accounts, setAccounts] = useState<EmailAccount[]>([])
    68→  const [presets, setPresets] = useState<EmailPreset[]>([])
    69→  const [loading, setLoading] = useState(false)
    70→  const [showAddForm, setShowAddForm] = useState(false)
    71→
    72→  // Form state
    73→  const [presetId, setPresetId] = useState<string>('gmail')
    74→  const [label, setLabel] = useState('')
    75→  const [email, setEmail] = useState('')
    76→  const [fromName, setFromName] = useState('')
    77→  const [username, setUsername] = useState('')
    78→  const [password, setPassword] = useState('')
    79→  const [imapHost, setImapHost] = useState('imap.gmail.com')
    80→  const [imapPort, setImapPort] = useState(993)
    81→  const [smtpHost, setSmtpHost] = useState('smtp.gmail.com')
    82→  const [smtpPort, setSmtpPort] = useState(465)
    83→  const [smtpSecure, setSmtpSecure] = useState(true)
    84→  const [submitting, setSubmitting] = useState(false)
    85→  const [formError, setFormError] = useState<string | null>(null)
    86→  const [formSuccess, setFormSuccess] = useState<string | null>(null)
    87→  const [deletingId, setDeletingId] = useState<string | null>(null)
    88→
    89→  const firstFieldRef = useRef<HTMLInputElement | null>(null)
    90→
    91→  /* ---------------- load accounts + presets ---------------- */
    92→  const loadAccounts = useCallback(async () => {
    93→    setLoading(true)
    94→    try {
    95→      const res = await fetch('/api/email/accounts')
    96→      const data = await res.json()
    97→      if (res.ok) {
    98→        setAccounts(data.accounts ?? [])
    99→        setPresets(data.presets ?? [])
   100→      }
   101→    } catch {
   102→      /* best-effort */
   103→    } finally {
   104→      setLoading(false)
   105→    }
   106→  }, [])
   107→
   108→  useEffect(() => {
   109→    if (open) {
   110→      loadAccounts()
   111→      setFormError(null)
   112→      setFormSuccess(null)
   113→      setShowAddForm(false)
   114→    }
   115→  }, [open, loadAccounts])
   116→
   117→  /* ---------------- escape + focus ---------------- */
   118→  useEffect(() => {
   119→    if (!open) return
   120→    const onKey = (e: KeyboardEvent) => {
   121→      if (e.key === 'Escape') onClose()
   122→    }
   123→    window.addEventListener('keydown', onKey)
   124→    return () => window.removeEventListener('keydown', onKey)
   125→  }, [open, onClose])
   126→
   127→  useEffect(() => {
   128→    if (open && showAddForm && firstFieldRef.current) {
   129→      const id = window.setTimeout(() => firstFieldRef.current?.focus(), 30)
   130→      return () => window.clearTimeout(id)
   131→    }
   132→  }, [open, showAddForm])
   133→
   134→  /* ---------------- preset selection ---------------- */
   135→  const applyPreset = useCallback((id: string) => {
   136→    setPresetId(id)
   137→    setFormError(null)
   138→    setFormSuccess(null)
   139→    const p = presets.find((x) => x.id === id)
   140→    if (!p || id === 'custom') return
   141→    setImapHost(p.imapHost)
   142→    setImapPort(p.imapPort)
   143→    setSmtpHost(p.smtpHost)
   144→    setSmtpPort(p.smtpPort)
   145→    setSmtpSecure(p.smtpSecure)
   146→    // Auto-derive label + username from email if empty
   147→    setEmail((prev) => prev)
   148→    setUsername((prev) => prev || email)
   149→  }, [presets, email])
   150→
   151→  /* ---------------- submit (connect account) ---------------- */
   152→  const submit = useCallback(async () => {
   153→    setFormError(null)
   154→    setFormSuccess(null)
   155→    if (!label.trim()) return setFormError('Give this account a label (e.g. "Work Gmail").')
   156→    if (!EMAIL_REGEX.test(email)) return setFormError('Enter a valid email address.')
   157→    if (!username.trim()) return setFormError('Username is required (usually your full email).')
   158→    if (!password) return setFormError('Password / app password is required.')
   159→    if (presetId === 'custom') {
   160→      if (!imapHost || !imapPort || !smtpHost || !smtpPort) {
   161→        return setFormError('Fill in the IMAP and SMTP host/port for your custom provider.')
   162→      }
   163→    }
   164→    setSubmitting(true)
   165→    try {
   166→      const res = await fetch('/api/email/accounts', {
   167→        method: 'POST',
   168→        headers: { 'Content-Type': 'application/json' },
   169→        body: JSON.stringify({
   170→          label: label.trim(),
   171→          email: email.trim(),
   172→          fromName: fromName.trim(),
   173→          imapHost,
   174→          imapPort,
   175→          smtpHost,
   176→          smtpPort,
   177→          smtpSecure,
   178→          username: username.trim(),
   179→          password,
   180→        }),
   181→      })
   182→      const data = await res.json()
   183→      if (!res.ok) {
   184→        return setFormError(data.error || 'Could not connect that account.')
   185→      }
   186→      const v = data.verification || {}
   187→      if (v.ok) {
   188→        setFormSuccess(v.message || 'Connected successfully.')
   189→        // Reset form + refresh list
   190→        setLabel('')
   191→        setFromName('')
   192→        setPassword('')
   193→        setShowAddForm(false)
   194→        await loadAccounts()
   195→        onAccountsChange?.()
   196→      } else {
   197→        setFormError(v.message || 'Verification failed — check your credentials.')
   198→      }
   199→    } catch (err: any) {
   200→      setFormError(err?.message || 'Network error — please try again.')
   201→    } finally {
   202→      setSubmitting(false)
   203→    }
   204→  }, [label, email, fromName, username, password, presetId, imapHost, imapPort, smtpHost, smtpPort, smtpSecure, loadAccounts, onAccountsChange])
   205→
   206→  /* ---------------- delete account ---------------- */
   207→  const removeAccount = useCallback(async (id: string) => {
   208→    setDeletingId(id)
   209→    try {
   210→      const res = await fetch(`/api/email/accounts/${id}`, { method: 'DELETE' })
   211→      if (res.ok) {
   212→        await loadAccounts()
   213→        onAccountsChange?.()
   214→      }
   215→    } catch {
   216→      /* best-effort */
   217→    } finally {
   218→      setDeletingId(null)
   219→    }
   220→  }, [loadAccounts, onAccountsChange])
   221→
   222→  const selectedPreset = presets.find((p) => p.id === presetId)
   223→  const connectedCount = accounts.filter((a) => a.status === 'connected').length
   224→
   225→  return (
   226→    <AnimatePresence>
   227→      {open && (
   228→        <motion.div
   229→          className="fixed inset-0 z-50 flex items-end justify-center bg-black/50 p-0 sm:items-center sm:p-4"
   230→          initial={{ opacity: 0 }}
   231→          animate={{ opacity: 1 }}
   232→          exit={{ opacity: 0 }}
   233→          onClick={onClose}
   234→        >
   235→          <motion.div
   236→            initial={{ y: '100%', opacity: 0 }}
   237→            animate={{ y: 0, opacity: 1 }}
   238→            exit={{ y: '100%', opacity: 0 }}
   239→            transition={{ type: 'spring', damping: 30, stiffness: 300 }}
   240→            onClick={(e) => e.stopPropagation()}
   241→            className="omni-scroll flex max-h-[92vh] w-full flex-col overflow-y-auto rounded-t-3xl border border-border bg-background sm:max-w-lg sm:rounded-3xl"
   242→            role="dialog"
   243→            aria-modal="true"
   244→            aria-label="Connect your apps and email"
   245→          >
   246→            {/* Header */}
   247→            <div className="sticky top-0 z-10 flex items-center justify-between border-b border-border bg-background/95 px-5 py-4 backdrop-blur">
   248→              <div className="flex items-center gap-2.5">
   249→                <span className="flex h-8 w-8 items-center justify-center rounded-full bg-gradient-to-br from-orange-500 to-rose-500 text-white">
   250→                  <Plus className="h-4 w-4" aria-hidden />
   251→                </span>
   252→                <div>
   253→                  <h2 className="text-base font-semibold">Connect</h2>
   254→                  <p className="text-[11px] text-muted-foreground">
   255→                    Link your email and apps to Nexus
   256→                  </p>
   257→                </div>
   258→              </div>
   259→              <button
   260→                onClick={onClose}
   261→                aria-label="Close"
   262→                className="flex h-8 w-8 items-center justify-center rounded-full text-muted-foreground transition hover:bg-secondary"
   263→              >
   264→                <X className="h-4 w-4" />
   265→              </button>
   266→            </div>
   267→
   268→            <div className="px-5 py-5">
   269→              {/* Connected apps grid */}
   270→              <section>
   271→                <h3 className="mb-2.5 text-[10px] font-semibold uppercase tracking-widest text-muted-foreground/60">
   272→                  Apps
   273→                </h3>
   274→                <div className="grid grid-cols-2 gap-2.5 sm:grid-cols-3">
   275→                  {/* Email — the active connector */}
   276→                  <div
   277→                    className={`relative flex flex-col items-start gap-2 overflow-hidden rounded-2xl border p-3 ${
   278→                      connectedCount > 0
   279→                        ? 'border-emerald-500/40 bg-emerald-500/5'
   280→                        : 'border-border bg-card'
   281→                    }`}
   282→                  >
   283→                    <span className="flex h-9 w-9 items-center justify-center rounded-xl bg-gradient-to-br from-orange-500 to-amber-500 text-white shadow-sm">
   284→                      <Mail className="h-4 w-4" aria-hidden />
   285→                    </span>
   286→                    <div>
   287→                      <p className="text-xs font-semibold">Email</p>
   288→                      <p className="text-[10px] text-muted-foreground">
   289→                        {connectedCount > 0 ? `${connectedCount} connected` : 'Not connected'}
   290→                      </p>
   291→                    </div>
   292→                    {connectedCount > 0 && (
   293→                      <span className="absolute right-2 top-2 flex h-4 w-4 items-center justify-center rounded-full bg-emerald-500 text-white">
   294→                        <CheckCircle2 className="h-2.5 w-2.5" aria-hidden />
   295→                      </span>
   296→                    )}
   297→                  </div>
   298→                  {/* Coming-soon apps */}
   299→                  {COMING_SOON_APPS.map((app) => {
   300→                    const Icon = app.icon
   301→                    return (
   302→                      <div
   303→                        key={app.id}
   304→                        className="relative flex flex-col items-start gap-2 overflow-hidden rounded-2xl border border-dashed border-border/60 bg-secondary/20 p-3 opacity-60"
   305→                      >
   306→                        <span className="flex h-9 w-9 items-center justify-center rounded-xl bg-secondary text-muted-foreground">
   307→                          <Icon className="h-4 w-4" aria-hidden />
   308→                        </span>
   309→                        <div>
   310→                          <p className="text-xs font-semibold">{app.label}</p>
   311→                          <p className="text-[10px] text-muted-foreground">Soon</p>
   312→                        </div>
   313→                      </div>
   314→                    )
   315→                  })}
   316→                </div>
   317→              </section>
   318→
   319→              {/* Email accounts list */}
   320→              <section className="mt-6">
   321→                <div className="mb-2.5 flex items-center justify-between">
   322→                  <h3 className="text-[10px] font-semibold uppercase tracking-widest text-muted-foreground/60">
   323→                    Email accounts
   324→                  </h3>
   325→                  <button
   326→                    onClick={() => {
   327→                      setShowAddForm((o) => !o)
   328→                      setFormError(null)
   329→                      setFormSuccess(null)
   330→                    }}
   331→                    className="flex items-center gap-1 rounded-full bg-primary px-2.5 py-1 text-[11px] font-medium text-primary-foreground transition hover:brightness-110"
   332→                  >
   333→                    <Plus className="h-3 w-3" aria-hidden />
   334→                    {showAddForm ? 'Cancel' : 'Add account'}
   335→                  </button>
   336→                </div>
   337→
   338→                {loading ? (
   339→                  <div className="flex items-center gap-2 rounded-xl border border-border bg-card px-4 py-4 text-xs text-muted-foreground">
   340→                    <Loader2 className="h-3.5 w-3.5 animate-spin" /> Loading…
   341→                  </div>
   342→                ) : accounts.length === 0 ? (
   343→                  <div className="rounded-xl border border-dashed border-border/70 bg-card/40 px-4 py-6 text-center">
   344→                    <Mail className="mx-auto mb-2 h-6 w-6 text-muted-foreground/40" aria-hidden />
   345→                    <p className="text-xs text-muted-foreground">
   346→                      No email accounts yet. Add one to send and read email from Nexus.
   347→                    </p>
   348→                  </div>
   349→                ) : (
   350→                  <ul className="space-y-2">
   351→                    {accounts.map((a) => {
   352→                      const ok = a.status === 'connected'
   353→                      return (
   354→                        <li
   355→                          key={a.id}
   356→                          className="flex items-start gap-3 rounded-xl border border-border bg-card px-3.5 py-3"
   357→                        >
   358→                          <span
   359→                            className={`mt-0.5 flex h-7 w-7 shrink-0 items-center justify-center rounded-full ${
   360→                              ok ? 'bg-emerald-500/15 text-emerald-600' : 'bg-destructive/15 text-destructive'
   361→                            }`}
   362→                          >
   363→                            {ok ? <CheckCircle2 className="h-3.5 w-3.5" /> : <XCircle className="h-3.5 w-3.5" />}
   364→                          </span>
   365→                          <div className="min-w-0 flex-1">
   366→                            <p className="truncate text-sm font-medium">{a.label}</p>
   367→                            <p className="truncate text-xs text-muted-foreground">{a.email}</p>
   368→                            <p className="mt-0.5 truncate text-[10px] text-muted-foreground/70">
   369→                              IMAP {a.imapHost} · SMTP {a.smtpHost}
   370→                            </p>
   371→                            {!ok && a.statusMessage && (
   372→                              <p className="mt-1 text-[10px] text-destructive/90">{a.statusMessage}</p>
   373→                            )}
   374→                          </div>
   375→                          <button
   376→                            onClick={() => removeAccount(a.id)}
   377→                            disabled={deletingId === a.id}
   378→                            aria-label="Remove account"
   379→                            className="flex h-7 w-7 shrink-0 items-center justify-center rounded-full text-muted-foreground transition hover:bg-destructive/10 hover:text-destructive disabled:opacity-50"
   380→                          >
   381→                            {deletingId === a.id ? (
   382→                              <Loader2 className="h-3.5 w-3.5 animate-spin" />
   383→                            ) : (
   384→                              <Trash2 className="h-3.5 w-3.5" />
   385→                            )}
   386→                          </button>
   387→                        </li>
   388→                      )
   389→                    })}
   390→                  </ul>
   391→                )}
   392→              </section>
   393→
   394→              {/* Add-account form (collapsible) */}
   395→              <AnimatePresence initial={false}>
   396→                {showAddForm && (
   397→                  <motion.section
   398→                    initial={{ height: 0, opacity: 0 }}
   399→                    animate={{ height: 'auto', opacity: 1 }}
   400→                    exit={{ height: 0, opacity: 0 }}
   401→                    className="overflow-hidden"
   402→                  >
   403→                    <div className="mt-4 rounded-2xl border border-border bg-card p-4">
   404→                      <h4 className="mb-3 flex items-center gap-2 text-sm font-semibold">
   405→                        <Mail className="h-4 w-4 text-primary" aria-hidden /> Connect an email account
   406→                      </h4>
   407→
   408→                      {/* Provider preset picker */}
   409→                      <Label className="mb-1.5 block text-[11px] text-muted-foreground">Provider</Label>
   410→                      <div className="mb-3 flex flex-wrap gap-1.5">
   411→                        {presets.map((p) => (
   412→                          <button
   413→                            key={p.id}
   414→                            type="button"
   415→                            onClick={() => applyPreset(p.id)}
   416→                            className={`rounded-full border px-3 py-1.5 text-xs font-medium transition ${
   417→                              presetId === p.id
   418→                                ? 'border-primary bg-primary/10 text-primary'
   419→                                : 'border-border bg-card text-muted-foreground hover:border-primary/40'
   420→                            }`}
   421→                          >
   422→                            {p.label}
   423→                          </button>
   424→                        ))}
   425→                      </div>
   426→
   427→                      {selectedPreset?.hint && (
   428→                        <p className="mb-3 flex items-start gap-1.5 rounded-lg bg-amber-500/10 px-3 py-2 text-[11px] text-amber-700">
   429→                          <ShieldCheck className="mt-0.5 h-3 w-3 shrink-0" aria-hidden />
   430→                          <span>{selectedPreset.hint}</span>
   431→                        </p>
   432→                      )}
   433→
   434→                      <div className="space-y-3">
   435→                        <Field>
   436→                          <Label htmlFor="acct-label" className="mb-1.5 block text-[11px] text-muted-foreground">
   437→                            Label
   438→                          </Label>
   439→                          <Input
   440→                            id="acct-label"
   441→                            ref={firstFieldRef}
   442→                            value={label}
   443→                            onChange={(e) => setLabel(e.target.value)}
   444→                            placeholder="Work Gmail"
   445→                            className="h-10"
   446→                          />
   447→                        </Field>
   448→                        <Field>
   449→                          <Label htmlFor="acct-email" className="mb-1.5 block text-[11px] text-muted-foreground">
   450→                            Email address
   451→                          </Label>
   452→                          <Input
   453→                            id="acct-email"
   454→                            type="email"
   455→                            value={email}
   456→                            onChange={(e) => {
   457→                              setEmail(e.target.value)
   458→                              if (!username || username === email) setUsername(e.target.value)
   459→                            }}
   460→                            placeholder="you@gmail.com"
   461→                            className="h-10"
   462→                          />
   463→                        </Field>
   464→                        <Field>
   465→                          <Label htmlFor="acct-fromname" className="mb-1.5 block text-[11px] text-muted-foreground">
   466→                            From name <span className="text-muted-foreground/60">(optional)</span>
   467→                          </Label>
   468→                          <Input
   469→                            id="acct-fromname"
   470→                            value={fromName}
   471→                            onChange={(e) => setFromName(e.target.value)}
   472→                            placeholder="Your name"
   473→                            className="h-10"
   474→                          />
   475→                        </Field>
   476→                        <div className="grid grid-cols-2 gap-3">
   477→                          <Field>
   478→                            <Label htmlFor="acct-user" className="mb-1.5 block text-[11px] text-muted-foreground">
   479→                              Username
   480→                            </Label>
   481→                            <Input
   482→                              id="acct-user"
   483→                              value={username}
   484→                              onChange={(e) => setUsername(e.target.value)}
   485→                              placeholder="you@gmail.com"
   486→                              className="h-10"
   487→                            />
   488→                          </Field>
   489→                          <Field>
   490→                            <Label htmlFor="acct-pass" className="mb-1.5 block text-[11px] text-muted-foreground">
   491→                              Password / App password
   492→                            </Label>
   493→                            <Input
   494→                              id="acct-pass"
   495→                              type="password"
   496→                              value={password}
   497→                              onChange={(e) => setPassword(e.target.value)}
   498→                              placeholder="••••••••"
   499→                              className="h-10"
   500→                            />
   501→                          </Field>
   502→                        </div>
   503→
   504→                        {/* Custom provider — show IMAP/SMTP fields */}
   505→                        {presetId === 'custom' && (
   506→                          <div className="rounded-lg border border-border/60 bg-secondary/30 p-3">
   507→                            <p className="mb-2 text-[11px] font-medium text-muted-foreground">
   508→                              Custom IMAP / SMTP settings
   509→                            </p>
   510→                            <div className="grid grid-cols-2 gap-2.5">
   511→                              <Field>
   512→                                <Label className="mb-1 block text-[10px] text-muted-foreground">IMAP host</Label>
   513→                                <Input value={imapHost} onChange={(e) => setImapHost(e.target.value)} className="h-9 text-xs" />
   514→                              </Field>
   515→                              <Field>
   516→                                <Label className="mb-1 block text-[10px] text-muted-foreground">IMAP port</Label>
   517→                                <Input
   518→                                  type="number"
   519→                                  value={imapPort}
   520→                                  onChange={(e) => setImapPort(parseInt(e.target.value || '993', 10))}
   521→                                  className="h-9 text-xs"
   522→                                />
   523→                              </Field>
   524→                              <Field>
   525→                                <Label className="mb-1 block text-[10px] text-muted-foreground">SMTP host</Label>
   526→                                <Input value={smtpHost} onChange={(e) => setSmtpHost(e.target.value)} className="h-9 text-xs" />
   527→                              </Field>
   528→                              <Field>

   583→              {/* Help footer */}

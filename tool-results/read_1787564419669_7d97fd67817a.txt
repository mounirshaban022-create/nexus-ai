   125→    setMessages(prev => [...prev, { id: `u-${Date.now()}`, role: 'user', content: msg }])

   128→    try {
   129→      // If a tool is pending, route to the tool engine
   130→      if (toolEngine.pendingTool) {
   131→        await toolEngine.execute(msg)
   132→        return

   134→      // Otherwise: plain chat stream
   135→      const res = await fetch('/api/chat', {
   136→        method: 'POST',
   137→        headers: { 'Content-Type': 'application/json' },
   138→        body: JSON.stringify({ message: msg }),
   139→      })
   140→      const reader = res.body!.getReader()
   141→      const decoder = new TextDecoder()
   142→      let buf = ''
   143→      for (;;) {
   144→        const { done, value } = await reader.read()
   145→        if (done) break
   146→        buf += decoder.decode(value, { stream: true })

   148→        buf = lines.pop() ?? ''
   149→        for (const line of lines) {
   150→          if (!line.trim()) continue

   152→            const e = JSON.parse(line)
   153→            if (e.type === 'assistant') {
   154→              setMessages(prev => [...prev, { id: `a-${Date.now()}`, role: 'assistant', content: e.content, attachments: e.attachments }])
   155→            }
   156→          } catch {}
   157→        }
   158→      }
   159→    } catch (err: any) {
   160→      setMessages(prev => [...prev, { id: `a-${Date.now()}`, role: 'assistant', content: `⚠️ ${err.message || 'Something went wrong.'}` }])
   161→    } finally {
   162→      setSending(false)
   163→    }

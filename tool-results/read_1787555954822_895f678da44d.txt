     8→/* ------------------------------------------------------------------ */
     9→/* Safe math expression evaluator (recursive descent — never eval())   */
    10→/* ------------------------------------------------------------------ */
    11→
    12→class MathParser {
    13→  private pos = 0
    14→
    15→  constructor(private readonly input: string) {}
    16→
    17→  parse(): number {
    18→    this.skipWs()
    19→    const value = this.parseExpr()
    20→    this.skipWs()
    21→    if (this.pos < this.input.length) throw new Error(`Unexpected character at position ${this.pos}`)
    22→    if (!Number.isFinite(value)) throw new Error('Result is not a finite number')
    23→    return value
    24→  }
    25→
    26→  private skipWs() {
    27→    while (this.pos < this.input.length && /\s/.test(this.input[this.pos])) this.pos++
    28→  }
    29→
    30→  private peek(): string {
    31→    return this.input[this.pos] ?? ''
    32→  }
    33→
    34→  private parseExpr(): number {
    35→    let left = this.parseTerm()
    36→    for (;;) {
    37→      this.skipWs()
    38→      const op = this.peek()
    39→      if (op === '+' || op === '-') {
    40→        this.pos++
    41→        const right = this.parseTerm()
    42→        left = op === '+' ? left + right : left - right
    43→      } else {
    44→        return left
    45→      }
    46→    }
    47→  }
    48→
    49→  private parseTerm(): number {
    50→    let left = this.parseFactor()
    51→    for (;;) {
    52→      this.skipWs()
    53→      const op = this.peek()
    54→      if (op === '*' || op === '/' || op === '%') {
    55→        this.pos++
    56→        const right = this.parseFactor()
    57→        if ((op === '/' || op === '%') && right === 0) throw new Error('Division by zero')
    58→        left = op === '*' ? left * right : op === '/' ? left / right : left % right
    59→      } else {
    60→        return left
    61→      }
    62→    }
    63→  }
    64→
    65→  private parseFactor(): number {
    66→    const base = this.parseUnary()
    67→    this.skipWs()
    68→    if (this.peek() === '^') {
    69→      this.pos++
    70→      const exponent = this.parseFactor()
    71→      return Math.pow(base, exponent)
    72→    }
    73→    return base
    74→  }
    75→
    76→  private parseUnary(): number {
    77→    this.skipWs()
    78→    if (this.peek() === '-') {
    79→      this.pos++
    80→      return -this.parseUnary()
    81→    }
    82→    if (this.peek() === '+') {
    83→      this.pos++
    84→      return this.parseUnary()
    85→    }
    86→    return this.parsePrimary()
    87→  }
    88→
    89→  private parsePrimary(): number {
    90→    this.skipWs()
    91→    const ch = this.peek()
    92→    if (ch === '(') {
    93→      this.pos++
    94→      const value = this.parseExpr()
    95→      this.skipWs()
    96→      if (this.peek() !== ')') throw new Error('Missing closing parenthesis')
    97→      this.pos++
    98→      return value
    99→    }
   100→    // number (int, decimal, scientific)
   101→    const numMatch = /^\d+(?:\.\d+)?(?:[eE][+-]?\d+)?/.exec(this.input.slice(this.pos))
   102→    if (!numMatch) throw new Error(`Expected a number at position ${this.pos}`)
   103→    this.pos += numMatch[0].length
   104→    return parseFloat(numMatch[0])
   105→  }
   106→}
   107→
   108→export function evaluateMathExpression(expression: string): number {
   109→  if (!/^[\d\s+\-*/%^().eE]+$/.test(expression)) {
   110→    throw new Error('Expression contains invalid characters')
   111→  }
   112→  if (expression.length > 200) throw new Error('Expression too long')
   113→  return new MathParser(expression).parse()
   114→}

   143→/** Blocks requests to private/internal networks (SSRF guard). */
   144→export function assertPublicUrl(rawUrl: string): URL {
   145→  let parsed: URL
   146→  try {
   147→    parsed = new URL(rawUrl)
   148→  } catch {
   149→    throw new Error('Invalid URL')
   150→  }
   151→  if (!/^https?:$/.test(parsed.protocol)) {
   152→    throw new Error('Only http(s) URLs are allowed')
   153→  }
   154→  const host = parsed.hostname.toLowerCase()
   155→  const privatePatterns = [
   156→    /^localhost$/i,
   157→    /^127\./,
   158→    /^10\./,
   159→    /^192\.168\./,
   160→    /^172\.(1[6-9]|2\d|3[01])\./,
   161→    /^169\.254\./,
   162→    /^0\./,
   163→    /^\[?::1\]?$/,
   164→    /^\[?fc00:/,
   165→    /^\[?fe80:/,
   166→    /^\[?fd/i,
   167→    /\.internal$/i,
   168→    /\.local$/i,
   169→  ]
   170→  if (privatePatterns.some((re) => re.test(host))) {
   171→    throw new Error('Requests to internal/private addresses are blocked')
   172→  }
   173→  return parsed
   174→}

   176→async function fetchJson(url: string, timeoutMs = 15000): Promise<unknown> {
   177→  const controller = new AbortController()
   178→  const timer = setTimeout(() => controller.abort(), timeoutMs)
   179→  try {
   180→    const res = await fetch(url, {
   181→      signal: controller.signal,
   182→      headers: { 'User-Agent': 'NEXUS-AI/1.0 (super-app connector)' },
   183→    })
   184→    if (!res.ok) throw new Error(`Upstream responded ${res.status}`)
   185→    return await res.json()
   186→  } finally {
   187→    clearTimeout(timer)
   188→  }
   189→}

  1351→const VALID_SIZES = ['1024x1024', '1344x768', '768x1344']

  1353→export function validateConnectorArgs(
  1354→  connector: ConnectorDefinition,
  1355→  args: Record<string, unknown>
  1356→): { ok: true; args: Record<string, unknown> } | { ok: false; error: string } {
  1357→  for (const param of connector.params) {
  1358→    const value = args[param.name]
  1359→    if (param.required && (value === undefined || value === null || String(value).trim() === '')) {
  1360→      return { ok: false, error: `Missing required parameter "${param.name}"` }
  1361→    }
  1362→  }
  1363→  // Reject unexpected parameters (strict schema — OWASP: validate expected input formats)
  1364→  const allowed = new Set(connector.params.map((p) => p.name))
  1365→  for (const key of Object.keys(args)) {
  1366→    if (!allowed.has(key)) return { ok: false, error: `Unknown parameter "${key}"` }
  1367→  }
  1368→  // Length guards
  1369→  for (const [key, value] of Object.entries(args)) {
  1370→    if (typeof value === 'string' && value.length > 3000) {
  1371→      return { ok: false, error: `Parameter "${key}" is too long` }
  1372→    }
  1373→  }
  1374→  if (connector.id === 'generate_image' && args.size && !VALID_SIZES.includes(String(args.size))) {
  1375→    return { ok: false, error: 'Invalid size' }
  1376→  }
  1377→  return { ok: true, args }
  1378→}

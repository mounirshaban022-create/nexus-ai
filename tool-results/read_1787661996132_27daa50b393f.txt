   124→export function NexusChat(props: NexusChatProps) {
   125→  const { pinnedAgent, onPinnedAgentChange, onSessionCreated } = props
   126→
   127→  const [messages, setMessages] = useState<NxMsg[]>([])
   128→  /** Mirror of `messages` for read-inside-callback checks (burst detection
   129→   *  reads the CURRENT bubble content inside the stream loop without
   130→   *  re-creating the send callback on every delta). */
   131→  const messagesRef = useRef<NxMsg[]>([])
   132→  useEffect(() => {
   133→    messagesRef.current = messages
   134→  }, [messages])
   135→  const [streaming, setStreaming] = useState(false)
   136→  const [sessionId, setSessionId] = useState<string | undefined>(props.sessionId)
   137→  const [input, setInput] = useState(props.prefill ?? '')
   138→  const [attach, setAttach] = useState<{ filename: string; dataUrl: string } | null>(null)
   139→  const [attachError, setAttachError] = useState('')
   140→  const [assign, setAssign] = useState<AgentAssignEvent | null>(null)
   141→
   142→  const scrollRef = useRef<HTMLDivElement>(null)
   143→  const textareaRef = useRef<HTMLTextAreaElement>(null)
   144→  const fileRef = useRef<HTMLInputElement>(null)
   145→  const stickToBottomRef = useRef(true)
   146→  const streamIdRef = useRef<string | null>(null)
   147→  const pendingDeltaRef = useRef('')
   148→  const flushTimerRef = useRef<ReturnType<typeof setInterval> | null>(null)
   149→  const abortRef = useRef<AbortController | null>(null)
   150→  const assignRef = useRef<AgentAssignEvent | null>(null)
   151→  const seqRef = useRef(0)
   152→  /** Live status line from the server ("Picking the right specialist…") —
   153→   *  shown next to the typing dots so waiting never feels dead. */
   154→  const [statusText, setStatusText] = useState('')
   155→  /** Typewriter reveal state — animates one-shot bursts word-by-word. */
   156→  const typewriterRef = useRef<{ timer: ReturnType<typeof setInterval> | null }>({ timer: null })
   157→  const scrollFrameRef = useRef<number | null>(null)
   158→
   159→  const nextId = (prefix: string) => `${prefix}-${Date.now()}-${seqRef.current++}`
   160→
   161→  /* ---------- session resume + one-time pin sync ---------- */
   162→  // The session to resume is captured at MOUNT: page.tsx remounts the chat
   163→  // (chatEpoch key) whenever the user switches sessions, and later
   164→  // props.sessionId changes (e.g. when THIS chat creates its own session)
   165→  // must NOT re-trigger a refetch that would clobber the live transcript
   166→  // (handoff pills, tool cards and attachments are stream-only state).
   167→  const [resumeId] = useState(props.sessionId)
   168→  useEffect(() => {
   169→    if (!resumeId) return
   170→    let cancelled = false
   171→    ;(async () => {
   172→      try {
   173→        const res = await fetch(`/api/chat/sessions/${encodeURIComponent(resumeId)}`)
   174→        if (!res.ok) return
   175→        const data = (await res.json().catch(() => null)) as { session?: unknown } | null
   176→        const s =
   177→          data && typeof data.session === 'object' ? (data.session as Record<string, unknown>) : null
   178→        if (cancelled || !s) return
   179→        const rowsRaw = Array.isArray(s.messages) ? s.messages : []
   180→        const sessionAgentSlug = typeof s.agentSlug === 'string' ? s.agentSlug : null
   181→        const rows: NxMsg[] = rowsRaw
   182→          .filter((m): m is Record<string, unknown> => !!m && typeof m === 'object')
   183→          .map((m, i): NxMsg => {
   184→            const role = m.role === 'user' ? 'user' : m.role === 'tool' ? 'tool' : 'assistant'
   185→            return {
   186→              id: typeof m.id === 'string' ? m.id : `m-${i}`,
   187→              role,
   188→              content: typeof m.content === 'string' ? m.content : '',
   189→              toolName: typeof m.toolName === 'string' ? m.toolName : null,
   190→              toolData: typeof m.toolData === 'string' ? m.toolData : null,
   191→              toolStatus: role === 'tool' ? 'done' : undefined,
   192→              agentSlug: role === 'assistant' ? sessionAgentSlug : undefined,
   193→              // NEXUS One: attachments persisted server-side (image/video/
   194→              // document/code cards) survive resume.
   195→              attachments: Array.isArray(m.attachments) ? m.attachments : undefined,
   196→            }
   197→          })
   198→        setMessages(rows)
   199→        setSessionId(typeof s.id === 'string' ? s.id : resumeId)
   200→        // Sync page-level pin state ONCE after the session loads.
   201→        onPinnedAgentChange(s.agentPinned === true && sessionAgentSlug ? sessionAgentSlug : null)
   202→      } catch {
   203→        /* failed to load — fall back to an empty transcript */
   204→      }
   205→    })()
   206→    return () => {
   207→      cancelled = true
   208→    }
   209→    // Mount-per-session resume (page.tsx remounts via chatEpoch); the pin
   210→    // sync must fire exactly once per loaded session.
   211→  }, [resumeId])
   212→
   213→  /* ---------- abort any live stream on unmount ---------- */
   214→  useEffect(() => {
   215→    return () => {
   216→      abortRef.current?.abort()
   217→      if (flushTimerRef.current) clearInterval(flushTimerRef.current)
   218→      if (typewriterRef.current.timer) clearInterval(typewriterRef.current.timer)
   219→      if (scrollFrameRef.current) cancelAnimationFrame(scrollFrameRef.current)
   220→    }
   221→  }, [])
   222→
   223→  /* ---------- smart auto-scroll (sticks while near the bottom) ----------
   224→   * SCROLL-GLITCH FIX — three hardening measures:
   225→   *   1. rAF throttling: the DOM height settles within one animation frame
   226→   *      before we pin to the bottom (no fighting mid-layout writes).
   227→   *   2. Hysteresis: stick again at ≤140px from the bottom, but only
   228→   *      un-stick beyond 260px — mobile rubber-banding at the bottom edge
   229→   *      no longer randomly detaches the auto-scroll.
   230→   *   3. `overflow-anchor:none` on .nx-scroll (globals.css) stops the
   231→   *      browser's own scroll-anchoring from fighting our scrollTop writes.
   232→   */
   233→  const scrollToEnd = useCallback(() => {
   234→    if (scrollFrameRef.current) cancelAnimationFrame(scrollFrameRef.current)
   235→    scrollFrameRef.current = requestAnimationFrame(() => {
   236→      scrollFrameRef.current = null
   237→      const el = scrollRef.current
   238→      if (!el || !stickToBottomRef.current) return
   239→      el.scrollTop = el.scrollHeight
   240→    })
   241→  }, [])
   242→
   243→  useEffect(() => {
   244→    if (!stickToBottomRef.current) return
   245→    scrollToEnd()
   246→  }, [messages, streaming, scrollToEnd])
   247→
   248→  /* Mobile keyboards + viewport resizes change the container height without
   249→   * a messages update — re-stick to the bottom so the newest content stays
   250→   * visible instead of being cut off under the composer. */
   251→  useEffect(() => {
   252→    const onResize = () => {
   253→      if (stickToBottomRef.current) scrollToEnd()
   254→    }
   255→    window.addEventListener('resize', onResize)
   256→    return () => window.removeEventListener('resize', onResize)
   257→  }, [scrollToEnd])
   258→
   259→  const onScroll = () => {
   260→    const el = scrollRef.current
   261→    if (!el) return
   262→    const distFromBottom = el.scrollHeight - el.scrollTop - el.clientHeight
   263→    // Hysteresis band: easily re-stick near the bottom, hard to un-stick.
   264→    if (distFromBottom <= 140) stickToBottomRef.current = true
   265→    else if (distFromBottom > 260) stickToBottomRef.current = false
   266→  }
   267→
   268→  /* ---------- textarea auto-resize (rows 1 → 6) ---------- */
   269→  useEffect(() => {
   270→    const el = textareaRef.current
   271→    if (!el) return
   272→    el.style.height = 'auto'
   273→    el.style.height = `${Math.min(el.scrollHeight, 144)}px`
   274→  }, [input])
   275→
   276→  /* ---------- streaming helpers ---------- */
   277→
   278→  const appendDelta = useCallback((delta: string) => {
   279→    const id = streamIdRef.current
   280→    if (!id || !delta) return
   281→    setMessages((prev) => prev.map((m) => (m.id === id ? { ...m, content: m.content + delta } : m)))
   282→  }, [])
   283→
   284→  /** Deltas are buffered and flushed every 80ms so react-markdown is not
   285→   * re-parsed on every single token — keeps long streams silky. */
   286→  const startDeltaFlusher = useCallback(() => {
   287→    if (flushTimerRef.current) return
   288→    flushTimerRef.current = setInterval(() => {
   289→      if (!pendingDeltaRef.current) return
   290→      const chunk = pendingDeltaRef.current
   291→      pendingDeltaRef.current = ''
   292→      appendDelta(chunk)
   293→    }, 80)
   294→  }, [appendDelta])
   295→
   296→  const stopDeltaFlusher = useCallback(() => {
   297→    if (flushTimerRef.current) {
   298→      clearInterval(flushTimerRef.current)
   299→      flushTimerRef.current = null
   300→    }
   301→    if (pendingDeltaRef.current) {
   302→      const chunk = pendingDeltaRef.current
   303→      pendingDeltaRef.current = ''
   304→      appendDelta(chunk)
   305→    }
   306→  }, [appendDelta])
   307→
   308→  /* ---------- typewriter reveal (the "alive" effect) ----------
   309→   * When the backend hands us a finished text in ONE burst (non-streaming
   310→   * fallback path, or a coalesced mega-delta), animating it word-by-word
   311→   * keeps the ChatGPT feel instead of a jarring wall of text. Reveal is
   312→   * chunked so even long answers finish in ~1.2s. */
   313→  const revealTypewriter = useCallback((id: string, fullText: string) => {
   314→    if (typewriterRef.current.timer) {
   315→      clearInterval(typewriterRef.current.timer)
   316→      typewriterRef.current.timer = null
   317→    }
   318→    if (!fullText) return
   319→    const total = fullText.length
   320→    // ~70 ticks × 16ms ≈ 1.1s regardless of length; min 2 chars/tick.
   321→    const chunk = Math.max(2, Math.ceil(total / 70))
   322→    let pos = 0
   323→    setMessages((prev) => prev.map((m) => (m.id === id ? { ...m, content: '' } : m)))
   324→    typewriterRef.current.timer = setInterval(() => {
   325→      pos = Math.min(total, pos + chunk)
   326→      const slice = fullText.slice(0, pos)
   327→      setMessages((prev) => prev.map((m) => (m.id === id ? { ...m, content: slice } : m)))
   328→      if (pos >= total) {
   329→        if (typewriterRef.current.timer) clearInterval(typewriterRef.current.timer)
   330→        typewriterRef.current.timer = null
   331→      }
   332→    }, 16)
   333→  }, [])
   334→
   335→  /** Patch the live tool chip that matches (toolName, index). */
   336→  const updateTool = useCallback(
   337→    (tool: string, index: number, patch: Partial<NxMsg>) => {
   338→      setMessages((prev) => {
   339→        let target = -1
   340→        for (let i = prev.length - 1; i >= 0; i--) {
   341→          const m = prev[i]
   342→          if (m.role === 'tool' && m.toolName === tool && m.toolIndex === index) {
   343→            target = i
   344→            break
   345→          }
   346→        }
   347→        if (target === -1) {
   348→          for (let i = prev.length - 1; i >= 0; i--) {
   349→            const m = prev[i]
   350→            if (m.role === 'tool' && m.toolName === tool && m.toolStatus === 'running') {
   351→              target = i
   352→              break
   353→            }
   354→          }
   355→        }
   356→        if (target === -1) return prev
   357→        const copy = [...prev]
   358→        copy[target] = { ...copy[target], ...patch }
   359→        return copy
   360→      })
   361→    },
   362→    []
   363→  )
   364→
   365→  /* ---------- send ---------- */
   366→
   367→  const send = useCallback(
   368→    async (text: string) => {
   369→      const trimmed = text.trim()
   370→      const hasAttach = attach != null
   371→      if (streaming) return
   372→      if (!trimmed && !hasAttach) return
   373→
   374→      const userMsg: NxMsg = {
   375→        id: nextId('local'),
   376→        role: 'user',
   377→        content: trimmed,
   378→        ...(hasAttach ? { attachName: attach!.filename } : {}),
   379→      }
   380→      setMessages((prev) => [...prev, userMsg])
   381→      setInput('')
   382→      setAttach(null)
   383→      setAttachError('')
   384→      setStreaming(true)
   385→      stickToBottomRef.current = true
   386→      setStatusText('')
   387→
   388→      abortRef.current = new AbortController()
   389→
   390→      // STREAM WATCHDOG (defense-in-depth): if the server ever fails to
   391→      // close the NDJSON stream, reader.read() hangs forever and the
   392→      // composer stays locked in "generating". This timer aborts the fetch
   393→      // after 90s of total silence and frees the UI. Declared OUTSIDE the
   394→      // try so the finally block can always disarm it.
   395→      let watchdog: ReturnType<typeof setTimeout> | null = null
   396→      const armWatchdog = () => {
   397→        if (watchdog) clearTimeout(watchdog)
   398→        watchdog = setTimeout(() => {
   399→          abortRef.current?.abort()
   400→        }, 90_000)
   401→      }
   402→      const disarmWatchdog = () => {
   403→        if (watchdog) clearTimeout(watchdog)
   404→        watchdog = null
   405→      }
   406→
   407→      try {
   408→        const res = await fetch('/api/chat', {
   409→          method: 'POST',
   410→          headers: { 'Content-Type': 'application/json' },
   411→          body: JSON.stringify({
   412→            message: trimmed,
   413→            ...(sessionId ? { sessionId } : {}),
   414→            // '' = auto-routing; a slug pins the session persona (idempotent).
   415→            agentSlug: pinnedAgent ?? '',
   416→            ...(hasAttach
   417→              ? { attachment: { filename: attach!.filename, dataUrl: attach!.dataUrl } }
   418→              : {}),
   419→          }),
   420→          signal: abortRef.current.signal,
   421→        })
   422→
   423→        if (!res.ok || !res.body) {
   424→          const data = (await res.json().catch(() => null)) as { error?: string } | null
   425→          throw new Error(data?.error || 'Chat request failed.')
   426→        }
   427→
   428→        // Consume the NDJSON stream.
   429→        const reader = res.body.getReader()
   430→        const decoder = new TextDecoder()
   431→        let buffer = ''
   432→        armWatchdog()
   433→
   434→        for (;;) {
   435→          const { done, value } = await reader.read()
   436→          if (done) break
   437→          armWatchdog()
   438→          buffer += decoder.decode(value, { stream: true })
   439→          const lines = buffer.split('\n')
   440→          buffer = lines.pop() ?? ''
   441→          for (const line of lines) {
   442→            const text = line.trim()
   443→            if (!text) continue
   444→            let event: ChatStreamEvent
   445→            try {
   446→              event = JSON.parse(text) as ChatStreamEvent
   447→            } catch {
   448→              continue // partial / malformed line — skip
   449→            }
   450→
   451→            switch (event.type) {
   452→              case 'user': {
   453→                setMessages((prev) =>
   454→                  prev.map((m) =>
   455→                    m.id === userMsg.id ? { ...m, id: event.id, content: event.content } : m
   456→                  )
   457→                )
   458→                break
   459→              }
   460→              case 'agent_assign': {
   461→                assignRef.current = event
   462→                setAssign(event)
   463→                setMessages((prev) => [
   464→                  ...prev,
   465→                  { id: nextId('assign'), role: 'agent', content: '', assign: event },
   466→                ])
   467→                break
   468→              }
   469→              case 'status': {
   470→                // Live server status line — displayed with the typing dots
   471→                // until the first delta lands.
   472→                setStatusText(event.message)
   473→                break
   474→              }
   475→              case 'assistant_start': {
   476→                streamIdRef.current = event.id
   477→                const slug = pinnedAgent ?? assignRef.current?.agentSlug ?? null
   478→                setMessages((prev) => [
   479→                  ...prev,
   480→                  { id: event.id, role: 'assistant', content: '', streaming: true, agentSlug: slug },
   481→                ])
   482→                break
   483→              }
   484→              case 'assistant_delta': {
   485→                setStatusText('')
   486→                const id = streamIdRef.current
   487→                // BURST DETECTION: a huge first delta means the backend
   488→                // handed us a finished text in one shot (non-streaming
   489→                // fallback) — reveal it with the typewriter instead of
   490→                // slamming a wall of text on screen.
   491→                if (id && event.delta.length > 160) {
   492→                  const bubbleEmpty = !messagesRef.current.find((m) => m.id === id)?.content
   493→                  if (bubbleEmpty && !typewriterRef.current.timer) {
   494→                    revealTypewriter(id, event.delta)
   495→                    break
   496→                  }
   497→                }
   498→                pendingDeltaRef.current += event.delta
   499→                startDeltaFlusher()
   500→                break
   501→              }
   502→              case 'assistant_end': {
   503→                stopDeltaFlusher()
   504→                const id = streamIdRef.current
   505→                streamIdRef.current = null
   506→                if (id) {
   507→                  const attachments = Array.isArray(event.attachments) ? event.attachments : []
   508→                  setMessages((prev) =>
   509→                    prev.map((m) =>
   510→                      m.id === id
   511→                        ? {
   512→                            ...m,
   513→                            streaming: false,
   514→                            ...(attachments.length > 0 ? { attachments } : {}),
   515→                          }
   516→                        : m
   517→                    )
   518→                  )
   519→                }
   520→                break
   521→              }
   522→              case 'assistant': {
   523→                // Full-message fallback (non-streaming providers).
   524→                stopDeltaFlusher()
   525→                const id = streamIdRef.current
   526→                streamIdRef.current = null
   527→                const attachments = Array.isArray(event.attachments) ? event.attachments : []
   528→                if (id) {
   529→                  // Typewriter reveal — the full text arrived in one shot;
   530→                  // animate it word-by-word so it feels alive.
   531→                  setMessages((prev) =>
   532→                    prev.map((m) =>
   533→                      m.id === id
   534→                        ? {
   535→                            ...m,
   536→                            streaming: false,
   537→                            ...(attachments.length > 0 ? { attachments } : {}),
   538→                          }
   539→                        : m
   540→                    )
   541→                  )
   542→                  revealTypewriter(id, event.content)
   543→                } else {
   544→                  const freshId = nextId('a')
   545→                  setMessages((prev) => [
   546→                    ...prev,
   547→                    {
   548→                      id: freshId,
   549→                      role: 'assistant',
   550→                      content: '',
   551→                      agentSlug: pinnedAgent ?? assignRef.current?.agentSlug ?? null,
   552→                      ...(attachments.length > 0 ? { attachments } : {}),
   553→                    },
   554→                  ])
   555→                  revealTypewriter(freshId, event.content)
   556→                }
   557→                break
   558→              }
   559→              case 'tool_start': {
   560→                setMessages((prev) => [
   561→                  ...prev,
   562→                  {
   563→                    id: nextId(`tool-${event.index}`),
   564→                    role: 'tool',
   565→                    content: '',
   566→                    toolName: event.tool,
   567→                    toolIndex: event.index,
   568→                    toolStatus: 'running',
   569→                    toolArgs: event.args,
   570→                  },
   571→                ])
   572→                break
   573→              }
   574→              case 'tool_progress': {
   575→                updateTool(event.tool, event.index, { toolMessage: event.message })
   576→                break
   577→              }
   578→              case 'tool_result': {
   579→                let data: string | null = null
   580→                try {
   581→                  data = JSON.stringify(event.result)
   582→                } catch {
   583→                  data = null
   584→                }
   585→                updateTool(event.tool, event.index, {
   586→                  toolStatus: event.ok ? 'done' : 'error',
   587→                  toolData: data,
   588→                })
   589→                break
   590→              }
   591→              case 'done': {
   592→                setSessionId(event.sessionId)
   593→                onSessionCreated(event.sessionId)
   594→                break
   595→              }
   596→              case 'error': {
   597→                throw new Error(event.message)
   598→              }
   599→            }
   600→          }
   601→        }
   602→      } catch (error) {
   603→        if (error instanceof DOMException && error.name === 'AbortError') return
   604→        setMessages((prev) => [
   605→          ...prev,
   606→          {
   607→            id: nextId('err'),
   608→            role: 'assistant',
   609→            content:
   610→              error instanceof Error
   611→                ? `Something went wrong: ${error.message}`
   612→                : 'Something went wrong. Please try again.',
   613→            isError: true,
   614→            agentSlug: pinnedAgent ?? assignRef.current?.agentSlug ?? null,
   615→          },
   616→        ])
   617→      } finally {
   618→        disarmWatchdog()
   619→        stopDeltaFlusher()
   620→        streamIdRef.current = null
   621→        setStreaming(false)
   622→        setStatusText('')
   623→        setMessages((prev) =>
   624→          prev.map((m) =>
   625→            m.streaming ? { ...m, streaming: false } : m.toolStatus === 'running' ? { ...m, toolStatus: 'done' } : m
   626→          )
   627→        )
   628→      }
   629→    },

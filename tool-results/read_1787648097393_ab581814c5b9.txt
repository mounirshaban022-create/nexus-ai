    53→
    54→export function ChatMode({ initialPrompt, onInitialPromptConsumed, headerSlot }: ChatModeProps) {
    55→  const { toast } = useToast()
    56→  const [messages, setMessages] = useState<ChatMessageItem[]>([])
    57→  const [sessionId, setSessionId] = useState<string | null>(null)
    58→  const [input, setInput] = useState('')
    59→  const [sending, setSending] = useState(false)
    60→  const [sessions, setSessions] = useState<ChatSessionItem[]>([])
    61→  const [historyOpen, setHistoryOpen] = useState(false)
    62→  const [thinkingMode, setThinkingMode] = useState(false)
    63→  const [liveToolSteps, setLiveToolSteps] = useState<Array<{ tool: string; args: Record<string, unknown>; status: 'running' | 'done' | 'error' }>>([])
    64→  // Streaming state: partial text of the in-flight assistant message
    65→  const [streamingText, setStreamingText] = useState('')
    66→  const [streamingAttachments, setStreamingAttachments] = useState<ChatAttachment[]>([])
    67→  // Stop support
    68→  const abortRef = useRef<AbortController | null>(null)
    69→  const streamerRef = useRef<{ stop: () => void; done: Promise<void> } | null>(null)
    70→  const [stopRequested, setStopRequested] = useState(false)
    71→
    72→  const scrollRef = useRef<HTMLDivElement>(null)
    73→  const sentInitialRef = useRef(false)
    74→  // Smart auto-scroll: only pin to bottom when the user hasn't scrolled away
    75→  const userScrolledUpRef = useRef(false)
    76→
    77→  const scrollToBottom = useCallback((smooth = true) => {
    78→    requestAnimationFrame(() => {
    79→      const el = scrollRef.current
    80→      if (!el || userScrolledUpRef.current) return // respect user's position
    81→      el.scrollTo({ top: el.scrollHeight, behavior: smooth ? 'smooth' : 'auto' })
    82→    })
    83→  }, [])
    84→
    85→  // Detect user scroll position
    86→  const onScroll = useCallback(() => {
    87→    const el = scrollRef.current
    88→    if (!el) return
    89→    const distanceFromBottom = el.scrollHeight - el.scrollTop - el.clientHeight
    90→    userScrolledUpRef.current = distanceFromBottom > 120 // user scrolled up significantly
    91→  }, [])
    92→
    93→  const refreshSessions = useCallback(async () => {
    94→    try {
    95→      const res = await fetch('/api/chat/sessions?kind=chat')
    96→      if (res.ok) {
    97→        const data = await res.json()
    98→        setSessions(data.sessions ?? [])
    99→      }
   100→    } catch {
   101→      /* non-fatal */
   102→    }
   103→  }, [])
   104→
   105→  useEffect(() => {
   106→    refreshSessions()
   107→  }, [refreshSessions])
   108→
   109→  /** Streams text word-by-word into the UI — the 'alive' ChatGPT feel.
   110→   * Cancelable via the returned stop() function. */
   111→  const streamWords = useCallback((fullText: string): { stop: () => void; done: Promise<void> } => {
   112→    const words = fullText.split(/(\s+)/) // keep whitespace tokens
   113→    let i = 0
   114→    let cancelled = false
   115→    let timer: ReturnType<typeof setInterval> | null = null
   116→
   117→    const done = new Promise<void>((resolve) => {
   118→      timer = setInterval(() => {
   119→        if (cancelled || stopRequested) {
   120→          if (timer) clearInterval(timer)
   121→          resolve()
   122→          return
   123→        }
   124→        // Emit 1-3 words per tick for natural pacing
   125→        const burst = words.slice(i, i + (Math.random() > 0.7 ? 2 : 1))
   126→        i += burst.length
   127→        setStreamingText(words.slice(0, i).join(''))
   128→        if (i >= words.length) {
   129→          if (timer) clearInterval(timer)
   130→          resolve()
   131→        }
   132→        // keep pinned while streaming (respects userScrolledUp)
   133→        requestAnimationFrame(() => {
   134→          const el = scrollRef.current
   135→          if (el && !userScrolledUpRef.current) {
   136→            el.scrollTop = el.scrollHeight
   137→          }
   138→        })
   139→      }, 24)
   140→    })
   141→
   142→    return {
   143→      stop: () => {
   144→        cancelled = true
   145→        if (timer) clearInterval(timer)
   146→      },
   147→      done,
   148→    }
   149→  }, [stopRequested])
   150→
   151→  const send = useCallback(
   152→    async (text: string) => {
   153→      const trimmed = text.trim()
   154→      if (!trimmed || sending) return
   155→
   156→      const userMsg: ChatMessageItem = {
   157→        id: `local-${Date.now()}`,
   158→        role: 'user',
   159→        content: trimmed,
   160→      }
   161→      setMessages((prev) => [...prev, userMsg])
   162→      setInput('')
   163→      setSending(true)
   164→      setLiveToolSteps([])
   165→      scrollToBottom()
   166→
   167→      // ============ ENGINE 1: PUTER (primary — 507 free models) ============
   168→      try {
   169→        const puterReady = await isPuterReady()
   170→        if (puterReady) {
   171→          // Stream Puter reply word-by-word (ChatGPT feel)
   172→          setStreamingText('')
   173→          const result = await puterChat(
   174→            `You are NEXUS AI, created by Mounir Shaaban. ${trimmed}`,
   175→            'gpt-5-nano'
   176→          )
   177→          if (result.ok && result.text) {
   178→            const streamer = streamWords(result.text)
   179→            streamerRef.current = streamer
   180→            await streamer.done
   181→            streamerRef.current = null
   182→            setStreamingText('')
   183→            setMessages((prev) => [
   184→              ...prev.map((m) => (m.id === userMsg.id ? { ...m, id: `u-${Date.now()}` } : m)),
   185→              { id: `a-${Date.now()}`, role: 'assistant', content: result.text },
   186→            ])
   187→            scrollToBottom()
   188→            return // Puter handled it — done!
   189→          }
   190→        }
   191→      } catch {
   192→        // Puter failed (not signed in / error) → fall through to server AI
   193→      }
   194→
   195→      // ============ ENGINE 2: SERVER AI (smart router: DeepSeek→OpenRouter→Z.ai) ============
   196→      setStopRequested(false)
   197→      abortRef.current = new AbortController()
   198→      try {
   199→        const res = await fetch('/api/chat', {
   200→          method: 'POST',
   201→          headers: {
   202→            'Content-Type': 'application/json',
   203→            ...(supabaseUserIdRef.current ? { 'x-supabase-user-id': supabaseUserIdRef.current } : {}),
   204→          },
   205→          body: JSON.stringify({ message: trimmed, sessionId, thinking: thinkingMode }),
   206→          signal: abortRef.current.signal,
   207→        })
   208→
   209→        if (!res.ok || !res.body) {
   210→          const data = await res.json().catch(() => null)
   211→          throw new Error(data?.error || 'Chat request failed.')
   212→        }
   213→
   214→        // Read the unified NDJSON stream (tool calls + final answer)
   215→        const reader = res.body.getReader()
   216→        const decoder = new TextDecoder()
   217→        let buffer = ''
   218→
   219→        for (;;) {
   220→          const { done, value } = await reader.read()
   221→          if (done) break
   222→          buffer += decoder.decode(value, { stream: true })
   223→          const lines = buffer.split('\n')
   224→          buffer = lines.pop() ?? ''
   225→          for (const line of lines) {
   226→            const text = line.trim()
   227→            if (!text) continue
   228→            try {
   229→              const event = JSON.parse(text) as ChatEvent
   230→              if (event.type === 'user') {
   231→                setMessages((prev) => prev.map((m) => (m.id === userMsg.id ? { id: event.id, role: 'user', content: event.content } : m)))
   232→              } else if (event.type === 'tool_start') {
   233→                setLiveToolSteps((prev) => [...prev, { tool: event.tool, args: event.args, status: 'running' }])
   234→                scrollToBottom()
   235→              } else if (event.type === 'tool_result') {
   236→                setLiveToolSteps((prev) =>
   237→                  prev.map((st, i) => (i === prev.length - 1 && st.tool === event.tool ? { ...st, status: event.ok ? 'done' : 'error' } : st))
   238→                )
   239→              } else if (event.type === 'assistant') {
   240→                setLiveToolSteps([])
   241→                const streamer = streamWords(event.content)
   242→                streamerRef.current = streamer
   243→                setStreamingAttachments(event.attachments ?? [])
   244→                await streamer.done
   245→                streamerRef.current = null
   246→                setStreamingText('')
   247→                setStreamingAttachments([])
   248→                setMessages((prev) => [
   249→                  ...prev.map((m) => (m.id === userMsg.id && m.id.startsWith('local-') ? { ...m, id: `u-${Date.now()}` } : m)),
   250→                  { id: `a-${Date.now()}`, role: 'assistant', content: event.content, attachments: event.attachments },
   251→                ])
   252→                scrollToBottom()
   253→              } else if (event.type === 'done') {
   254→                setSessionId(event.sessionId)
   255→              } else if (event.type === 'error') {
   256→                throw new Error(event.message)
   257→              }
   258→            } catch (parseErr) {
   259→              if (parseErr instanceof Error && !parseErr.message.includes('JSON')) throw parseErr
   260→            }
   261→          }
   262→        }
   263→        refreshSessions()
   264→      } catch (error) {
   265→        // ============ ENGINE 3: PUTER AS LAST RESORT (if signed in) ============
   266→        try {
   267→          const result = await puterChat(
   268→            `You are NEXUS AI, created by Mounir Shaaban. ${trimmed}`,
   269→            'gpt-5-nano'
   270→          )
   271→          if (result.ok && result.text) {
   272→            setMessages((prev) => [
   273→              ...prev.map((m) => (m.id === userMsg.id ? { ...m, id: `u-${Date.now()}` } : m)),
   274→              { id: `a-${Date.now()}`, role: 'assistant', content: result.text },
   275→            ])
   276→            setInput('')
   277→            scrollToBottom()
   278→            return
   279→          }
   280→        } catch {
   281→          /* all engines failed */
   282→        }
   283→        toast({
   284→          title: 'Message failed',
   285→          description: error instanceof Error ? error.message : 'Please try again.',
   286→          variant: 'destructive',
   287→        })
   288→        setMessages((prev) => prev.filter((m) => m.id !== userMsg.id))
   289→        setInput(trimmed)
   290→      } finally {
   291→        setSending(false)
   292→        setLiveToolSteps([])
   293→        setStreamingText('')
   294→        setStreamingAttachments([])
   295→      }
   296→    },
   297→    [sessionId, sending, scrollToBottom, refreshSessions, toast, thinkingMode]
   298→  )
   299→
   300→  // Fire the initial prompt as soon as one arrives and the composer is idle
   301→  useEffect(() => {
   302→    if (initialPrompt && !sentInitialRef.current && !sending) {
   303→      sentInitialRef.current = true
   304→      const p = initialPrompt
   305→      onInitialPromptConsumed?.()
   306→      send(p)
   307→    }
   308→  }, [initialPrompt, send, sending, onInitialPromptConsumed])
   309→
   310→  const regenerate = useCallback(async () => {
   311→    if (!sessionId || sending) return
   312→    setSending(true)
   313→    // Optimistically drop the trailing assistant message locally
   314→    setMessages((prev) => {
   315→      const copy = [...prev]
   316→      while (copy.length && copy[copy.length - 1].role === 'assistant') copy.pop()
   317→      return copy
   318→    })
   319→    scrollToBottom()
   320→    try {
   321→      const res = await fetch('/api/chat/regenerate', {
   322→        method: 'POST',
   323→        headers: { 'Content-Type': 'application/json' },
   324→        body: JSON.stringify({ sessionId }),
   325→      })
   326→      const data = await res.json()
   327→      if (!res.ok) throw new Error(data.error || 'Regeneration failed.')
   328→      setMessages((prev) => [...prev, data.reply])
   329→      scrollToBottom()
   330→    } catch (error) {
   331→      toast({
   332→        title: 'Regeneration failed',
   333→        description: error instanceof Error ? error.message : 'Please try again.',
   334→        variant: 'destructive',
   335→      })
   336→    } finally {
   337→      setSending(false)
   338→    }
   339→  }, [sessionId, sending, scrollToBottom, toast])
   340→
   341→  const { artifact, openArtifact, closeArtifact } = useArtifact()
   342→  const [puterActive, setPuterActive] = useState(false)
   343→  const supabaseUserIdRef = useRef<string | null>(null)
   344→
   345→  // Load Supabase user ID for cloud sync
   346→  useEffect(() => {
   347→    getCurrentUser().then((u) => {
   348→      supabaseUserIdRef.current = u?.id ?? null
   349→    }).catch(() => {})
   350→  }, [])
   351→
   352→  // Check Puter sign-in state once — hide activation banner if already active
   353→  useEffect(() => {
   354→    isPuterReady().then(setPuterActive).catch(() => {})
   355→  }, [])
   356→  const [editingId, setEditingId] = useState<string | null>(null)
   357→  const [editText, setEditText] = useState('')
   358→  const [feedbackGiven, setFeedbackGiven] = useState<Record<string, 'up' | 'down'>>({})
   359→
   360→  const copyText = useCallback(async (text: string) => {
   361→    try {
   362→      await navigator.clipboard.writeText(text)
   363→      toast({ title: 'Copied to clipboard' })
   364→    } catch {
   365→      toast({ title: 'Copy failed', variant: 'destructive' })
   366→    }
   367→  }, [toast])
   368→
   369→  const feedback = useCallback(
   370→    (msgId: string, kind: 'up' | 'down') => {
   371→      setFeedbackGiven((prev) => ({ ...prev, [msgId]: kind }))
   372→      toast({
   373→        title: kind === 'up' ? 'Thanks for the feedback!' : 'Feedback noted — thanks!',
   374→        duration: 2000,
   375→      })
   376→    },
   377→    [toast]
   378→  )
   379→
   380→  /** Edit a sent message: truncate history after it, resend as new message. */
   381→  const submitEdit = useCallback(
   382→    (msgId: string) => {
   383→      const edited = editText.trim()
   384→      setEditingId(null)
   385→      if (!edited) return
   386→      // Truncate messages after the edited one
   387→      const idx = messages.findIndex((m) => m.id === msgId)
   388→      if (idx === -1) return
   389→      setMessages((prev) => prev.slice(0, idx))
   390→      send(edited)
   391→    },
   392→    [editText, messages, send]
   393→  )
   394→
   395→  const stopGeneration = useCallback(() => {
   396→    setStopRequested(true)
   397→    abortRef.current?.abort()
   398→    streamerRef.current?.stop()
   399→    const partial = streamingText
   400→    if (partial.trim()) {
   401→      setMessages((prev) => [...prev, { id: `a-${Date.now()}`, role: 'assistant', content: partial + '\n\n*(stopped)*' }])
   402→    }
   403→    setStreamingText('')
   404→    setStreamingAttachments([])
   405→    setSending(false)
   406→    setLiveToolSteps([])
   407→  }, [streamingText])
   408→
   409→  const loadSession = useCallback(
   410→    async (id: string) => {
   411→      if (sending) return
   412→      setHistoryOpen(false)
   413→      try {
   414→        const res = await fetch(`/api/chat/sessions/${id}`)
   415→        if (!res.ok) throw new Error('Conversation not found.')
   416→        const data = await res.json()
   417→        setSessionId(data.session.id)
   418→        setMessages(
   419→          (data.session.messages as ChatMessageItem[]).map((m) => ({
   420→            ...m,
   421→            thinking: m.thinking ?? null,
   422→          }))
   423→        )
   424→        setTimeout(() => scrollToBottom(false), 60)
   425→      } catch (error) {
   426→        toast({
   427→          title: 'Could not open conversation',
   428→          description: error instanceof Error ? error.message : undefined,
   429→          variant: 'destructive',
   430→        })
   431→      }
   432→    },
   433→    [sending, scrollToBottom, toast]
   434→  )
   435→
   436→  const deleteSession = useCallback(
   437→    async (id: string, e: React.MouseEvent) => {
   438→      e.stopPropagation()
   439→      try {
   440→        const res = await fetch(`/api/chat/sessions/${id}`, { method: 'DELETE' })
   441→        if (!res.ok) throw new Error('Delete failed.')
   442→        setSessions((prev) => prev.filter((s) => s.id !== id))
   443→        if (sessionId === id) {
   444→          setSessionId(null)
   445→          setMessages([])
   446→        }
   447→        toast({ title: 'Conversation deleted' })
   448→      } catch {
   449→        toast({ title: 'Could not delete conversation', variant: 'destructive' })
   450→      }
   451→    },
   452→    [sessionId, toast]
   453→  )
   454→
   455→  const startNew = () => {
   456→    if (sending) return
   457→    setSessionId(null)
   458→    setMessages([])
   459→    setInput('')
   460→  }
   461→
   462→  const activeSession = sessions.find((s) => s.id === sessionId)
   463→
   464→  return (
   465→    <div className="flex h-full min-h-0 flex-col">
   466→      {/* Chat header — ChatGPT style: ability picker + actions */}
   467→      <header className="flex items-center justify-between gap-3 border-b border-border/50 px-5 py-3 sm:px-8">
   468→        <div className="flex min-w-0 items-center gap-2">
   469→          {headerSlot}
   470→          <div className="hidden min-w-0 sm:block">
   471→            <p className="truncate text-sm text-muted-foreground">
   472→              {messages.length > 0
   473→                ? activeSession?.title ?? 'Conversation'
   474→                : 'Ask anything — full Markdown answers'}
   475→            </p>
   476→          </div>
   477→        </div>
   478→        <div className="flex shrink-0 items-center gap-2">
   479→          <Sheet open={historyOpen} onOpenChange={setHistoryOpen}>
   480→            <SheetTrigger asChild>
   481→              <Button variant="outline" size="sm" className="gap-2 rounded-lg">
   482→                <History className="h-4 w-4" />
   483→                <span className="hidden sm:inline">History</span>
   484→                {sessions.length > 0 && (
   485→                  <span className="rounded-full bg-secondary px-1.5 text-[11px] font-semibold text-muted-foreground">
   486→                    {sessions.length}
   487→                  </span>
   488→                )}
   489→              </Button>
   490→            </SheetTrigger>
   491→            <SheetContent side="left" className="flex w-80 flex-col gap-0 p-0 sm:w-96">
   492→              <SheetHeader className="border-b border-border/60 px-4 py-4">
   493→                <SheetTitle className="flex items-center gap-2 text-base">
   494→                  <History className="h-4 w-4 text-primary" /> Conversations
   495→                </SheetTitle>
   496→              </SheetHeader>
   497→              <div className="border-b border-border/60 p-3">
   498→                <Button
   499→                  onClick={() => {
   500→                    startNew()
   501→                    setHistoryOpen(false)
   502→                  }}
   503→                  className="w-full gap-2 rounded-lg bg-primary text-primary-foreground hover:brightness-110"
   504→                >
   505→                  <Plus className="h-4 w-4" /> New conversation
   506→                </Button>
   507→              </div>
   508→              <ScrollArea className="omni-scroll flex-1">
   509→                <div className="flex flex-col gap-1 p-2">
   510→                  {sessions.length === 0 && (
   511→                    <p className="px-3 py-8 text-center text-sm text-muted-foreground">
   512→                      No conversations yet.
   513→                    </p>
   514→                  )}
   515→                  {sessions.map((s) => (
   516→                    <div
   517→                      key={s.id}
   518→                      className={`group flex w-full items-start gap-2 rounded-lg border px-3 py-2.5 text-left transition ${
   519→                        s.id === sessionId
   520→                          ? 'border-border bg-secondary/60'
   521→                          : 'border-transparent hover:border-border/60 hover:bg-secondary/50'
   522→                      }`}
   523→                    >
   524→                      <button
   525→                        onClick={() => loadSession(s.id)}
   526→                        className="min-w-0 flex-1 text-left"
   527→                        aria-label={`Open conversation: ${s.title}`}
   528→                      >
   529→                        <p className="truncate text-sm font-medium">{s.title}</p>
   530→                        <p className="mt-0.5 truncate text-xs text-muted-foreground">
   531→                          {s.preview || '…'}
   532→                        </p>
   533→                        <p className="mt-1 text-[11px] text-muted-foreground/70">
   534→                          {new Date(s.updatedAt).toLocaleDateString(undefined, {
   535→                            month: 'short',
   536→                            day: 'numeric',
   537→                            hour: '2-digit',
   538→                            minute: '2-digit',
   539→                          })}
   540→                        </p>
   541→                      </button>
   542→                      <button
   543→                        aria-label={`Delete conversation: ${s.title}`}
   544→                        onClick={(e) => deleteSession(s.id, e)}
   545→                        className="mt-0.5 shrink-0 rounded-md p-1.5 text-muted-foreground/50 opacity-0 transition hover:bg-destructive/15 hover:text-destructive focus-visible:opacity-100 group-hover:opacity-100"
   546→                      >
   547→                        <Trash2 className="h-3.5 w-3.5" />
   548→                      </button>
   549→                    </div>
   550→                  ))}
   551→                </div>
   552→              </ScrollArea>
   553→            </SheetContent>
   554→          </Sheet>
   555→          <Button
   556→            variant="outline"
   557→            size="sm"
   558→            onClick={startNew}

   829→function CopyButton({ text }: { text: string }) {
   830→  const [copied, setCopied] = useState(false)
   831→  return (
   832→    <button
   833→      onClick={async () => {
   834→        try {
   835→          await navigator.clipboard.writeText(text)
   836→          setCopied(true)
   837→          setTimeout(() => setCopied(false), 1600)
   838→        } catch {
   839→          /* clipboard unavailable */
   840→        }
   841→      }}
   842→      className="flex items-center gap-1 rounded-md px-1.5 py-0.5 text-[11px] text-muted-foreground transition hover:bg-secondary hover:text-foreground"
   843→      aria-label="Copy message"
   844→    >
   845→      {copied ? (
   846→        <>
   847→          <Check className="h-3 w-3 text-emerald-400" aria-hidden /> Copied
   848→        </>
   849→      ) : (
   850→        <>
   851→          <Copy className="h-3 w-3" aria-hidden /> Copy
   852→        </>
   853→      )}
   854→    </button>
   855→  )
   856→}

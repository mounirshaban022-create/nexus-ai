   139→function formatMemberSince(iso: string | null | undefined, lang: 'en' | 'ar'): string {
   140→  if (!iso) return ''
   141→  try {
   142→    return new Date(iso).toLocaleDateString(lang === 'ar' ? 'ar-EG' : 'en-US', {
   143→      month: 'long',
   144→      year: 'numeric',
   145→    })
   146→  } catch {
   147→    return ''
   148→  }
   149→}

   151→function isValidAvatarUrl(url: string | null | undefined): url is string {
   152→  if (!url) return false
   153→  return url.startsWith('/') || url.startsWith('http') || url.startsWith('data:')
   154→}

   297→function EmailConnectDialog({
   298→  open,
   299→  onOpenChange,
   300→  presets,
   301→  onConnected,
   302→}: {
   303→  open: boolean
   304→  onOpenChange: (open: boolean) => void
   305→  presets: EmailPresetRow[]
   306→  onConnected: () => void | Promise<void>
   307→}) {
   308→  const { toast } = useToast()
   309→  const { t } = useI18n()
   310→
   311→  const [presetId, setPresetId] = useState('gmail')
   312→  const [email, setEmail] = useState('')
   313→  const [password, setPassword] = useState('')
   314→  const [imapHost, setImapHost] = useState('imap.gmail.com')
   315→  const [imapPort, setImapPort] = useState('993')
   316→  const [smtpHost, setSmtpHost] = useState('smtp.gmail.com')
   317→  const [smtpPort, setSmtpPort] = useState('465')
   318→  const [smtpSecure, setSmtpSecure] = useState(true)
   319→  const [submitting, setSubmitting] = useState(false)
   320→  const [error, setError] = useState<string | null>(null)
   321→
   322→  /* Fresh form every time the dialog opens (Gmail defaults). */
   323→  useEffect(() => {
   324→    if (!open) return
   325→    setPresetId('gmail')
   326→    setEmail('')
   327→    setPassword('')
   328→    setImapHost('imap.gmail.com')
   329→    setImapPort('993')
   330→    setSmtpHost('smtp.gmail.com')
   331→    setSmtpPort('465')
   332→    setSmtpSecure(true)
   333→    setError(null)
   334→  }, [open])
   335→
   336→  const selectedPreset = presets.find((p) => p.id === presetId)
   337→
   338→  const applyPreset = (p: EmailPresetRow) => {
   339→    setPresetId(p.id)
   340→    setError(null)
   341→    if (p.id === 'custom') return
   342→    setImapHost(p.imapHost)
   343→    setImapPort(String(p.imapPort))
   344→    setSmtpHost(p.smtpHost)
   345→    setSmtpPort(String(p.smtpPort))
   346→    setSmtpSecure(p.smtpSecure)
   347→  }
   348→
   349→  const submit = async (e: React.FormEvent) => {
   350→    e.preventDefault()
   351→    if (submitting) return
   352→    setError(null)
   353→
   354→    if (!EMAIL_REGEX.test(email.trim())) {
   355→      setError('Enter a valid email address.')
   356→      return
   357→    }
   358→    if (!password) {
   359→      setError(t('settings.password') + ' — ' + (langPasswordHint(selectedPreset?.id) ?? ''))
   360→      return
   361→    }
   362→    const iPort = parseInt(imapPort, 10)
   363→    const sPort = parseInt(smtpPort, 10)
   364→    if (!imapHost.trim() || !smtpHost.trim() || !(iPort >= 1 && iPort <= 65535) || !(sPort >= 1 && sPort <= 65535)) {
   365→      setError('Fill in the IMAP and SMTP host/port for your provider.')
   366→      return
   367→    }
   368→
   369→    const label =
   370→      selectedPreset && selectedPreset.id !== 'custom'
   371→        ? selectedPreset.label
   372→        : email.trim().split('@')[1] || 'Email account'
   373→
   374→    setSubmitting(true)
   375→    try {
   376→      const res = await fetch('/api/email/accounts', {
   377→        method: 'POST',
   378→        headers: { 'Content-Type': 'application/json' },
   379→        body: JSON.stringify({
   380→          label,
   381→          email: email.trim(),
   382→          imapHost: imapHost.trim(),
   383→          imapPort: iPort,
   384→          smtpHost: smtpHost.trim(),
   385→          smtpPort: sPort,
   386→          smtpSecure,
   387→          username: email.trim(),
   388→          password,
   389→        }),
   390→      })
   391→      const data = (await res.json().catch(() => ({}))) as {
   392→        error?: string
   393→        verification?: { ok?: boolean; message?: string }
   394→      }
   395→      if (!res.ok) {
   396→        setError(data.error || 'Could not connect that account.')
   397→        return
   398→      }
   399→      const verification = data.verification ?? {}
   400→      if (verification.ok) {
   401→        toast({
   402→          title: t('settings.emailConnected'),
   403→          description: t('settings.connectionOk'),
   404→        })
   405→        onOpenChange(false)
   406→        await onConnected()
   407→      } else {
   408→        setError(
   409→          `${t('settings.connectionFailed')} — ${verification.message || 'check your credentials.'}`
   410→        )
   411→        // The failed attempt is saved with an error status — refresh the
   412→        // list underneath so the row (and its status message) is visible.
   413→        await onConnected()
   414→      }
   415→    } catch {
   416→      setError('Network error — please try again.')
   417→    } finally {
   418→      setSubmitting(false)
   419→    }
   420→  }
   421→
   422→  function langPasswordHint(id: string | undefined): string {
   423→    if (id === 'gmail') return 'use a Google App Password'
   424→    if (id === 'outlook') return 'use a Microsoft App Password'
   425→    return 'required'
   426→  }
   427→
   428→  return (
   429→    <Dialog open={open} onOpenChange={onOpenChange}>
   430→      <DialogContent className="max-h-[88dvh] overflow-y-auto border-white/10 bg-[#101012] text-zinc-100 sm:max-w-lg">
   431→        <DialogHeader>
   432→          <DialogTitle className="font-display flex items-center gap-2 text-left text-lg tracking-tight">
   433→            <span
   434→              className="flex h-8 w-8 items-center justify-center rounded-xl"
   435→              style={{ background: tint('#ff5a5f', 0.14), color: '#ff5a5f' }}
   436→              aria-hidden
   437→            >
   438→              <Mail className="h-4 w-4" />
   439→            </span>
   440→            {t('settings.emailConnect')}
   441→          </DialogTitle>
   442→          <DialogDescription className="text-left text-xs leading-relaxed text-zinc-500">
   443→            {t('settings.emailDesc')}
   444→          </DialogDescription>
   445→        </DialogHeader>
   446→
   447→        <form onSubmit={submit} className="space-y-4">
   448→          {/* Provider presets */}
   449→          <div>
   450→            <span className={fieldLabelClass} aria-hidden>
   451→              Gmail · Outlook · IMAP
   452→            </span>
   453→            <div className="flex flex-wrap gap-1.5" role="group" aria-label="Email provider presets">
   454→              {(presets.length > 0
   455→                ? presets
   456→                : [
   457→                    {
   458→                      id: 'gmail',
   459→                      label: 'Gmail',
   460→                      imapHost: 'imap.gmail.com',
   461→                      imapPort: 993,
   462→                      smtpHost: 'smtp.gmail.com',
   463→                      smtpPort: 465,
   464→                      smtpSecure: true,
   465→                      hint: GMAIL_DEFAULT_HINT,
   466→                    },
   467→                  ]
   468→              ).map((p) => (
   469→                <PresetPill key={p.id} active={presetId === p.id} onClick={() => applyPreset(p)}>
   470→                  {p.id === 'custom' ? 'Custom IMAP' : p.label.split(' ')[0]}
   471→                </PresetPill>
   472→              ))}
   473→            </div>
   474→          </div>
   475→
   476→          {/* Credentials */}
   477→          <label className="block">
   478→            <span className={fieldLabelClass}>{t('settings.email')}</span>
   479→            <input
   480→              type="email"
   481→              value={email}
   482→              onChange={(e) => setEmail(e.target.value)}
   483→              placeholder={t('settings.emailHint')}
   484→              autoComplete="email"
   485→              className={inputClass}
   486→            />
   487→          </label>
   488→
   489→          <label className="block">
   490→            <span className={fieldLabelClass}>{t('settings.password')}</span>
   491→            <input
   492→              type="password"
   493→              value={password}
   494→              onChange={(e) => setPassword(e.target.value)}
   495→              placeholder="•••• •••• •••• ••••"
   496→              autoComplete="new-password"
   497→              className={inputClass}
   498→            />
   499→          </label>
   500→
   501→          {/* Preset hint */}
   502→          <p className="flex items-start gap-2 rounded-xl border border-[#f5a623]/20 bg-[#f5a623]/[0.07] px-3.5 py-2.5 text-[11px] leading-relaxed text-[#f5c46a]">
   503→            <Info className="mt-0.5 h-3.5 w-3.5 shrink-0" aria-hidden />
   504→            {selectedPreset?.hint ?? GMAIL_DEFAULT_HINT}
   505→          </p>
   506→
   507→          {/* Server settings */}
   508→          <div className="grid grid-cols-1 gap-3 sm:grid-cols-[1fr_5.5rem]">
   509→            <label className="block">
   510→              <span className={fieldLabelClass}>{t('settings.imapHost')}</span>
   511→              <input
   512→                type="text"
   513→                value={imapHost}
   514→                onChange={(e) => setImapHost(e.target.value)}
   515→                placeholder="imap.gmail.com"
   516→                spellCheck={false}
   517→                className={inputClass}
   518→              />
   519→            </label>
   520→            <label className="block">
   521→              <span className={fieldLabelClass}>{t('settings.port')}</span>
   522→              <input
   523→                type="number"
   524→                value={imapPort}
   525→                onChange={(e) => setImapPort(e.target.value)}
   526→                min={1}
   527→                max={65535}
   528→                className={inputClass}
   529→              />
   530→            </label>
   531→            <label className="block">
   532→              <span className={fieldLabelClass}>{t('settings.smtpHost')}</span>
   533→              <input
   534→                type="text"
   535→                value={smtpHost}
   536→                onChange={(e) => setSmtpHost(e.target.value)}
   537→                placeholder="smtp.gmail.com"
   538→                spellCheck={false}
   539→                className={inputClass}
   540→              />
   541→            </label>
   542→            <label className="block">
   543→              <span className={fieldLabelClass}>{t('settings.port')}</span>

   549→                max={65535}
   550→                className={inputClass}
   551→              />
   552→            </label>
   553→          </div>
   554→
   555→          {error && (
   556→            <p
   557→              role="alert"
   558→              className="rounded-xl border border-red-500/20 bg-red-500/10 px-3.5 py-2.5 text-xs leading-relaxed text-red-300"
   559→            >
   560→              {error}
   561→            </p>
   562→          )}
   563→
   564→          <div className="flex gap-2.5 pt-1">
   565→            <button
   566→              type="button"
   567→              onClick={() => onOpenChange(false)}
   568→              className="min-h-[44px] flex-1 rounded-xl border border-white/10 px-4 text-sm font-medium text-zinc-400 transition hover:border-white/25 hover:text-zinc-200"
   569→            >
   570→              {t('settings.cancel')}
   571→            </button>
   572→            <button
   573→              type="submit"
   574→              disabled={submitting}
   575→              className="nx-gradient-surface flex min-h-[44px] flex-[2] items-center justify-center gap-2 rounded-xl px-4 text-sm font-semibold disabled:opacity-60"
   576→            >
   577→              {submitting ? (

   588→            </button>
   589→          </div>
   590→        </form>
   591→      </DialogContent>
   592→    </Dialog>
   593→  )

   800→            <button

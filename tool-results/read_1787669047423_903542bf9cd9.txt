     3→/**
     4→ * NEXUS One — full-screen premium voice conversation overlay.
     5→ *
     6→ * FAIL-SOFT VOICE CHAIN (the "works perfectly" part):
     7→ *  INPUT  : Web Speech API (live transcript as you speak)
     8→ *         → MediaRecorder + in-browser VAD → server ASR on the recorded blob
     9→ *         → always-available text input (mic denied / exotic browsers).
    10→ *  THINK  : POST /api/voice/turn { message | audio, history, voice, lang }.
    11→ *  SPEAK  : inline audio from the turn response → selected Kokoro offline
    12→ *           voice (only if the in-browser model loaded) → POST /api/tts →
    13→ *           browser speechSynthesis. No engine failure can dead-end a turn.
    14→ *
    15→ * Hands-free: after each spoken reply the overlay auto-restarts listening
    16→ * until the user mutes or closes it. Voice + language choices persist to
    17→ * localStorage; the Arabic UI defaults to an Arabic neural voice (mirrors
    18→ * the server's pickVoiceForLanguage('ar') override in /api/voice/turn).
    19→ *
    20→ * The orb is the BrandMark wrapped in layered gradient glow rings whose
    21→ * motion maps to the current state, with a real mic-driven waveform strip
    22→ * (getUserMedia analyser; synthetic motion when the analyser is missing).
    23→ */

    87→function base64ToBlob(base64: string, type: string): Blob {
    88→  const binary = atob(base64)
    89→  const bytes = new Uint8Array(binary.length)
    90→  for (let i = 0; i < binary.length; i++) bytes[i] = binary.charCodeAt(i)
    91→  return new Blob([bytes], { type })
    92→}

   275→  const speak = useCallback(
   276→    async (text: string, inlineAudio?: string | null, audioFormat?: string): Promise<void> => {
   277→      const epoch = ++speakEpochRef.current
   278→      const alive = () => openRef.current && speakEpochRef.current === epoch
   279→      setStateSafe('speaking')
   280→      setCaption(text)

   283→        if (objectUrlRef.current) URL.revokeObjectURL(objectUrlRef.current)
   284→        objectUrlRef.current = url

   292→          const url = URL.createObjectURL(res.blob)
   293→          trackUrl(url)
   294→          await playUrl(url, epoch)
   295→          return
   296→        }
   297→        // Kokoro failed → silently fall through to the server chain.
   298→      }
   299→      if (!alive()) return
   300→

   302→      if (inlineAudio) {
   303→        try {
   304→          const blob = base64ToBlob(inlineAudio, audioFormat === 'mp3' ? 'audio/mpeg' : 'audio/wav')
   305→          if (blob.size > 100) {
   306→            const url = URL.createObjectURL(blob)
   307→            trackUrl(url)
   308→            await playUrl(url, epoch)
   309→            return
   310→          }
   311→        } catch {
   312→          /* fall through */
   313→        }
   314→        if (!alive()) return
   315→      }
   316→
   317→      // 3 — dedicated TTS endpoint
   318→      try {
   319→        const res = await fetch(`/api/tts?lang=${encodeURIComponent(uiLangRef.current)}`, {
   320→          method: 'POST',
   321→          headers: { 'Content-Type': 'application/json' },
   322→          body: JSON.stringify({ text, voice: serverVoiceRef.current, speed: 1.0 }),
   323→          signal: turnControllerRef.current?.signal ?? undefined,
   324→        })
   325→        if (res.ok) {
   326→          const blob = await res.blob()
   327→          if (blob.size > 100 && alive()) {
   328→            const url = URL.createObjectURL(blob)
   329→            trackUrl(url)
   330→            await playUrl(url, epoch)
   331→            return
   332→          }
   333→        }
   334→      } catch {
   335→        /* fall through */
   336→      }
   337→      if (!alive()) return
   338→
   339→      // 4 — browser built-in synthesizer (last resort)
   340→      try {
   341→        await new Promise<void>((resolve) => {
   342→          const utter = new SpeechSynthesisUtterance(text)
   343→          utter.lang = langRef.current
   344→          utter.onend = () => resolve()
   345→          utter.onerror = () => resolve()
   346→          window.speechSynthesis.speak(utter)
   347→          window.setTimeout(resolve, Math.min(30_000, text.length * 90))
   348→        })
   349→      } catch {
   350→        /* the text is still in the conversation log */
   351→      }
   352→    },
   353→    [playUrl, setStateSafe]

   361→    const body: Record<string, unknown> = {
   362→      history: historyRef.current.slice(-6),
   363→      language: langRef.current.split('-')[0],

   372→      method: 'POST',

   376→    })

   441→        }

   445→        await speak(reply, data.audio, data.audioFormat)
   446→      } catch (e) {
   447→        if ((e as Error)?.name === 'AbortError') return
   448→        const msg = e instanceof Error ? e.message : 'Something went wrong — try again.'
   449→        setError(msg)
   450→        toast({ title: 'Voice error', description: msg, variant: 'destructive' })

   453→        if (openRef.current && !mutedRef.current && !micDeniedRef.current) {
   454→          startListeningRef.current?.() // hands-free loop: listen again
   455→        } else {
   456→          setStateSafe('idle')
   457→        }
   458→      }
   459→    },

   468→  /* ASR fallback — MediaRecorder + VAD → server ASR                  */
   469→  /* (used when Web Speech is missing or thrashing)                   */

   471→
   472→  const startAsrLoop = useCallback(async (): Promise<void> => {
   473→    const stream = micStreamRef.current
   474→    if (!stream || typeof MediaRecorder === 'undefined' || busyRef.current) return
   475→    if (!openRef.current || mutedRef.current) return
   476→    const token = ++asrTokenRef.current
   477→    const alive = () =>
   478→      asrTokenRef.current === token && openRef.current && !mutedRef.current && !busyRef.current
   479→
   480→    try {
   481→      const ctx = ensureAudioCtx()
   482→      if (!ctx || ctx.state === 'closed') throw new Error('Audio context unavailable')
   483→
   484→      const source = ctx.createMediaStreamSource(stream)
   485→      const analyser = ctx.createAnalyser()
   486→      analyser.fftSize = 1024
   487→      source.connect(analyser)
   488→
   489→      const mimeType = MediaRecorder.isTypeSupported('audio/webm') ? 'audio/webm' : ''
   490→      const rec = new MediaRecorder(stream, mimeType ? { mimeType } : undefined)
   491→      mediaRecorderRef.current = rec
   492→      const chunks: Blob[] = []
   493→      rec.ondataavailable = (e) => {
   494→        if (e.data.size > 0) chunks.push(e.data)
   495→      }
   496→      rec.start(250)
   497→      setStateSafe('listening')
   498→      setError('')
   499→      setCaption('')
   500→      netErrRef.current = 0
   501→
   502→      // VAD — resolve(true) once speech is followed by ~1.5s of silence.
   503→      const spoke = await new Promise<boolean>((resolve) => {
   504→        const data = new Uint8Array(analyser.fftSize)
   505→        let speechStart: number | null = null
   506→        let lastSound = performance.now()
   507→        const startedAt = performance.now()
   508→        const THRESHOLD = 0.008
   509→        const SILENCE_MS = 1500
   510→        const MIN_SPEECH = 450
   511→        const MAX_WAIT = 12_000
   512→        let raf = 0
   513→        const loop = () => {
   514→          if (!alive()) {
   515→            cancelAnimationFrame(raf)
   516→            resolve(false)
   517→            return
   518→          }
   519→          analyser.getByteTimeDomainData(data)
   520→          let sum = 0
   521→          for (let i = 0; i < data.length; i++) {
   522→            const v = (data[i] - 128) / 128
   523→            sum += v * v
   524→          }
   525→          const rms = Math.sqrt(sum / data.length)
   526→          const now = performance.now()
   527→          if (rms > THRESHOLD) {
   528→            if (speechStart === null) speechStart = now
   529→            lastSound = now
   530→          }
   531→          if (
   532→            speechStart !== null &&
   533→            now - lastSound > SILENCE_MS &&
   534→            now - speechStart > MIN_SPEECH
   535→          ) {
   536→            cancelAnimationFrame(raf)
   537→            resolve(true)
   538→            return
   539→          }
   540→          if (speechStart === null && now - startedAt > MAX_WAIT) {
   541→            cancelAnimationFrame(raf)
   542→            resolve(false)
   543→            return
   544→          }
   545→          if (speechStart !== null && now - speechStart > MAX_WAIT) {
   546→            cancelAnimationFrame(raf)
   547→            resolve(true)
   548→            return
   549→          }
   550→          raf = requestAnimationFrame(loop)
   551→        }

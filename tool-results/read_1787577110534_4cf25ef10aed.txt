    10→const FILES_DIR = path.join(process.cwd(), 'generated-images') // shared generated-files dir
    11→
    12→/**
    13→ * Office Studio generation endpoint.
    14→ * The client sends a structured document (title + blocks) and the file type;
    15→ * we build a real .docx / .xlsx / .pptx / .md file and return its URL.
    16→ */
    17→
    18→type Block =
    19→  | { type: 'heading'; text: string; level?: number }
    20→  | { type: 'paragraph'; text: string }
    21→  | { type: 'bullets'; items: string[] }
    22→  | { type: 'table'; rows: string[][] }
    23→  | { type: 'slide'; title: string; bullets: string[] }
    24→
    25→const requestSchema = z.object({
    26→  format: z.enum(['docx', 'xlsx', 'pptx', 'md']),
    27→  title: z.string().min(1).max(200),
    28→  theme: z.enum(['nexus', 'executive', 'sunset', 'rosewood']).optional(),
    29→  blocks: z
    30→    .array(
    31→      z.union([
    32→        z.object({
    33→          type: z.literal('heading'),
    34→          text: z.string().max(500),
    35→          level: z.number().int().min(1).max(4).optional(),
    36→        }),
    37→        z.object({ type: z.literal('paragraph'), text: z.string().max(4000) }),
    38→        z.object({ type: z.literal('bullets'), items: z.array(z.string().max(1000)).max(30) }),
    39→        z.object({
    40→          type: z.literal('table'),
    41→          rows: z.array(z.array(z.string().max(500)).max(12)).max(40),
    42→        }),
    43→        z.object({
    44→          type: z.literal('slide'),
    45→          title: z.string().max(200),
    46→          bullets: z.array(z.string().max(500)).max(12),
    47→        }),
    48→      ])
    49→    )
    50→    .min(1)
    51→    .max(60),
    52→})

    98→function themeFromTitle(title: string): DocTheme {
    99→  const t = title.toLowerCase()
   100→  if (/budget|finance|invoice|cost|revenue|bank/.test(t)) return THEMES.executive
   101→  if (/pitch|startup|creative|marketing|brand/.test(t)) return THEMES.sunset
   102→  if (/love|story|recipe|travel|food|health/.test(t)) return THEMES.rosewood
   103→  return THEMES.nexus
   104→}

   106→const MIME_TYPES: Record<string, string> = {
   107→  docx: 'application/vnd.openxmlformats-officedocument.wordprocessingml.document',

   113→export async function POST(req: NextRequest) {
   114→  try {
   115→    const limit = rateLimit(`office-create:${clientKey(req)}`, 15, 60_000)
   116→    if (!limit.ok) {
   117→      return NextResponse.json(
   118→        { error: `Too many generations. Retry in ${limit.retryAfterSeconds}s.` },
   119→        { status: 429 }
   120→      )
   121→    }
   122→
   123→    const parsed = requestSchema.safeParse(await req.json())
   124→    if (!parsed.success) {
   125→      return NextResponse.json({ error: 'Invalid document structure.' }, { status: 400 })
   126→    }
   127→    const { format, title, blocks } = parsed.data
   128→    // Theme: explicit choice, or auto-picked from the title content
   129→    const themeKey = parsed.data.theme ?? (() => {
   130→      const t = title.toLowerCase()
   131→      if (/budget|finance|invoice|cost|revenue|bank/.test(t)) return 'executive'
   132→      if (/pitch|startup|creative|marketing|brand/.test(t)) return 'sunset'
   133→      if (/love|story|recipe|travel|food|health/.test(t)) return 'rosewood'
   134→      return 'nexus'
   135→    })()
   136→
   137→    await mkdir(FILES_DIR, { recursive: true })
   138→    const id = randomUUID()
   139→    let filename: string
   140→    let buffer: Buffer
   141→
   142→    if (format === 'docx') {
   143→      const {
   144→        Document,
   145→        Packer,
   146→        Paragraph,
   147→        TextRun,
   148→        HeadingLevel,
   149→        Table,
   150→        TableRow,
   151→        TableCell,
   152→        WidthType,
   153→        AlignmentType,
   154→        BorderStyle,
   155→        ShadingType,
   156→        Footer,
   157→        PageNumber,
   158→      } = await import('docx')
   159→      const theme = THEMES[themeKey] ?? THEMES.nexus
   160→
   161→      const children: Array<InstanceType<typeof Paragraph> | InstanceType<typeof Table>> = []
   162→
   163→      /* ---- Cover header block ---- */
   164→      children.push(
   165→        new Paragraph({
   166→          children: [
   167→            new TextRun({ text: 'NEXUS AI DOCUMENT', bold: true, size: 18, color: theme.accent }),
   168→          ],
   169→          spacing: { after: 120 },
   170→        }),
   171→        new Paragraph({
   172→          children: [new TextRun({ text: title, bold: true, size: 56, color: theme.primaryDark })],
   173→          spacing: { after: 200 },
   174→        }),
   175→        new Paragraph({
   176→          children: [
   177→            new TextRun({
   178→              text: new Date().toLocaleDateString('en-US', { year: 'numeric', month: 'long', day: 'numeric' }),
   179→              size: 22, color: theme.textMuted, italics: true,
   180→            }),
   181→          ],
   182→          spacing: { after: 300 },
   183→          border: { bottom: { style: BorderStyle.SINGLE, size: 12, color: theme.primary, space: 8 } },
   184→        })
   185→      )
   186→
   187→      /* ---- Content ---- */
   188→      for (const block of blocks) {
   189→        if (block.type === 'heading') {
   190→          const levels = [HeadingLevel.HEADING_1, HeadingLevel.HEADING_2, HeadingLevel.HEADING_3, HeadingLevel.HEADING_4]
   191→          children.push(
   192→            new Paragraph({
   193→              text: block.text,
   194→              heading: levels[(block.level ?? 2) - 1] ?? HeadingLevel.HEADING_2,
   195→              spacing: { before: 280, after: 140 },
   196→            })
   197→          )
   198→        } else if (block.type === 'paragraph') {
   199→          children.push(
   200→            new Paragraph({
   201→              children: [new TextRun({ text: block.text, size: 24, color: theme.text })],
   202→              spacing: { after: 160, line: 340 },
   203→            })
   204→          )
   205→        } else if (block.type === 'bullets') {
   206→          for (const item of block.items) {
   207→            children.push(
   208→              new Paragraph({
   209→                children: [new TextRun({ text: item, size: 24, color: theme.text })],
   210→                bullet: { level: 0 },
   211→                spacing: { after: 100 },
   212→              })
   213→            )
   214→          }
   215→        } else if (block.type === 'table' && block.rows.length > 0) {
   216→          children.push(
   217→            new Table({
   218→              width: { size: 100, type: WidthType.PERCENTAGE },
   219→              rows: block.rows.map(
   220→                (row, rIdx) =>
   221→                  new TableRow({
   222→                    tableHeader: rIdx === 0,
   223→                    children: row.map(
   224→                      (cell) =>
   225→                        new TableCell({
   226→                          shading: rIdx === 0
   227→                            ? { type: ShadingType.CLEAR, fill: theme.primary, color: 'auto' }
   228→                            : rIdx % 2 === 1
   229→                              ? { type: ShadingType.CLEAR, fill: theme.surface, color: 'auto' }
   230→                              : undefined,
   231→                          margins: { top: 100, bottom: 100, left: 140, right: 140 },
   232→                          children: [
   233→                            new Paragraph({
   234→                              children: [
   235→                                new TextRun({
   236→                                  text: cell,
   237→                                  bold: rIdx === 0,
   238→                                  size: 22,
   239→                                  color: rIdx === 0 ? theme.white : theme.text,
   240→                                }),
   241→                              ],
   242→                            }),
   243→                          ],
   244→                        })
   245→                    ),
   246→                  })
   247→              ),
   248→            })
   249→          )
   250→          children.push(new Paragraph({ text: '', spacing: { after: 160 } }))
   251→        }
   252→      }
   253→
   254→      /* ---- Footer line ---- */
   255→      children.push(
   256→        new Paragraph({
   257→          children: [
   258→            new TextRun({ text: '— Generated by NEXUS AI · ', size: 18, color: theme.textMuted }),
   259→            new TextRun({ text: theme.name, size: 18, color: theme.textMuted, italics: true }),
   260→            new TextRun({ text: ' theme —', size: 18, color: theme.textMuted }),
   261→          ],
   262→          spacing: { before: 400 },
   263→          alignment: AlignmentType.CENTER,
   264→        })
   265→      )
   266→
   267→      const doc = new Document({
   268→        styles: {
   269→          default: {
   270→            heading1: { run: { color: theme.primaryDark, bold: true, size: 36 } },
   271→            heading2: { run: { color: theme.primary, bold: true, size: 30 } },
   272→            heading3: { run: { color: theme.primary, bold: true, size: 26 } },
   273→          },
   274→        },
   275→        sections: [{
   276→          children,
   277→          footers: {
   278→            default: new Footer({
   279→              children: [
   280→                new Paragraph({
   281→                  alignment: AlignmentType.CENTER,
   282→                  children: [
   283→                    new TextRun({ text: 'Page ', size: 18, color: theme.textMuted }),
   284→                    new TextRun({ children: [PageNumber.CURRENT], size: 18, color: theme.textMuted }),
   285→                  ],
   286→                }),
   287→              ],
   288→            }),
   289→          },
   290→        }],
   291→      })
   292→      buffer = Buffer.from(await Packer.toBuffer(doc))
   293→      filename = `${id}.docx`
   294→    } else if (format === 'xlsx') {
   295→      const ExcelJS = (await import('exceljs')).default
   296→      const theme = THEMES[themeKey] ?? THEMES.nexus
   297→      const wb = new ExcelJS.Workbook()
   298→      wb.creator = 'NEXUS AI'
   299→      wb.created = new Date()
   300→
   301→      const tables = blocks.filter((b): b is Extract<Block, { type: 'table' }> => b.type === 'table')
   302→
   303→      /* ---- Overview sheet ---- */
   304→      const overview = wb.addWorksheet('Overview', {
   305→        properties: { defaultRowHeight: 20 },
   306→      })
   307→      // Title banner (merged)
   308→      overview.mergeCells(1, 1, 1, 2)
   309→      const titleCell = overview.getCell(1, 1)
   310→      titleCell.value = title
   311→      titleCell.font = { bold: true, size: 16, color: { argb: 'FF' + theme.white } }
   312→      titleCell.fill = { type: 'pattern', pattern: 'solid', fgColor: { argb: 'FF' + theme.primary } }
   313→      titleCell.alignment = { vertical: 'middle', horizontal: 'left', indent: 1 }
   314→      overview.getRow(1).height = 32
   315→
   316→      // Subtitle
   317→      overview.mergeCells(2, 1, 2, 2)
   318→      const subCell = overview.getCell(2, 1)
   319→      subCell.value = 'Generated by NEXUS AI · ' + theme.name + ' theme · ' + new Date().toLocaleDateString()
   320→      subCell.font = { size: 10, color: { argb: 'FF' + theme.textMuted }, italic: true }
   321→
   322→      // Header row
   323→      const headerRow = overview.getRow(4)
   324→      headerRow.values = ['Section', 'Content']
   325→      headerRow.eachCell((cell) => {
   326→        cell.font = { bold: true, color: { argb: 'FF' + theme.white }, size: 11 }
   327→        cell.fill = { type: 'pattern', pattern: 'solid', fgColor: { argb: 'FF' + theme.primary } }
   328→        cell.alignment = { vertical: 'middle', indent: 1 }
   329→      })
   330→      overview.getRow(4).height = 22
   331→
   332→      let r = 5
   333→      for (const block of blocks) {
   334→        if (block.type === 'heading') {
   335→          overview.mergeCells(r, 1, r, 2)
   336→          const c = overview.getCell(r, 1)
   337→          c.value = block.text.toUpperCase()
   338→          c.font = { bold: true, size: 11, color: { argb: 'FF' + theme.primaryDark } }
   339→          c.fill = { type: 'pattern', pattern: 'solid', fgColor: { argb: 'FF' + theme.surfaceAlt } }
   340→          r++
   341→        } else if (block.type === 'paragraph') {
   342→          overview.getCell(r, 1).value = 'Text'
   343→          overview.getCell(r, 2).value = block.text
   344→          r++
   345→        } else if (block.type === 'bullets') {
   346→          for (const item of block.items) {
   347→            overview.getCell(r, 1).value = '•'
   348→            overview.getCell(r, 2).value = item
   349→            r++
   350→          }
   351→        } else if (block.type === 'slide') {
   352→          overview.getCell(r, 1).value = 'Slide: ' + block.title
   353→          overview.getCell(r, 2).value = block.bullets.join(' · ')
   354→          r++
   355→        }
   356→      }
   357→
   358→      // Zebra striping + borders on data rows
   359→      for (let row = 5; row < r; row++) {
   360→        const excelRow = overview.getRow(row)
   361→        if (row % 2 === 0) {
   362→          excelRow.eachCell((cell) => {
   363→            cell.fill = { type: 'pattern', pattern: 'solid', fgColor: { argb: 'FF' + theme.surface } }
   364→          })
   365→        }
   366→        excelRow.eachCell((cell) => {
   367→          cell.border = {
   368→            bottom: { style: 'thin', color: { argb: 'FF' + theme.border } },
   369→          }
   370→          if (cell.font && cell.font.bold) return
   371→          cell.font = { size: 11, color: { argb: 'FF' + theme.text } }
   372→        })
   373→      }
   374→      overview.getColumn(1).width = 28
   375→      overview.getColumn(2).width = 90
   376→      overview.views = [{ state: 'frozen', ySplit: 4 }]
   377→
   378→      /* ---- Data sheets ---- */
   379→      tables.forEach((t, i) => {
   380→        const sheet = wb.addWorksheet(`Data ${i + 1}`)
   381→        const header = sheet.getRow(1)
   382→        header.values = t.rows[0]
   383→        header.eachCell((cell) => {
   384→          cell.font = { bold: true, color: { argb: 'FF' + theme.white }, size: 11 }
   385→          cell.fill = { type: 'pattern', pattern: 'solid', fgColor: { argb: 'FF' + theme.primary } }
   386→          cell.alignment = { vertical: 'middle', horizontal: 'center', wrapText: true }
   387→        })
   388→        sheet.getRow(1).height = 24
   389→
   390→        for (let ri = 1; ri < t.rows.length; ri++) {
   391→          const row = sheet.getRow(ri + 1)
   392→          row.values = t.rows[ri]
   393→          if (ri % 2 === 0) {
   394→            row.eachCell((cell) => {
   395→              cell.fill = { type: 'pattern', pattern: 'solid', fgColor: { argb: 'FF' + theme.surface } }
   396→            })
   397→          }
   398→          row.eachCell((cell) => {
   399→            cell.border = { bottom: { style: 'thin', color: { argb: 'FF' + theme.border } } }
   400→            cell.font = { size: 11, color: { argb: 'FF' + theme.text } }
   401→          })
   402→        }
   403→
   404→        // Auto-width
   405→        const colCount = t.rows[0]?.length ?? 0
   406→        for (let c = 1; c <= colCount; c++) {
   407→          let maxLen = 10
   408→          for (const row of t.rows) {
   409→            const v = row[c - 1] ?? ''
   410→            maxLen = Math.max(maxLen, String(v).length + 2)
   411→          }
   412→          sheet.getColumn(c).width = Math.min(maxLen, 42)
   413→        }
   414→        sheet.views = [{ state: 'frozen', ySplit: 1 }]
   415→      })
   416→
   417→      buffer = Buffer.from(await wb.xlsx.writeBuffer())
   418→      filename = `${id}.xlsx`
   419→    } else if (format === 'pptx') {
   420→      const PptxGenJS = (await import('pptxgenjs')).default
   421→      const pptx = new PptxGenJS()
   422→      pptx.layout = 'LAYOUT_16x9'
   423→      pptx.title = title
   424→      const theme = THEMES[themeKey] ?? THEMES.nexus
   425→      const hex = (c: string) => c
   426→
   427→      /* ---- Master slides ---- */
   428→      pptx.defineSlideMaster({
   429→        title: 'NEXUS_TITLE',
   430→        background: { color: theme.primaryDark },
   431→        objects: [
   432→          // Top accent bar
   433→          { rect: { x: 0, y: 0, w: 10, h: 0.09, fill: { color: theme.accent } } },
   434→          // Decorative circle cluster (right)
   435→          { oval: { x: 8.7, y: 0.5, w: 1.7, h: 1.7, fill: { color: theme.primary }, line: { color: theme.accent, width: 0.75 } } },
   436→          { oval: { x: 9.35, y: 1.25, w: 0.85, h: 0.85, fill: { color: theme.accent } } },
   437→          // Bottom brand footer
   438→          { rect: { x: 0, y: 5.16, w: 10, h: 0.24, fill: { color: theme.primary } } },
   439→          { text: { text: 'NEXUS AI  ·  ' + title.slice(0, 60), options: { x: 0.5, y: 5.13, w: 7, h: 0.3, fontSize: 9, color: theme.surfaceAlt, align: 'left' } } },
   440→        ],
   441→      })
   442→
   443→      pptx.defineSlideMaster({
   444→        title: 'NEXUS_CONTENT',
   445→        background: { color: theme.white },
   446→        objects: [
   447→          // Left accent rail with gradient feel (two-tone)
   448→          { rect: { x: 0, y: 0, w: 0.14, h: 2.9, fill: { color: theme.primary } } },
   449→          { rect: { x: 0, y: 2.9, w: 0.14, h: 2.73, fill: { color: theme.accent } } },
   450→          // Corner decorative circles (subtle)
   451→          { oval: { x: 8.85, y: 4.35, w: 1.9, h: 1.9, fill: { color: theme.surface }, line: { color: theme.border, width: 0.5 } } },
   452→          { oval: { x: 9.5, y: 4.95, w: 0.75, h: 0.75, fill: { color: theme.surfaceAlt } } },
   453→          // Bottom-right brand chip
   454→          { rect: { x: 8.35, y: 0.32, w: 1.1, h: 0.34, fill: { color: theme.surface }, line: { color: theme.border, width: 0.5 } } },
   455→          { text: { text: 'NEXUS', options: { x: 8.35, y: 0.32, w: 1.1, h: 0.34, fontSize: 10, bold: true, color: theme.primary, align: 'center', fontFace: 'Segoe UI' } } },
   456→          // Footer
   457→          { text: { text: 'NEXUS AI  ·  ' + title.slice(0, 50), options: { x: 0.55, y: 5.3, w: 6, h: 0.25, fontSize: 8, color: theme.textMuted, align: 'left' } } },
   458→        ],
   459→        slideNumber: { x: 9.3, y: 5.3, w: 0.45, h: 0.25, fontSize: 9, color: theme.primary, bold: true },
   460→      })
   461→
   462→      pptx.defineSlideMaster({
   463→        title: 'NEXUS_SECTION',
   464→        background: { color: theme.primary },
   465→        objects: [
   466→          { rect: { x: 0, y: 0, w: 10, h: 0.09, fill: { color: theme.accent } } },
   467→          { oval: { x: -1.2, y: 3.4, w: 4.4, h: 4.4, fill: { color: theme.primaryDark } } },
   468→          { oval: { x: 8.2, y: -1.6, w: 3.6, h: 3.6, fill: { color: theme.accent, transparency: 55 } } },
   469→        ],
   470→      })
   471→
   472→      /* ---- Title slide ---- */
   473→      const titleSlide = pptx.addSlide({ masterName: 'NEXUS_TITLE' })
   474→      titleSlide.addText(title, {
   475→        x: 0.55, y: 1.85, w: 8.9, h: 1.35,
   476→        fontSize: 38, bold: true, color: theme.white, fontFace: 'Segoe UI',
   477→      })
   478→      titleSlide.addText('— ' + new Date().toLocaleDateString('en-US', { year: 'numeric', month: 'long', day: 'numeric' }) + ' —', {

   484→      const slides = blocks.filter((b): b is Extract<Block, { type: 'slide' }> => b.type === 'slide')
   485→      const otherBlocks = blocks.filter((b) => b.type !== 'slide')
   486→
   487→      const addSlideBlock = (slideTitle: string, bulletItems: string[]) => {

   489→

   494→        })
   495→        slide.addText(slideTitle, {
   496→          x: 0.55, y: 0.62, w: 6.2, h: 0.9,
   497→          fontSize: 24, bold: true, color: theme.primaryDark, fontFace: 'Segoe UI',
   498→        })
   499→        slide.addShape('rect', { x: 0.58, y: 1.52, w: 0.85, h: 0.05, fill: { color: theme.accent } })
   500→
   501→        // Detect a headline stat from bullets (e.g. "$2.4M" / "40%")
   502→        const statMatch = bulletItems
   503→          .map((b) => b.match(/([$€£]\d[\d.,]*\s?[MBKmbk]?|\d+(?:\.\d+)?%)/))
   504→          .find(Boolean)
   505→        const statValue = statMatch?.[1] ?? null
   506→
   507→        const hasStat = Boolean(statValue)
   508→
   509→        // Bullets (left column; narrower when stat panel present)
   510→        const textW = hasStat ? 5.4 : 8.3
   511→        slide.addText(
   512→          bulletItems.slice(0, 5).map((b, i) => ({
   513→            text: b.replace(/([$€£]\d[\d.,]*\s?[MBKmbk]?|\d+(?:\.\d+)?%)/, '$1'),
   514→            options: {
   515→              bullet: false,
   516→              color: theme.text,
   517→              breakLine: true,
   518→              paraSpaceAfter: 10,
   519→            },
   520→          })),
   521→          {
   522→            x: 0.9, y: 1.85, w: textW, h: 3.1,
   523→            fontSize: 14.5, lineSpacingMultiple: 1.25, fontFace: 'Segoe UI',
   524→          }
   525→        )
   526→        // Custom square markers for each bullet line
   527→        bulletItems.slice(0, 5).forEach((_, i) => {
   528→          slide.addShape('rect', {
   529→            x: 0.58, y: 2.02 + i * 0.52, w: 0.09, h: 0.09,
   530→            fill: { color: i % 2 === 0 ? theme.primary : theme.accent },
   531→          })
   532→        })
   533→
   534→        // Right panel: big stat card or decorative block
   535→        if (hasStat) {
   536→          slide.addShape('roundRect', {
   537→            x: 6.55, y: 1.85, w: 2.9, h: 2.9, rectRadius: 0.12,
   538→            fill: { color: theme.surface }, line: { color: theme.border, width: 0.75 },
   539→          })
   540→          slide.addText(statValue!, {
   541→            x: 6.55, y: 2.5, w: 2.9, h: 1.0,
   542→            fontSize: 44, bold: true, color: theme.primary, align: 'center', fontFace: 'Segoe UI',
   543→          })
   544→          slide.addText('KEY METRIC', {
   545→            x: 6.55, y: 3.65, w: 2.9, h: 0.3,
   546→            fontSize: 10, bold: true, color: theme.textMuted, align: 'center', charSpacing: 3,
   547→          })
   548→        } else {
   549→          slide.addShape('ellipse', {
   550→            x: 7.75, y: 2.35, w: 1.55, h: 1.55,
   551→            fill: { color: theme.surface }, line: { color: theme.border, width: 0.75 },
   552→          })
   553→          slide.addShape('ellipse', {
   554→            x: 8.3, y: 2.9, w: 0.55, h: 0.55,
   555→            fill: { color: theme.accent },
   556→          })
   557→        }
   558→      }
   559→
   560→      const addSectionDivider = (text: string) => {
   561→        const slide = pptx.addSlide({ masterName: 'NEXUS_SECTION' })
   562→        slide.addText(text, {
   563→          x: 0.9, y: 2.15, w: 8.2, h: 1.0,
   564→          fontSize: 30, bold: true, color: theme.white, fontFace: 'Segoe UI',
   565→        })
   566→        slide.addShape('rect', { x: 0.95, y: 3.2, w: 1.6, h: 0.06, fill: { color: theme.accent } })
   567→      }
   568→
   569→      const addTableSlide = (heading: string, rows: string[][]) => {
   570→        const slide = pptx.addSlide({ masterName: 'NEXUS_CONTENT' })
   571→        slide.addText(heading, {
   572→          x: 0.55, y: 0.45, w: 8.9, h: 0.8,
   573→          fontSize: 24, bold: true, color: theme.primaryDark, fontFace: 'Segoe UI',
   574→        })
   575→        slide.addShape('rect', { x: 0.58, y: 1.25, w: 1.15, h: 0.055, fill: { color: theme.accent } })
   576→        const tableRows = rows.map((row, i) => ({
   577→          options: {
   578→            bold: i === 0,
   579→            color: i === 0 ? theme.white : theme.text,
   580→            fill: { color: i === 0 ? theme.primary : i % 2 === 1 ? theme.surface : theme.white },
   581→            fontSize: 13,
   582→            valign: 'middle' as const,
   583→          },
   584→          text: row.map((c) => c.replace(/\|/g, '/')),
   585→        }))
   586→        slide.addTable(tableRows, {
   587→          x: 0.55, y: 1.6, w: 8.9, fontSize: 13,
   588→          border: { pt: 0.5, color: theme.border },
   589→          rowH: 0.42, margin: 0.08,
   590→        })
   591→      }
   592→
   593→      if (slides.length > 0) {
   594→        for (const sl of slides) {
   595→          if (sl.title.startsWith('~') || sl.bullets.length === 0) {
   596→            addSectionDivider(sl.title.replace(/^~\s*/, ''))
   597→          } else {
   598→            addSlideBlock(sl.title, sl.bullets)
   599→          }
   600→        }
   601→      }
   602→
   603→      // Auto-convert other blocks
   604→      let currentBullets: string[] = []
   605→      let currentTitle = ''
   606→      const flush = () => {
   607→        if (currentBullets.length === 0) return
   608→        addSlideBlock(currentTitle || title, currentBullets.slice(0, 6))
   609→        currentBullets = []
   610→      }
   611→      for (const block of otherBlocks) {
   612→        if (block.type === 'heading') {
   613→          flush()
   614→          currentTitle = block.text
   615→        } else if (block.type === 'bullets') {
   616→          currentBullets.push(...block.items)
   617→        } else if (block.type === 'paragraph') {
   618→          const sentences = block.text.match(/[^.!?]+[.!?]+/g) ?? [block.text]
   619→          currentBullets.push(...sentences.slice(0, 5).map((x) => x.trim()))
   620→        } else if (block.type === 'table' && block.rows.length > 0) {
   621→          flush()
   622→          addTableSlide(currentTitle || 'Data', block.rows)
   623→        }
   624→        if (currentBullets.length >= 6) flush()
   625→      }
   626→      flush()
   627→
   628→      /* ---- Closing slide ---- */
   629→      const closing = pptx.addSlide({ masterName: 'NEXUS_TITLE' })
   630→      closing.addText('Thank You', {
   631→        x: 0.55, y: 2.1, w: 8.9, h: 1.0,
   632→        fontSize: 40, bold: true, color: theme.white, align: 'center',
   633→      })
   634→      closing.addText('Generated by NEXUS AI · ' + theme.name + ' theme', {
   635→        x: 0.55, y: 3.2, w: 8.9, h: 0.4,
   636→        fontSize: 13, color: theme.surfaceAlt, align: 'center', italic: true,
   637→      })
   638→
   639→      buffer = Buffer.from((await pptx.write({ outputType: 'nodebuffer' })) as ArrayBuffer)
   640→      filename = `${id}.pptx`
   641→    } else {
   642→      // Markdown
   643→      const lines: string[] = [`# ${title}`, '']
   644→      for (const block of blocks) {
   645→        if (block.type === 'heading') {
   646→          lines.push(`${'#'.repeat(Math.min(block.level ?? 2, 4) + 1)} ${block.text}`, '')
   647→        } else if (block.type === 'paragraph') {
   648→          lines.push(block.text, '')
   649→        } else if (block.type === 'bullets') {
   650→          block.items.forEach((i) => lines.push(`- ${i}`))
   651→          lines.push('')
   652→        } else if (block.type === 'table') {
   653→          if (block.rows.length > 0) {
   654→            lines.push(`| ${block.rows[0].join(' | ')} |`)
   655→            lines.push(`| ${block.rows[0].map(() => '---').join(' | ')} |`)
   656→            block.rows.slice(1).forEach((r) => lines.push(`| ${r.join(' | ')} |`))
   657→            lines.push('')
   658→          }
   659→        } else if (block.type === 'slide') {
   660→          lines.push(`## ${block.title}`)
   661→          block.bullets.forEach((b) => lines.push(`- ${b}`))
   662→          lines.push('')
   663→        }
   664→      }
   665→      buffer = Buffer.from(lines.join('\n'), 'utf8')
   666→      filename = `${id}.md`
   667→    }
   668→
   669→    await writeFile(path.join(FILES_DIR, filename), buffer)
   670→
   671→    return NextResponse.json({
   672→      file: {
   673→        url: `/api/office/file/${id}`,
   674→        format,
   675→        title,
   676→        filename: `${title.replace(/[^a-zA-Z0-9 _-]/g, '').slice(0, 60) || 'document'}.${format}`,
   677→        size: buffer.length,
   678→        mimeType: MIME_TYPES[format],
   679→      },
   680→    })
   681→  } catch (error) {
   682→    console.error('[api/office/create] POST error:', error)
   683→    const message = error instanceof Error ? error.message : 'Document generation failed.'
   684→    return NextResponse.json({ error: message }, { status: 500 })
   685→  }
   686→}

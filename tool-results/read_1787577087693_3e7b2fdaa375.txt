    94→function AuthGate({ isGuest, onContinueAsGuest, language, onToggleLanguage }: {
    95→  isGuest: boolean
    96→  onContinueAsGuest: () => void
    97→  language: 'en' | 'ar'
    98→  onToggleLanguage: () => void
    99→}) {
   100→  const { user } = useAuth()
   101→  const [authOpen, setAuthOpen] = useState(false)
   102→  const [authMode, setAuthMode] = useState<'signin' | 'signup'>('signin')
   103→  const [legalPage, setLegalPage] = useState<'privacy' | 'terms' | null>(null)
   104→
   105→  const openAuth = (mode: 'signin' | 'signup') => {
   106→    setAuthMode(mode)
   107→    setAuthOpen(true)
   108→  }
   109→
   110→  // Listen for "open legal page" requests (Privacy / Terms)
   111→  useEffect(() => {
   112→    const handler = (e: Event) => {
   113→      const tp = (e as CustomEvent<'privacy' | 'terms'>).detail
   114→      if (tp === 'privacy' || tp === 'terms') setLegalPage(tp)
   115→    }
   116→    window.addEventListener('nexus:open-legal', handler as EventListener)
   117→    return () => window.removeEventListener('nexus:open-legal', handler as EventListener)
   118→  }, [])
   119→
   120→  // If the user is signed in OR has chosen guest mode, show the app.
   121→  if (user || isGuest) {
   122→    return <NexusApp />
   123→  }
   124→
   125→  // Otherwise, show the dedicated auth landing page.
   126→  return (
   127→    <>
   128→      <AuthLanding
   129→        onSignIn={() => openAuth('signin')}
   130→        onSignUp={() => openAuth('signup')}
   131→        onContinueAsGuest={onContinueAsGuest}
   132→        onOpenPrivacy={() => setLegalPage('privacy')}
   133→        onOpenTerms={() => setLegalPage('terms')}
   134→        language={language}
   135→        onToggleLanguage={onToggleLanguage}
   136→      />
   137→      <AuthModal open={authOpen} onClose={() => setAuthOpen(false)} initialMode={authMode} />
   138→      {legalPage !== null && (
   139→        <LegalPage type={legalPage} onClose={() => setLegalPage(null)} language={language} />
   140→      )}
   141→    </>
   142→  )
   143→}

   211→    useCallback((running: boolean, label?: string) => {

   214→  )
   215→
   216→  // Send: routes through tool engine if a tool is pending, else plain chat.
   217→  // Allows empty text when a tool with a file attachment is pending — e.g.
   218→  // user picks a PDF via "Upload file" and just hits Send to get a summary,
   219→  // without being forced to type a question first.
   220→  const send = useCallback(async (text?: string) => {
   221→    const msg = (text ?? input).trim()
   222→    const hasPendingFile = !!toolEngine.pendingFile
   223→    const hasPendingTool = !!toolEngine.pendingTool
   224→    // Block only when there's truly nothing to send
   225→    if (sending) return
   226→    if (!msg && !hasPendingFile) return
   227→    // For plain chat (no tool), still require text
   228→    if (!msg && !hasPendingTool) return
   229→    // Display the user's text (or the file name as a placeholder)
   230→    const userDisplay = msg || (hasPendingFile ? `📄 ${toolEngine.pendingFile?.name ?? ''}`.trim() : '')
   231→    if (userDisplay) {
   232→      setMessages(prev => [...prev, { id: `u-${Date.now()}`, role: 'user', content: userDisplay }])
   233→    }
   234→    setInput('')
   235→    setSending(true)
   236→    try {
   237→      // If a tool is pending, route to the tool engine
   238→      if (hasPendingTool) {
   239→        await toolEngine.execute(msg)
   240→        return
   241→      }
   242→      // Otherwise: plain chat stream
   243→      // Phase 1 P2: when an artifact is open, pass its current state to the
   244→      // server so the AI can emit ARTIFACT_PATCH directives for targeted
   245→      // edits instead of regenerating the whole document via create_document.
   246→      const openArtifactPayload = openArtifactState
   247→        ? {
   248→            artifactId: openArtifactState.artifactId ?? openArtifactState.id,
   249→            type: openArtifactState.type,
   250→            title: openArtifactState.title,
   251→            // Send the CURRENT version content (latest from AI patches +
   252→            // user edits). The ArtifactPanel holds the version stack but
   253→            // the chat request only needs the latest content for the prompt.
   254→            content: (openArtifactState.content ?? '').slice(0, 20000),
   255→          }
   256→        : null
   257→      // Phase 1 P3: include the session id (for continuity) and the active project
   258→      // id (for new sessions being created inside a project). When resuming an
   259→      // existing session, sessionId is set and projectId is omitted — the
   260→      // server reads session.projectId as the authoritative binding. When
   261→      // starting a new conversation in a project, sessionId is null and
   262→      // projectId is set — the server creates a new session bound to that
   263→      // project.
   264→      const res = await fetch('/api/chat', {
   265→        method: 'POST',
   266→        headers: { 'Content-Type': 'application/json' },
   267→        body: JSON.stringify({
   268→          message: msg,
   269→          language,
   270→          sessionId: currentChatSessionId,
   271→          projectId: activeProjectId,
   272→          openArtifact: openArtifactPayload,
   273→        }),
   274→      })
   275→      const reader = res.body!.getReader()
   276→      const decoder = new TextDecoder()
   277→      let buf = ''
   278→      let streamingId: string | null = null
   279→
   280→      // Helper: process one NDJSON line. Returns true if a streaming bubble
   281→      // was opened (so the caller can hide the shimmer placeholder).
   282→      const handleLine = (line: string) => {
   283→        const trimmed = line.trim()
   284→        if (!trimmed) return
   285→        try {
   286→          const e = JSON.parse(trimmed)
   287→          // Streaming chat protocol: assistant_start opens a bubble, deltas append, end finalizes.
   288→          if (e.type === 'assistant_start') {
   289→            // Capture streamingId in a local const so the setMessages updater
   290→            // closure below references the SAME value even if streamingId is
   291→            // later mutated by another event before React runs the updater.
   292→            const id = `a-${Date.now()}`
   293→            streamingId = id
   294→            setStreamingActive(true)
   295→            setMessages(prev => [...prev, { id, role: 'assistant', content: '', attachments: [] }])
   296→          } else if (e.type === 'assistant_delta') {
   297→            const delta = e.delta ?? ''
   298→            if (!delta) return
   299→            // Capture the current streamingId locally so the updater sees the
   300→            // correct value even if streamingId has since been mutated.
   301→            const id = streamingId
   302→            setMessages(prev => {
   303→              if (!id) return prev
   304→              const next = [...prev]
   305→              const idx = next.findIndex(m => m.id === id)
   306→              if (idx >= 0) {
   307→                const m = next[idx]
   308→                next[idx] = { ...m, content: (m.content || '') + delta }
   309→              }
   310→              return next
   311→            })
   312→          } else if (e.type === 'assistant_end') {
   313→            // Capture the current streamingId locally so the updater sees the
   314→            // correct value even if streamingId has since been mutated. This
   315→            // is the bug that previously caused the second assistant_end's
   316→            // attachments to be applied to the wrong bubble (or to no bubble
   317→            // because streamingId had been nulled).
   318→            const id = streamingId
   319→            if (id) {
   320→              const atts = e.attachments ?? []
   321→              setStreamingActive(false)
   322→              setMessages(prev => {
   323→                const next = [...prev]
   324→                const idx = next.findIndex(m => m.id === id)
   325→                if (idx >= 0) {
   326→                  const m = next[idx]
   327→                  // Drop an empty-content bubble if no deltas arrived AND no attachments
   328→                  // (e.g. when the model only emitted a TOOL_CALL with no prelude)
   329→                  if (!m.content && atts.length === 0) {
   330→                    next.splice(idx, 1)
   331→                  } else {
   332→                    next[idx] = { ...m, attachments: atts }
   333→                  }
   334→                }
   335→                return next
   336→              })
   337→              streamingId = null
   338→            }
   339→          } else if (e.type === 'assistant') {
   340→            // Legacy non-streamed path (creator-identity shortcut, etc.)
   341→            setMessages(prev => [...prev, { id: `a-${Date.now()}`, role: 'assistant', content: e.content, attachments: e.attachments }])
   342→          } else if (e.type === 'artifact_patch') {
   343→            // Phase 1 P2: AI applied a targeted find/replace edit to the
   344→            // open artifact. Queue it for the ArtifactPanel to apply +
   345→            // push as a new version (preserves undo/redo history).
   346→            enqueuePatch({
   347→              artifactId: e.artifactId,
   348→              find: e.find,
   349→              replace: e.replace,
   350→              note: e.note,
   351→            })
   352→          } else if (e.type === 'done') {
   353→            // Phase 1 P3: capture the sessionId returned by the server so the
   354→            // next message resumes this session (with its message history +
   355→            // project context) instead of creating a new one each turn.
   356→            if (typeof e.sessionId === 'string' && e.sessionId) {
   357→              setCurrentChatSessionId(e.sessionId)
   358→            }
   359→          }
   360→        } catch {}
   361→      }
   362→
   363→      for (;;) {
   364→        const { done, value } = await reader.read()
   365→        if (done) break
   366→        buf += decoder.decode(value, { stream: true })
   367→        const lines = buf.split('\n')
   368→        buf = lines.pop() ?? '' // keep the last (possibly partial) line for next chunk
   369→        for (const line of lines) handleLine(line)
   370→      }
   371→      // FLUSH: the reader's final chunk often contains a partial line (no
   372→      // trailing '\n' because the stream was closed) AND the TextDecoder
   373→      // may still hold an incomplete UTF-8 sequence (common with Arabic,
   374→      // where chars are 2+ bytes). Both need to be flushed here, otherwise
   375→      // the last 1-2 tokens (e.g. "تك اليوم؟" in an Arabic reply) get lost
   376→      // and the assistant message looks truncated mid-word.
   377→      buf += decoder.decode() // flush remaining UTF-8 bytes
   378→      if (buf.trim()) handleLine(buf)
   379→      buf = ''
   380→    } catch (err: any) {
   381→      setStreamingActive(false)
   382→      setMessages(prev => [...prev, { id: `a-${Date.now()}`, role: 'assistant', content: `⚠️ ${err.message || 'Something went wrong.'}` }])
   383→    } finally {
   384→      setSending(false)
   385→      setStreamingActive(false)
   386→    }
   387→  }, [input, sending, toolEngine, openArtifactState, enqueuePatch, currentChatSessionId, activeProjectId])

   109→   109→  /** Streams text word-by-word into the UI — the 'alive' ChatGPT feel.

   112→   112→    const words = fullText.split(/(\s+)/) // keep whitespace tokens
   113→   113→    let i = 0
   114→   114→    let cancelled = false
   115→   115→    let timer: ReturnType<typeof setInterval> | null = null
   116→   116→
   117→   117→    const done = new Promise<void>((resolve) => {
   118→   118→      timer = setInterval(() => {
   119→   119→        if (cancelled || stopRequested) {
   120→   120→          if (timer) clearInterval(timer)
   121→   121→          resolve()
   122→   122→          return
   123→   123→        }
   124→   124→        // Emit 1-3 words per tick for natural pacing
   125→   125→        const burst = words.slice(i, i + (Math.random() > 0.7 ? 2 : 1))
   126→   126→        i += burst.length
   127→   127→        setStreamingText(words.slice(0, i).join(''))
   128→   128→        if (i >= words.length) {
   129→   129→          if (timer) clearInterval(timer)
   130→   130→          resolve()
   131→   131→        }
   132→   132→        // keep pinned while streaming (respects userScrolledUp)
   133→   133→        requestAnimationFrame(() => {
   134→   134→          const el = scrollRef.current
   135→   135→          if (el && !userScrolledUpRef.current) {
   136→   136→            el.scrollTop = el.scrollHeight
   137→   137→          }
   138→   138→        })
   139→   139→      }, 24)
   140→   140→    })
   141→   141→
   142→   142→    return {
   143→   143→      stop: () => {
   144→   144→        cancelled = true
   145→   145→        if (timer) clearInterval(timer)
   146→   146→      },
   147→   147→      done,
   148→   148→    }
   149→   149→  }, [stopRequested])
   150→   150→
   151→   151→  const send = useCallback(
   152→   152→    async (text: string) => {
   153→   153→      const trimmed = text.trim()

   155→   155→
   156→   156→      const userMsg: ChatMessageItem = {
   157→   157→        id: `local-${Date.now()}`,
   158→   158→        role: 'user',
   159→   159→        content: trimmed,
   160→   160→      }
   161→   161→      setMessages((prev) => [...prev, userMsg])
   162→   162→      setInput('')
   163→   163→      setSending(true)
   164→   164→      setLiveToolSteps([])
   165→   165→      scrollToBottom()

   167→   167→      // ============ ENGINE 1: PUTER (primary — 507 free models) ============
   168→   168→      try {
   169→   169→        const puterReady = await isPuterReady()

   171→   171→          // Stream Puter reply word-by-word (ChatGPT feel)
   172→   172→          setStreamingText('')
   173→   173→          const result = await puterChat(
   174→   174→            `You are NEXUS AI, created by Mounir Shaaban. ${trimmed}`,
   175→   175→            'gpt-5-nano'
   176→   176→          )
   177→   177→          if (result.ok && result.text) {
   178→   178→            const streamer = streamWords(result.text)
   179→   179→            streamerRef.current = streamer
   180→   180→            await streamer.done
   181→   181→            streamerRef.current = null
   182→   182→            setStreamingText('')

   184→   184→              ...prev.map((m) => (m.id === userMsg.id ? { ...m, id: `u-${Date.now()}` } : m)),
   185→   185→              { id: `a-${Date.now()}`, role: 'assistant', content: result.text },
   186→   186→            ])
   187→   187→            scrollToBottom()
   188→   188→            return // Puter handled it — done!
   189→   189→          }
   190→   190→        }
   191→   191→      } catch {
   192→   192→        // Puter failed (not signed in / error) → fall through to server AI
   193→   193→      }
   194→   194→
   195→   195→      // ============ ENGINE 2: SERVER AI (smart router: DeepSeek→OpenRouter→Z.ai) ============
   196→   196→      setStopRequested(false)
   197→   197→      abortRef.current = new AbortController()
   198→   198→      try {
   199→   199→        const res = await fetch('/api/chat', {
   200→   200→          method: 'POST',
   201→   201→          headers: {
   202→   202→            'Content-Type': 'application/json',
   203→   203→            ...(supabaseUserIdRef.current ? { 'x-supabase-user-id': supabaseUserIdRef.current } : {}),
   204→   204→          },
   205→   205→          body: JSON.stringify({ message: trimmed, sessionId, thinking: thinkingMode }),
   206→   206→          signal: abortRef.current.signal,
   207→   207→        })
   208→   208→
   209→   209→        if (!res.ok || !res.body) {
   210→   210→          const data = await res.json().catch(() => null)
   211→   211→          throw new Error(data?.error || 'Chat request failed.')
   212→   212→        }
   213→   213→
   214→   214→        // Read the unified NDJSON stream (tool calls + final answer)
   215→   215→        const reader = res.body.getReader()
   216→   216→        const decoder = new TextDecoder()
   217→   217→        let buffer = ''
   218→   218→
   219→   219→        for (;;) {
   220→   220→          const { done, value } = await reader.read()
   221→   221→          if (done) break
   222→   222→          buffer += decoder.decode(value, { stream: true })
   223→   223→          const lines = buffer.split('\n')
   224→   224→          buffer = lines.pop() ?? ''
   225→   225→          for (const line of lines) {
   226→   226→            const text = line.trim()
   227→   227→            if (!text) continue
   228→   228→            try {
   229→   229→              const event = JSON.parse(text) as ChatEvent
   230→   230→              if (event.type === 'user') {
   231→   231→                setMessages((prev) => prev.map((m) => (m.id === userMsg.id ? { id: event.id, role: 'user', content: event.content } : m)))
   232→   232→              } else if (event.type === 'tool_start') {
   233→   233→                setLiveToolSteps((prev) => [...prev, { tool: event.tool, args: event.args, status: 'running' }])
   234→   234→                scrollToBottom()
   235→   235→              } else if (event.type === 'tool_result') {
   236→   236→                setLiveToolSteps((prev) =>
   237→   237→                  prev.map((st, i) => (i === prev.length - 1 && st.tool === event.tool ? { ...st, status: event.ok ? 'done' : 'error' } : st))
   238→   238→                )
   239→   239→              } else if (event.type === 'assistant') {
   240→   240→                setLiveToolSteps([])
   241→   241→                const streamer = streamWords(event.content)
   242→   242→                streamerRef.current = streamer
   243→   243→                setStreamingAttachments(event.attachments ?? [])
   244→   244→                await streamer.done
   245→   245→                streamerRef.current = null
   246→   246→                setStreamingText('')
   247→   247→                setStreamingAttachments([])
   248→   248→                setMessages((prev) => [
   249→   249→                  ...prev.map((m) => (m.id === userMsg.id && m.id.startsWith('local-') ? { ...m, id: `u-${Date.now()}` } : m)),
   250→   250→                  { id: `a-${Date.now()}`, role: 'assistant', content: event.content, attachments: event.attachments },
   251→   251→                ])
   252→   252→                scrollToBottom()
   253→   253→              } else if (event.type === 'done') {
   254→   254→                setSessionId(event.sessionId)
   255→   255→              } else if (event.type === 'error') {
   256→   256→                throw new Error(event.message)
   257→   257→              }
   258→   258→            } catch (parseErr) {
   259→   259→              if (parseErr instanceof Error && !parseErr.message.includes('JSON')) throw parseErr
   260→   260→            }
   261→   261→          }
   262→   262→        }
   263→   263→        refreshSessions()
   264→   264→      } catch (error) {
   265→   265→        // ============ ENGINE 3: PUTER AS LAST RESORT (if signed in) ============
   266→   266→        try {
   267→   267→          const result = await puterChat(
   268→   268→            `You are NEXUS AI, created by Mounir Shaaban. ${trimmed}`,
   269→   269→            'gpt-5-nano'
   270→   270→          )
   271→   271→          if (result.ok && result.text) {
   272→   272→            setMessages((prev) => [
   273→   273→              ...prev.map((m) => (m.id === userMsg.id ? { ...m, id: `u-${Date.now()}` } : m)),
   274→   274→              { id: `a-${Date.now()}`, role: 'assistant', content: result.text },
   275→   275→            ])
   276→   276→            setInput('')
   277→   277→            scrollToBottom()
   278→   278→            return
   279→   279→          }
   280→   280→        } catch {
   281→   281→          /* all engines failed */
   282→   282→        }
   283→   283→        toast({
   284→   284→          title: 'Message failed',
   285→   285→          description: error instanceof Error ? error.message : 'Please try again.',
   286→   286→          variant: 'destructive',
   287→   287→        })
   288→   288→        setMessages((prev) => prev.filter((m) => m.id !== userMsg.id))
   289→   289→        setInput(trimmed)
   290→   290→      } finally {
   291→   291→        setSending(false)
   292→   292→        setLiveToolSteps([])
   293→   293→        setStreamingText('')
   294→   294→        setStreamingAttachments([])
   295→   295→      }
   296→   296→    },
   297→   297→    [sessionId, sending, scrollToBottom, refreshSessions, toast, thinkingMode]
   298→   298→  )
   299→   299→
   300→   300→  // Fire the initial prompt as soon as one arrives and the composer is idle
   301→   301→  useEffect(() => {
   302→   302→    if (initialPrompt && !sentInitialRef.current && !sending) {

   304→   304→      const p = initialPrompt

   307→   307→    }
   308→   308→  }, [initialPrompt, send, sending, onInitialPromptConsumed])
   309→   309→
   310→   310→  const regenerate = useCallback(async () => {
   311→   311→    if (!sessionId || sending) return
   312→   312→    setSending(true)

   315→   315→      const copy = [...prev]
   316→   316→      while (copy.length && copy[copy.length - 1].role === 'assistant') copy.pop()
   317→   317→      return copy
   318→   318→    })
   319→   319→    scrollToBottom()
   320→   320→    try {
   321→   321→      const res = await fetch('/api/chat/regenerate', {
   322→   322→        method: 'POST',
   323→   323→        headers: { 'Content-Type': 'application/json' },
   324→   324→        body: JSON.stringify({ sessionId }),
   325→   325→      })
   326→   326→      const data = await res.json()
   327→   327→      if (!res.ok) throw new Error(data.error || 'Regeneration failed.')
   328→   328→      setMessages((prev) => [...prev, data.reply])
   329→   329→      scrollToBottom()
   330→   330→    } catch (error) {
   331→   331→      toast({
   332→   332→        title: 'Regeneration failed',
   333→   333→        description: error instanceof Error ? error.message : 'Please try again.',
   334→   334→        variant: 'destructive',
   335→   335→      })
   336→   336→    } finally {
   337→   337→      setSending(false)
   338→   338→    }
   339→   339→  }, [sessionId, sending, scrollToBottom, toast])
   340→   340→
   341→   341→  const { artifact, openArtifact, closeArtifact } = useArtifact()
   342→   342→  const [puterActive, setPuterActive] = useState(false)

   348→   348→      supabaseUserIdRef.current = u?.id ?? null

   350→   350→  }, [])
   351→   351→
   352→   352→  // Check Puter sign-in state once — hide activation banner if already active

   354→   354→    isPuterReady().then(setPuterActive).catch(() => {})
   355→   355→  }, [])
   356→   356→  const [editingId, setEditingId] = useState<string | null>(null)
   357→   357→  const [editText, setEditText] = useState('')
   358→   358→  const [feedbackGiven, setFeedbackGiven] = useState<Record<string, 'up' | 'down'>>({})
   359→   359→
   360→   360→  const copyText = useCallback(async (text: string) => {

   373→   373→        title: kind === 'up' ? 'Thanks for the feedback!' : 'Feedback noted — thanks!',

   402→   402→    }
   403→   403→    setStreamingText('')
   404→   404→    setStreamingAttachments([])
   405→   405→    setSending(false)
   406→   406→    setLiveToolSteps([])

   409→   409→  const loadSession = useCallback(
   410→   410→    async (id: string) => {
   411→   411→      if (sending) return
   412→   412→      setHistoryOpen(false)
   413→   413→      try {
   414→   414→        const res = await fetch(`/api/chat/sessions/${id}`)

   416→   416→        const data = await res.json()
   417→   417→        setSessionId(data.session.id)
   418→   418→        setMessages(
   419→   419→          (data.session.messages as ChatMessageItem[]).map((m) => ({
   420→   420→            ...m,
   421→   421→            thinking: m.thinking ?? null,
   422→   422→          }))
   423→   423→        )
   424→   424→        setTimeout(() => scrollToBottom(false), 60)
   425→   425→      } catch (error) {
   426→   426→        toast({
   427→   427→          title: 'Could not open conversation',
   428→   428→          description: error instanceof Error ? error.message : undefined,
   429→   429→          variant: 'destructive',
   430→   430→        })
   431→   431→      }
   432→   432→    },
   433→   433→    [sending, scrollToBottom, toast]
   434→   434→  )
   435→   435→
   436→   436→  const deleteSession = useCallback(
   437→   437→    async (id: string, e: React.MouseEvent) => {
   438→   438→      e.stopPropagation()

   440→   440→        const res = await fetch(`/api/chat/sessions/${id}`, { method: 'DELETE' })
   441→   441→        if (!res.ok) throw new Error('Delete failed.')
   442→   442→        setSessions((prev) => prev.filter((s) => s.id !== id))
   443→   443→        if (sessionId === id) {
   444→   444→          setSessionId(null)
   445→   445→          setMessages([])
   446→   446→        }
   447→   447→        toast({ title: 'Conversation deleted' })
   448→   448→      } catch {
   449→   449→        toast({ title: 'Could not delete conversation', variant: 'destructive' })
   450→   450→      }
   451→   451→    },
   452→   452→    [sessionId, toast]
   453→   453→  )
   454→   454→
   455→   455→  const startNew = () => {
   456→   456→    if (sending) return
   457→   457→    setSessionId(null)
   458→   458→    setMessages([])
   459→   459→    setInput('')
   460→   460→  }
   461→   461→
   462→   462→  const activeSession = sessions.find((s) => s.id === sessionId)
   463→   463→
   464→   464→  return (
   465→   465→    <div className="flex h-full min-h-0 flex-col">
   466→   466→      {/* Chat header — ChatGPT style: ability picker + actions */}
   467→   467→      <header className="flex items-center justify-between gap-3 border-b border-border/50 px-5 py-3 sm:px-8">

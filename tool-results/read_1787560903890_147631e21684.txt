    59→    59→export function VoiceModeOverlay({ open, onClose }: { open: boolean; onClose: () => void }) {
    60→    60→  const { toast } = useToast()
    61→    61→  const [state, setState] = useState<VoiceState>('idle')
    62→    62→  const [iframeBlocked, setIframeBlocked] = useState(false)
    63→    63→  const [audioUnlocked, setAudioUnlocked] = useState(false)
    64→    64→  const [lang, setLang] = useState('en-US')
    65→    65→  const [ttsVoice, setTtsVoice] = useState('en-US-AriaNeural')
    66→    66→  const [interim, setInterim] = useState('')
    67→    67→  const [micLevel, setMicLevel] = useState(0)
    68→    68→  const [turns, setTurns] = useState<Turn[]>([])
    69→    69→  const [showVoicePicker, setShowVoicePicker] = useState(false)
    70→    70→  const [showLangPicker, setShowLangPicker] = useState(false)
    71→    71→  const [error, setError] = useState('')
    72→    72→
    73→    73→  const recognitionRef = useRef<any>(null)
    74→    74→  const historyRef = useRef<Array<{ role: string; content: string }>>([])
    75→    75→  const audioRef = useRef<HTMLAudioElement | null>(null)
    76→    76→  const stateRef = useRef<VoiceState>('idle')
    77→    77→  const audioCtxRef = useRef<AudioContext | null>(null)
    78→    78→  const sourceRef = useRef<AudioBufferSourceNode | null>(null)
    79→    79→  const scrollRef = useRef<HTMLDivElement | null>(null)
    80→    80→
    81→    81→  const setStateSafe = useCallback((s: VoiceState) => {
    82→    82→    stateRef.current = s
    83→    83→    setState(s)
    84→    84→  }, [])
    85→    85→
    86→    86→  const stopSpeaking = useCallback(() => {
    87→    87→    if (audioRef.current) { audioRef.current.pause(); audioRef.current = null }
    88→    88→    if (sourceRef.current) { try { sourceRef.current.stop() } catch {} sourceRef.current = null }
    89→    89→  }, [])
    90→    90→
    91→    91→  const stopRecognition = useCallback(() => {
    92→    92→    if (recognitionRef.current) {
    93→    93→      try {
    94→    94→        const pulse = (recognitionRef.current as any).__pulse
    95→    95→        if (pulse) clearInterval(pulse)
    96→    96→        recognitionRef.current.onend = null
    97→    97→        recognitionRef.current.onresult = null
    98→    98→        recognitionRef.current.onerror = null
    99→    99→        recognitionRef.current.stop()
   100→   100→      } catch {}
   101→   101→      recognitionRef.current = null
   102→   102→      setMicLevel(0)
   103→   103→    }
   104→   104→  }, [])
   105→   105→
   106→   106→  /* ---------- WebAudio unlock (autoplay policy) ---------- */
   107→   107→  const unlockAudio = useCallback(() => {
   108→   108→    try {
   109→   109→      if (!audioCtxRef.current || audioCtxRef.current.state === 'closed') {
   110→   110→        const AudioCtx = window.AudioContext || (window as any).webkitAudioContext
   111→   111→        audioCtxRef.current = new AudioCtx()
   112→   112→      }
   113→   113→      if (audioCtxRef.current.state === 'suspended') {
   114→   114→        audioCtxRef.current.resume().catch(() => {})
   115→   115→      }
   116→   116→      setAudioUnlocked(true)
   117→   117→    } catch {}
   118→   118→  }, [])
   119→   119→
   120→   120→  const playWebAudio = useCallback(async (arrayBuffer: ArrayBuffer): Promise<void> => {
   121→   121→    unlockAudio()
   122→   122→    const ctx = audioCtxRef.current
   123→   123→    if (!ctx || ctx.state === 'closed') throw new Error('Audio context unavailable')
   124→   124→    if (ctx.state === 'suspended') await ctx.resume().catch(() => {})
   125→   125→    const audioBuffer = await ctx.decodeAudioData(arrayBuffer)
   126→   126→    if (sourceRef.current) { try { sourceRef.current.stop() } catch {} }
   127→   127→    await new Promise<void>((resolve) => {
   128→   128→      const source = ctx.createBufferSource()
   129→   129→      source.buffer = audioBuffer
   130→   130→      source.connect(ctx.destination)
   131→   131→      sourceRef.current = source
   132→   132→      source.onended = () => resolve()
   133→   133→      try { source.start() } catch { resolve() }
   134→   134→    })
   135→   135→  }, [unlockAudio])
   136→   136→
   137→   137→  /* ---------- SPEAK (premium multi-layer TTS) ---------- */
   138→   138→  const speak = useCallback(async (text: string): Promise<void> => {
   139→   139→    // Layer 1: Kokoro in-browser neural voice (premium, iframe-proof)
   140→   140→    try {
   141→   141→      const kokoroResult = await kokoroSpeak(text)
   142→   142→      if (kokoroResult.ok && kokoroResult.blob) {
   143→   143→        setStateSafe('speaking')
   144→   144→        await playWebAudio(await kokoroResult.blob.arrayBuffer())
   145→   145→        return
   146→   146→      }
   147→   147→    } catch { /* continue to Edge TTS */ }
   148→   148→
   149→   149→    // Layer 2: Edge neural TTS via our API
   150→   150→    try {
   151→   151→      setStateSafe('speaking')
   152→   152→      const r = await safeJsonFetch('/api/tts', {
   153→   153→        method: 'POST',
   154→   154→        headers: { 'Content-Type': 'application/json' },
   155→   155→        body: JSON.stringify({ text, voice: ttsVoice, speed: 1.0 }),
   156→   156→      }, { timeoutMs: 30_000, label: 'Voice synthesis' })
   157→   157→      if (!r.ok) throw new Error(r.error || 'TTS failed')
   158→   158→      // r.data is { audio: base64, format } OR raw audio blob?
   159→   159→      // /api/tts returns audio bytes directly — re-fetch as blob if JSON.
   160→   160→      // Handle both: if JSON with audio field, decode; else fetch as blob.
   161→   161→      let arrayBuffer: ArrayBuffer
   162→   162→      if (r.data && typeof r.data === 'object' && r.data.audio) {
   163→   163→        const b64 = r.data.audio
   164→   164→        const byteStr = atob(b64)
   165→   165→        const bytes = new Uint8Array(byteStr.length)
   166→   166→        for (let i = 0; i < byteStr.length; i++) bytes[i] = byteStr.charCodeAt(i)
   167→   167→        arrayBuffer = bytes.buffer
   168→   168→      } else {
   169→   169→        // Fallback: fetch as blob directly
   170→   170→        const blobRes = await fetch('/api/tts', {
   171→   171→          method: 'POST',
   172→   172→          headers: { 'Content-Type': 'application/json' },
   173→   173→          body: JSON.stringify({ text, voice: ttsVoice, speed: 1.0 }),
   174→   174→        })
   175→   175→        const blob = await blobRes.blob()
   176→   176→        arrayBuffer = await blob.arrayBuffer()
   177→   177→      }
   178→   178→      await playWebAudio(arrayBuffer)
   179→   179→    } catch {
   180→   180→      // Layer 3: browser speechSynthesis (works in restricted contexts)
   181→   181→      try {
   182→   182→        await new Promise<void>((resolve) => {
   183→   183→          const utter = new SpeechSynthesisUtterance(text)
   184→   184→          utter.lang = lang
   185→   185→          utter.onend = () => resolve()
   186→   186→          utter.onerror = () => resolve()
   187→   187→          speechSynthesis.speak(utter)
   188→   188→          setTimeout(resolve, Math.min(30000, text.length * 100))
   189→   189→        })
   190→   190→      } catch {}
   191→   191→    }
   192→   192→  }, [ttsVoice, lang, playWebAudio, setStateSafe])
   193→   193→
   194→   194→  /* ---------- THINK (one LLM turn via /api/voice/turn) ---------- */
   195→   195→  const think = useCallback(async (userText: string): Promise<string> => {
   196→   196→    const r = await safeJsonFetch<any>('/api/voice/turn', {
   197→   197→      method: 'POST',
   198→   198→      headers: { 'Content-Type': 'application/json' },
   199→   199→      body: JSON.stringify({
   200→   200→        message: userText,
   201→   201→        history: historyRef.current.slice(-6),
   202→   202→        language: lang.split('-')[0],
   203→   203→        voice: ttsVoice,
   204→   204→        audio: 'UklGRiQAAABXQVZFZm10IBAAAAABAAEAQB8AAEAfAAABAAgAZGF0YQAAAAA=',
   205→   205→      }),
   206→   206→    }, { timeoutMs: 60_000, label: 'AI reply' })
   207→   207→    if (!r.ok || !r.data?.reply) throw new Error(r.error || 'AI had trouble responding. Try again.')
   208→   208→    return r.data.reply
   209→   209→  }, [lang, ttsVoice])
   210→   210→
   211→   211→  /* ---------- HANDLE FINAL TRANSCRIPT ---------- */
   212→   212→  const handleUtterance = useCallback(async (text: string) => {
   213→   213→    const clean = text.trim()
   214→   214→    if (!clean) return
   215→   215→    stopRecognition()
   216→   216→    setStateSafe('thinking')
   217→   217→    setInterim('')
   218→   218→    setError('')
   219→   219→    try {
   220→   220→      historyRef.current.push({ role: 'user', content: clean })
   221→   221→      const reply = await think(clean)
   222→   222→      historyRef.current.push({ role: 'assistant', content: reply })
   223→   223→      setTurns(prev => [...prev, { id: crypto.randomUUID(), user: clean, reply }])
   224→   224→      await speak(reply)
   225→   225→    } catch (e) {
   226→   226→      setError(e instanceof Error ? e.message : 'Something went wrong')
   227→   227→      toast({ title: 'Voice error', description: 'Could not get a response.', variant: 'destructive' })
   228→   228→    } finally {
   229→   229→      startListening()
   230→   230→    }
   231→   231→  }, [think, speak, toast, setStateSafe, stopRecognition])
   232→   232→
   233→   233→  /* ---------- ASR FALLBACK (record + upload) ---------- */
   234→   234→  const startAsrFallback = useCallback(() => {
   235→   235→    try {
   236→   236→      startAsrMode({ lang, ttsVoice, historyRef, setStateSafe, setTurns, setInterim, setError })
   237→   237→    } catch {
   238→   238→      setError('Voice is not available in this browser.')
   239→   239→      setStateSafe('idle')
   240→   240→    }
   241→   241→  }, [lang, ttsVoice, setStateSafe])
   242→   242→
   243→   243→  /* ---------- LISTEN (Web Speech API) ---------- */
   244→   244→  const startListening = useCallback(() => {
   245→   245→    const SR = (window as any).SpeechRecognition || (window as any).webkitSpeechRecognition
   246→   246→    if (!SR) {
   247→   247→      // No Web Speech — go straight to ASR fallback
   248→   248→      startAsrFallback()
   249→   249→      return
   250→   250→    }
   251→   251→    stopSpeaking()
   252→   252→    const rec = new SR()
   253→   253→    rec.lang = lang
   254→   254→    rec.continuous = true
   255→   255→    rec.interimResults = true
   256→   256→    rec.onresult = (event: any) => {
   257→   257→      let finalTranscript = ''
   258→   258→      let interimText = ''
   259→   259→      for (let i = event.resultIndex; i < event.results.length; i++) {
   260→   260→        const r = event.results[i]
   261→   261→        if (r.isFinal) finalTranscript += r[0].transcript
   262→   262→        else interimText += r[0].transcript
   263→   263→      }
   264→   264→      if (interimText) {
   265→   265→        setInterim(interimText)
   266→   266→        setMicLevel(0.3 + Math.random() * 0.4)
   267→   267→      }
   268→   268→      if (finalTranscript.trim()) handleUtterance(finalTranscript)
   269→   269→    }
   270→   270→    rec.onerror = (event: any) => {
   271→   271→      if (event.error === 'not-allowed') {
   272→   272→        const inIframe = window.self !== window.top
   273→   273→        setError(inIframe
   274→   274→          ? 'This preview frame blocks the microphone. Open the app in a new tab for voice.'
   275→   275→          : 'Microphone blocked — allow mic access in your browser settings.')
   276→   276→        if (inIframe) setIframeBlocked(true)
   277→   277→        setStateSafe('idle')
   278→   278→      } else if (['audio-capture', 'service-not-allowed', 'network'].includes(event.error)) {
   279→   279→        stopRecognition()
   280→   280→        setStateSafe('idle')
   281→   281→        setTimeout(() => startAsrFallback(), 50)
   282→   282→      }
   283→   283→    }
   284→   284→    rec.onend = () => {
   285→   285→      if (stateRef.current === 'listening' && recognitionRef.current) {
   286→   286→        try { recognitionRef.current.start() } catch {}
   287→   287→      }
   288→   288→    }
   289→   289→    recognitionRef.current = rec
   290→   290→    try {
   291→   291→      rec.start()
   292→   292→      setStateSafe('listening')
   293→   293→      setError('')
   294→   294→      const pulse = setInterval(() => setMicLevel(v => Math.max(0.05, v * 0.85)), 150)
   295→   295→      ;(rec as any).__pulse = pulse
   296→   296→    } catch {}
   297→   297→  }, [lang, handleUtterance, stopSpeaking, setStateSafe, startAsrFallback, stopRecognition])
   298→   298→
   299→   299→  /* ---------- CONTROLS ---------- */
   300→   300→  const toggle = useCallback(() => {
   301→   301→    const s = stateRef.current
   302→   302→    if (s === 'idle') { unlockAudio(); startListening() }
   303→   303→    else if (s === 'listening') { stopRecognition(); stopAsrMode(); setStateSafe('idle'); setInterim('') }
   304→   304→    else if (s === 'speaking') { stopSpeaking(); stopAsrMode(); startListening() }
   305→   305→    else { stopSpeaking(); stopRecognition(); stopAsrMode(); setStateSafe('idle') }
   306→   306→  }, [unlockAudio, startListening, stopRecognition, stopSpeaking, setStateSafe])
   307→   307→
   308→   308→  const endConversation = useCallback(() => {
   309→   309→    stopRecognition(); stopAsrMode(); stopSpeaking()
   310→   310→    setStateSafe('idle'); setTurns([]); setInterim('')
   311→   311→    historyRef.current = []
   312→   312→  }, [stopRecognition, stopSpeaking, setStateSafe])
   313→   313→
   314→   314→  /* ---------- AUTO-SCROLL TRANSCRIPT ---------- */
   315→   315→  useEffect(() => {
   316→   316→    if (scrollRef.current) scrollRef.current.scrollTop = scrollRef.current.scrollHeight
   317→   317→  }, [turns, interim])
   318→   318→
   319→   319→  /* ---------- CLEANUP ON CLOSE ---------- */
   320→   320→  useEffect(() => {
   321→   321→    if (!open) {
   322→   322→      stopRecognition(); stopAsrMode(); stopSpeaking()
   323→   323→      setStateSafe('idle')
   324→   324→    }
   325→   325→  }, [open])
   326→   326→
   327→   327→  useEffect(() => () => { stopRecognition(); stopSpeaking() }, [stopRecognition, stopSpeaking])
   328→   328→
   329→   329→  const stateMeta: Record<VoiceState, { label: string; sub: string }> = {
   330→   330→    idle: { label: 'Tap to talk', sub: 'Natural conversation — I hear you as you speak' },
   331→   331→    listening: { label: 'Listening…', sub: 'Speak naturally — I will respond when you pause' },
   332→   332→    thinking: { label: 'Thinking…', sub: 'Crafting your reply' },
   333→   333→    speaking: { label: 'Speaking', sub: 'Tap to interrupt' },
   334→   334→  }
   335→   335→  const meta = stateMeta[state]
   336→   336→
   337→   337→  const voicesForLang = (code: string) => {
   338→   338→    const langPrefix = code.split('-')[0]
   339→   339→    const edge = EDGE_VOICES.filter(v => v.id.startsWith(langPrefix))
   340→   340→    return [...NEXUS_VOICES, ...edge]
   341→   341→  }
   342→   342→
   343→   343→  return (
   344→   344→    <AnimatePresence>
   345→   345→      {open && (
   346→   346→        <motion.div
   347→   347→          initial={{ opacity: 0 }}
   348→   348→          animate={{ opacity: 1 }}
   349→   349→          exit={{ opacity: 0 }}
   350→   350→          transition={{ duration: 0.25 }}
   351→   351→          className="fixed inset-0 z-50 flex flex-col bg-background"
   352→   352→          role="dialog"
   353→   353→          aria-modal="true"
   354→   354→          aria-label="Nexus Voice Mode"
   355→   355→        >
   356→   356→          {/* Ambient gradient backdrop */}
   357→   357→          <div className="pointer-events-none absolute inset-0 overflow-hidden" aria-hidden>
   358→   358→            <div className="absolute left-1/2 top-1/3 h-[60vh] w-[60vh] -translate-x-1/2 -translate-y-1/2 rounded-full bg-gradient-to-br from-primary/25 via-rose-500/10 to-transparent blur-3xl" />
   359→   359→            <div className="absolute bottom-0 left-0 h-[40vh] w-[40vh] rounded-full bg-gradient-to-tr from-primary/10 to-transparent blur-3xl" />
   360→   360→          </div>
   361→   361→
   362→   362→          {/* Top bar */}
   363→   363→          <div className="relative z-10 flex items-center justify-between px-5 pt-[max(1rem,env(safe-area-inset-top))] pb-3">
   364→   364→            <div className="flex items-center gap-2">
   365→   365→              <Image src="/nexus-header-logo.png" alt="Nexus" width={96} height={32} className="h-7 w-auto" />
   366→   366→              <span className="rounded-full bg-primary/10 px-2 py-0.5 text-[10px] font-semibold uppercase tracking-widest text-primary">Voice</span>
   367→   367→            </div>
   368→   368→            <button onClick={onClose} aria-label="Close voice mode"
   369→   369→              className="flex h-10 w-10 items-center justify-center rounded-full bg-secondary/60 transition hover:bg-secondary">
   370→   370→              <X className="h-5 w-5" />
   371→   371→            </button>
   372→   372→          </div>
   373→   373→
   374→   374→          {/* Voice picker sheet */}
   375→   375→          <AnimatePresence>
   376→   376→            {showVoicePicker && (
   377→   377→              <motion.div
   378→   378→                initial={{ opacity: 0, y: 20 }}
   379→   379→                animate={{ opacity: 1, y: 0 }}
   380→   380→                exit={{ opacity: 0, y: 20 }}
   381→   381→                className="absolute inset-x-3 top-16 z-20 mx-auto max-w-md"
   382→   382→              >
   383→   383→                <div className="omni-scroll max-h-[60vh] overflow-y-auto rounded-3xl border border-border bg-card/95 p-4 shadow-2xl backdrop-blur-xl">
   384→   384→                  <div className="mb-3 flex items-center justify-between">
   385→   385→                    <p className="text-xs font-semibold uppercase tracking-widest text-muted-foreground">Premium voices</p>
   386→   386→                    <button onClick={() => setShowVoicePicker(false)} className="text-muted-foreground hover:text-foreground"><X className="h-4 w-4" /></button>
   387→   387→                  </div>
   388→   388→                  <div className="space-y-4">
   389→   389→                    <div>
   390→   390→                      <p className="mb-2 text-[10px] font-medium uppercase tracking-widest text-muted-foreground/70">Nexus voices</p>
   391→   391→                      <div className="space-y-1">
   392→   392→                        {NEXUS_VOICES.map(v => (
   393→   393→                          <VoiceRow key={v.id} v={v} active={ttsVoice === v.id} onPick={() => { setTtsVoice(v.id); setShowVoicePicker(false); toast({ title: `Voice: ${v.label}`, description: v.language }) }} />
   394→   394→                        ))}
   395→   395→                      </div>
